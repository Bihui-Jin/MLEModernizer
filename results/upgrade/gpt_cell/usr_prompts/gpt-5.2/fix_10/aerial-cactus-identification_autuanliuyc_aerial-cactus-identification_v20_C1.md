import numpy as np
import torch
from fastai.vision.all import ImageDataLoaders, imagenet_stats, Resize, Normalize

np.random.seed(42)

data = ImageDataLoaders.from_df(
    train_df,
    path=root,
    folder="train",
    valid_pct=0.01,
    seed=42,
    fn_col="id",
    label_col="has_cactus",
    item_tfms=Resize(SZ),
    batch_tfms=[*tsfm, Normalize.from_stats(*imagenet_stats)],
    bs=BS,
    device=torch.device("cuda:0"),
    test_df=test_df,
    test_folder="test",
)
