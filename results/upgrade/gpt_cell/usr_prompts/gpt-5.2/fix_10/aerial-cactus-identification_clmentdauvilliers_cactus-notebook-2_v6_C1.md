with ZipFile(path + "train.zip", "r") as zipper:
    names = zipper.namelist()

    train_prefix = None

    for candidate in ("train/", "aerial-cactus-identification/train/"):
        if any(n.startswith(candidate) for n in names):
            train_prefix = candidate
            break

    if train_prefix is None:
        for n in names:
            if "/train/" in n:
                train_prefix = n.split("/train/")[0] + "/train/"
                break

    if train_prefix is None:
        ids = files_dataframe["id"].astype(str).tolist()
        names_set = set(names)
        if any(img_id in names_set for img_id in ids):
            train_prefix = ""
        else:
            raise KeyError(
                "Could not locate train images inside train.zip (neither under a '/train/' prefix nor at zip root)."
            )

    members = []
    names_set = set(names)
    for img_id in files_dataframe["id"].astype(str).tolist():
        member = train_prefix + img_id
        if member in names_set:
            members.append(member)

    if len(members) == 0:
        raise KeyError(
            "No training images matched ids from train.csv inside train.zip; check zip structure."
        )

    zipper.extractall("./training/", members)

with ZipFile(path + "test.zip", "r") as zipper:
    zipper.extractall("./test/")
