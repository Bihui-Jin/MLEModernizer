from fastai.vision.all import aug_transforms, ImageDataLoaders, imagenet_stats

    return aug_transforms(**kwargs)

    folder="train/train",
    csv_fname="train.csv",
    valid_pct=0.2,
    seed=42,
    item_tfms=None,
    batch_tfms=get_transforms(),
    size=32,
    bs=bs,
    normalize=imagenet_stats,
    test="test/test",
)
