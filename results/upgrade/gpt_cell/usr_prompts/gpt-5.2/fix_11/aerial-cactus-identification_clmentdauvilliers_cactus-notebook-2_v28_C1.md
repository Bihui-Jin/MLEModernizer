with ZipFile(path + "train.zip", "r") as zipper:
    archive_names = zipper.namelist()
    members = []
    for p in training_files.tolist():
        matches = [n for n in archive_names if n.endswith(p)]
        if not matches:
            fname = os.path.basename(p)
            matches = [
                n for n in archive_names if n.endswith("/" + fname) or n == fname
            ]
        if matches:
            members.append(matches[0])
        else:
            raise KeyError(
                f"There is no item ending with {p!r} (or its filename) in the archive"
            )
    zipper.extractall("./training/", members)

import skimage.exposure as exposure


def preprocess(img):
    p2, p98 = np.percentile(img, (3, 97))
    img_rescale = exposure.rescale_intensity(img, in_range=(p2, p98))
