```
id,has_cactus

0017242f54ececa4512b4d7937d1e21e.jpg,0.5
001ee6d8564003107853118ab87df407.jpg,0.5
etc.
```

This dataset contains a large number of 32 x 32 thumbnail images containing aerial photos of a cactus. The file name of an image corresponds to its `id`.

- **train/** - the training set images
- **test/** - the test set images (you must predict the labels of these)
- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

## === cell 0
import numpy as np # linear algebra
import pandas as pd # data processing, CSV file I/O (e.g. pd.read_csv)

import os
print(os.listdir("../input"))


## === cell 1
from fastai import *
from fastai.vision import *
import torch


## === cell 2
from pathlib import Path

data_folder = Path("../input")


## === cell 3
train_df = pd.read_csv("../input/train.csv")
test_df = pd.read_csv("../input/sample_submission.csv")


## === cell 4
from fastai.vision.all import *

test_files = (data_folder / "test" / "test") / test_df["id"].values

trfm = dict(
    do_flip=True,
    flip_vert=True,
    max_rotate=10.0,
    max_zoom=1.1,
    max_lighting=0.2,
    max_warp=0.2,
    p_affine=0.75,
    p_lighting=0.75,
)

dblock = DataBlock(
    blocks=(ImageBlock, CategoryBlock),
    get_x=ColReader("id", pref=str(data_folder / "train" / "train") + os.sep),
    get_y=ColReader("has_cactus"),
    splitter=RandomSplitter(valid_pct=0.01, seed=42),
    item_tfms=Resize(128),
    batch_tfms=[*aug_transforms(**trfm), Normalize.from_stats(*imagenet_stats)],
)

train_img = dblock.dataloaders(
    train_df, bs=64, device=torch.device("cuda" if torch.cuda.is_available() else "cpu")
)
train_img.test = train_img.test_dl(test_files)


## === cell 5
train_img.show_batch(nrows=3, figsize=(7, 6))


## === cell 6
classes = getattr(train_img.train_ds, "vocab", None)
if classes is None:
    classes = getattr(train_img, "vocab", None)
print(classes)
print("No. of classes : {}".format(train_img.c))


## === cell 7
learner = cnn_learner(train_img, models.densenet161, metrics=[accuracy, error_rate])


## === cell 8
lr_min = learner.lr_find()
learner.lr_find()
