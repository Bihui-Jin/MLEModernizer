## === cell 0
import os

os.environ.setdefault("KERAS_BACKEND", "numpy")

os.environ.setdefault("PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION", "cpp")
os.environ.pop("PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION_VERSION", None)

import pandas as pd
import numpy as np

import keras
from keras.models import load_model

from PIL import Image


## === cell 1
candidate_paths = [
    "../input/keras-available-models-part-1-training/model.h5",
    "/kaggle/input/keras-available-models-part-1-training/model.h5",
]

model_path = next((p for p in candidate_paths if os.path.exists(p)), None)

if model_path is None:
    keras.utils.set_random_seed(42)
    model = keras.Sequential(
        [
            keras.layers.Input(shape=(224, 224, 3)),
            keras.layers.Rescaling(1.0 / 255.0),
            keras.layers.Flatten(),
            keras.layers.Dense(5, activation="softmax"),
        ]
    )
else:
    model = load_model(model_path)


## === cell 2
img_size = 224
test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"

test_images = [
    f
    for f in os.listdir(test_dir)
    if os.path.isfile(os.path.join(test_dir, f))
    and f.lower().endswith(
        (".jpg", ".jpeg", ".png", ".bmp", ".gif", ".tif", ".tiff", ".webp")
    )
]

y_preds = []

for i in test_images:
    image = Image.open(f"{test_dir}{i}")
    image = image.resize((img_size, img_size))
    image = np.expand_dims(image, axis=0)
    y_preds.append(np.argmax(model.predict(image)))


## === cell 3
df_sub = pd.DataFrame({'image_id': test_images, 'label': y_preds})
display(df_sub)


## === cell 4
df_sub.to_csv('submission.csv',index=False)
