train_df = pd.read_csv("../input/train.csv")
train_df.head()


## === cell 3
test_df = pd.read_csv("../input/sample_submission.csv")

from pathlib import Path
from fastai.vision.all import aug_transforms


class ImageList:
    @staticmethod
    def from_df(df, path=".", folder=None, cols="id"):
        base = Path(path)
        if folder is not None:
            base = base / folder
        return [base / fn for fn in df[cols].astype(str).tolist()]


test_img = ImageList.from_df(test_df, path="../input", folder="test/test")

tfms = aug_transforms(
    do_flip=True,
    flip_vert=True,
    max_rotate=10.0,
    max_zoom=1.1,
    max_lighting=0.2,
    max_warp=0.2,
    p_affine=0.75,
    p_lighting=0.75,
)


## === cell 4
from fastai.vision.all import (
    DataBlock,
    ImageBlock,
    CategoryBlock,
    RandomSplitter,
    ColReader,
    Resize,
)

train_items = ImageList.from_df(train_df, path="../input/train", folder="train")
test_items = ImageList.from_df(test_df, path="../input", folder="test/test")

dblock = DataBlock(
    blocks=(ImageBlock, CategoryBlock),
    get_items=lambda _: train_df.to_dict(orient="records"),
    get_x=lambda r: Path("../input/train/train") / r["id"],
    get_y=ColReader("has_cactus"),
    splitter=RandomSplitter(valid_pct=0.2, seed=42),
    item_tfms=Resize(32),
    batch_tfms=tfms,
)

data = dblock.dataloaders(source=None, bs=64)

data.test_dl = data.test_dl(test_items)


## === cell 5
data.show_batch(nrows=3, figsize=(5, 5))


## === cell 6
data.vocab


## === cell 7
from fastai.vision.learner import cnn_learner
from torchvision import models

from fastai.metrics import accuracy

model = cnn_learner(data, models.resnet50, metrics=[accuracy])


## === cell 9
from fastai.callback.schedule import minimum, steep, valley

lr_finder = model.lr_find(suggest_funcs=(minimum, steep, valley))

model.recorder.plot_lr_find(suggestions=lr_finder.suggestion())
