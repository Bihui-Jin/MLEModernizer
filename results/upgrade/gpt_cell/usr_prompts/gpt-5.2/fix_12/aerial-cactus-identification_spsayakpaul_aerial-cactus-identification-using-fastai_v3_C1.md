from fastai.vision.augment import aug_transforms


def get_transforms(
    do_flip=True,
    flip_vert=False,
    max_rotate=10.0,
    max_zoom=1.1,
    max_lighting=0.2,
    max_warp=0.2,
    p_affine=0.75,
    p_lighting=0.75,
):
    train_tfms = aug_transforms(
        do_flip=do_flip,
        flip_vert=flip_vert,
        max_rotate=max_rotate,
        max_zoom=max_zoom,
        max_lighting=max_lighting,
        max_warp=max_warp,
        p_affine=p_affine,
        p_lighting=p_lighting,
    )
    valid_tfms = []
    return train_tfms, valid_tfms


trfm = get_transforms(

    max_zoom=1.1,

from fastai.vision.all import (
    DataBlock,
    ImageBlock,
    CategoryBlock,
    ColReader,
    RandomSplitter,
    Resize,
    Normalize,
    imagenet_stats,
    PILImage,
)

test_path = Path(test_dir)

    blocks=(ImageBlock, CategoryBlock),
    get_x=ColReader("id", pref=str(train_path) + os.sep),
