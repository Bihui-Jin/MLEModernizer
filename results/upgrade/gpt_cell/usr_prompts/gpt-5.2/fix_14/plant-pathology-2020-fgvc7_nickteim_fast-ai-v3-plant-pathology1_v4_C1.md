from fastai.vision.all import *

class _V2ItemList:
    def __init__(self, items, path):
        self.items = list(items)
        self.path = Path(path)
        self._splitter = None

    def split_by_rand_pct(self, valid_pct=0.2, seed=42):
        self._splitter = RandomSplitter(valid_pct=valid_pct, seed=seed)
        return self

    def label_from_df(self, cols, label_cls=None):
        path = self.path
        items = self.items

        def get_x(o):
            return o

        def _image_id_from_path(p):
            return Path(p).stem

        def get_y(o):
            img_id = _image_id_from_path(o)
            row = df.loc[df["image_id"] == img_id, cols].iloc[0]
            return [c for c in cols if int(row[c]) == 1]

        splitter = (
            self._splitter
            if self._splitter is not None
            else RandomSplitter(valid_pct=0.2, seed=42)
        )

        dblock = DataBlock(
            blocks=(ImageBlock, MultiCategoryBlock(vocab=list(cols))),
            get_items=lambda _: items,
            get_x=get_x,
            get_y=get_y,
            splitter=splitter,
        )
        return dblock.dataloaders(source=path)


class ImageList:
    @classmethod
    def from_df(cls, df_in, path, folder="", suffix="", cols="image_id"):
        p = Path(path) / folder
        items = [p / f"{x}{suffix}" for x in df_in[cols].astype(str).tolist()]
        return _V2ItemList(items, path)

    @classmethod
    def from_csv(cls, path, csv_name, folder="", suffix=""):
        df_in = pd.read_csv(Path(path) / csv_name)
        return cls.from_df(df_in, path, folder=folder, suffix=suffix, cols="image_id")
