from fastai.vision.augment import aug_transforms

    do_flip=True,
    flip_vert=True,
    max_rotate=10.0,
    max_zoom=1.1,
    max_lighting=0.2,
    max_warp=0.2,
    p_affine=0.75,
    p_lighting=0.75,
)


## === cell 8
SZ=128
BS=64


## === cell 9
import numpy as np
import torch

from fastai.vision.all import ImageDataLoaders, Resize, Normalize, imagenet_stats

np.random.seed(42)

data = ImageDataLoaders.from_df(
    df=train_df,
    path=root / "train" / "train",
    fn_col="id",
    label_col="has_cactus",
    valid_pct=0.01,
    seed=42,
    item_tfms=Resize(SZ),
    batch_tfms=[*tsfm, Normalize.from_stats(*imagenet_stats)],
    bs=BS,
)

data.test = data.test_dl(test_set)

data

data.show_batch(nrows=3, figsize=(6, 6))


## === cell 12

## === cell 13
