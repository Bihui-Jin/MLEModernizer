def load_images(path):
    image_list = []
    for pth in path:
        img = cv2.imread(pth, 0) # read grayscale image
        img = cv2.resize(img, (input_shape[1], input_shape[0]))
        img = img / 255.
        img = np.expand_dims(img, axis=-1)
        image_list.append(img)
    return image_list
