        mx = np.unravel_index(rank.argmin(), rank.shape)
        w = (m - n) / (m + n / 10)
        m_gmean += (
            w
            * (
                np.log(concat_sub.iloc[:, mx[0] + 1])
                + np.log(concat_sub.iloc[:, mx[1] + 1])
            )
            / 2
        )
        s += w
        rank[mx] = 1
    m_gmean = np.exp(m_gmean / s).clip(0.0, 1.0)
