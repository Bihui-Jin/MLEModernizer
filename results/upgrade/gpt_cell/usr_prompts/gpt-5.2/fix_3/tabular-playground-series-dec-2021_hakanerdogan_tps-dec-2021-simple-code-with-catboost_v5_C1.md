## === cell 0
import numpy as np  # linear algebra
import pandas as pd  # data processing, CSV file I/O (e.g. pd.read_csv)

import os



## === cell 1
train_path = "../input/tabular-playground-series-dec-2021/train.csv"
test_path = "../input/tabular-playground-series-dec-2021/test.csv"
sub_path = "../input/tabular-playground-series-dec-2021/sample_submission.csv"

train_cols = pd.read_csv(train_path, nrows=0).columns.tolist()
test_cols = pd.read_csv(test_path, nrows=0).columns.tolist()

feature_cols = [c for c in train_cols if c not in ("Id", "Cover_Type")]
usecols_train = ["Id"] + feature_cols + ["Cover_Type"]
usecols_test = ["Id"] + feature_cols

dtype_train = {c: np.float32 for c in feature_cols}
dtype_train["Id"] = np.int32
dtype_train["Cover_Type"] = np.int8

dtype_test = {c: np.float32 for c in feature_cols}
dtype_test["Id"] = np.int32

train = pd.read_csv(train_path, usecols=usecols_train, dtype=dtype_train)
test = pd.read_csv(test_path, usecols=usecols_test, dtype=dtype_test)
sample_submission = pd.read_csv(sub_path)



## === cell 2
y_train = train["Cover_Type"].to_numpy()
X_train = train.drop(["Id", "Cover_Type"], axis=1)
X_test = test.drop(["Id"], axis=1)



## === cell 3
from sklearn import preprocessing

scaler = preprocessing.StandardScaler(copy=False)
X_train = scaler.fit_transform(X_train.to_numpy(dtype=np.float32, copy=False))
X_test = scaler.transform(X_test.to_numpy(dtype=np.float32, copy=False))



## === cell 4
from catboost import CatBoostClassifier

params = dict(verbose=0)
try:
    clf_CatBoostClassifier = CatBoostClassifier(**params, task_type="GPU")
    clf_CatBoostClassifier.fit(X_train, y_train)
except Exception:
    clf_CatBoostClassifier = CatBoostClassifier(**params, task_type="CPU")
    clf_CatBoostClassifier.fit(X_train, y_train)

sample_submission.iloc[:, 1] = clf_CatBoostClassifier.predict(X_test)
sample_submission.to_csv("submission_CatBoostClassifier.csv", index=False)
