## === cell 10
folds = KFold(n_splits=5, shuffle=True, random_state=2021)

predictions = np.zeros(len(test))

_classes = np.sort(y.unique())
class_to_idx = {c: i for i, c in enumerate(_classes)}
idx_to_class = {i: c for c, i in class_to_idx.items()}
y_mapped = y.map(class_to_idx).astype(int)

for fold, (trn_idx, val_idx) in enumerate(folds.split(train)):
    print(f"Fold: {fold}")
    X_train, X_test = train.iloc[trn_idx], train.iloc[val_idx]
    y_train, y_test = y_mapped.iloc[trn_idx], y_mapped.iloc[val_idx]

    model = XGBClassifier(tree_method="gpu_hist")

    model.fit(
        X_train,
        y_train,
        eval_set=[(X_test, y_test)],
        early_stopping_rounds=400,
        verbose=False,
    )
    pred = model.predict(X_test)
    acc = accuracy_score(y_test, pred)
    print(f" accuracy_score: {acc}")
    print("-" * 50)

    fold_pred_idx = model.predict(test).astype(int)
    fold_pred_labels = np.vectorize(idx_to_class.get)(fold_pred_idx).astype(float)
    predictions += fold_pred_labels / folds.n_splits
