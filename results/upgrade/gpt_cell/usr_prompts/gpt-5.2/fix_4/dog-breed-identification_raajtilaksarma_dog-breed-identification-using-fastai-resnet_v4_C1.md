def get_cv_idxs(n, cv_idx=0, val_pct=0.2, seed=42):
    rng = np.random.RandomState(seed + int(cv_idx))
    idxs = np.arange(n)
    rng.shuffle(idxs)
    n_val = int(round(n * val_pct))
    return idxs[:n_val]


val_idxs = get_cv_idxs(n)  # random 20% data for validation set
