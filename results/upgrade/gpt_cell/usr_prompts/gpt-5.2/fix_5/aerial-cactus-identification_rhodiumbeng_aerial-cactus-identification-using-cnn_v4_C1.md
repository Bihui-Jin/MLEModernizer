def prep_cnn_data(df, n_x, n_c, path):
    """load image data into tensors for CNN"""
    tensors = np.zeros((df.shape[0], n_x, n_x, n_c))
    for i in range(df.shape[0]):
        pic = load_img(path+df.iloc[i]['id'])
        pic_array = img_to_array(pic)
        tensors[i,:] = pic_array
    tensors = tensors / 255.
    return tensors
