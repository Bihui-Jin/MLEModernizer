data_folder = Path("../input")
train_df = pd.read_csv("../input/train.csv")

## === cell 5
from fastai.vision.all import *

train_path = data_folder / "train"
test_path = data_folder / "test"

batch_tfms = aug_transforms(
    do_flip=True,
    flip_vert=True,
    max_rotate=10.0,
    max_zoom=1.1,
    max_lighting=0.2,
    max_warp=0.2,
    p_affine=0.75,
    p_lighting=0.75,
)

train_img = ImageDataLoaders.from_df(
    train_df,
    path=train_path,
    folder="train",
    valid_pct=0.01,
    seed=42,
    fn_col="id",
    label_col="has_cactus",
    item_tfms=Resize(128),
    batch_tfms=[*batch_tfms, Normalize.from_stats(*imagenet_stats)],
    bs=64,
)

train_img.test_dl(
    test_df["id"].tolist(), with_labels=False, path=test_path, folder="test"
)


## === cell 6
learn50 = cnn_learner(train_img, models.densenet161, metrics=[error_rate, accuracy], model_dir="/tmp/model/")


## === cell 7

if hasattr(_rec, "plot_lr_find"):
    _rec.plot_lr_find()
else:
    _rec.plot()


## === cell 8
