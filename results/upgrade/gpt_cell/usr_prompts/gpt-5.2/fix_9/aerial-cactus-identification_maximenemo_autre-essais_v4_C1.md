def load_image_32x32_rgb(path):
    img = Image.open(path).convert("RGB")
    arr = np.asarray(img, dtype=np.uint8)  # (32, 32, 3)
    return arr


def _safe_extract_zip(zip_path, dest_dir):
    os.makedirs(dest_dir, exist_ok=True)
    with zipfile.ZipFile(zip_path, "r") as z:
        z.extractall(dest_dir)


def _find_images_dir(root, dirname):
    for cur_root, dirs, _files in os.walk(root):
        if os.path.basename(cur_root) == dirname:
            return cur_root
    raise FileNotFoundError(
        f"Could not find directory named '{dirname}' under '{root}'"
    )
