main_model_preprocess = transforms.Compose([
    transforms.CenterCrop(800),
    transforms.Resize((518, 518)),
    transforms.ToTensor(),  
    transforms.Normalize(mean=[0.5,0.5,0.5], std=[0.5,0.5,0.5])
])

def second_model_preprocess(image_np):
    image = Image.fromarray(image_np.astype('uint8'), 'RGB')
    image = image.resize((224, 224))
    image = np.array(image) / 255.0
    image = np.expand_dims(image, axis=0)
    return image

feature_description = {
    "image": tf.io.FixedLenFeature([], tf.string),
    "image_name": tf.io.FixedLenFeature([], tf.string)
}

model1 = load_model("/kaggle/input/densenet/keras/default/1/DenseNet (1).keras")
model2 = load_model("/kaggle/input/abc/keras/default/1/newModel7.keras")

path = "/kaggle/input/cassava-leaf-disease-classification/test_tfrecords"
tfrecs = os.listdir(path)

combined_probs = []
prediction = []
image_ids = []

for tfrec in tfrecs:
    raw_dataset = tf.data.TFRecordDataset(f"{path}/{tfrec}")

    for raw_record in raw_dataset:
        
        parsed_record = tf.io.parse_single_example(raw_record, feature_description)

        image_bytes = parsed_record["image"]
        image = tf.io.decode_jpeg(image_bytes)

        image_name = parsed_record["image_name"].numpy().decode("utf-8")
        image_ids.append(image_name)
        image_np = image.numpy()
        
        image = second_model_preprocess(image_np)
        
        output_tf = model1.predict(image, verbose=0)
        probs1 = output_tf[0]
        
        output_tf = model2.predict(image, verbose=0)
        probs2 = output_tf[0]
        
        combined_probs.append(np.concatenate((probs1, probs2)))
