def tf_transforms(image):
    image = tf.image.resize(image, [512, 512])
    image = tf.cast(image, tf.float32) / 255.0
    mean = [0.5, 0.5, 0.5]
    std = [0.5, 0.5, 0.5]
    image = (image - mean) / std
    image = np.expand_dims(image, axis=0)
    return image


torch_transforms = transforms.Compose([
    transforms.Resize((512, 512)),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5])
])

    transforms.Resize((518, 518)),

image_dir = '/kaggle/input/cassava-leaf-disease-classification/test_images'
image_ids = os.listdir(image_dir)

combined_output = []
prediction  = []

count = 0
length = len(image_ids)
for image_id in image_ids:
    image = Image.open(f"{image_dir}/{image_id}").convert("RGB")

    image1 = tf_transforms(image)

    image2 = torch_transforms(image)
    image2 = image2.unsqueeze(0).to(device)

    image3 = torch_transforms_VIT(image)
    image3 = image3.unsqueeze(0).to(device)

    train_prediction_1 = np.array(model1.predict(image1,verbose=False)[0])

    with torch.no_grad():
        train_prediction_2 = np.array(model2(image2).cpu()[0])
        train_prediction_3 = np.array(model3(image3).cpu()[0])
        del image

    combined_output.append(np.concatenate((train_prediction_1,train_prediction_2, train_prediction_3)))
    
    count+=1
    print(f"Count:{count}/{length}", end="\r")
