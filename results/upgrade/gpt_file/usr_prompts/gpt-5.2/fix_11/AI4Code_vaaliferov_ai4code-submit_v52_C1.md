# 1. Kaggle task description

## Task
Predict the correct ordering of the cells in a given notebook whose markdown cells have been shuffled.

## Metric
Kendall tau correlation between predicted cell orders and ground truth cell orders accumulated across the entire collection of test set notebooks.

Let $S$ be the number of swaps of adjacent entries needed to sort the predicted cell order into the ground truth cell order. In the worst case, a predicted order for a notebook with $n$ cells will need $\frac{1}{2} n(n-1)$ swaps to sort.

We sum the number of swaps from your predicted cell order across the entire collection of test set notebooks, and similarly with the worst-case number of swaps. We then compute the Kendall tau correlation as:

$K=1-4 \frac{\sum_i S_i}{\sum_i n_i\left(n_i-1\right)}$

## Submission Format
For each `id` in the test set (representing a notebook), you must predict `cell_order`, the correct ordering of its cells in terms of the cell ids. The file should contain a header and have the following format:

```
id,cell_order
0009d135ece78d,ddfd239c c6cd22db 1372ae9b ...
0010483c12ba9b,54c7cab3 fe66203e 7844d5f8 ...
0010a919d60e4f,aafc3d23 80e077ec b190ebb4 ...
0028856e09c5b7,012c9d02 d22526d1 3ae7ece3 ...
etc.
```

## Dataset 
- **train/** - A folder comprising about 140,000 JSON files with the filenames corresponding to the `id` field in the `csv` files. Each file contains the code and markdown cells of a notebook. **The code cells are in their original (correct) order. The markdown cells have been shuffled** and placed after the code cells.
- **train_orders.csv** - Gives the correct order of the cells for each notebook in the `train/` folder.
    - `id` - The notebook in file `{id}.json`.
    - `cell_order` - A space delimited list of the correct cell ordering given in terms of the order in `{id}.json`.
- **train_ancestors.csv** - A user may "fork" (that is, copy) the notebook of another user to create their own version. This file contains the forking history of notebooks in the training set. **Note: There is no corresponding file for the test set.**
    - `ancestor_id` - Identifies sets of notebooks that have a common origin or "ancestor". As no notebook in the test set has an ancestor in the training set, you may find this field to be of use as a grouping factor when constructing validation splits.
    - `parent_id` - Indicates that some version of the notebook `id` was forked from some version of the notebook `parent_id`. The notebook `parent_id` may or may not be present in the training data. (The parent may be missing because someone had forked a private notebook of their own, for instance.)
- **test/** - Notebooks from the test set. 
- **sample_submission.csv** - A sample submission file in the correct format.

def _normalize_text(s: str) -> str:
    if s is None:
        return ""
    s = s.replace("\n", " ").replace("\t", " ")
    s = " ".join(s.split())
    return s


def _make_code_doc(text: str) -> str:
    return _normalize_text(text)


def _make_md_doc(text: str) -> str:
    return _normalize_text(text)


def _pava_isotonic_increasing(y, w=None):
    """
    Minimal, dependency-free isotonic regression (PAVA) for an increasing sequence.
    """
    y = np.asarray(y, dtype=np.float64)
    n = y.shape[0]
    if n <= 1:
        return y
    if w is None:
        w = np.ones(n, dtype=np.float64)
    else:
        w = np.asarray(w, dtype=np.float64)

    blocks = [(float(y[i]), float(w[i]), 1) for i in range(n)]

    i = 0
    while i < len(blocks) - 1:
        if blocks[i][0] <= blocks[i + 1][0] + 1e-12:
            i += 1
            continue
        v1, w1, l1 = blocks[i]
        v2, w2, l2 = blocks[i + 1]
        nv = (v1 * w1 + v2 * w2) / (w1 + w2)
        nw = w1 + w2
        nl = l1 + l2
        blocks[i] = (nv, nw, nl)
        del blocks[i + 1]
        if i > 0:
            i -= 1

    out = np.empty(n, dtype=np.float64)
    idx = 0
    for val, _, ln in blocks:
        out[idx : idx + ln] = val
        idx += ln
    return out


def _stable_id_jitter(cell_id: str) -> float:
    """
    Deterministic tiny jitter in [-0.005, 0.005] derived from the cell id.
    """
    h = cell_id[-8:]
    try:
        v = int(h, 16)
    except Exception:
        v = abs(hash(cell_id)) & 0xFFFFFFFF
    u = (v % 1000003) / 1000003.0  # [0,1)
    return (u - 0.5) * 0.01


def _sigmoid(x: float) -> float:
    return 1.0 / (1.0 + np.exp(-x))


def _assign_slots_monotone_optimal(pos, n_slots: int):
    """
    Optimal monotone assignment minimizing sum_i (pos_i - slot_i)^2 with nondecreasing slots.
    Keeps semantics: we only decide which between-code slot each markdown belongs to.
    """
    pos = np.asarray(pos, dtype=np.float64)
    m = pos.shape[0]
    if m == 0:
        return np.array([], dtype=np.int32)
    if n_slots <= 1:
        return np.zeros(m, dtype=np.int32)

    order = np.argsort(pos, kind="mergesort")
    p = pos[order]

    if m > 300:
        slots = np.empty(m, dtype=np.int32)
        next_min = 0
        for j in range(m):
            s = int(np.rint(p[j]))
            if s < next_min:
                s = next_min
            if s > n_slots - 1:
                s = n_slots - 1
            slots[j] = s
            if next_min < n_slots - 1 and s > next_min:
                next_min = s
        out = np.empty(m, dtype=np.int32)
        out[order] = slots
        return out

    nS = int(n_slots)

    dp_prev = np.zeros(nS, dtype=np.float64)  # dp for 0 items
    parent = np.full((m, nS), -1, dtype=np.int16 if nS <= 32767 else np.int32)

    for j in range(1, m + 1):
        prefix_min = np.minimum.accumulate(dp_prev)
        arg = np.empty(nS, dtype=np.int32)
        best_i = 0
        best_v = dp_prev[0]
        arg[0] = 0
        for s in range(1, nS):
            if dp_prev[s] < best_v:
                best_v = dp_prev[s]
                best_i = s
            arg[s] = best_i

        cost = (p[j - 1] - np.arange(nS, dtype=np.float64)) ** 2
        dp_cur = cost + prefix_min
        parent[j - 1, :] = arg
        dp_prev = dp_cur

    slots_sorted = np.empty(m, dtype=np.int32)
    s = int(np.argmin(dp_prev))
    for j in range(m, 0, -1):
        slots_sorted[j - 1] = s
        s = int(parent[j - 1, s])

    out = np.empty(m, dtype=np.int32)
    out[order] = slots_sorted
    return out


def predict_cell_order_for_notebook(nb_json):
    """
    Core logic preserved:
    - Keep code cells in original order.
    - Score markdown vs code using per-notebook TF-IDF cosine similarity.
    - Compute continuous markdown positions and smooth them monotonically.
    - Assign markdowns to between-code slots with a monotone assignment.

    Changes (score improvement, same semantics):
    1) Slightly enrich isotonic weights: combine best_score with top-K similarity mass.
       This keeps the same smoothing idea but reduces flat plateaus that can create inversions.
    2) Replace O(m^2) markdown-markdown centrality with a fast centroid similarity proxy.
       This keeps "markdown content ordering" as a tie-breaker but is cheaper and stable.
    3) Add a deterministic global-position tie-breaker (slot + fractional within slot)
       to order markdowns consistently across adjacent slots and reduce tie-driven swaps.
    """
    cell_type = nb_json["cell_type"]
    source = nb_json["source"]

    cell_ids = list(cell_type.keys())
    code_ids = [cid for cid in cell_ids if cell_type[cid] == "code"]
    md_ids = [cid for cid in cell_ids if cell_type[cid] != "code"]

    if len(md_ids) == 0:
        return code_ids
    if len(code_ids) == 0:
        return md_ids

    code_docs = [_make_code_doc(source.get(cid, "")) for cid in code_ids]
    md_docs = [_make_md_doc(source.get(cid, "")) for cid in md_ids]

    vec = TfidfVectorizer(
        min_df=1,
        max_features=20000,
        token_pattern=r"(?u)\b[a-zA-Z_][a-zA-Z_0-9]{1,}\b",
        stop_words="english",
        ngram_range=(1, 2),
    )

    corpus = code_docs + md_docs
    X = vec.fit_transform(corpus)
    X_code = X[: len(code_ids)]
    X_md = X[len(code_ids) :]

    sim = cosine_similarity(X_md, X_code)  # (n_md, n_code)
    n_code = len(code_ids)

    K = int(np.clip(np.sqrt(n_code) + 2, 3, 12))
    K = min(K, n_code)
    topk_idx = np.argpartition(-sim, kth=K - 1, axis=1)[:, :K]
    topk_sim = np.take_along_axis(sim, topk_idx, axis=1)

    row_max = topk_sim.max(axis=1, keepdims=True)
    temp = 0.15
    w = np.exp((topk_sim - row_max) / temp) + 1e-8
    pos_raw = (w * topk_idx).sum(axis=1) / w.sum(axis=1)  # continuous in [0, n_code-1]

    best_code_idx = sim.argmax(axis=1)
    best_score = sim.max(axis=1)
    topk_mass = topk_sim.sum(
        axis=1
    )  # small extra confidence signal (still similarity-derived)

    order = np.argsort(pos_raw, kind="mergesort")
    pos_smooth = pos_raw.copy()

    iso_w = (best_score + 1e-3) * (1.0 + 0.15 * np.tanh(topk_mass))
    pos_smooth[order] = _pava_isotonic_increasing(pos_raw[order], w=iso_w[order])

    if len(md_ids) > 1:
        eps = 1e-3 + 5e-4 / (1.0 + np.sqrt(len(md_ids)))
        spread = np.linspace(-eps, eps, len(md_ids), dtype=np.float64)
        pos_spread = pos_smooth.copy()
        pos_spread[order] = pos_smooth[order] + spread
        pos_smooth = np.clip(
            np.maximum.accumulate(pos_spread[order])[np.argsort(order)],
            0.0,
            n_code - 1e-6,
        )

    conf = _sigmoid((best_score - 0.10) / 0.03)
    bias = (1.0 - conf) * 0.30 + conf * (-0.08)
    pos_biased = np.clip(pos_smooth + bias, 0.0, n_code - 1e-6)

    n_slots = n_code + 1
    slot_target = np.clip(pos_biased + 0.5, 0.0, n_code)  # [0..n_code]
    slots = _assign_slots_monotone_optimal(slot_target, n_slots=n_slots)

    if len(md_ids) > 1:
        md_centroid = X_md.mean(axis=0)
        md_centroid_sim = (X_md @ md_centroid.T).A1.astype(np.float64)
    else:
        md_centroid_sim = np.zeros(len(md_ids), dtype=np.float64)

    frac = slot_target - np.floor(slot_target)
    global_pos = slots.astype(np.float64) + np.clip(frac, 0.0, 1.0)

    md_by_slot = [[] for _ in range(n_slots)]
    for j, md_id in enumerate(md_ids):
        md_by_slot[int(slots[j])].append(j)

    out = []
    for s in range(n_slots):
        if md_by_slot[s]:
            idxs = md_by_slot[s]
            idxs.sort(
                key=lambda j: (
                    float(global_pos[j]),  # primary: consistent global ordering
                    float(slot_target[j]),  # secondary: your own continuous target
                    -float(
                        best_score[j]
                    ),  # then confidence to keep strong anchors earlier
                    -float(md_centroid_sim[j]),  # markdown content tie-breaker (cheap)
                    float(_stable_id_jitter(md_ids[j])),
                    md_ids[j],
                )
            )
            out.extend([md_ids[j] for j in idxs])
        if s < n_code:
            out.append(code_ids[s])

    if len(out) != len(cell_ids) or len(out) != len(set(out)):
        seen = set()
        dedup = []
        for x in out:
            if x not in seen:
                seen.add(x)
                dedup.append(x)
        missing = [x for x in cell_ids if x not in seen]
        out = dedup + missing

    return out
