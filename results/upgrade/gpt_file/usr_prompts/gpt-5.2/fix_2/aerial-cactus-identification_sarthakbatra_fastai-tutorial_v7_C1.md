# 1. Kaggle task description

## Task
Create a classifier to predict whether an image contains a cactus.

## Metric
Area under the ROC curve.

## Submission Format
For each ID in the test set, you must predict a probability for the `has_cactus` variable. The file should contain a header and have the following format:

```
id,has_cactus
000940378805c44108d287872b2f04ce.jpg,0.5
0017242f54ececa4512b4d7937d1e21e.jpg,0.5
001ee6d8564003107853118ab87df407.jpg,0.5
etc.
```

## Dataset
This dataset contains a large number of 32 x 32 thumbnail images containing aerial photos of a cactus. The file name of an image corresponds to its `id`.

- **train/** - the training set images
- **test/** - the test set images (you must predict the labels of these)
- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

import os

import torch

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)
if torch.cuda.is_available():
    torch.cuda.manual_seed_all(SEED)

print("Listing ../input:")
print(os.listdir("../input"))



## === cell 1
import sys, subprocess


def _pip_install(pkg):
    subprocess.check_call([sys.executable, "-m", "pip", "install", "-q", pkg])


try:
    import fastai

    ver = getattr(fastai, "__version__", "")
except Exception:
    ver = ""

need_fastai1 = True
if ver:
    need_fastai1 = not ver.startswith("1.")

if need_fastai1:
    _pip_install("fastai==1.0.61")

from fastai import *
from fastai.vision import *



## === cell 2
bs = 64



## === cell 3
path = Path("../input/aerial-cactus-identification")

path_train = path / "train"
path_test = path / "test"

print("path:", path)
print("path_train exists:", path_train.exists(), path_train)
print("path_test exists:", path_test.exists(), path_test)



## === cell 4
labels_df = pd.read_csv(path / "train.csv")
test_df = pd.read_csv(path / "sample_submission.csv")

print(labels_df.head())
print(test_df.head())
print("Train rows:", len(labels_df), "Test rows:", len(test_df))



## === cell 5
np.random.seed(SEED)

assert "id" in labels_df.columns and "has_cactus" in labels_df.columns
assert "id" in test_df.columns and "has_cactus" in test_df.columns

test_items = ImageList.from_df(test_df, path=path_test, cols="id")

data = (
    ImageList.from_df(labels_df, path=path_train, cols="id")
    .split_by_rand_pct(0.05, seed=SEED)
    .label_from_df(cols="has_cactus")
    .add_test(test_items)
    .transform(get_transforms(flip_vert=True, max_warp=0.0), size=128)
    .databunch(path=path, bs=bs)
    .normalize(imagenet_stats)
)

print(data)



## === cell 6
try:
    data.show_batch(rows=3, figsize=(10, 8))
except Exception as e:
    print("show_batch skipped:", repr(e))



## === cell 7
print("Classes:", data.classes)



## === cell 8
learn = cnn_learner(data, models.resnet101, metrics=accuracy, model_dir="/tmp/model/")
print(learn.model.__class__.__name__)



## === cell 9
try:
    learn.lr_find()
except Exception as e:
    print("lr_find skipped due to:", repr(e))



## === cell 10
try:
    learn.recorder.plot()
except Exception as e:
    print("recorder.plot skipped:", repr(e))



## === cell 11
lr = 3e-2



## === cell 12
learn.fit_one_cycle(5, slice(lr))



## === cell 13
preds, _ = learn.get_preds(ds_type=DatasetType.Test)
print("preds shape:", preds.shape)



## === cell 14
print("First 5 probs for class 1:", preds[:5, 1])



## === cell 15
test_df["has_cactus"] = preds[:, 1].cpu().numpy().astype(np.float64)
print(test_df.head())



## === cell 16
sub_path = Path("submission.csv")
