assert os.path.isdir(TEST_IMG_DIR), f"Missing test_images dir: {TEST_IMG_DIR}"

sample_sub.head(), sample_sub.shape



## === cell 1
IM_SIZE = 512
IMNET_MEAN = np.array([0.485, 0.456, 0.406], dtype=np.float32)
IMNET_STD = np.array([0.229, 0.224, 0.225], dtype=np.float32)


def preprocess_cv2_bgr(img_bgr, out_size=IM_SIZE):
    if img_bgr is None:
        raise ValueError("cv2.imread returned None (corrupt/missing image).")
    img_rgb = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)
    img_rgb = cv2.resize(img_rgb, (out_size, out_size), interpolation=cv2.INTER_AREA)
    x = img_rgb.astype(np.float32) / 255.0
    x = (x - IMNET_MEAN) / IMNET_STD
    x = np.transpose(x, (2, 0, 1))  # CHW
    return torch.from_numpy(x)


class InferSet(Dataset):
    def __init__(self, img_dir, image_ids, out_size=IM_SIZE):
        self.img_dir = img_dir
        self.image_ids = list(image_ids)
        self.out_size = out_size

    def __len__(self):
        return len(self.image_ids)

    def __getitem__(self, idx):
        fn = self.image_ids[idx]
        path = os.path.join(self.img_dir, fn)
        img = cv2.imread(path)
        x = preprocess_cv2_bgr(img, out_size=self.out_size)
        return x, fn


infer_ds = InferSet(TEST_IMG_DIR, sample_sub["image_id"].values, out_size=IM_SIZE)
infer_loader = DataLoader(
    infer_ds,
    batch_size=32,
    shuffle=False,
    num_workers=2,
    pin_memory=torch.cuda.is_available(),
)

xb, fnb = next(iter(infer_loader))
xb.shape, fnb[:3]



## === cell 2
model = torchvision.models.resnet50(
    weights=torchvision.models.ResNet50_Weights.IMAGENET1K_V2
)
model.fc = nn.Linear(model.fc.in_features, 5)
model = model.to(device)

weights_path = "../input/wwwwwww/weight_epoch_18.pth"
if os.path.isfile(weights_path):
    state = torch.load(weights_path, map_location="cpu")
    if isinstance(state, dict) and "model" in state:
        state = state["model"]
    missing, unexpected = model.load_state_dict(state, strict=False)
    print("Loaded weights:", weights_path)
    print("Missing keys:", len(missing), "Unexpected keys:", len(unexpected))
else:
    print(
        "Weights not found; using ImageNet-pretrained backbone with random 5-class head:",
        weights_path,
    )

model.eval()



## === cell 3
all_fns = []
all_preds = []

with torch.no_grad():
    for x, fns in infer_loader:
        x = x.to(device, non_blocking=True)
        logits = model(x)
        preds = torch.argmax(logits, dim=1).detach().cpu().numpy().astype(int)

        all_fns.extend(list(fns))
        all_preds.extend(list(preds))

pred_df = pd.DataFrame({"image_id": all_fns, "label": all_preds})

sub = sample_sub[["image_id"]].merge(pred_df, on="image_id", how="left")

sub["label"] = sub["label"].fillna(0).astype(int)

assert len(sub) == len(sample_sub), "Submission length mismatch vs sample_submission."
assert list(sub.columns) == ["image_id", "label"], "Wrong submission columns."

out_path = "./submission.csv"
sub.to_csv(out_path, index=False)
