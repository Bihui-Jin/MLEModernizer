def load_resnet50_model_safely(model_path: str) -> tf.keras.Model:
    try:
        model = tf.keras.models.load_model(model_path, compile=False)
        return model
    except Exception as e:
        print("Primary load_model failed, attempting fallback. Error:", repr(e))

        base = tf.keras.applications.ResNet50(
            include_top=False,
            weights=None,
            input_shape=(IMG_SIZE[0], IMG_SIZE[1], 3),
            pooling="avg",
        )
        x = keras.layers.Dropout(0.2)(base.output)
        out = keras.layers.Dense(5, activation="softmax")(x)
        model = keras.Model(inputs=base.input, outputs=out)

        model.load_weights(model_path)
        return model

sample_sub = pd.read_csv(SAMPLE_SUB_PATH)
assert {"image_id", "label"}.issubset(
    sample_sub.columns
), "Unexpected sample_submission columns"

preprocess_input = tf.keras.applications.resnet50.preprocess_input


def load_and_preprocess_image(path: str) -> tf.Tensor:
    img = keras.preprocessing.image.load_img(path)  # RGB by default
    img = img_to_array(img).astype(np.float32)
    img = tf.image.resize(img, IMG_SIZE, method="bilinear")
    img = preprocess_input(
        img
    )  # expects float32 in 0-255 RGB, converts to ResNet50 expected space
    return img




## === cell 3
preds = []
image_ids = sample_sub["image_id"].tolist()

for i in range(0, len(image_ids), BATCH_SIZE):
    batch_ids = image_ids[i : i + BATCH_SIZE]
    batch_imgs = [
        load_and_preprocess_image(os.path.join(TEST_IMG_DIR, img_id))
        for img_id in batch_ids
    ]
    batch = tf.stack(batch_imgs, axis=0)  # (B, H, W, 3)

    probs = model.predict(batch, verbose=0)
    batch_preds = np.argmax(probs, axis=1).astype(int).tolist()
    preds.extend(batch_preds)

assert len(preds) == len(sample_sub), "Prediction length mismatch"

my_submission = pd.DataFrame({"image_id": image_ids, "label": preds})
