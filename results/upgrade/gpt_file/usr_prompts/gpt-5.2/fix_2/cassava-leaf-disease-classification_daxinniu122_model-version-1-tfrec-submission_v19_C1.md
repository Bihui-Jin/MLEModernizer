preds = []

for image in sample_sub["image_id"].tolist():
    img = keras.preprocessing.image.load_img(os.path.join(TEST_IMG_DIR, image))
    img = img_to_array(img)
    img = tf.reshape(tf.image.resize(img, (512, 512)), (-1, 512, 512, 3))

    prediction1 = model1.predict(img, verbose=0) * alpha_1
    prediction2 = model2.predict(img, verbose=0) * alpha_2
    prediction3 = model3.predict(img, verbose=0) * alpha_3

    prediction = prediction1 + prediction2 + prediction3
    preds.append(int(np.argmax(prediction, axis=-1)[0]))

my_submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": preds})
my_submission.to_csv(SUB_PATH, index=False)
