import os
import pandas as pd
import numpy as np
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras.preprocessing.image import img_to_array


model1 = tf.keras.models.load_model("../input/f-models/ResNet50_f.h5")
model2 = tf.keras.models.load_model("../input/f-models/VGG19_f.h5")
model3 = tf.keras.models.load_model("../input/f-models/MobileNetV3L_f.h5")

norm_constant = 0.87 + 0.91 + 0.6
alpha_1 = 0.91 / norm_constant
alpha_2 = 0.87 / norm_constant
alpha_3 = 0.6 / norm_constant

preds = []

sample_sub = pd.read_csv(
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)


def _infer_hw(m, fallback=512):
    try:
        shp = m.input_shape  # e.g., (None, H, W, 3)
        h = int(shp[1]) if shp is not None and shp[1] is not None else fallback
        w = int(shp[2]) if shp is not None and shp[2] is not None else fallback
        return h, w
    except Exception:
        return fallback, fallback


H1, W1 = _infer_hw(model1, 512)
H2, W2 = _infer_hw(model2, 512)
H3, W3 = _infer_hw(model3, 512)

test_img_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"

for image in sample_sub.image_id:
    img = keras.preprocessing.image.load_img(os.path.join(test_img_dir, image))
    img = img_to_array(img).astype(np.float32)

    img1 = tf.reshape(tf.image.resize(img, (H1, W1)), (-1, H1, W1, 3))
    img2 = tf.reshape(tf.image.resize(img, (H2, W2)), (-1, H2, W2, 3))
    img3 = tf.reshape(tf.image.resize(img, (H3, W3)), (-1, H3, W3, 3))

    prediction1 = model1.predict(img1, verbose=0) * alpha_1
    prediction2 = model2.predict(img2, verbose=0) * alpha_2
    prediction3 = model3.predict(img3, verbose=0) * alpha_3

    prediction = prediction1 + prediction2 + prediction3
    preds.append(int(np.argmax(prediction, axis=1)[0]))

my_submission = pd.DataFrame({"image_id": sample_sub.image_id, "label": preds})
my_submission.to_csv("/kaggle/working/submission.csv", index=False)
