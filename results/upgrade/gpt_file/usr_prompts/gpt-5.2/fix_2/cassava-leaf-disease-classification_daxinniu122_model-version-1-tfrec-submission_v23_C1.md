import os
import pandas as pd
import numpy as np
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras.preprocessing.image import img_to_array


tf.random.set_seed(42)
np.random.seed(42)

TEST_DIR = "/kaggle/input/cassava-leaf-disease-classification/test_images"
SAMPLE_SUB_PATH = (
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)
OUT_PATH = "/kaggle/working/submission.csv"

MODEL1_PATH = "../input/f-models/ResNet50_f.h5"
MODEL2_PATH = "../input/f-models/VGG19_f.h5"
MODEL3_PATH = "../input/f-models/MobileNetV3L_f.h5"

model1 = tf.keras.models.load_model(MODEL1_PATH, compile=False)
model2 = tf.keras.models.load_model(MODEL2_PATH, compile=False)
model3 = tf.keras.models.load_model(MODEL3_PATH, compile=False)

sample_sub = pd.read_csv(SAMPLE_SUB_PATH)

preds = []

for image in sample_sub["image_id"].values:
    img = keras.preprocessing.image.load_img(os.path.join(TEST_DIR, image))
    img = img_to_array(img)  # float32 HWC
    img = tf.image.resize(img, (512, 512))
    img = tf.reshape(img, (-1, 512, 512, 3))

    prediction1 = model1.predict(img, verbose=0)
    prediction2 = model2.predict(img, verbose=0)
    prediction3 = model3.predict(img, verbose=0)

    prediction = prediction1 + prediction2 + prediction3
    preds.append(int(np.argmax(prediction, axis=1)[0]))

my_submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": preds})
my_submission.to_csv(OUT_PATH, index=False)
