def safe_load_model(path):
    if not os.path.exists(path):
        return None
    custom_objects = {}
    try:
        from tensorflow_hub import KerasLayer  # noqa: F401
        import tensorflow_hub as hub  # noqa: F401

        custom_objects["KerasLayer"] = KerasLayer
    except Exception:

        class _MissingKerasLayer(tf.keras.layers.Layer):
            def __init__(self, *args, **kwargs):
                super().__init__()
                raise RuntimeError(
                    "tensorflow_hub.KerasLayer is required to load this model in this environment."
                )

        custom_objects["KerasLayer"] = _MissingKerasLayer

    try:
        return tf.keras.models.load_model(
            path, compile=False, custom_objects=custom_objects
        )
    except Exception as e:
        print(f"WARNING: Failed to load model at {path}: {type(e).__name__}: {e}")
        return None


model1 = safe_load_model("../input/f-models/ResNet50_f.h5")
model2 = safe_load_model("../input/f-models/VGG19_f.h5")
model3 = safe_load_model(
    "../input/f-models/MobileNetV3L_f.h5"
)  # kept for core-logic parity (unused below)

## === cell 3
preds = []

fallback_label = 0

for image in sample_sub.image_id.values:
    img_path = os.path.join(TEST_IMG_DIR, image)
    img = keras.preprocessing.image.load_img(img_path)
    img = img_to_array(img)

    img = tf.reshape(tf.image.resize(img, (512, 512)), (-1, 512, 512, 3))

    if (model1 is None) or (model2 is None):
        preds.append(fallback_label)
        continue

    prediction1 = model1.predict(img, verbose=0) * alpha_1
    prediction2 = model2.predict(img, verbose=0) * alpha_2
    prediction = prediction1 + prediction2  # + prediction3 (kept commented as original)
    preds.append(int(np.argmax(prediction, axis=1)[0]))

my_submission = pd.DataFrame({"image_id": sample_sub.image_id, "label": preds})
my_submission.to_csv(OUT_PATH, index=False)
print(f"Wrote submission: {OUT_PATH} shape={my_submission.shape}")
