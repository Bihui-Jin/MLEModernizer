## === cell 0

import os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras.preprocessing.image import smart_resize, img_to_array

tf.random.set_seed(42)
np.random.seed(42)

INPUT_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TEST_IMG_DIR = os.path.join(INPUT_DIR, "test_images")
SAMPLE_SUB_PATH = os.path.join(INPUT_DIR, "sample_submission.csv")
SUB_PATH = "/kaggle/working/submission.csv"

sample_sub = pd.read_csv(SAMPLE_SUB_PATH)

model = tf.keras.applications.EfficientNetB3(
    include_top=True,
    weights="imagenet",
    classes=1000,
    input_shape=(300, 300, 3),
)

x = model.layers[-2].output  # global average pooled features
outputs = keras.layers.Dense(5, activation="softmax", name="cassava_head")(x)
model = keras.Model(inputs=model.input, outputs=outputs)

preprocess = tf.keras.applications.efficientnet.preprocess_input

preds = []

for image_id in sample_sub["image_id"].values:
    img = keras.preprocessing.image.load_img(os.path.join(TEST_IMG_DIR, image_id))
    img = img_to_array(img)
    img = smart_resize(img, (300, 300))
    img = tf.reshape(img, (-1, 300, 300, 3))
    img = preprocess(img)

    prediction = model.predict(img, verbose=0)
    preds.append(int(np.argmax(prediction, axis=1)[0]))

my_submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": preds})
my_submission.to_csv(SUB_PATH, index=False)
