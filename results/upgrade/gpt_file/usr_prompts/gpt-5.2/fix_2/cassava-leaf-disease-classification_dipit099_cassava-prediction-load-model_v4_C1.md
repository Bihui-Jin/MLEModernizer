## === cell 0
import os
import pandas as pd
from PIL import Image
from datetime import datetime

import numpy as np

import torch
import torch.nn as nn
import torchvision.transforms as transforms
from torch.utils.data import Dataset, DataLoader
from torchvision.models import vgg16, VGG16_Weights



## === cell 1
num_classes = 5


class CNN(nn.Module):
    def __init__(self, num_classes=5):
        super().__init__()
        self.conv_layers = vgg16(
            weights=VGG16_Weights.IMAGENET1K_V1
        )  # outputs 1000 logits
        self.dense_layers = nn.Sequential(
            nn.Linear(1000, 256),
            nn.ReLU(),
            nn.Linear(256, 64),
            nn.ReLU(),
            nn.Linear(64, num_classes),
        )

    def forward(self, x):
        out = self.conv_layers(x)  # (B,1000)
        out = out.view(out.size(0), -1)  # keep as in original
        out = self.dense_layers(out)  # (B,5)
        return out




## === cell 2
device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
print("device:", device)

model = CNN(num_classes=num_classes).to(device)
model.eval()
print("model type:", type(model))



## === cell 3
model.eval()



## === cell 4
transform = transforms.Compose(
    [
        transforms.Resize((224, 224)),
        transforms.ToTensor(),
        transforms.Normalize(mean=(0.485, 0.456, 0.406), std=(0.229, 0.224, 0.225)),
    ]
)

BATCH_SIZE = 32



## === cell 5
test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"


class TestDataset(Dataset):
    def __init__(self, root_dir, image_names=None, transform=None):
        self.root_dir = root_dir
        self.transform = transform
        if image_names is None:
            self.image_names = sorted(
                [
                    f
                    for f in os.listdir(root_dir)
                    if os.path.isfile(os.path.join(root_dir, f))
                ]
            )
        else:
            self.image_names = list(image_names)

    def __len__(self):
        return len(self.image_names)

    def __getitem__(self, idx):
        img_name = os.path.join(self.root_dir, self.image_names[idx])
        image = Image.open(img_name).convert("RGB")
        if self.transform:
            image = self.transform(image)
        return image, self.image_names[idx]


sample_path = "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
sample_sub = pd.read_csv(sample_path)
test_image_names = sample_sub["image_id"].tolist()

test_dataset = TestDataset(
    root_dir=test_dir, image_names=test_image_names, transform=transform
)
test_loader = DataLoader(
    test_dataset,
    batch_size=BATCH_SIZE,
    shuffle=False,
    num_workers=2,
    pin_memory=torch.cuda.is_available(),
)

print("Number of files in the test dataset:", len(test_dataset))



## === cell 6
model.eval()

predictions = []
img_names = []

t0 = datetime.now()

with torch.no_grad():
    for images, file_names in test_loader:
        images = images.to(device, non_blocking=True)
        outputs = model(images)
        preds = torch.argmax(outputs, dim=1)

        predictions.extend(preds.detach().cpu().numpy().tolist())
        img_names.extend(list(file_names))

submission_df = pd.DataFrame({"image_id": img_names, "label": predictions})

submission_df["image_id"] = submission_df["image_id"].astype(str)
submission_df["label"] = submission_df["label"].astype(int)

out_path = "/kaggle/working/submission.csv"
submission_df.to_csv(out_path, index=False)

dt = datetime.now() - t0
