## === cell 0
import pandas as pd
import numpy as np
import tensorflow as tf
from tensorflow.keras.models import load_model
from PIL import Image
import os



def second_model_preprocess(image_np: np.ndarray) -> np.ndarray:
    image = Image.fromarray(image_np.astype("uint8"), "RGB")
    image = image.resize((224, 224))
    image = np.array(image, dtype=np.float32) / 255.0
    image = np.expand_dims(image, axis=0)
    return image


feature_description = {
    "image": tf.io.FixedLenFeature([], tf.string),
    "image_name": tf.io.FixedLenFeature([], tf.string),
}

model2 = load_model("/kaggle/input/new-model8/keras/default/1/newModel8.keras")

path = "/kaggle/input/cassava-leaf-disease-classification/test_tfrecords"
tfrecs = sorted([f for f in os.listdir(path) if f.endswith(".tfrec")])

prediction = []
image_ids = []

for tfrec in tfrecs:
    raw_dataset = tf.data.TFRecordDataset(os.path.join(path, tfrec))

    for parsed in raw_dataset.map(
        lambda x: tf.io.parse_single_example(x, feature_description)
    ).as_numpy_iterator():
        image_bytes = parsed["image"]
        image_name = parsed["image_name"].decode("utf-8")

        image = tf.io.decode_jpeg(image_bytes).numpy()

        image_ids.append(image_name)
        x = second_model_preprocess(image)
        probs2 = model2.predict(x, verbose=0)[0]
        prediction.append(int(np.argmax(probs2)))
