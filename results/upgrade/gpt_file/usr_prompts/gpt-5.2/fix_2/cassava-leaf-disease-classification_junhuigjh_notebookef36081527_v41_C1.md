## === cell 0
import os
import numpy as np
import pandas as pd

import torch
import torch.nn as nn
from torchvision import transforms
from torchvision.io import read_image, ImageReadMode



class CustomResNet50(nn.Module):
    def __init__(self, base_model):
        super(CustomResNet50, self).__init__()
        self.base_model = nn.Sequential(
            *list(base_model.children())[:-1]
        )  # Remove the final FC layer
        self.fc1 = nn.Linear(base_model.fc.in_features, 512)  # First dense layer
        self.dropout = nn.Dropout(0.3)
        self.fc2 = nn.Linear(512, 5)  # Output layer for classification

    def forward(self, x):
        x = self.base_model(x)
        x = x.view(x.size(0), -1)  # Flatten
        x = self.fc1(x)
        x = self.dropout(x)
        x = self.fc2(x)
        return x


main_model_preprocess = transforms.Compose(
    [
        transforms.Resize((512, 512)),
        transforms.ConvertImageDtype(
            torch.float32
        ),  # scales uint8 [0,255] -> float [0,1]
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

model_path = "/kaggle/input/resnet50_70_512x512_more/pytorch/default/1/Resnet50_70_512x512_more.pth"
model = torch.load(model_path, map_location=device)
model.to(device)
model.eval()

test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
sample_path = "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"

sample_sub = pd.read_csv(sample_path)
image_ids = sample_sub["image_id"].tolist()

missing = [
    img for img in image_ids[:10] if not os.path.exists(os.path.join(test_dir, img))
]
if missing:
    raise FileNotFoundError(
        f"Some expected test images were not found under {test_dir}. Example missing: {missing[:3]}"
    )

predictions = []

with torch.no_grad():
    for img_name in image_ids:
        img_path = os.path.join(test_dir, img_name)

        img = read_image(img_path, mode=ImageReadMode.RGB)

        x = main_model_preprocess(img).unsqueeze(0).to(device, non_blocking=True)
        logits = model(x)
        pred = int(torch.argmax(logits, dim=1).item())
        predictions.append(pred)

assert len(predictions) == len(
    image_ids
), "Prediction count does not match image_id count."
