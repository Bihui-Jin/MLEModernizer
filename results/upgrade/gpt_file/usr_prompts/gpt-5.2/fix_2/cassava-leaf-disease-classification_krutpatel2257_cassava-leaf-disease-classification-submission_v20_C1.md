## === cell 4
to_tensor = transforms.ToTensor()


def load_rgb_image(path: str) -> np.ndarray:
    img = Image.open(path).convert("RGB")
    return np.array(img)




## === cell 5
sample_sub = pd.read_csv(sample_sub_path)
assert set(sample_sub.columns) >= {
    "image_id",
    "label",
}, "Unexpected sample submission columns"

predictions = []
tta_n = 5  # keep original loop count

with torch.no_grad():
    for _, sample_row in sample_sub.iterrows():
        img_path = os.path.join(test_images_path, sample_row.image_id)
        image_pred = None

        for _ in range(tta_n):
            image_np = load_rgb_image(img_path)
            aug_np = sub_aug(image=image_np)["image"]  # float32 HWC after Normalize
            image_t = to_tensor(aug_np).to(DEVICE)  # CHW float32

            outputs = model(image_t.unsqueeze(0))  # [1,5]
            if image_pred is None:
                image_pred = outputs
            else:
                image_pred = image_pred + outputs

        image_pred = image_pred / float(tta_n)

        pred_label = int(torch.argmax(image_pred, dim=1).item())
        predictions.append([sample_row.image_id, pred_label])

sub_df = pd.DataFrame(predictions, columns=["image_id", "label"])
sub_df.to_csv("submission.csv", index=False)
print(sub_df.head())
print("Wrote submission.csv with shape:", sub_df.shape)
