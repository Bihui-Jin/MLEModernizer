## === cell 2
class CassavaDataset(VisionDataset):
    """Custom dataset for the Cassava test data.

    Returns:
        image_tensor, filename
    """

    def __init__(self, data_dir, transform=None, images=None):
        super().__init__(root=data_dir)
        self.transform = transform

        if images is None:
            self.images = sorted(os.listdir(data_dir))
        else:
            self.images = list(images)

    def __getitem__(self, idx):
        filename = self.images[idx]
        img_path = os.path.join(self.root, filename)

        img = Image.open(img_path).convert("RGB")

        if self.transform:
            img = self.transform(img)

        return img, filename

    def __len__(self):
        return len(self.images)




## === cell 3
test_transforms = weights.transforms()

sample_df = pd.read_csv(sample_sub_path)

)

## === cell 4
all_names = []
all_preds = []

model.eval()
with torch.no_grad():
    for batch_idx, (inputs, filenames) in enumerate(test_loader):
        inputs = inputs.to(device, non_blocking=True)
        logits = model(inputs)
        probs = normalizer(logits)
        pred_labels = torch.argmax(probs, dim=1).tolist()

        all_names.extend(list(filenames))
        all_preds.extend(pred_labels)

assert len(all_names) == len(

assert set(all_names) == set(
    test_image_ids
), "Predicted filenames do not match sample submission filenames."
