# 5. Code solution

## === cell 0
import os
import random
import numpy as np
import pandas as pd

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
os.environ["PYTHONHASHSEED"] = str(SEED)

ROOT_DIR = "/kaggle/input/cassava-leaf-disease-classification"
print("ROOT_DIR exists:", os.path.exists(ROOT_DIR))
print("ROOT_DIR sample:", os.listdir(ROOT_DIR)[:10])



## === cell 1
import tensorflow as tf
from tensorflow.keras import layers, models

print("TensorFlow:", tf.__version__)

os.environ["TF_CPP_MIN_LOG_LEVEL"] = "2"

try:
    tf.keras.utils.set_random_seed(SEED)
    tf.config.experimental.enable_op_determinism()
except Exception as e:
    print("Determinism not fully enabled:", repr(e))



## === cell 2
from PIL import Image

TRAIN_CSV = os.path.join(ROOT_DIR, "train.csv")
SAMPLE_SUB = os.path.join(ROOT_DIR, "sample_submission.csv")
TRAIN_DIR = os.path.join(ROOT_DIR, "train_images")
TEST_DIR = os.path.join(ROOT_DIR, "test_images")

train_df = pd.read_csv(TRAIN_CSV)
sample_df = pd.read_csv(SAMPLE_SUB)

print(train_df.head())
print(sample_df.head())
print("Train rows:", len(train_df), "Test rows:", len(sample_df))



## === cell 3
IMG_SIZE = 300
size = (IMG_SIZE, IMG_SIZE)
NUM_CLASSES = 5


def load_image(path):
    img = Image.open(path).convert("RGB")
    img = img.resize(size)
    arr = np.asarray(img, dtype=np.float32) / 255.0  # normalize to [0,1]
    return arr


X = np.zeros((len(train_df), IMG_SIZE, IMG_SIZE, 3), dtype=np.float32)
y = train_df["label"].astype(np.int64).values

for i, fn in enumerate(train_df["image_id"].values):
    X[i] = load_image(os.path.join(TRAIN_DIR, fn))
    if (i + 1) % 3000 == 0:
        print(f"Loaded {i+1}/{len(train_df)}")

print("X shape:", X.shape, "y shape:", y.shape, "y min/max:", y.min(), y.max())



## === cell 4
idx = np.arange(len(X))
np.random.shuffle(idx)
split = int(0.9 * len(idx))
tr_idx, va_idx = idx[:split], idx[split:]

X_train, y_train = X[tr_idx], y[tr_idx]
X_val, y_val = X[va_idx], y[va_idx]

print("Train:", X_train.shape, "Val:", X_val.shape)



## === cell 5
new_model = models.Sequential(
    [
        layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3)),
        layers.Conv2D(16, 3, activation="relu"),
        layers.MaxPooling2D(),
        layers.Conv2D(32, 3, activation="relu"),
        layers.MaxPooling2D(),
        layers.Conv2D(64, 3, activation="relu"),
        layers.MaxPooling2D(),
        layers.Flatten(),
        layers.Dense(128, activation="relu"),
        layers.Dropout(0.3),
        layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

new_model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)

new_model.summary()



## === cell 6
history = new_model.fit(
    X_train, y_train, validation_data=(X_val, y_val), epochs=3, batch_size=32, verbose=2
)



## === cell 7
test_images = sample_df["image_id"].values.tolist()

preds = []
for i, image_id in enumerate(test_images):
    img_path = os.path.join(TEST_DIR, image_id)
    img = load_image(img_path)
    img = np.expand_dims(img, axis=0)
    p = new_model.predict(img, verbose=0).argmax(axis=1)[0]
    preds.append(int(p))
    if (i + 1) % 500 == 0:
        print(f"Predicted {i+1}/{len(test_images)}")

print("Preds length:", len(preds), "Expected:", len(test_images))
print("First 10 preds:", preds[:10])



## === cell 8
sub = pd.DataFrame({"image_id": test_images, "label": preds})

assert sub.shape[0] == sample_df.shape[0]
assert list(sub.columns) == ["image_id", "label"]

print(sub.head())
sub.to_csv("submission.csv", index=False)
print("Wrote submission.csv with rows:", len(sub))
