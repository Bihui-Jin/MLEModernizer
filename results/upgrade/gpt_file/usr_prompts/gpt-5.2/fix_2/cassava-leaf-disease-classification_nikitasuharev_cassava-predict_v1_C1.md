## === cell 0
import os
import math
import random
import numpy as np
import pandas as pd
import tensorflow as tf

print("Tensorflow version " + tf.__version__)

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
tf.random.set_seed(SEED)



## === cell 1
DATA_ROOT = "../input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
TRAIN_IMG_DIR = os.path.join(DATA_ROOT, "train_images")
TEST_IMG_DIR = os.path.join(DATA_ROOT, "test_images")
SAMPLE_SUB = os.path.join(DATA_ROOT, "sample_submission.csv")

IMG_SIZE = 512  # preserve original core logic: resize to 512x512
BATCH_SIZE = 16  # conservative to avoid OOM at 512x512 on Kaggle CPU/GPU



## === cell 2
import cv2


def load_image(jpeg_path, image_id):
    img = cv2.imread(os.path.join(jpeg_path, image_id))
    if img is None:
        raise FileNotFoundError(
            f"Could not read image: {os.path.join(jpeg_path, image_id)}"
        )
    img = img.astype(np.float32) / 255.0
    img = cv2.resize(img, (IMG_SIZE, IMG_SIZE))[:, :, ::-1]  # BGR->RGB
    return img


def test_generator(filepath, paths, batch_size=32):
    n = len(paths)
    for i in range(0, n, batch_size):
        batch_paths = paths[i : i + batch_size]
        batch = [load_image(filepath, p) for p in batch_paths]
        yield np.stack(batch, axis=0)




## === cell 3
submission = pd.read_csv(SAMPLE_SUB)
test_ids = submission["image_id"].values
print("Test images:", len(test_ids))



## === cell 4
MODEL_PATH = "../input/model-trained/my_model.h5"

model = None
if os.path.exists(MODEL_PATH):
    model = tf.keras.models.load_model(MODEL_PATH)
    print(f"Loaded model from: {MODEL_PATH}")
else:
    print(
        f"Pretrained model not found at {MODEL_PATH}. Training a fallback model from train_images/train.csv..."
    )

    train_df = pd.read_csv(TRAIN_CSV)
    train_ids = train_df["image_id"].values
    train_labels = train_df["label"].values.astype(np.int32)

    idx = np.arange(len(train_ids))
    rng = np.random.default_rng(SEED)
    rng.shuffle(idx)
    split = int(0.9 * len(idx))
    tr_idx, va_idx = idx[:split], idx[split:]

    tr_ids, tr_y = train_ids[tr_idx], train_labels[tr_idx]
    va_ids, va_y = train_ids[va_idx], train_labels[va_idx]

    def train_generator(image_dir, ids, labels, batch_size=32, shuffle=True):
        n = len(ids)
        order = np.arange(n)
        while True:
            if shuffle:
                rng = np.random.default_rng(SEED)
                rng.shuffle(order)
            for i in range(0, n, batch_size):
                batch_idx = order[i : i + batch_size]
                batch_ids = ids[batch_idx]
                x = np.stack(
                    [load_image(image_dir, img_id) for img_id in batch_ids], axis=0
                )
                y = labels[batch_idx]
                yield x, y

    steps_per_epoch = math.ceil(len(tr_ids) / BATCH_SIZE)
    val_steps = math.ceil(len(va_ids) / BATCH_SIZE)

    inputs = tf.keras.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
    x = tf.keras.layers.Conv2D(16, 3, padding="same", activation="relu")(inputs)
    x = tf.keras.layers.MaxPooling2D()(x)
    x = tf.keras.layers.Conv2D(32, 3, padding="same", activation="relu")(x)
    x = tf.keras.layers.MaxPooling2D()(x)
    x = tf.keras.layers.Conv2D(64, 3, padding="same", activation="relu")(x)
    x = tf.keras.layers.MaxPooling2D()(x)
    x = tf.keras.layers.GlobalAveragePooling2D()(x)
    x = tf.keras.layers.Dense(64, activation="relu")(x)
    outputs = tf.keras.layers.Dense(5, activation="softmax")(x)

    model = tf.keras.Model(inputs, outputs)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )

    history = model.fit(
        train_generator(
            TRAIN_IMG_DIR, tr_ids, tr_y, batch_size=BATCH_SIZE, shuffle=True
        ),
        validation_data=train_generator(
            TRAIN_IMG_DIR, va_ids, va_y, batch_size=BATCH_SIZE, shuffle=False
        ),
        steps_per_epoch=steps_per_epoch,
        validation_steps=val_steps,
        epochs=2,
        verbose=2,
    )



## === cell 5
ds_test = test_generator(TEST_IMG_DIR, test_ids, batch_size=BATCH_SIZE)
test_steps = math.ceil(len(test_ids) / BATCH_SIZE)

preds = model.predict(ds_test, steps=test_steps, verbose=1)
pred_labels = preds.argmax(axis=1).astype(np.int32)

pred_labels = pred_labels[: len(submission)]

submission["label"] = pred_labels
submission.to_csv("submission.csv", index=False)
print("Wrote submission.csv with shape:", submission.shape)
