## === cell 1
NUM_CLASSES = 5
IMG_SIZE = 512

base = keras.applications.ResNet50(
    include_top=False,
    weights="imagenet",
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
    pooling="avg",
)
x = keras.layers.Dense(NUM_CLASSES, activation="softmax")(base.output)
final_model2 = keras.Model(inputs=base.input, outputs=x)




## === cell 2
from tensorflow.keras.applications.resnet50 import preprocess_input


def load_and_preprocess_image(path, img_size=512):
    img = Image.open(path).convert("RGB")
    img = img.resize((img_size, img_size), resample=Image.BILINEAR)
    arr = np.asarray(img, dtype=np.float32)
    arr = np.expand_dims(arr, axis=0)
    arr = preprocess_input(arr)
    return arr




## === cell 3
sub_df = pd.read_csv(SAMPLE_SUB)
test_images = sub_df["image_id"].tolist()

predictions = []
for image_id in test_images:
    img_path = os.path.join(TEST_IMG_DIR, image_id)
    x = load_and_preprocess_image(img_path, img_size=IMG_SIZE)
    proba = final_model2.predict(x, verbose=0)
    pred = int(np.argmax(proba, axis=1)[0])
    predictions.append(pred)

assert len(predictions) == len(
