    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False

## === cell 2
def get_tta(image_size, p=0.5):
    imagenet_stats = {"mean": [0.485, 0.456, 0.406], "std": [0.229, 0.224, 0.225]}
    augs = A.Compose(
        [
            A.Resize(600, 600),
            A.CenterCrop(image_size, image_size),
            A.HorizontalFlip(p=p),
            A.Normalize(mean=imagenet_stats["mean"], std=imagenet_stats["std"]),
            ToTensorV2(),
        ]
    )
    return augs




## === cell 3
class CassavaLeafDataset(Dataset):
    def __init__(self, root_dir, transforms, csv_path):
        self.root_dir = root_dir
        self.transform = transforms
        self.dataframe = pd.read_csv(csv_path)
        if "image_id" not in self.dataframe.columns:
            raise ValueError(f"CSV at {csv_path} must contain 'image_id' column.")
        self.image_ids = self.dataframe["image_id"].astype(str).tolist()

    def __len__(self):
        return len(self.image_ids)

    def __getitem__(self, idx):
        image_id = self.image_ids[idx]
        img_name = os.path.join(self.root_dir, image_id)
        image = cv2.imread(img_name)
        if image is None:
            raise FileNotFoundError(f"Could not read image: {img_name}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = self.transform(image=image)["image"]
        dummy_label = 0
        return image, dummy_label

## === cell 5
weights = models.ResNeXt50_32X4D_Weights.IMAGENET1K_V2
model = models.resnext50_32x4d(weights=weights)
model.fc = nn.Linear(in_features=2048, out_features=5, bias=True)
model.to(device)
model.eval()

print(model.__class__.__name__, "ready")



## === cell 6
preds = []
with torch.no_grad():
    for images, _ in test_loader:
        images = images.to(device, non_blocking=True)
        outputs = model(images)
        predicted = outputs.argmax(dim=1)
        preds.extend(predicted.detach().cpu().tolist())

print("Num predictions:", len(preds))
