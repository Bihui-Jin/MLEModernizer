test_images = glob.glob(
    "../input/cassava-leaf-disease-classification/test_images/*.jpg"
)

if len(test_images) == 0:
    raise FileNotFoundError(
        "No test images found. Expected jpg files under "
        "../input/cassava-leaf-disease-classification/test_images/"
    )

df_test = pd.DataFrame(test_images, columns=["path"])


def make_test_gen(batch_size=64):
    my_test_idg = ImageDataGenerator(
        rotation_range=90,
        brightness_range=(0.2, 0.4),
        zoom_range=0.6,
        horizontal_flip=True,
        vertical_flip=True,
    )
    test_gen = my_test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=batch_size,
        seed=SEED,
        shuffle=False,
        class_mode=None,
        target_size=(512, 512),
    )
    return test_gen




## === cell 4
pred_list = []
