weight_path = "/kaggle/input/resnet50-trasnfer-learning-on-tpu/cassava_base.h5"

my_model = None
if os.path.exists(weight_path):
    my_model = tf.keras.models.load_model(weight_path)
else:
    base = tf.keras.applications.ResNet50(
        include_top=False,
        weights="imagenet",
        input_shape=(256, 256, 3),
        pooling="avg",
    )
    x = base.output
    x = tfl.Dropout(0.2)(x)
    outputs = tfl.Dense(5, activation="softmax")(x)
    my_model = tf.keras.Model(inputs=base.input, outputs=outputs)
    my_model.compile(
        optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
    )



## === cell 3
test_glob = "/kaggle/input/cassava-leaf-disease-classification/test_images/*.jpg"
test_images = glob.glob(test_glob)

if len(test_images) == 0:
    test_glob = "/kaggle/data/cassava-leaf-disease-classification/test_images/*.jpg"
    test_images = glob.glob(test_glob)

if len(test_images) == 0:
    raise FileNotFoundError(
        "No test images found. Tried:\n"
        f" - /kaggle/input/cassava-leaf-disease-classification/test_images/*.jpg\n"
        f" - /kaggle/data/cassava-leaf-disease-classification/test_images/*.jpg"
    )

df_test = pd.DataFrame(test_images, columns=["path"])

preprocess = tf.keras.applications.resnet50.preprocess_input


def make_test_gen(batch_size=16):
    my_test_idg = ImageDataGenerator(preprocessing_function=preprocess)
    test_gen = my_test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=batch_size,
        seed=SEED,
        shuffle=False,
        class_mode=None,
        target_size=(256, 256),
    )
    return test_gen




## === cell 4
pred_list = []

for i in range(1):
    test_gen = make_test_gen(batch_size=16)
    pred_test = my_model.predict(test_gen, verbose=True)
    pred_list.append(pred_test.tolist())

pred_test = np.mean(pred_list, axis=0)
pred_test_labels = np.argmax(pred_test, axis=-1)

final_submission = df_test.copy()
final_submission["image_id"] = final_submission.path.str.split("/").str[-1]
final_submission["label"] = pred_test_labels.astype(int)

final_csv = final_submission[["image_id", "label"]]

sample_path = "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
if os.path.exists(sample_path):
    sample = pd.read_csv(sample_path)
    final_csv = sample[["image_id"]].merge(final_csv, on="image_id", how="left")
    if final_csv["label"].isna().any():
        final_csv["label"] = final_csv["label"].fillna(0).astype(int)

final_csv.to_csv("submission.csv", index=False)
