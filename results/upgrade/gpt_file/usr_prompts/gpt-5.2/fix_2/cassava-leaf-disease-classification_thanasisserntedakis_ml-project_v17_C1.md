def build_base_model(app_model_fn, name):
    backbone = app_model_fn(
        include_top=False,
        weights="imagenet",
        input_shape=(IMG_SIZE, IMG_SIZE, 3),
        pooling="avg",
    )
    backbone.trainable = False  # inference only
    x = backbone.output
    out = Dense(5, activation="softmax", name=f"{name}_cls")(x)
    model = Model(backbone.input, out, name=name)
    return model


resnet = build_base_model(tf.keras.applications.ResNet50, "resnet50")
mobilenet = build_base_model(tf.keras.applications.MobileNetV2, "mobilenetv2")
densenet = build_base_model(tf.keras.applications.DenseNet121, "densenet121")
efficientnet = build_base_model(tf.keras.applications.EfficientNetB0, "efficientnetb0")

models = [resnet, mobilenet, densenet, efficientnet]

final_model = Sequential(
    [
        Input(shape=(20,)),
        Dense(64, activation="relu"),
        Dense(5, activation="softmax"),
    ],
    name="final_stacker",
)

final_model.compile(
    loss="categorical_crossentropy", optimizer="adam", metrics=["accuracy"]
)




## === cell 3
def get_final_model_input(models, data_gen, df_test):
    base_predictions = []
    for current_model in models:
        preds = current_model.predict(data_gen, verbose=1)
        base_predictions.append(preds)

    x_test = []
    for i in range(len(df_test.index)):
        new_row = []
        for prediction in base_predictions:
            new_row.append(prediction[i])
        x_test.append(new_row)

    return np.array(x_test, dtype=np.float32)
