class TestDataset(Dataset):
    def __init__(self, df, transform=None):
        self.df = df.reset_index(drop=True)
        self.file_names = self.df["image_id"].values
        self.transform = transform

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        file_name = self.file_names[idx]
        file_path = os.path.join(data_path, "test_images", str(file_name))
        image = cv2.imread(file_path)
        if image is None:
            raise FileNotFoundError(f"Could not read image: {file_path}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        if self.transform:
            augmented = self.transform(image=image)
            image = augmented["image"]
        return image




## === cell 4
def get_transforms(*, data):
    if data == "test":
        return Compose(
            [
                Resize(380, 380),
                Normalize(
                    mean=[0.485, 0.456, 0.406],
                    std=[0.229, 0.224, 0.225],
                ),
                ToTensorV2(),
            ]
        )
    raise ValueError(f"Unknown data={data}")




## === cell 5
from torchvision.models import resnext50_32x4d, ResNeXt50_32X4D_Weights


def build_model(num_classes=5):
    weights = ResNeXt50_32X4D_Weights.IMAGENET1K_V2
    m = resnext50_32x4d(weights=weights)
    in_features = m.fc.in_features
    m.fc = nn.Linear(in_features, num_classes)
    return m


model = build_model(num_classes=5)
model.to(device)
model.eval()

test_dataset = TestDataset(test, transform=get_transforms(data="test"))
test_loader = DataLoader(
    test_dataset,
    batch_size=32,
    shuffle=False,
    num_workers=2,
    pin_memory=torch.cuda.is_available(),
)


def preds(model, dataloader, device):
    model.eval()
    pred_list = []
    with torch.no_grad():
        for images in tqdm(dataloader, total=len(dataloader)):
            images = images.to(device, non_blocking=True)
            outputs = model(images)
            pred_list.extend(outputs.argmax(dim=1).detach().cpu().numpy().tolist())
    return pred_list


pred = preds(model, test_loader, device)
print("num preds:", len(pred), "num test rows:", len(test))
print("pred label counts:", Counter(pred))
