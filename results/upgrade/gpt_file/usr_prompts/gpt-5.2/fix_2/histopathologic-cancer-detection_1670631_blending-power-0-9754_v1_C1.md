# 1. Kaggle task description

## Task
Given a dataset of images from digital pathology scans, predict if the center 32x32px region of a patch contains at least one pixel of tumor tissue. Tumor tissue in the outer region of the patch does not influence the label. 

## Metric
Area under the ROC curve.

## Submission Format
For each `id` in the test set, you must predict a probability that center 32x32px region of a patch contains at least one pixel of tumor tissue. The file should contain a header and have the following format:

```
id,label
0b2ea2a822ad23fdb1b5dd26653da899fbd2c0d5,0
95596b92e5066c5c52466c90b69ff089b39f2737,0
248e6738860e2ebcf6258cdc1f32f299e0c76914,0
etc.
```

## Dataset
Files are named with an image `id`. The `train_labels.csv` file provides the ground truth for the images in the `train` folder. You are predicting the labels for the images in the `test` folder.

sub_template = pd.read_csv(sample_path)
sub_template = sub_template[["id", "label"]].copy()

df2 = sub_template.copy()




## === cell 1
def _read_image_rgb(path):
    try:
        import cv2  # type: ignore

        im = cv2.imread(path, cv2.IMREAD_COLOR)
        if im is None:
            raise ValueError("cv2.imread returned None")
        im = cv2.cvtColor(im, cv2.COLOR_BGR2RGB)
        return im
    except Exception:
        from PIL import Image  # type: ignore

        with Image.open(path) as img:
            return np.array(img.convert("RGB"))


def center_brightness_score(img_rgb):
    """
    Compute a simple probability-like score from the center 32x32 region.
    Core semantics align with the competition label definition (center region matters).
    """
    h, w = img_rgb.shape[:2]
    cy, cx = h // 2, w // 2
    half = 16
    crop = img_rgb[cy - half : cy + half, cx - half : cx + half, :]

    gray = (
        0.2989 * crop[..., 0] + 0.5870 * crop[..., 1] + 0.1140 * crop[..., 2]
    ).astype(np.float32)

    mean_brightness = float(gray.mean()) / 255.0
    score = 1.0 - mean_brightness  # higher => darker => more likely tumor

    score = 1.0 / (1.0 + np.exp(-(score - 0.5) * 6.0))
    return float(np.clip(score, 0.0, 1.0))




## === cell 2
ids = sub_template["id"].astype(str).tolist()

preds = np.zeros(len(ids), dtype=np.float32)
missing = 0

for i, img_id in enumerate(ids):
    img_path = os.path.join(test_dir, f"{img_id}.tif")
    if not os.path.exists(img_path):
        missing += 1
        preds[i] = 0.0
        continue
    img = _read_image_rgb(img_path)
    preds[i] = center_brightness_score(img)

df1 = pd.DataFrame({"id": ids, "label": preds})

df1 = df2[["id"]].merge(df1, on="id", how="left")
df1["label"] = df1["label"].fillna(0.0).astype(float)

df1["label"] = 0.8 * df1["label"].values + 0.2 * df2["label"].astype(float).values

assert df1.shape[0] == df2.shape[0]
