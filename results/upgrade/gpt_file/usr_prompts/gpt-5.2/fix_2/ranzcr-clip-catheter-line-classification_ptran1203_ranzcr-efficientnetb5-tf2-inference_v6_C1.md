def triple_image(image):
    return tf.concat([image] * 3, axis=-1)


def decode_jpeg_from_path(path):
    img_bytes = tf.io.read_file(path)
    image = tf.image.decode_jpeg(img_bytes, channels=1)
    image = tf.image.resize(image, (H, W))
    image = triple_image(image)
    return image


def preprocess(images, labels):
    images = tf.cast(images, tf.float32) / 255.0
    images = (images - mean) / std
    return images, labels


def get_model(
    baseline_weight=None, init_weight=None, lr=0.001, optimizer=tf.optimizers.Adam
):
    base_model = tf.keras.applications.EfficientNetB3(
        include_top=False, input_shape=(W, H, 3), pooling="avg", weights=baseline_weight
    )
    base_out = base_model.output
    out = tf.keras.layers.Dropout(0.3)(base_out)
    out = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(out)
    model = tf.keras.models.Model(inputs=base_model.input, outputs=out)

    model.compile(
        optimizer=optimizer(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC()],
    )

    if init_weight and tf.io.gfile.exists(init_weight):
        try:
            model.load_weights(init_weight)
            print(f"Weight loaded from {init_weight}")
        except Exception as e:
            print(f"Load weight from {init_weight} failed, {e}")
    else:
        print(
            f"Init weight not found at {init_weight}; running with baseline weights={baseline_weight}."
        )
    return model




## === cell 2
test_files = sorted([f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")])
image_ids = [os.path.splitext(f)[0] for f in test_files]
test_paths = [os.path.join(test_dir, f) for f in test_files]

paths_ds = tf.data.Dataset.from_tensor_slices(
    (test_paths, tf.zeros((len(test_paths), N_CLASSES), dtype=tf.float32))
)
test_data = paths_ds.map(
    lambda p, y: (decode_jpeg_from_path(p), y), num_parallel_calls=autotune
)
test_data = (
    test_data.batch(16).map(preprocess, num_parallel_calls=autotune).prefetch(autotune)
)

sub_df = pd.read_csv(sample_sub_path)
assert (
    "StudyInstanceUID" in sub_df.columns
), "sample_submission.csv missing StudyInstanceUID"

for c in target_cols:
    if c not in sub_df.columns:
        sub_df[c] = 0.0

model = get_model(None, weight_path)

preds = model.predict(test_data, verbose=1)
preds = np.asarray(preds, dtype=np.float32)

assert preds.shape[0] == len(
    image_ids
), f"Prediction rows {preds.shape[0]} != num test ids {len(image_ids)}"
assert preds.shape[1] == N_CLASSES, f"Prediction cols {preds.shape[1]} != {N_CLASSES}"

pred_df = pd.DataFrame(preds, columns=target_cols)
pred_df.insert(0, "StudyInstanceUID", image_ids)

sub_df = sub_df.drop(
    columns=[c for c in target_cols if c in sub_df.columns], errors="ignore"
)
sub_df = sub_df.merge(pred_df, on="StudyInstanceUID", how="left")

sub_df[target_cols] = sub_df[target_cols].fillna(0.5).clip(0.0, 1.0)

sub_df.to_csv("submission.csv", index=False)
