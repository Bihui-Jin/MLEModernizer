# 5. Code solution

## === cell 0
import os
import random
import numpy as np
import pandas as pd

os.environ.setdefault("PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION", "python")
os.environ.setdefault("PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION_VERSION", "2")

import tensorflow as tf

SEED = 42
os.environ["PYTHONHASHSEED"] = str(SEED)
random.seed(SEED)
np.random.seed(SEED)
tf.random.set_seed(SEED)

print("TF version:", tf.__version__)
print("Eager:", tf.executing_eagerly())



## === cell 1
from sklearn.model_selection import train_test_split
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.applications.efficientnet import preprocess_input
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D, Dropout
from tensorflow.keras.models import Model
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau

DATA_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(DATA_DIR, "train.csv")
SAMPLE_SUB = os.path.join(DATA_DIR, "sample_submission.csv")
TRAIN_IMG_DIR = os.path.join(DATA_DIR, "train_images")
TEST_IMG_DIR = os.path.join(DATA_DIR, "test_images")

train_df = pd.read_csv(TRAIN_CSV)
sample_df = pd.read_csv(SAMPLE_SUB)

train_df["path"] = train_df["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))

train_df["label"] = train_df["label"].astype(int)
assert (
    train_df["path"].map(os.path.exists).mean() > 0.99
), "Training image paths look wrong."
assert set(train_df["label"].unique()).issubset(
    set(range(5))
), "Labels must be in {0,1,2,3,4}."
train_df["label_str"] = train_df["label"].astype(str)

train_split, valid_split = train_test_split(
    train_df,
    test_size=0.2,
    random_state=SEED,
    stratify=train_df["label"],
)

IMG_SIZE = (224, 224)
BATCH_SIZE = 32
NUM_CLASSES = 5

datagen_aug = ImageDataGenerator(
    preprocessing_function=preprocess_input,
    rotation_range=45,
    width_shift_range=0.2,
    height_shift_range=0.2,
    shear_range=0.2,
    zoom_range=0.2,
    horizontal_flip=True,
    vertical_flip=True,
    fill_mode="nearest",
)

datagen_valid = ImageDataGenerator(preprocessing_function=preprocess_input)

train_gen = datagen_aug.flow_from_dataframe(
    dataframe=train_split,
    x_col="path",
    y_col="label_str",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="sparse",
    shuffle=True,
    seed=SEED,
)

valid_gen = datagen_valid.flow_from_dataframe(
    dataframe=valid_split,
    x_col="path",
    y_col="label_str",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="sparse",
    shuffle=False,
)

callbacks = [
    EarlyStopping(monitor="val_loss", patience=3, restore_best_weights=True),
    ReduceLROnPlateau(
        monitor="val_loss", patience=2, factor=0.5, min_lr=1e-6, verbose=1
    ),
]

idx_to_classname = {v: k for k, v in train_gen.class_indices.items()}
print("class_indices:", train_gen.class_indices)



## === cell 2
base = EfficientNetB0(
    include_top=False, weights="imagenet", input_shape=(IMG_SIZE[0], IMG_SIZE[1], 3)
)
base.trainable = False  # minimal/training-stable

x = GlobalAveragePooling2D()(base.output)
x = Dropout(0.2)(x)
out = Dense(NUM_CLASSES, activation="softmax")(x)

model = Model(inputs=base.input, outputs=out)
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()



## === cell 3
steps_per_epoch = int(np.ceil(len(train_split) / BATCH_SIZE))
validation_steps = int(np.ceil(len(valid_split) / BATCH_SIZE))

history = model.fit(
    train_gen,
    steps_per_epoch=steps_per_epoch,
    validation_data=valid_gen,
    validation_steps=validation_steps,
    epochs=10,
    callbacks=callbacks,
    verbose=1,
)

base.trainable = True
for layer in base.layers[:-20]:
    layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-5),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

history_ft = model.fit(
    train_gen,
    steps_per_epoch=steps_per_epoch,
    validation_data=valid_gen,
    validation_steps=validation_steps,
    epochs=5,
    callbacks=callbacks,
    verbose=1,
)



## === cell 4
sample_df["path"] = sample_df["image_id"].apply(lambda x: os.path.join(TEST_IMG_DIR, x))

missing = sample_df["path"].map(os.path.exists)
if missing.mean() < 0.99:
    bad = sample_df.loc[~missing, "path"].head(5).tolist()
    raise FileNotFoundError(f"Some test images not found. Example missing paths: {bad}")

test_gen = datagen_valid.flow_from_dataframe(
    dataframe=sample_df,
    x_col="path",
    y_col=None,
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode=None,
    shuffle=False,
)

test_steps = int(np.ceil(len(sample_df) / BATCH_SIZE))
probs = model.predict(test_gen, steps=test_steps, verbose=1)
pred_indices = np.argmax(probs, axis=1).astype(int)

pred_labels = np.array([int(idx_to_classname[i]) for i in pred_indices], dtype=int)

submission_df = pd.DataFrame(
    {"image_id": sample_df["image_id"].values, "label": pred_labels}
)
submission_path = "/kaggle/working/submission.csv"
submission_df.to_csv(submission_path, index=False)

print("Wrote:", submission_path)
print(submission_df.head())
print(submission_df["label"].value_counts().sort_index())
