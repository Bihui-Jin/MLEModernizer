## === cell 0
path = "/kaggle/input/cassava-leaf-disease-classification/"



## === cell 1
import os
import numpy as np
import pandas as pd
from fastai.vision.all import *
from fastai.callback.tracker import SaveModelCallback

set_seed(42, reproducible=True)

df = pd.read_csv(os.path.join(path, "train.csv"))

df_train = df.sample(int(0.7 * len(df)), random_state=42)
df_valid = df[~df.image_id.isin(df_train.image_id)].copy()

caminho = os.path.join(path, "train_images")

valid_idx = df.index[~df.image_id.isin(df_train.image_id)].to_numpy()

dls = ImageDataLoaders.from_df(
    df,
    path=caminho,
    fn_col="image_id",
    label_col="label",
    valid_idx=valid_idx,
    item_tfms=Resize(224),
    bs=64,
)



## === cell 2
import shutil
from pathlib import Path

models_dir = Path("./models")
models_dir.mkdir(parents=True, exist_ok=True)

found = None
for root, _, files in os.walk("/kaggle/input"):
    if "best_model.pth" in files:
        found = os.path.join(root, "best_model.pth")
        break

if found is not None:
    shutil.copy(found, models_dir / "best_model.pth")
    print(f"Copied pretrained weights from: {found} -> {models_dir/'best_model.pth'}")
else:
    print(
        "No best_model.pth found under /kaggle/input; will train a small model to enable submission."
    )

print("Models dir contents:", os.listdir(models_dir))



## === cell 3
learn = cnn_learner(
    dls, resnet18, metrics=[error_rate, accuracy], pretrained=False, path="./"
)
save_callback = SaveModelCallback(
    fname="best_model", with_opt=True, monitor="valid_loss", reset_on_fit=False
)

model_path = models_dir / "best_model.pth"
if model_path.exists():
    learn.load("best_model")
else:
    learn.fit_one_cycle(1, 1e-3, cbs=[save_callback])
    learn.load("best_model")



## === cell 4
img = dls.train_ds[0][0]
learn.predict(img)



## === cell 5
submission_df = pd.read_csv(os.path.join(path, "sample_submission.csv"))

test_img_dir = os.path.join(path, "test_images")
test_data_path = (
    submission_df["image_id"].apply(lambda x: os.path.join(test_img_dir, x)).tolist()
)

tst_dl = learn.dls.test_dl(test_data_path)

preds, _ = learn.tta(dl=tst_dl, n=10)

pred_labels = preds.argmax(dim=1).cpu().numpy().astype(int)

submission_df["label"] = pred_labels
submission_df



## === cell 6
submission_df.to_csv("submission.csv", index=False)
print("Wrote submission.csv with shape:", submission_df.shape)
print(submission_df.head())
