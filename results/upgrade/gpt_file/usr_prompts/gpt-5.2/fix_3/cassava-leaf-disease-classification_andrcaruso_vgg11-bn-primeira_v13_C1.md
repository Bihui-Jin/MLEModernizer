## === cell 0
import os
import numpy as np
import pandas as pd
from fastai.vision.all import *
from fastai.callback.tracker import SaveModelCallback

path = "../input/cassava-leaf-disease-classification/"

set_seed(42, reproducible=True)

df = pd.read_csv(os.path.join(path, "train.csv"))

df_train = df.sample(int(0.7 * len(df)), random_state=42)
df_valid = df[~df.image_id.isin(df_train.image_id)].copy()

df2 = df.copy()
df2["is_valid"] = df2.image_id.isin(df_valid.image_id)

caminho = os.path.join(path, "train_images")
dls = ImageDataLoaders.from_df(
    df2,
    path=caminho,
    fn_col="image_id",
    label_col="label",
    valid_col="is_valid",
    item_tfms=Resize(224),
    bs=64,
)



## === cell 1

os.makedirs("/kaggle/working/models", exist_ok=True)
print("Models directory:", "/kaggle/working/models")
print("Existing files:", os.listdir("/kaggle/working/models"))



## === cell 2
learn = cnn_learner(
    dls,
    resnet18,
    metrics=[error_rate, accuracy],
    pretrained=False,
    path="/kaggle/working",
)

save_callback = SaveModelCallback(
    fname="best_model", with_opt=True, monitor="valid_loss", reset_on_fit=False
)

ckpt_path = "/kaggle/working/models/best_model.pth"
if os.path.exists(ckpt_path):
    learn.load("best_model")
    print("Loaded existing checkpoint:", ckpt_path)
else:
    learn.fit_one_cycle(3, 1e-3, cbs=[save_callback])
    if os.path.exists(ckpt_path):
        print("Saved checkpoint:", ckpt_path)
    else:
        print(
            "Warning: checkpoint not found after training; continuing without explicit load."
        )



## === cell 3
img = dls.train_ds[0][0]
learn.predict(img)



## === cell 4
submission_df = pd.read_csv(os.path.join(path, "sample_submission.csv"))
test_img_dir = os.path.join(path, "test_images")
test_data_path = submission_df["image_id"].apply(
    lambda x: os.path.join(test_img_dir, x)
)

tst_dl = learn.dls.test_dl(test_data_path)

preds_tta, _ = learn.tta(dl=tst_dl, n=10)

preds_np = preds_tta.detach().cpu().numpy()
submission_df["label"] = np.argmax(preds_np, axis=1).astype(int)

submission_df.head(), submission_df.shape



## === cell 5
submission_path = "submission.csv"
submission_df.to_csv(submission_path, index=False)
print("Wrote:", submission_path)
