base_model = keras.applications.ResNet50(
    include_top=False,
    weights="imagenet",
    input_shape=(IMG_HEIGHT, IMG_WIDTH, 3),
)
base_model.trainable = (
    False  # inference-only pipeline (no training code existed originally)
)

inputs = keras.Input(shape=(IMG_HEIGHT, IMG_WIDTH, 3), name="image")

x = base_model(x, training=False)
x = layers.GlobalAveragePooling2D()(x)
outputs = layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = keras.Model(inputs, outputs)

model.summary()



## === cell 9


@tf.function(reduce_retracing=True)
def _predict_batch(x):
    return model(x, training=False)


def _load_image_tf(image_path):
    img_bytes = tf.io.read_file(image_path)
    img = tf.image.decode_jpeg(img_bytes, channels=3)  # uint8
    img = tf.image.resize(
        img,
        [IMG_HEIGHT, IMG_WIDTH],
        method=tf.image.ResizeMethod.LANCZOS3,
        antialias=True,
    )
    img = tf.cast(img, tf.float32)  # match previous np.float32
    return img  # (H, W, 3)


def get_augmented_images_batch(image_id):
    image_path = tf.constant(os.path.join(TEST_DIR, image_id))
    img = _load_image_tf(image_path)  # (H, W, 3)

    a0 = img
    a1 = tf.image.flip_left_right(img)
    a2 = tf.image.flip_up_down(img)
    a3 = tf.image.rot90(img, k=1)
    a4 = tf.image.rot90(img, k=2)
    a5 = tf.image.rot90(img, k=3)

    batch = tf.stack([a0, a1, a2, a3, a4, a5], axis=0)  # (6, H, W, 3)
    return batch




## === cell 10
test_results = []
image_ids = test_df["image_id"].values

test_results = [None] * len(image_ids)

for i, image_id in enumerate(image_ids):
    image_batch = get_augmented_images_batch(image_id)  # (6, H, W, 3)
    probs = _predict_batch(image_batch)  # (6, NUM_CLASSES)
    mean_prob = tf.reduce_mean(probs, axis=0)  # (NUM_CLASSES,)
    pred_label = int(tf.argmax(mean_prob, axis=0).numpy())

    test_results[i] = {"image_id": image_id, "label": pred_label}

test_results_df = pd.DataFrame(test_results)

submission = sample_sub[["image_id"]].merge(test_results_df, on="image_id", how="left")
submission["label"] = submission["label"].fillna(0).astype(int)

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Wrote {submission_path} with shape {submission.shape}")
