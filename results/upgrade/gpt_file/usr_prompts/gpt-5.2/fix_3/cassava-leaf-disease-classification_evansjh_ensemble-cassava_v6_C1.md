def _discover_h5_paths():
    explicit = [model_path_1, model_path_2, model_path_3, model_path_4, model_path_5]
    roots = [
        "/kaggle/input/combinedmodel3",
        "/kaggle/input/bestmodel_550_2",
        "/kaggle/input/bestmodel_8878",
    ]

    found = []
    for p in explicit:
        if os.path.isfile(p) and p.lower().endswith(".h5"):
            found.append(p)

    for r in roots:
        if os.path.isdir(r):
            for dirpath, _, filenames in os.walk(r):
                for fn in filenames:
                    if fn.lower().endswith(".h5"):
                        found.append(os.path.join(dirpath, fn))

    seen = set()
    out = []
    for p in found:
        if p not in seen:
            seen.add(p)
            out.append(p)
    return out


def _infer_input_size(model):
    try:
        shp = model.input_shape
        if isinstance(shp, list):
            shp = shp[0]
        if len(shp) == 4 and shp[1] and shp[2]:
            return (int(shp[1]), int(shp[2]))
    except Exception:
        pass
    return (550, 550)


h5_paths = _discover_h5_paths()

models = []
missing = []
for path in h5_paths:
    try:
        m = load_model(path, compile=False)

    except Exception as e:
        missing.append((path, f"load_error: {repr(e)}"))

image_ids = sample_csv["image_id"].tolist()

image_predictions = []
no_model_fallback_label = 3  # used only if zero models load

for image_id in image_ids:
    image_path = os.path.join(test_image_dir, image_id)

    if (not os.path.exists(image_path)) or (len(models) == 0):
        final_predicted_class = no_model_fallback_label
    else:
        model_predictions = []
        for model, input_size, _path in models:
            img = load_img(image_path, target_size=input_size)
            img_array = img_to_array(img) / 255.0
            img_array = np.expand_dims(img_array, axis=0)

            preds = model.predict(img_array, verbose=0)
            predicted_class = int(np.argmax(preds, axis=1)[0])
            model_predictions.append(predicted_class)

        final_predicted_class = Counter(model_predictions).most_common(1)[0][0]

    image_predictions.append((image_id, int(final_predicted_class)))

submission_df = pd.DataFrame(image_predictions, columns=["image_id", "label"])
