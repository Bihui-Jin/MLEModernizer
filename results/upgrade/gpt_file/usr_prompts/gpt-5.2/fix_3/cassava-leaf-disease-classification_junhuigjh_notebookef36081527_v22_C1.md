## === cell 0
import os
import sys
import numpy as np
import pandas as pd

os.environ["PYTHONHASHSEED"] = "42"
np.random.seed(42)

DATA_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TEST_IMG_DIR = os.path.join(DATA_DIR, "test_images")
SAMPLE_SUB_PATH = os.path.join(DATA_DIR, "sample_submission.csv")

print("Python:", sys.version)
print("Test images dir exists:", os.path.isdir(TEST_IMG_DIR))
print("Sample submission exists:", os.path.exists(SAMPLE_SUB_PATH))



## === cell 1
import torch
import torchvision
from torchvision import transforms
from PIL import Image

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Torch:", torch.__version__)
print("Torchvision:", torchvision.__version__)
print("Device:", device)

transform = transforms.Compose(
    [
        transforms.Resize(256, interpolation=transforms.InterpolationMode.BILINEAR),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)

weights = torchvision.models.EfficientNet_B0_Weights.IMAGENET1K_V1
model = torchvision.models.efficientnet_b0(weights=weights)
in_features = model.classifier[1].in_features
model.classifier[1] = torch.nn.Linear(in_features, 5)

torch.manual_seed(42)
torch.nn.init.normal_(model.classifier[1].weight, mean=0.0, std=0.02)
torch.nn.init.zeros_(model.classifier[1].bias)

model = model.to(device)
model.eval()



## === cell 2
sample_sub = pd.read_csv(SAMPLE_SUB_PATH)
image_ids = sample_sub["image_id"].tolist()

preds = []
missing = 0

with torch.inference_mode():
    for image_id in image_ids:
        img_path = os.path.join(TEST_IMG_DIR, image_id)
        if not os.path.exists(img_path):
            missing += 1
            preds.append(0)
            continue

        img = Image.open(img_path).convert("RGB")
        x = transform(img).unsqueeze(0).to(device)
        logits = model(x)
        pred = int(torch.argmax(logits, dim=1).item())
        preds.append(pred)

print("Predictions:", len(preds), "Missing files:", missing)



## === cell 3
sub = pd.DataFrame({"image_id": image_ids, "label": preds})
sub["label"] = sub["label"].astype(int)
sub.to_csv("submission.csv", index=False)

print("Wrote submission.csv with shape:", sub.shape)
print(sub.head())



## === cell 4
print("success")
