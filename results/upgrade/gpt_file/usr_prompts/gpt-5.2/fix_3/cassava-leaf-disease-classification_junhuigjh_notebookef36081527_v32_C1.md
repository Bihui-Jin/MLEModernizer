## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image

import torch
from torchvision import transforms, models

DATA_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TEST_IMG_DIR = os.path.join(DATA_DIR, "test_images")
SAMPLE_SUB_PATH = os.path.join(DATA_DIR, "sample_submission.csv")

torch.set_grad_enabled(False)

main_model_preprocess = transforms.Compose(
    [
        transforms.Resize((518, 518)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

MODEL_PATH = "/kaggle/input/resnet50_60/pytorch/default/1/Resnet50_60.pth"


def _build_fallback_model(num_classes: int = 5) -> torch.nn.Module:
    m = models.resnet50(weights=models.ResNet50_Weights.DEFAULT)
    in_feats = m.fc.in_features
    m.fc = torch.nn.Linear(in_feats, num_classes)
    return m


model = None
if os.path.exists(MODEL_PATH):
    obj = torch.load(MODEL_PATH, map_location=device)
    if isinstance(obj, torch.nn.Module):
        model = obj
    elif isinstance(obj, dict):
        state_dict = None
        for k in ("state_dict", "model_state_dict", "model", "net", "weights"):
            if k in obj and isinstance(obj[k], dict):
                state_dict = obj[k]
                break
        if state_dict is None:
            state_dict = obj
        model = _build_fallback_model(num_classes=5)
        fixed_state = {}
        for k, v in state_dict.items():
            nk = k.replace("module.", "")
            fixed_state[nk] = v
        missing, unexpected = model.load_state_dict(fixed_state, strict=False)
        if missing or unexpected:
            print(
                f"Warning: loaded checkpoint with missing={len(missing)} unexpected={len(unexpected)} keys."
            )
else:
    print(
        f"Warning: MODEL_PATH not found: {MODEL_PATH}. Using torchvision ResNet50 fallback."
    )
    model = _build_fallback_model(num_classes=5)

model.to(device)
model.eval()

sample_sub = pd.read_csv(SAMPLE_SUB_PATH)
image_ids = sample_sub["image_id"].tolist()

prediction = []



## === cell 1
missing = 0
for img_id in image_ids:
    img_path = os.path.join(TEST_IMG_DIR, img_id)
    if not os.path.exists(img_path):
        missing += 1
        prediction.append(0)
        continue

    image_pil = Image.open(img_path).convert("RGB")
    image_tensor = main_model_preprocess(image_pil).unsqueeze(0).to(device)

    output = model(image_tensor)

    if output.ndim == 1:
        output = output.unsqueeze(0)

    pred = int(torch.argmax(output, dim=1).item())
    prediction.append(pred)

if missing:
    print(f"Warning: {missing} test images missing; filled with label 0.")



## === cell 2
submission = pd.DataFrame({"image_id": image_ids, "label": prediction})

assert submission.shape[0] == len(image_ids), "Submission row count mismatch."
assert list(submission.columns) == ["image_id", "label"], "Submission columns mismatch."

submission.to_csv("submission.csv", index=False)
print("Wrote submission.csv with shape:", submission.shape)
print(submission.head())
