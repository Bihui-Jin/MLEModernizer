## === cell 0
import os
import random
import numpy as np
import pandas as pd

import torch
import torch.nn as nn
from torchvision import transforms
from torchvision.io import read_image, ImageReadMode
from torchvision.transforms import functional as TF
from torchvision.models import resnet50, ResNet50_Weights


SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)
torch.cuda.manual_seed_all(SEED)
torch.backends.cudnn.deterministic = True
torch.backends.cudnn.benchmark = False


class CustomResNet50(nn.Module):
    def __init__(self, base_model):
        super(CustomResNet50, self).__init__()
        self.base_model = nn.Sequential(
            *list(base_model.children())[:-1]
        )  # Remove final FC
        self.fc1 = nn.Linear(base_model.fc.in_features, 512)
        self.dropout = nn.Dropout(0.3)
        self.fc2 = nn.Linear(512, 5)

    def forward(self, x):
        x = self.base_model(x)
        x = x.view(x.size(0), -1)
        x = self.fc1(x)
        x = self.dropout(x)
        x = self.fc2(x)
        return x


class TensorResize:
    def __init__(self, size):
        self.size = size  # (H, W)

    def __call__(self, img: torch.Tensor) -> torch.Tensor:
        return TF.resize(img, self.size, antialias=True)


main_model_preprocess = transforms.Compose(
    [
        TensorResize((512, 512)),
        transforms.ConvertImageDtype(torch.float32),  # uint8 [0,255] -> float [0,1]
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

base = resnet50(weights=ResNet50_Weights.IMAGENET1K_V2)
model = CustomResNet50(base).to(device)
model.eval()

test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
sample_path = "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"

sample_sub = pd.read_csv(sample_path)
image_ids = sample_sub["image_id"].tolist()

missing = [
    img for img in image_ids[:10] if not os.path.exists(os.path.join(test_dir, img))
]
if missing:
    raise FileNotFoundError(
        f"Some expected test images were not found under {test_dir}. Example missing: {missing[:3]}"
    )

predictions = []
with torch.no_grad():
    for img_name in image_ids:
        img_path = os.path.join(test_dir, img_name)
        img = read_image(img_path, mode=ImageReadMode.RGB)  # CHW uint8

        x = main_model_preprocess(img).unsqueeze(0).to(device, non_blocking=True)
        logits = model(x)
        pred = int(torch.argmax(logits, dim=1).item())
        predictions.append(pred)

assert len(predictions) == len(
    image_ids
), "Prediction count does not match image_id count."



## === cell 1
submission = pd.DataFrame({"image_id": image_ids, "label": predictions})
submission.to_csv("submission.csv", index=False)

req_cols = ["image_id", "label"]
assert list(submission.columns) == req_cols, f"Submission columns must be {req_cols}"
assert len(submission) == len(
    sample_sub
), f"Expected {len(sample_sub)} rows (test set size), got {len(submission)}"

print(submission.head())
print("Wrote submission.csv with shape:", submission.shape)
