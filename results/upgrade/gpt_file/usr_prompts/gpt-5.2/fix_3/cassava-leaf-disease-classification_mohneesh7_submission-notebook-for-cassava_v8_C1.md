## === cell 0
import os
import json
import numpy as np
import pandas as pd
import tensorflow as tf

from PIL import Image

SEED = 42
tf.keras.utils.set_random_seed(SEED)
np.random.seed(SEED)

print("TF version:", tf.__version__)

DATA_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(DATA_DIR, "train.csv")
TRAIN_IMG_DIR = os.path.join(DATA_DIR, "train_images")
TEST_IMG_DIR = os.path.join(DATA_DIR, "test_images")
SAMPLE_SUB = os.path.join(DATA_DIR, "sample_submission.csv")

assert os.path.exists(TRAIN_CSV), f"Missing {TRAIN_CSV}"
assert os.path.exists(TRAIN_IMG_DIR), f"Missing {TRAIN_IMG_DIR}"
assert os.path.exists(TEST_IMG_DIR), f"Missing {TEST_IMG_DIR}"
assert os.path.exists(SAMPLE_SUB), f"Missing {SAMPLE_SUB}"

train_df = pd.read_csv(TRAIN_CSV)
sample_df = pd.read_csv(SAMPLE_SUB)

print("train_df:", train_df.shape, train_df.columns.tolist())
print("sample_df:", sample_df.shape, sample_df.columns.tolist())

NUM_CLASSES = train_df["label"].nunique()
print("NUM_CLASSES:", NUM_CLASSES)



## === cell 1
IMG_SIZE = 380
BATCH_SIZE = 16

preprocess = tf.keras.applications.efficientnet.preprocess_input


@tf.function
def load_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE], method="bilinear", antialias=True)
    img = tf.cast(img, tf.float32)
    img = preprocess(img)
    return img


_DS_OPTIONS = tf.data.Options()
_DS_OPTIONS.experimental_deterministic = True
_DS_OPTIONS.experimental_optimization.map_parallelization = True


def make_train_ds(df, shuffle=True, cache_path=None):
    paths = (TRAIN_IMG_DIR + "/" + df["image_id"].astype(str)).to_numpy()
    labels = df["label"].to_numpy(np.int32)

    ds = tf.data.Dataset.from_tensor_slices((paths, labels))
    if shuffle:
        ds = ds.shuffle(
            buffer_size=min(len(df), 4096), seed=SEED, reshuffle_each_iteration=True
        )

    def _map(p, y):
        x = load_image(p)
        return x, y

    ds = ds.map(_map, num_parallel_calls=tf.data.AUTOTUNE, deterministic=True)

    if cache_path is not None:
        ds = ds.cache(cache_path)
    else:
        ds = ds.cache()

    ds = ds.batch(BATCH_SIZE, drop_remainder=False)
    ds = ds.prefetch(tf.data.AUTOTUNE)
    ds = ds.with_options(_DS_OPTIONS)
    return ds


def make_test_ds(image_ids):
    image_ids = pd.Series(image_ids, dtype="string")
    paths = (TEST_IMG_DIR + "/" + image_ids).to_numpy()

    ds = tf.data.Dataset.from_tensor_slices(paths)

    def _map(p):
        x = load_image(p)
        return x

    ds = ds.map(_map, num_parallel_calls=tf.data.AUTOTUNE, deterministic=True)

    ds = ds.cache()
    ds = ds.batch(BATCH_SIZE, drop_remainder=False).prefetch(tf.data.AUTOTUNE)
    ds = ds.with_options(_DS_OPTIONS)
    return ds


from sklearn.model_selection import train_test_split

train_split, val_split = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=SEED
)

train_cache = "train_cache.tfdata"
val_cache = "val_cache.tfdata"

train_ds = make_train_ds(train_split, shuffle=True, cache_path=train_cache)
val_ds = make_train_ds(val_split, shuffle=False, cache_path=val_cache)

base = tf.keras.applications.EfficientNetB4(
    include_top=False, weights="imagenet", input_shape=(IMG_SIZE, IMG_SIZE, 3)
)
base.trainable = False

inputs = tf.keras.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = base(inputs, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
x = tf.keras.layers.Dropout(0.2)(x)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
    jit_compile=True,
)

model.summary()



## === cell 2
EPOCHS_HEAD = 2
history = model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS_HEAD, verbose=1)

base.trainable = True
fine_tune_at = int(len(base.layers) * 0.85)
for layer in base.layers[:fine_tune_at]:
    layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-5),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
    jit_compile=True,
)

EPOCHS_FT = 1
history_ft = model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS_FT, verbose=1)



## === cell 3
test_image_ids = sample_df["image_id"].tolist()
test_ds = make_test_ds(test_image_ids)

probs = model.predict(test_ds, verbose=1)
pred_labels = probs.argmax(axis=1).astype(int)

sub = pd.DataFrame({"image_id": test_image_ids, "label": pred_labels})
assert sub.shape[0] == sample_df.shape[0]
assert list(sub.columns) == ["image_id", "label"]
assert sub["label"].between(0, NUM_CLASSES - 1).all()

sub.to_csv("submission.csv", index=False)
print("Wrote submission.csv with shape:", sub.shape)
print(sub.head())
