## === cell 0
from fastai.vision.all import *
import pandas as pd
import json
from functools import partial



## === cell 1
path = Path("/kaggle/input/cassava-leaf-disease-classification/")
path.ls()



## === cell 2
train_df = pd.read_csv(path / "train.csv")
with open(path / "label_num_to_disease_map.json") as f:
    label_dict = json.load(f)
label_dict = {int(k): v for k, v in label_dict.items()}

df = train_df.set_index("image_id")
labels = df.to_dict()["label"]


def get_label(labels, x):
    x = Path(x)
    return labels[x.name]


def get_data(labels, bs=64, presize=500, resize=384):
    tfms = [
        Rotate(90),
        Warp(magnitude=0.4, p=1.0),
        Zoom(min_zoom=0.9, max_zoom=1.3),
        Brightness(max_lighting=0.5),
        Flip(),
        Contrast(),
        Resize(resize),
    ]
    comp = setup_aug_tfms(tfms)

    splitter = ColSplitter(col="is_valid")

    _df = train_df.copy()
    set_seed(42, reproducible=True)
    _df["is_valid"] = False
    valid_idx = []
    for c, g in _df.groupby("label"):
        idx = g.sample(frac=0.2, random_state=42).index
        valid_idx.extend(idx.tolist())
    _df.loc[valid_idx, "is_valid"] = True

    is_valid_map = dict(zip(_df["image_id"].values, _df["is_valid"].values))

    def _get_items(p):
        return get_image_files(p)

    def _splitter(items):
        valid = [
            i for i, o in enumerate(items) if is_valid_map.get(Path(o).name, False)
        ]
        train = [
            i for i, o in enumerate(items) if not is_valid_map.get(Path(o).name, False)
        ]
        return train, valid

    return DataBlock(
        blocks=(ImageBlock, CategoryBlock),
        get_items=_get_items,
        get_y=partial(get_label, labels),
        splitter=_splitter,
        item_tfms=[Resize(presize)],
        batch_tfms=[*comp, Normalize.from_stats(*imagenet_stats)],
    ).dataloaders(path / "train_images", bs=bs, num_workers=2)




## === cell 3
set_seed(42, reproducible=True)

dls = get_data(labels, bs=128, presize=384, resize=384)

learn = cnn_learner(dls, models.resnet50, metrics=[accuracy], pretrained=True)

test_files = get_image_files(path / "test_images")
test_dl = dls.test_dl(test_files)

predictions = []

ext_ckpt_dir = Path("/kaggle/input/resnet50/models")
loaded_any = False
for fold in range(3):
    ckpt = ext_ckpt_dir / f"resnet50-full-fold_{fold}.pth"
    if ckpt.exists():
        learn.load(ckpt.with_suffix(""))  # fastai expects path without .pth
        preds, _ = learn.tta(dl=test_dl, n=6)
        predictions.append(preds)
        loaded_any = True

if not loaded_any:
    learn.fine_tune(5, base_lr=3e-3)
    preds, _ = learn.tta(dl=test_dl, n=6)
    predictions.append(preds)



## === cell 4
avg_preds = torch.mean(torch.stack(predictions), dim=0)
pred_labels = torch.argmax(avg_preds, dim=1).cpu().numpy()

sample_sub = pd.read_csv(path / "sample_submission.csv")
test_fn = [p.name for p in test_files]
pred_map = dict(zip(test_fn, pred_labels.astype(int)))
ordered_labels = sample_sub["image_id"].map(pred_map).astype(int).values



## === cell 5
submission = pd.DataFrame(
    {"image_id": sample_sub["image_id"].values, "label": ordered_labels}
)
submission = submission[["image_id", "label"]]
assert len(submission) == len(sample_sub)
submission.to_csv("submission.csv", index=False)
