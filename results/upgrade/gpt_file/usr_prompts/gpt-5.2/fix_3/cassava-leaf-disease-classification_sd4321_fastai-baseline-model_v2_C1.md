## === cell 0
import os
import random
import numpy as np
import pandas as pd

import fastai
from fastai.vision.all import *



## === cell 1
path = Path("../input/cassava-leaf-disease-classification")
image_path = path / "train_images"
training_data_file = path / "train.csv"
sample_submission_file = path / "sample_submission.csv"
test_image_path = path / "test_images"

assert training_data_file.exists(), f"Missing: {training_data_file}"
assert sample_submission_file.exists(), f"Missing: {sample_submission_file}"
assert image_path.exists(), f"Missing: {image_path}"
assert test_image_path.exists(), f"Missing: {test_image_path}"



## === cell 2
seed = 42
set_seed(seed, reproducible=True)
torch.backends.cudnn.deterministic = True
torch.backends.cudnn.benchmark = False

train_df = pd.read_csv(training_data_file)
train_df["label"] = train_df["label"].astype(int)

img_size = 224
bs = 32


def get_x(r):
    return image_path / r["image_id"]


def get_y(r):
    return r["label"]


item_tfms = [Resize(img_size)]
batch_tfms = [Normalize.from_stats(*imagenet_stats)]

tiny_train_frac = 0.01  # ~187 images
splitter = RandomSplitter(valid_pct=0.2, seed=seed)

datablockk = DataBlock(
    blocks=(ImageBlock, CategoryBlock),
    splitter=splitter,
    get_x=get_x,
    get_y=get_y,
    item_tfms=item_tfms,
    batch_tfms=batch_tfms,
)

dls_full = datablockk.dataloaders(train_df, bs=bs)

train_items = dls_full.items
n_tiny = max(50, int(len(train_items) * tiny_train_frac))
rng = np.random.default_rng(seed)
tiny_idxs = rng.choice(len(train_items), size=n_tiny, replace=False)
train_df_tiny = train_items.iloc[tiny_idxs].reset_index(drop=True)

dls = datablockk.dataloaders(train_df_tiny, bs=bs)

print("Full train rows:", len(train_df), "| Tiny train rows used:", len(train_df_tiny))



## === cell 3
learn = vision_learner(dls, resnet18, metrics=accuracy)

learn.fine_tune(2)



## === cell 4
sub = pd.read_csv(sample_submission_file)
sub["image_id"] = sub["image_id"].astype(str)

test_files_ordered = [test_image_path / fn for fn in sub["image_id"].tolist()]
test_dl = dls.test_dl(test_files_ordered)

preds, _ = learn.get_preds(dl=test_dl)
pred_labels = preds.argmax(dim=1).cpu().numpy().astype(int)

sub["label"] = pred_labels
sub.to_csv("submission.csv", index=False)

print(sub.head())
print("Wrote submission.csv with shape:", sub.shape)
print("submission.csv exists:", Path("submission.csv").exists())
