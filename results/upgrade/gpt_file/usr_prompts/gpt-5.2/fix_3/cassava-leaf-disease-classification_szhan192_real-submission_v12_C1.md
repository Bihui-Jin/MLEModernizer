sample_sub_path = (
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)
test_img_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"

sample_sub = pd.read_csv(sample_sub_path)
image_ids = sample_sub["image_id"].tolist()

BATCH_SIZE = 32
preds = []

for start in range(0, len(image_ids), BATCH_SIZE):
    batch_ids = image_ids[start : start + BATCH_SIZE]
    batch_imgs = []

    for image in batch_ids:
        img = load_img(os.path.join(test_img_dir, image))
        img = img_to_array(img)
        img = smart_resize(img, (256, 256))
        batch_imgs.append(img)

    batch_arr = np.stack(batch_imgs, axis=0)  # (B, 256, 256, 3)
    prediction = model.predict(batch_arr, verbose=0)
    preds.extend(np.argmax(prediction, axis=-1).astype(int).tolist())

preds = preds[: len(image_ids)]

my_submission = pd.DataFrame({"image_id": image_ids, "label": preds})
out_path = "/kaggle/working/submission.csv"
