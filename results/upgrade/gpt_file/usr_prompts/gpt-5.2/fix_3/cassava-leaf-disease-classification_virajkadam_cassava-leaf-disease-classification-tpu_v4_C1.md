train_df = pd.read_csv(TRAIN_CSV)

num_classes = int(train_df["label"].nunique())
train_df["filepath"] = train_df["image_id"].apply(lambda x: os.path.join(TRAIN_DIR, x))

train_df.head(), sample_sub.head(), num_classes


## === cell 2
pretrained_path = "../input/notebook841c84bbfb/cassava_xception_v1.h5"

model = None
if os.path.exists(pretrained_path):
    model = keras.models.load_model(pretrained_path, compile=False)

model is not None


## === cell 3
IMG_SIZE = (512, 512)
BATCH_SIZE = 8

val_frac = 0.1
train_df_shuf = train_df.sample(frac=1.0, random_state=SEED).reset_index(drop=True)
n_val = int(len(train_df_shuf) * val_frac)
val_df = train_df_shuf.iloc[:n_val].copy()
trn_df = train_df_shuf.iloc[n_val:].copy()

_xcep_preprocess = keras.applications.xception.preprocess_input


def decode_and_resize(path, label=None):
    img_bytes = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img_bytes, channels=3)
    img = tf.image.resize(img, IMG_SIZE, method="bilinear")
    img = tf.cast(img, tf.float32)
    img = _xcep_preprocess(img)  # expects [0..255] float, outputs scaled for Xception
    if label is None:
        return img
    return img, tf.cast(label, tf.int32)


def make_ds(df, training):
    ds = tf.data.Dataset.from_tensor_slices((df["filepath"].values, df["label"].values))
    if training:
        ds = ds.shuffle(2048, seed=SEED, reshuffle_each_iteration=True)
    ds = ds.map(decode_and_resize, num_parallel_calls=tf.data.AUTOTUNE)
    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
    return ds


def make_test_ds(image_ids):
    paths = [os.path.join(TEST_DIR, iid) for iid in image_ids]
    ds = tf.data.Dataset.from_tensor_slices(paths)
    ds = ds.map(
        lambda p: decode_and_resize(p, None), num_parallel_calls=tf.data.AUTOTUNE
    )
    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
    return ds


train_ds = make_ds(trn_df, training=True)
val_ds = make_ds(val_df, training=False)

if model is None:
    base = keras.applications.Xception(
        include_top=False,
        weights="imagenet",
        input_shape=(IMG_SIZE[0], IMG_SIZE[1], 3),
        pooling="avg",
    )
    base.trainable = False

    inputs = keras.Input(shape=(IMG_SIZE[0], IMG_SIZE[1], 3))
    x = base(inputs, training=False)
    outputs = keras.layers.Dense(num_classes, activation="softmax")(x)
    model = keras.Model(inputs, outputs)

    model.compile(
        optimizer=keras.optimizers.Adam(learning_rate=1e-3),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )

    EPOCHS = 2
    history = model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS, verbose=2)


## === cell 4
test_ds = make_test_ds(sample_sub["image_id"].values)

probs = model.predict(test_ds, verbose=1)
preds = probs.argmax(axis=1).astype(int)

submission = pd.DataFrame({"image_id": sample_sub["image_id"].values, "label": preds})
submission = submission[["image_id", "label"]]
submission.to_csv("submission.csv", index=False)
