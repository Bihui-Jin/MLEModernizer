## === cell 0
import os
import glob
import numpy as np
import pandas as pd

import torch
from torchvision import transforms, models
from tqdm import tqdm
from PIL import Image

test_data_directory = "/kaggle/input/cassava-leaf-disease-classification/test_images"
sample_submission_path = (
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)
model_path = "/kaggle/input/vit_l_cassava/pytorch/default/1/model_weights_3.pth"

image_size = 518
num_classes = 5

val_transforms = transforms.Compose(
    [
        transforms.Resize((image_size, image_size)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Device:", device)


def find_fallback_weights_path(preferred_path: str) -> str | None:
    """
    Fix: original code hard-fails if the external weights dataset isn't attached.
    Minimal change: if preferred weights are missing, return None and we will use
    torchvision pretrained weights instead (same core model family: ViT).
    """
    if os.path.isfile(preferred_path):
        return preferred_path

    candidates = []
    for pattern in [
        "/kaggle/input/**/*.pth",
        "/kaggle/input/**/*.pt",
    ]:
        candidates.extend(glob.glob(pattern, recursive=True))

    def score(p):
        base = os.path.basename(p).lower()
        s = 0
        if "cassava" in p.lower():
            s += 2
        if "vit" in base:
            s += 2
        if "weight" in base or "weights" in base:
            s += 2
        if "model" in base:
            s += 1
        return s

    candidates = sorted(set(candidates), key=lambda p: (score(p), p), reverse=True)
    if not candidates:
        return None
    return candidates[0]


def torch_load_state_dict(path: str, map_location):
    """
    Fix: torch.load(weights_only=...) may not be supported consistently across environments.
    Try with weights_only first, fall back to plain torch.load.
    """
    try:
        return torch.load(path, map_location=map_location, weights_only=True)
    except TypeError:
        return torch.load(path, map_location=map_location)


weights_path = find_fallback_weights_path(model_path)
if weights_path is None:
    print(
        "No external .pth/.pt weights found under /kaggle/input; "
        "falling back to torchvision pretrained ViT weights."
    )
else:
    print(f"Using weights: {weights_path}")



## === cell 1

if weights_path is None:
    model = models.vit_h_14(
        weights=models.ViT_H_14_Weights.IMAGENET1K_SWAG_E2E_V1, image_size=image_size
    )
    model.heads.head = torch.nn.Linear(model.heads.head.in_features, num_classes)
else:
    model = models.vit_h_14(weights=None, image_size=image_size)
    model.heads.head = torch.nn.Linear(model.heads.head.in_features, num_classes)

    state = torch_load_state_dict(weights_path, map_location="cpu")

    if (
        isinstance(state, dict)
        and "state_dict" in state
        and isinstance(state["state_dict"], dict)
    ):
        state_dict = state["state_dict"]
    elif (
        isinstance(state, dict)
        and "model" in state
        and isinstance(state["model"], dict)
    ):
        state_dict = state["model"]
    else:
        state_dict = state

    missing, unexpected = model.load_state_dict(state_dict, strict=False)
    if missing or unexpected:
        print(
            f"Warning: load_state_dict non-strict load. Missing keys: {len(missing)}, unexpected keys: {len(unexpected)}"
        )

model = model.to(device)
model.eval()



## === cell 2
sample_sub = pd.read_csv(sample_submission_path)
image_ids = sample_sub["image_id"].tolist()

test_predictions = []

with torch.no_grad():
    for image_name in tqdm(image_ids, desc="Test"):
        image_path = os.path.join(test_data_directory, image_name)
        if not os.path.isfile(image_path):
            raise FileNotFoundError(f"Test image not found: {image_path}")

        image = Image.open(image_path).convert("RGB")
        image = val_transforms(image).unsqueeze(0).to(device)

        output = model(image)
        predicted_class = int(torch.argmax(output, dim=1).item())
        test_predictions.append(predicted_class)

submission_df = pd.DataFrame({"image_id": image_ids, "label": test_predictions})

submission_df = submission_df[["image_id", "label"]]
submission_df.to_csv("submission.csv", index=False)

print("Submission file created: submission.csv")
print(submission_df.head())
print("Rows:", len(submission_df))
