def _find_savedmodel_dir(base_path: str) -> str:
    """Return a directory that contains saved_model.pb(.txt); try common Kaggle layouts."""
    if base_path and tf.io.gfile.exists(base_path):
        if tf.io.gfile.exists(
            os.path.join(base_path, "saved_model.pb")
        ) or tf.io.gfile.exists(os.path.join(base_path, "saved_model.pbtxt")):
            return base_path

    parent = os.path.dirname(base_path) if base_path else "../input/efficientb7"
    if not parent:
        parent = "../input/efficientb7"
    if not tf.io.gfile.exists(parent):
        return base_path

    candidates = []
    for root, dirs, files in tf.io.gfile.walk(parent):
        if "saved_model.pb" in files or "saved_model.pbtxt" in files:
            candidates.append(root)
    if candidates:
        candidates = sorted(candidates, key=lambda p: (p.count(os.sep), len(p)))
        return candidates[0]
    return base_path


efficientB7 = _find_savedmodel_dir(efficientB7)
print("Resolved EfficientNetB7 SavedModel path:", efficientB7)

def _resolve_weights_path(path: str) -> str:
    """Accept either a checkpoint prefix or a directory; return a loadable weights path."""
    if not path or not tf.io.gfile.exists(path):
        return path

    if tf.io.gfile.isdir(path):
        ckpt = tf.train.latest_checkpoint(path)
        return ckpt if ckpt is not None else path

    return path

        image = tf.io.decode_jpeg(input_img, channels=3)

        tensor_image = tf.image.resize(image, [image_dims[0], image_dims[1]])
