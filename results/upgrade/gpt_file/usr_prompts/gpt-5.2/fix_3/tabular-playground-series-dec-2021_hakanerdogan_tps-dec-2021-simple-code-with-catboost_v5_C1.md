# 5. Code solution

## === cell 0
import numpy as np
import pandas as pd
import os




## === cell 1
train_path = "/kaggle/input/tabular-playground-series-dec-2021/train.csv"
test_path = "/kaggle/input/tabular-playground-series-dec-2021/test.csv"
sub_path = "/kaggle/input/tabular-playground-series-dec-2021/sample_submission.csv"

_train_cols = pd.read_csv(train_path, nrows=0).columns.tolist()
_test_cols = pd.read_csv(test_path, nrows=0).columns.tolist()

feature_cols = [c for c in _train_cols if c not in ("Id", "Cover_Type")]
usecols_train = ["Cover_Type"] + feature_cols
usecols_test = feature_cols

dtype_train = {c: np.int32 for c in feature_cols}
dtype_train["Cover_Type"] = np.int8
dtype_test = {c: np.int32 for c in feature_cols}

train = pd.read_csv(
    train_path,
    usecols=usecols_train,
    dtype=dtype_train,
    engine="c",
    low_memory=False,
)
test = pd.read_csv(
    test_path,
    usecols=usecols_test,
    dtype=dtype_test,
    engine="c",
    low_memory=False,
)
sample_submission = pd.read_csv(sub_path, engine="c")



## === cell 2
y_train = train["Cover_Type"].astype(np.int32).to_numpy()
X_train = train[feature_cols].to_numpy()
X_test = test[feature_cols].to_numpy()



## === cell 3
from sklearn.preprocessing import StandardScaler

scaler = StandardScaler(copy=False)
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)



## === cell 4
from catboost import CatBoostClassifier

clf_CatBoostClassifier = CatBoostClassifier(
    verbose=0, task_type="CPU", loss_function="MultiClass", random_seed=42
)

clf_CatBoostClassifier.fit(X_train, y_train)

pred = clf_CatBoostClassifier.predict(X_test)
pred = np.asarray(pred).reshape(-1).astype(int)

sample_submission["Cover_Type"] = pred
sample_submission.to_csv("submission_CatBoostClassifier.csv", index=False)
