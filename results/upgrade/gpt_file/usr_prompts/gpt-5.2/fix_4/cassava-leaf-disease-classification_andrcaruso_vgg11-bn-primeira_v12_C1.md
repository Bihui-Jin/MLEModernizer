import os

valid_idx = df.index[~df.image_id.isin(df_train.image_id)].to_numpy()

dls = ImageDataLoaders.from_df(
    df,
    path=caminho,
    fn_col="image_id",
    label_col="label",
    valid_idx=valid_idx,
    item_tfms=Resize(224),
    batch_tfms=aug_transforms(size=224, min_scale=0.75),
    bs=64,
)
