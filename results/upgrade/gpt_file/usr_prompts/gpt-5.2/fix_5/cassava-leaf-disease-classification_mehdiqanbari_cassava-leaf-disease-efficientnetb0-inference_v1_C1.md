# 5. Code solution

## === cell 0
import os

os.environ.pop("PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION", None)

import random
import numpy as np
import pandas as pd
import tensorflow as tf

from tensorflow.keras.layers import GlobalAveragePooling2D, Dense
from tensorflow.keras.models import Sequential
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.callbacks import ModelCheckpoint, ReduceLROnPlateau
from tensorflow.keras.optimizers import Adam

SEED = 42
os.environ["PYTHONHASHSEED"] = str(SEED)
random.seed(SEED)
np.random.seed(SEED)
tf.random.set_seed(SEED)

try:
    tf.config.threading.set_intra_op_parallelism_threads(0)
    tf.config.threading.set_inter_op_parallelism_threads(0)
except Exception:
    pass

AUTOTUNE = tf.data.AUTOTUNE

try:
    tf.config.optimizer.set_jit(True)
except Exception:
    pass

try:
    tf.config.experimental.enable_op_determinism()
except Exception:
    pass




## === cell 1
BATCH_SIZE = 16
IMG_SIZE = (224, 224)
EPOCHS = 5  # Keep runtime reasonable while producing a non-random, trainable model.

DATA_ROOT = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
TRAIN_DIR = os.path.join(DATA_ROOT, "train_images")
TEST_DIR = os.path.join(DATA_ROOT, "test_images")
SAMPLE_SUB = os.path.join(DATA_ROOT, "sample_submission.csv")




## === cell 2
def build_model():
    base_model = EfficientNetB0(weights=None, include_top=False)

    model = Sequential(
        [
            tf.keras.layers.Input(shape=(*IMG_SIZE, 3)),
            base_model,
            GlobalAveragePooling2D(),
            Dense(5, activation="softmax"),
        ]
    )

    model.compile(
        optimizer=Adam(learning_rate=1e-3),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
        jit_compile=True,
    )
    return model


model = build_model()
model.summary()




## === cell 3
train_df = pd.read_csv(TRAIN_CSV)
train_df["image_path"] = TRAIN_DIR + "/" + train_df["image_id"].astype(str)

try:
    existing = set(os.listdir(TRAIN_DIR))
    train_df = train_df[train_df["image_id"].isin(existing)].reset_index(drop=True)
except Exception:
    train_df = train_df[train_df["image_path"].map(os.path.isfile)].reset_index(
        drop=True
    )

num_classes = 5


def _decode_and_resize(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE, method=tf.image.ResizeMethod.BILINEAR)
    img = tf.cast(img, tf.float32) / 255.0
    return img


CACHE_TRAIN = "/kaggle/working/tfdata_cache_train"
CACHE_VAL = "/kaggle/working/tfdata_cache_val"
CACHE_TEST = "/kaggle/working/tfdata_cache_test"


def _make_ds(paths, labels=None, training=False, cache_path=None):
    if labels is None:
        ds = tf.data.Dataset.from_tensor_slices(paths)
        ds = ds.map(_decode_and_resize, num_parallel_calls=AUTOTUNE, deterministic=True)
    else:
        ds = tf.data.Dataset.from_tensor_slices((paths, labels))

        def _map_xy(p, y):
            x = _decode_and_resize(p)
            y = tf.one_hot(y, depth=num_classes)
            return x, y

        ds = ds.map(_map_xy, num_parallel_calls=AUTOTUNE, deterministic=True)

    if training:
        ds = ds.shuffle(
            buffer_size=min(8192, len(paths)), seed=SEED, reshuffle_each_iteration=True
        )

    ds = ds.batch(BATCH_SIZE, drop_remainder=False)

    if cache_path is not None:
        ds = ds.cache(cache_path)
    else:
        ds = ds.cache()

    opts = tf.data.Options()
    opts.experimental_slack = True
    ds = ds.with_options(opts)

    ds = ds.prefetch(AUTOTUNE)
    return ds


paths_all = train_df["image_path"].to_numpy()
labels_all = train_df["label"].to_numpy(dtype=np.int32)

rng = np.random.RandomState(SEED)
perm = rng.permutation(len(train_df))
val_size = int(round(0.1 * len(train_df)))
val_idx = perm[:val_size]
train_idx = perm[val_size:]

train_paths = paths_all[train_idx]
train_labels = labels_all[train_idx]
val_paths = paths_all[val_idx]
val_labels = labels_all[val_idx]

train_ds = _make_ds(train_paths, train_labels, training=True, cache_path=CACHE_TRAIN)
val_ds = _make_ds(val_paths, val_labels, training=False, cache_path=CACHE_VAL)

ckpt_path = "/kaggle/working/best_model.weights.h5"
callbacks = [
    ModelCheckpoint(
        ckpt_path,
        monitor="val_accuracy",
        mode="max",
        save_best_only=True,
        save_weights_only=True,
        verbose=1,
    ),
    ReduceLROnPlateau(
        monitor="val_loss", factor=0.5, patience=1, min_lr=1e-6, verbose=1
    ),
]

history = model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=EPOCHS,
    callbacks=callbacks,
    verbose=1,
)

if os.path.exists(ckpt_path):
    model.load_weights(ckpt_path)




## === cell 4
sample_sub = pd.read_csv(SAMPLE_SUB)
test_files = sample_sub["image_id"].astype(str).tolist()
test_paths = np.array([os.path.join(TEST_DIR, f) for f in test_files], dtype=object)

test_ds = _make_ds(test_paths, labels=None, training=False, cache_path=CACHE_TEST)

predictions = model.predict(test_ds, verbose=1)
predicted_labels = np.argmax(predictions, axis=1).astype(int)

assert len(predicted_labels) == len(sample_sub), (
    len(predicted_labels),
    len(sample_sub),
)

sub = sample_sub.copy()
sub["label"] = predicted_labels

sub.to_csv("submission.csv", index=False)
print("Submission file created: submission.csv")
print(sub.head())
