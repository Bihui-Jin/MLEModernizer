def stratified_split(df, label_col="label", val_frac=0.1, seed=SEED):
    rng = np.random.default_rng(seed)
    val_mask = np.zeros(len(df), dtype=bool)
    for _, idx in df.groupby(label_col).indices.items():
        idx = np.asarray(idx, dtype=np.int64)
        rng.shuffle(idx)
        n_val = max(1, int(round(len(idx) * val_frac)))
        val_mask[idx[:n_val]] = True
    val_df = df.loc[val_mask].reset_index(drop=True)
    trn_df = df.loc[~val_mask].reset_index(drop=True)
    return trn_df, val_df

IMG_SIZE = 224
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE


def _decode_resize_only(path):
    img_bytes = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img_bytes, channels=3)
    img.set_shape([None, None, 3])
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE], method="bilinear")
    img = tf.cast(img, tf.float32) / 255.0
    img.set_shape([IMG_SIZE, IMG_SIZE, 3])
    return img


def _decode_resize_batch(paths):
    return tf.map_fn(
        _decode_resize_only,
        paths,
        fn_output_signature=tf.float32,
        parallel_iterations=AUTOTUNE,
        swap_memory=True,
    )


def _augment_only(img):
    img = tf.image.random_flip_left_right(img, seed=SEED)
    img = tf.image.random_flip_up_down(img, seed=SEED)
    return img


def make_decoded_ds(df):
    paths = df["filepath"].astype(str).to_numpy()
    labels = df["label"].astype(np.int32).to_numpy()
    ds = tf.data.Dataset.from_tensor_slices((paths, labels))

    options = tf.data.Options()
    options.experimental_deterministic = True
    options.experimental_optimization.apply_default_optimizations = True
    try:
        options.threading.private_threadpool_size = 0
        options.threading.max_intra_op_parallelism = 0
    except Exception:
        pass
    ds = ds.with_options(options)

    ds = ds.batch(BATCH_SIZE, drop_remainder=False)
    ds = ds.map(
        lambda p, y: (_decode_resize_batch(p), tf.cast(y, tf.int32)),
        num_parallel_calls=AUTOTUNE,
        deterministic=True,
    )
    ds = ds.unbatch()

    ds = ds.cache()  # in-memory decoded cache
    ds = ds.apply(tf.data.experimental.ignore_errors())
    return ds


def make_train_ds_from_decoded(decoded_ds, n_items):
    ds = decoded_ds.shuffle(
        buffer_size=min(int(n_items), 4096), seed=SEED, reshuffle_each_iteration=True
    )
    ds = ds.map(
        lambda x, y: (_augment_only(x), y),
        num_parallel_calls=AUTOTUNE,
        deterministic=True,
    )
    ds = ds.batch(BATCH_SIZE, drop_remainder=True)
    ds = ds.prefetch(AUTOTUNE)
    return ds


def make_eval_ds_from_decoded(decoded_ds):
    ds = decoded_ds.batch(BATCH_SIZE, drop_remainder=False)
    ds = ds.prefetch(AUTOTUNE)
    return ds


train_decoded = make_decoded_ds(trn_df)
val_decoded = make_decoded_ds(val_df)


def _materialize_cache(decoded_ds, n_items, name):
    n = 0
    for _ in decoded_ds.take(n_items):
        n += 1
    print(f"Materialized {name} cache elements:", n)


_materialize_cache(train_decoded, len(trn_df), "train_decoded")
