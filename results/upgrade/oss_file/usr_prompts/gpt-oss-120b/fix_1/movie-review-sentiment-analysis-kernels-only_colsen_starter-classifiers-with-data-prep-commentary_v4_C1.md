## === cell 0
import numpy as np 
import pandas as pd 
from nltk.tokenize import TweetTokenizer
from nltk.util import ngrams
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.feature_extraction.text import HashingVectorizer
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.linear_model import LogisticRegression
from sklearn.svm import LinearSVC
from sklearn.multiclass import OneVsRestClassifier
from sklearn.pipeline import FeatureUnion
from xgboost import XGBClassifier


## === cell 1
train = pd.read_csv('../input/train.tsv', sep="\t")
test = pd.read_csv('../input/test.tsv', sep="\t")
sub = pd.read_csv('../input/sampleSubmission.csv', sep=",")


## === cell 2
train.head()


## === cell 3
tfhash = [("tfidf", TfidfVectorizer(stop_words='english')),
       ("hashing", HashingVectorizer (stop_words='english'))]
train_vectorized = FeatureUnion(tfhash).fit_transform(train.Phrase)
test_vectorized = FeatureUnion(tfhash).transform(test.Phrase)


## === cell 4
y = train['Sentiment']

xgb = XGBClassifier()
scores = cross_val_score(xgb, train_vectorized, y, scoring='accuracy', n_jobs=-1, cv=3)
print('Cross-validation mean accuracy {0:.2f}%, std {1:.2f}.'.format(np.mean(scores) * 100, np.std(scores) * 100))


## === cell 5
logreg = LogisticRegression()
ovr = OneVsRestClassifier(logreg)
ovr.fit(train_vectorized, y)
scores = cross_val_score(ovr, train_vectorized, y, scoring='accuracy', n_jobs=-1, cv=3)
print('Cross-validation mean accuracy {0:.2f}%, std {1:.2f}.'.format(np.mean(scores) * 100, np.std(scores) * 100))


## === cell 6
sub['Sentiment'] = ovr.predict(test_vectorized) 
sub.to_csv("ovr.csv", index=False)
