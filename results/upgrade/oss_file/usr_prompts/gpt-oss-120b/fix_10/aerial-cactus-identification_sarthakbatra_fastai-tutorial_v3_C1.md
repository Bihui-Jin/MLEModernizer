# 1. Kaggle task description

## Task
Create a classifier to predict whether an image contains a cactus.

## Metric
Area under the ROC curve.

## Submission Format
For each ID in the test set, you must predict a probability for the `has_cactus` variable. The file should contain a header and have the following format:

```
id,has_cactus
000940378805c44108d287872b2f04ce.jpg,0.5
0017242f54ececa4512b4d7937d1e21e.jpg,0.5
001ee6d8564003107853118ab87df407.jpg,0.5
etc.
```

## Dataset
This dataset contains a large number of 32 x 32 thumbnail images containing aerial photos of a cactus. The file name of an image corresponds to its `id`.

- **train/** - the training set images
- **test/** - the test set images (you must predict the labels of these)
- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

from pathlib import Path
import numpy as np
import pandas as pd
import torch
import torch.nn as nn

from fastai.vision.all import *
from fastai.data.block import RegressionBlock
from fastai.metrics import RocAuc

print("input root contents:", list(Path("../input").iterdir()))




## === cell 1
base_path = Path("../input")
if (base_path / "aerial-cactus-identification").exists():
    base_path = base_path / "aerial-cactus-identification"

train_path = base_path / "train"
test_path = base_path / "test"

if not test_path.is_dir():
    fallback = base_path.parent / "test"
    if fallback.is_dir():
        test_path = fallback
    else:
        raise FileNotFoundError(f"Test image directory not found at {test_path}")

train_df = pd.read_csv(base_path / "train.csv")
test_df = pd.read_csv(base_path / "sample_submission.csv")

print("Train CSV head:")
print(train_df.head())
print("\nTest CSV head:")
print(test_df.head())
print(f"Resolved train image path: {train_path}")
print(f"Resolved test image path: {test_path}")




## === cell 2
bs = 64
dls = ImageDataLoaders.from_df(
    train_df,
    path=train_path,
    fn_col="id",
    label_col="has_cactus",
    valid_pct=0.2,
    seed=42,
    bs=bs,
    item_tfms=Resize(32),
    batch_tfms=aug_transforms(flip_vert=True, max_warp=0.0),
    y_block=RegressionBlock,  # treat label as a float (0.0 or 1.0)
)

print(dls)




## === cell 3
custom_loss = lambda pred, target: nn.BCEWithLogitsLoss()(
    pred.squeeze(-1), target.float()
)

learn = cnn_learner(
    dls,
    models.resnet50,
    loss_func=custom_loss,
    metrics=RocAuc(),
).to_fp16()  # mixed precision for speed

lr = 1e-3
epochs = 10




## === cell 4
learn.fit_one_cycle(epochs, lr)




## === cell 5
test_dl = dls.test_dl(test_df["id"], path=test_path)

logits, _ = learn.get_preds(dl=test_dl)

probs = torch.sigmoid(logits).cpu().numpy().squeeze()
if probs.ndim > 1:
    probs = probs.reshape(-1)

submission = pd.DataFrame({"id": test_df["id"], "has_cactus": probs})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
