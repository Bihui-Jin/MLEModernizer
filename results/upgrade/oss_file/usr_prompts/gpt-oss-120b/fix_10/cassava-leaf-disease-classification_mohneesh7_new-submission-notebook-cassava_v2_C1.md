# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.layers import GlobalAveragePooling2D, Dense
from tensorflow.keras.models import Model
from tensorflow.keras.optimizers import Adam
from sklearn.model_selection import train_test_split

if tf.config.list_physical_devices("GPU"):
    from tensorflow.keras.mixed_precision import set_global_policy

    set_global_policy("mixed_float16")




## === cell 1
BASE = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE, "train_images")
TEST_IMG_DIR = os.path.join(BASE, "test_images")
TRAIN_CSV = os.path.join(BASE, "train.csv")
TEST_CSV = os.path.join(BASE, "sample_submission.csv")  # contains test image ids

IMG_SIZE = 224
BATCH_SIZE = 64
AUTOTUNE = tf.data.AUTOTUNE

train_df = pd.read_csv(TRAIN_CSV)
train_df, val_df = train_test_split(
    train_df, test_size=0.1, random_state=42, stratify=train_df["label"]
)

data_augmentation = keras.Sequential(
    [
        keras.layers.RandomFlip("horizontal"),
        keras.layers.RandomRotation(20 / 360.0),
        keras.layers.RandomTranslation(0.1, 0.1),
        keras.layers.RandomZoom(0.2),
    ]
)


def _decode_and_resize(image_id, label):
    """Read JPEG, decode, resize and preprocess; returns (image, label)."""
    path = tf.strings.join([TRAIN_IMG_DIR, "/", image_id])
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = keras.applications.efficientnet.preprocess_input(img)
    return img, label


def _augment(image, label):
    """Apply the shared augmentation pipeline."""
    return data_augmentation(image), label


def _decode_val(image_id, label):
    """Decoding for validation (no augmentation)."""
    path = tf.strings.join([TRAIN_IMG_DIR, "/", image_id])
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = keras.applications.efficientnet.preprocess_input(img)
    return img, label


options = tf.data.Options()
options.experimental_deterministic = False

train_ds = (
    tf.data.Dataset.from_tensor_slices(
        (train_df["image_id"].values, train_df["label"].values)
    )
    .map(_decode_and_resize, num_parallel_calls=AUTOTUNE)
    .cache()  # cache decoded & resized images
    .map(_augment, num_parallel_calls=AUTOTUNE)
    .shuffle(10000, seed=42)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
    .with_options(options)
)

val_ds = (
    tf.data.Dataset.from_tensor_slices(
        (val_df["image_id"].values, val_df["label"].values)
    )
    .map(_decode_val, num_parallel_calls=AUTOTUNE)
    .cache()
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
    .with_options(options)
)

base_model = EfficientNetB0(
    include_top=False, weights="imagenet", input_shape=(IMG_SIZE, IMG_SIZE, 3)
)
base_model.trainable = False  # freeze for few epochs

x = base_model.output
x = GlobalAveragePooling2D()(x)
output = Dense(5, activation="softmax", dtype="float32")(x)
model = Model(inputs=base_model.input, outputs=output)

model.compile(
    optimizer=Adam(learning_rate=1e-3),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(train_ds, epochs=3, validation_data=val_ds, verbose=2)




## === cell 2
test_df = pd.read_csv(TEST_CSV)  # contains image_id column


def _process_test(image_id):
    path = tf.strings.join([TEST_IMG_DIR, "/", image_id])
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = keras.applications.efficientnet.preprocess_input(img)
    return img


test_ds = (
    tf.data.Dataset.from_tensor_slices(test_df["image_id"].values)
    .map(lambda img_id: _process_test(img_id), num_parallel_calls=AUTOTUNE)
    .cache()  # cache test set (small) is still fine
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
    .with_options(options)
)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_df["image_id"], "label": pred_labels})




## === cell 3
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
