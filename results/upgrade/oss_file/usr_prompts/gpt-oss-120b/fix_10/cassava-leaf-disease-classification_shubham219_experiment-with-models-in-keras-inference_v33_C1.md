train_df["path"] = train_images_dir + train_df["image_id"].astype(str)




## === cell 2
def load_and_preprocess(df, img_size):
    """
    Load images listed in df['path'], resize to img_size, normalize to [0,1],
    and flatten to 1‑D vectors. Uses a thread pool to parallelize I/O‑bound work.
    Returns a NumPy array of shape (n_samples, img_h*img_w*3).
    """
    paths = df["path"].values
    n_samples = len(paths)
    h, w = img_size
    result = np.empty((n_samples, h * w * 3), dtype=np.float32)

    def _process(idx_path):
        idx, p = idx_path
        img = Image.open(p).convert("RGB")
        img = img.resize(img_size, Image.BILINEAR)
        arr = np.asarray(img, dtype=np.float32) / 255.0
        result[idx] = arr.ravel()

    max_workers = min(16, os.cpu_count() or 1)
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        executor.map(_process, enumerate(paths))
    return result




## === cell 3
X = load_and_preprocess(train_df, IMG_SIZE)
y = train_df["label"].values

X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.2, random_state=SEED, stratify=y
)



## === cell 4
model = make_pipeline(
    StandardScaler(),
    MLPClassifier(
        hidden_layer_sizes=(512, 256),  # deeper network for richer representation
        activation="relu",
        solver="adam",
        batch_size=64,
        max_iter=200,  # more training steps for convergence
        random_state=SEED,
        n_iter_no_change=10,  # patience for early stopping
        warm_start=True,  # continue training on full data without re‑initialising
    ),
)

model.fit(X_train, y_train)
val_acc = model.score(X_val, y_val)
print(f"Validation accuracy: {val_acc:.4f}")




## === cell 5
test_images = glob.glob(
