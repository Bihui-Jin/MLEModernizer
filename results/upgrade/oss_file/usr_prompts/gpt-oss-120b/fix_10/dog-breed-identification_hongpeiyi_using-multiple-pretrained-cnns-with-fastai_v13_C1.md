## === cell 0
from fastai.vision.all import *
import pandas as pd
import torch
import torch.nn as nn
from pathlib import Path
from sklearn.model_selection import StratifiedShuffleSplit

labels = pd.read_csv("/kaggle/input/dog-breed-identification/labels.csv")
labels["id"] = labels["id"].apply(lambda x: f"{x}.jpg")



## === cell 1
split = StratifiedShuffleSplit(n_splits=1, test_size=0.2, random_state=42)
train_idx, valid_idx = next(split.split(labels, labels["breed"]))
labels["is_valid"] = False
labels.loc[valid_idx, "is_valid"] = True



## === cell 2
base_path = Path("/kaggle/input/dog-breed-identification")
train_path = base_path / "train"

dls = ImageDataLoaders.from_df(
    df=labels,
    path=train_path,
    fn_col="id",
    label_col="breed",
    valid_col="is_valid",
    item_tfms=RandomResizedCrop(460, min_scale=0.3),
    batch_tfms=[*aug_transforms(size=300), Normalize.from_stats(*imagenet_stats)],
    bs=32,
)



## === cell 3
class_counts = labels["breed"].value_counts().reindex(dls.vocab).fillna(0)
weights = (len(labels) / (len(dls.vocab) * class_counts)).values
class_weights = torch.tensor(weights, dtype=torch.float32).to(dls.device)



## === cell 4
learn = cnn_learner(
    dls,
    resnet50,
    loss_func=nn.CrossEntropyLoss(weight=class_weights),
    metrics=[accuracy],
    pretrained=True,
).to_fp16()



## === cell 5
learn.fine_tune(6, base_lr=1e-3)



## === cell 6
test_path = base_path / "test"
test_files = get_image_files(test_path)
test_dl = dls.test_dl(test_files, bs=32)



## === cell 7
preds, _ = learn.get_preds(dl=test_dl)



## === cell 8
sub = pd.DataFrame({"id": [p.stem for p in test_files]})
breed_cols = list(dls.vocab)  # convert CategoryMap to list of column names
sub[breed_cols] = preds.softmax(dim=1).cpu().numpy()
sub.to_csv("submission.csv", index=False)



## === cell 9
print("Submission file saved as submission.csv")
