def load_dicom_image(path, img_size=IMAGE_SIZE, voi_lut=True, rotate=0):
    """
    Load a single DICOM file, apply VOI LUT if required, optionally rotate,
    and resize to (img_size, img_size). Returns a 2‑D NumPy array.
    """
    if not os.path.exists(path):
        return np.zeros((img_size, img_size), dtype=np.float32)

    dicom = pydicom.dcmread(path)
    data = dicom.pixel_array
    if voi_lut:
        data = apply_voi_lut(dicom, dicom)

    if rotate > 0:
        rot_choices = [
            0,
            cv2.ROTATE_90_CLOCKWISE,
            cv2.ROTATE_90_COUNTERCLOCKWISE,
            cv2.ROTATE_180,
        ]
        data = cv2.rotate(data, rot_choices[rotate])

    data = cv2.resize(data, (img_size, img_size))
    return data.astype(np.float32)


def load_dicom_images_3d(
    scan_id,
    num_imgs=NUM_IMAGES,
    img_size=IMAGE_SIZE,
    mri_type="FLAIR",
    split="test",
    rotate=0,
):
    """
    Load a stack of DICOM slices for a given scan ID and MRI modality.
    If no files are found, returns an array of zeros with the expected shape.
    """
    pattern = os.path.join(data_directory, split, scan_id, mri_type, "*.dcm")
    files = sorted(
        glob.glob(pattern),
        key=lambda var: [
            int(x) if x.isdigit() else x for x in re.findall(r"[^0-9]|[0-9]+", var)
        ],
    )
    if not files:
        return np.zeros((1, img_size, img_size, num_imgs), dtype=np.float32)

    middle = len(files) // 2
    half = num_imgs // 2
    p1 = max(0, middle - half)
    p2 = min(len(files), middle + half)
    selected = files[p1:p2]

    slice_imgs = [load_dicom_image(f, rotate=rotate) for f in selected]
    img3d = np.stack(slice_imgs, axis=-1)  # shape (img_size, img_size, N)
    if img3d.shape[-1] < num_imgs:
        pad_width = ((0, 0), (0, 0), (0, num_imgs - img3d.shape[-1]))
        img3d = np.pad(img3d, pad_width, mode="constant")
    if img3d.max() > img3d.min():
        img3d = (img3d - img3d.min()) / (img3d.max() - img3d.min())
    return np.expand_dims(img3d, axis=0)  # shape (1, img_size, img_size, num_imgs)
