# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from sklearn.preprocessing import LabelEncoder
from xgboost import XGBClassifier

np.random.seed(42)
os.environ["PYTHONHASHSEED"] = "0"




## === cell 1
train_path = "/kaggle/input/tabular-playground-series-dec-2021/train.csv"
test_path = "/kaggle/input/tabular-playground-series-dec-2021/test.csv"

all_train_cols = pd.read_csv(train_path, nrows=0).columns.tolist()
all_test_cols = pd.read_csv(test_path, nrows=0).columns.tolist()

train_dtype = {c: np.float32 for c in all_train_cols if c not in ["Cover_Type", "Id"]}
train_dtype.update({"Cover_Type": np.int32, "Id": np.int32})

test_dtype = {c: np.float32 for c in all_test_cols if c != "Id"}
test_dtype.update({"Id": np.int32})

train_df = pd.read_csv(train_path, dtype=train_dtype)
test_df = pd.read_csv(test_path, dtype=test_dtype)




## === cell 2
X = train_df.drop(columns=["Cover_Type", "Id"]).values
Y_raw = train_df["Cover_Type"]

le = LabelEncoder()
Y = le.fit_transform(Y_raw)  # encode labels to 0-(num_classes-1)

num_classes = len(le.classes_)
model_xgbc = XGBClassifier(
    n_estimators=600,
    n_jobs=4,
    learning_rate=0.05,
    max_depth=8,
    subsample=0.8,
    colsample_bytree=0.8,
    objective="multi:softprob",
    num_class=num_classes,
    tree_method="hist",
    eval_metric="merror",
    early_stopping_rounds=None,
    verbose=0,
)

model_xgbc.fit(X, Y, verbose=False)




## === cell 3
test_features = test_df.drop(columns=["Id"]).values  # already float32
y_pred_encoded = model_xgbc.predict(test_features)
y_predict_xgbc = le.inverse_transform(y_pred_encoded)




## === cell 4
result = pd.DataFrame()
result["Id"] = test_df["Id"]
result["Cover_Type"] = y_predict_xgbc.astype(int)




## === cell 5
print("Result shape:", result.shape)




## === cell 6
result.to_csv("/kaggle/working/submission.csv", index=False)
print("Done")
