## === cell 0
import os, numpy as np, pandas as pd, torch
from pathlib import Path
from fastai.vision.all import *
from fastai.data.block import RegressionBlock  # correct block for float labels
from fastai.metrics import RocAuc
from fastai.losses import BCEWithLogitsLossFlat
import torch.nn as nn

print("Root input contents:", os.listdir("../input"))

BASE_PATH = Path("../input/aerial-cactus-identification")
train_path = BASE_PATH / "train"
test_path = BASE_PATH / "test"

df_train = pd.read_csv(BASE_PATH / "train.csv")
submission_template = pd.read_csv(BASE_PATH / "sample_submission.csv")

test_files = [p for p in test_path.iterdir() if p.is_file()]



## === cell 1
cactus_block = DataBlock(
    blocks=(ImageBlock, RegressionBlock),  # image + float label
    get_x=ColReader(
        "id", pref=str(train_path) + "/"
    ),  # prepend folder to filename as string
    get_y=ColReader("has_cactus"),  # 0/1 labels as floats
    splitter=RandomSplitter(valid_pct=0.2, seed=47),
    item_tfms=Resize(32),
)

dls = cactus_block.dataloaders(df_train, bs=128)



## === cell 2
learn = cnn_learner(
    dls,
    models.resnet50,
    loss_func=BCEWithLogitsLossFlat(),  # suitable for binary labels
    metrics=RocAuc(),
    model_dir="/tmp/model/",
)
learn.fine_tune(5, base_lr=1e-3)  # slightly longer training for better AUC

valid_auc = learn.validate()[1]
print(f"Validation ROC‑AUC: {valid_auc:.5f}")



## === cell 3
test_files = sorted(
    [p for p in test_path.iterdir() if p.is_file()]
)  # deterministic order
test_dl = learn.dls.test_dl(test_files)  # create a test dataloader
preds, _ = learn.get_preds(dl=test_dl)  # raw logits
probs = torch.sigmoid(preds).numpy().flatten()  # convert to probabilities

submission = pd.DataFrame({"id": [p.name for p in test_files], "has_cactus": probs})
submission.to_csv("submission.csv", index=False)
print("Submission file saved with shape:", submission.shape)
