## === cell 0
import os, sys, tempfile, hashlib

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
os.environ["PROTOCOL_BUFFERS_PYTHON_PIP"] = "1"

for mod in list(sys.modules):
    if mod.startswith("google.protobuf"):
        del sys.modules[mod]

import re, random
import numpy as np, pandas as pd
import tensorflow as tf
from sklearn.model_selection import train_test_split

tf.config.optimizer.set_jit(True)

gpus = tf.config.list_physical_devices("GPU")
if gpus:
    for gpu in gpus:
        tf.config.experimental.set_memory_growth(gpu, True)

tf.config.threading.set_inter_op_parallelism_threads(4)
tf.config.threading.set_intra_op_parallelism_threads(4)




## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")

IMAGE_SIZE = (224, 224)
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE
NUM_CLASSES = 5




## === cell 2
train_df = pd.read_csv(TRAIN_CSV)
train_df["filepath"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)

train_paths, val_paths, train_labels, val_labels = train_test_split(
    train_df["filepath"].values,
    train_df["label"].values,
    test_size=0.1,
    stratify=train_df["label"],
    random_state=42,
)


def decode_resize(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMAGE_SIZE)
    img = img / 255.0
    return img


def augment(img):
    img = tf.image.random_flip_left_right(img)
    img = tf.image.random_flip_up_down(img)
    img = tf.image.random_brightness(img, max_delta=0.1)
    img = tf.image.random_contrast(img, 0.9, 1.1)
    return img


def build_dataset(paths, labels, shuffle=True, do_augment=False, cache=False):
    ds = tf.data.Dataset.from_tensor_slices((paths, labels))
    ds = ds.map(
        lambda p, l: (decode_resize(p), tf.cast(l, tf.int32)),
        num_parallel_calls=AUTOTUNE,
    )
    if cache:
        unique_cache_dir = tempfile.mkdtemp()
        ds = ds.cache(unique_cache_dir)
    if do_augment:
        ds = ds.map(
            lambda img, l: (augment(img), l),
            num_parallel_calls=AUTOTUNE,
        )
    if shuffle:
        ds = ds.shuffle(buffer_size=1000, seed=42)
    options = tf.data.Options()
    options.experimental_deterministic = False
    ds = ds.with_options(options)

    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds


train_ds = build_dataset(
    train_paths, train_labels, shuffle=True, do_augment=True, cache=True
)

val_ds = build_dataset(
    val_paths, val_labels, shuffle=False, do_augment=False, cache=False
)




## === cell 3
tf.keras.mixed_precision.set_global_policy("mixed_float16")

base_model = tf.keras.applications.EfficientNetB0(
    input_shape=(*IMAGE_SIZE, 3), include_top=False, weights="imagenet"
)
base_model.trainable = False  # freeze base for initial training

inputs = tf.keras.Input(shape=(*IMAGE_SIZE, 3))
x = base_model(inputs, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
x = tf.keras.layers.Dropout(0.2)(x)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax", dtype="float32")(x)

model = tf.keras.Model(inputs, outputs)
model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(train_ds, validation_data=val_ds, epochs=20, verbose=2)

base_model.trainable = True
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)
model.fit(train_ds, validation_data=val_ds, epochs=20, verbose=2)




## === cell 4
test_filenames = tf.io.gfile.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
test_ids = [os.path.basename(p) for p in test_filenames]

test_ds = tf.data.Dataset.from_tensor_slices(test_filenames)
test_ds = (
    test_ds.map(lambda p: decode_resize(p), num_parallel_calls=AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

prob = model.predict(test_ds)
pred_labels = np.argmax(prob, axis=-1).astype(int)




## === cell 5
submission_path = "submission.csv"
submission_df = pd.DataFrame({"image_id": test_ids, "label": pred_labels})
submission_df.to_csv(submission_path, index=False)

print(submission_df.head())
