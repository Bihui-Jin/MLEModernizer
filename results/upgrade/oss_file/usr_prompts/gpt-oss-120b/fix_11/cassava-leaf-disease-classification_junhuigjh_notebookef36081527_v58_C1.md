# 5. Code solution

## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image
import torch

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")




## === cell 1
train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"

train_df = pd.read_csv(train_csv_path)




## === cell 2
def image_stats(image_path):
    """Return a richer set of color statistics for an image.
    Features (13‑dim):
    - mean R,G,B
    - std R,G,B
    - median R,G,B          <-- new
    - ratio G/R, ratio G/B
    - overall brightness (mean of RGB means)
    """
    try:
        img = Image.open(image_path).convert("RGB")
        arr = np.asarray(img, dtype=np.float32) / 255.0  # HxWx3
        mean = arr.mean(axis=(0, 1))  # (3,)
        std = arr.std(axis=(0, 1))  # (3,)
        median = np.median(arr, axis=(0, 1))  # (3,)
        eps = 1e-6
        ratio_g_r = mean[1] / (mean[0] + eps)
        ratio_g_b = mean[1] / (mean[2] + eps)
        brightness = mean.mean()
        return np.concatenate(
            [mean, std, median, [ratio_g_r, ratio_g_b, brightness]]
        )  # (13,)
    except Exception:
        return np.full(13, 0.5, dtype=np.float32)


train_features = []
train_labels = []

for _, row in train_df.iterrows():
    img_path = os.path.join(train_images_dir, row["image_id"])
    if os.path.exists(img_path):
        feats = image_stats(img_path)
        train_features.append(feats)
        train_labels.append(int(row["label"]))

train_features = np.stack(train_features) if train_features else np.empty((0, 13))
train_labels = np.array(train_labels, dtype=int)




## === cell 3
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier

scaler = StandardScaler()
train_features_scaled = scaler.fit_transform(train_features)

rf_clf = RandomForestClassifier(
    n_estimators=1000,  # increased trees for a modest gain
    max_depth=None,
    random_state=42,
    n_jobs=-1,
    class_weight="balanced",
)

rf_clf.fit(train_features_scaled, train_labels)




## === cell 4
image_paths = []
image_names = []
for root, _, files in os.walk(test_image_dir):
    for f in files:
        if f.lower().endswith(".jpg"):
            image_paths.append(os.path.join(root, f))
            image_names.append(f)

test_features = []
valid_ids = []  # keep only ids for which we can read the image

for img_name, img_path in zip(image_names, image_paths):
    if os.path.exists(img_path):
        test_features.append(image_stats(img_path))
        valid_ids.append(img_name)

test_features = np.stack(test_features) if test_features else np.empty((0, 13))
test_features_scaled = scaler.transform(test_features)  # same scaling as training




## === cell 5
predictions = rf_clf.predict(test_features_scaled).astype(int).tolist()

if len(predictions) < len(image_names):
    most_common = int(pd.Series(train_labels).mode()[0])
    missing_count = len(image_names) - len(predictions)
    predictions.extend([most_common] * missing_count)
    valid_ids = image_names




## === cell 6
submission = pd.DataFrame({"image_id": valid_ids, "label": predictions})
submission_path = "submission.csv"  # Kaggle expects this in the working directory
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path} with {len(submission)} rows.")
