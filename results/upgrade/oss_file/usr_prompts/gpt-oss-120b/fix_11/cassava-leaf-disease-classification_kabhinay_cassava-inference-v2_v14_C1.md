## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"

TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUBMIT = os.path.join(BASE_PATH, "sample_submission.csv")

train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(int)


def mean_rgb(image_path):
    """Return the mean RGB vector of the image at image_path."""
    try:
        with Image.open(image_path) as img:
            img = img.convert("RGB")
            arr = np.asarray(img, dtype=np.float32) / 255.0
            return arr.mean(axis=(0, 1))  # shape (3,)
    except Exception:
        return np.array([0.5, 0.5, 0.5], dtype=np.float32)


class_vectors = {}
for label in train_df["label"].unique():
    class_vectors[label] = []

print("Computing mean RGB for training images...")
for _, row in train_df.iterrows():
    img_path = os.path.join(TRAIN_IMG_DIR, row["image_id"])
    vec = mean_rgb(img_path)
    class_vectors[row["label"]].append(vec)

class_centroids = {}
for label, vectors in class_vectors.items():
    if vectors:
        class_centroids[label] = np.mean(vectors, axis=0)
    else:
        class_centroids[label] = np.array([0.5, 0.5, 0.5], dtype=np.float32)

print("Class centroids (RGB) computed for", len(class_centroids), "classes.")

test_files = sorted(os.listdir(TEST_IMG_DIR))
test_df = pd.DataFrame({"image_id": test_files})
print("Number of test images:", len(test_df))




## === cell 1
def nearest_label(rgb_vec):
    best_label = None
    best_dist = float("inf")
    for label, centroid in class_centroids.items():
        dist = np.linalg.norm(rgb_vec - centroid)
        if dist < best_dist:
            best_dist = dist
            best_label = label
    return best_label


print("Predicting test set labels...")
pred_labels = []
for img_name in test_df["image_id"]:
    img_path = os.path.join(TEST_IMG_DIR, img_name)
    rgb = mean_rgb(img_path)
    pred = nearest_label(rgb)
    pred_labels.append(int(pred))

submission = pd.DataFrame({"image_id": test_df["image_id"], "label": pred_labels})

output_path = "/kaggle/working/submission.csv"
submission.to_csv(output_path, index=False)
print("Submission written to:", output_path)
