# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, optimizers, applications
from tensorflow.keras import mixed_precision

if tf.config.list_physical_devices("GPU"):
    policy = mixed_precision.Policy("mixed_float16")
    mixed_precision.set_global_policy(policy)

tf.get_logger().setLevel("ERROR")
tf.random.set_seed(42)




## === cell 1
base_path = "/kaggle/input/cassava-leaf-disease-classification"
if not os.path.isdir(base_path):
    base_path = "../input/cassava-leaf-disease-classification"

train_dir = os.path.join(base_path, "train_images")
test_dir = os.path.join(base_path, "test_images")
train_csv_path = os.path.join(base_path, "train.csv")
sample_submission_path = os.path.join(base_path, "sample_submission.csv")

train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(str)

train_filepaths = (
    train_df["image_id"].apply(lambda x: os.path.join(train_dir, x)).values
)
labels_int = train_df["label"].astype(int).values

num_samples = len(train_filepaths)
rng = np.random.default_rng(seed=42)
perm = rng.permutation(num_samples)
split_idx = int(0.8 * num_samples)
train_idx, val_idx = perm[:split_idx], perm[split_idx:]

train_paths, train_labels = train_filepaths[train_idx], labels_int[train_idx]
val_paths, val_labels = train_filepaths[val_idx], labels_int[val_idx]

BATCH_SIZE = 64
AUTOTUNE = tf.data.AUTOTUNE
NUM_CLASSES = 5


def _parse_image(filename, label):
    image = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [224, 224])
    image = tf.cast(image, tf.float32) / 255.0
    label = tf.one_hot(label, NUM_CLASSES)
    return image, label


def _parse_image_test(filename):
    image = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [224, 224])
    image = tf.cast(image, tf.float32) / 255.0
    return image




## === cell 2
options = tf.data.Options()
options.experimental_deterministic = False

train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.shuffle(1000, seed=42, reshuffle_each_iteration=False)
train_ds = train_ds.map(_parse_image, num_parallel_calls=AUTOTUNE, deterministic=False)
train_ds = train_ds.with_options(options)
train_ds = train_ds.cache()
train_ds = train_ds.batch(BATCH_SIZE, drop_remainder=True).prefetch(AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
val_ds = val_ds.map(_parse_image, num_parallel_calls=AUTOTUNE, deterministic=False)
val_ds = val_ds.with_options(options)
val_ds = val_ds.cache()
val_ds = val_ds.batch(BATCH_SIZE, drop_remainder=True).prefetch(AUTOTUNE)




## === cell 3
base_model = applications.EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3), pooling="avg"
)
base_model.trainable = True

model = models.Sequential([base_model, layers.Dense(NUM_CLASSES, activation="softmax")])

model.compile(
    optimizer=optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)




## === cell 4
model.fit(
    train_ds,
    epochs=30,
    validation_data=val_ds,
    verbose=2,
)




## === cell 5
import glob

test_paths = sorted(glob.glob(os.path.join(test_dir, "**", "*.jpg"), recursive=True))
test_filenames = [os.path.basename(p) for p in test_paths]

test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(
    _parse_image_test, num_parallel_calls=AUTOTUNE, deterministic=False
)
test_ds = test_ds.with_options(options)
test_ds = test_ds.batch(BATCH_SIZE, drop_remainder=False).prefetch(AUTOTUNE)

pred_probs = model.predict(test_ds, verbose=1)
pred_indices = np.argmax(pred_probs, axis=1).astype(str)

submission = pd.DataFrame({"image_id": test_filenames, "label": pred_indices})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
