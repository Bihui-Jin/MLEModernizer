## === cell 0
import os
import warnings
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.metrics import accuracy_score
from scipy.stats import mode
from PIL import Image
from concurrent.futures import ThreadPoolExecutor
import multiprocessing
import joblib

warnings.filterwarnings("ignore")

SEED = 42
np.random.seed(SEED)

BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TEST_CSV = os.path.join(BASE_PATH, "sample_submission.csv")  # column template

df_train = pd.read_csv(TRAIN_CSV)
df_train["filepath"] = df_train["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)




## === cell 1
def load_and_preprocess(image_path, size=(64, 64)):
    """Load an image, resize, and flatten to a 1‑D uint8 array."""
    try:
        img = Image.open(image_path).convert("RGB")
        img = img.resize(size)
        return np.asarray(img, dtype=np.uint8).flatten()
    except Exception:
        return np.zeros(size[0] * size[1] * 3, dtype=np.uint8)


def parallel_preprocess(paths, size=(64, 64)):
    """
    Process images in parallel using threads (PIL releases the GIL) and
    pre‑allocate the output array to avoid repeated allocations.
    """
    n = len(paths)
    dim = size[0] * size[1] * 3
    out = np.empty((n, dim), dtype=np.uint8)

    max_workers = min(multiprocessing.cpu_count(), 8)  # limit to avoid oversubscription
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        for i, arr in enumerate(executor.map(load_and_preprocess, paths, chunksize=1)):
            out[i] = arr
    return out


TRAIN_CACHE_PATH = "train_features.npy"
if os.path.exists(TRAIN_CACHE_PATH):
    X_train = np.load(TRAIN_CACHE_PATH, mmap_mode="r")
else:
    X_train = parallel_preprocess(df_train["filepath"].values)
    np.save(TRAIN_CACHE_PATH, X_train)  # persist for future runs
y_train = df_train["label"].values




## === cell 2
rf_clf = RandomForestClassifier(
    n_estimators=100,  # preserved original setting
    max_depth=None,
    random_state=SEED,
    n_jobs=-1,
    class_weight="balanced",
)

gb_clf = GradientBoostingClassifier(
    n_estimators=80,  # preserved original setting
    learning_rate=0.1,
    max_depth=3,
    random_state=SEED,
)

rf_clf.fit(X_train, y_train)
gb_clf.fit(X_train, y_train)




## === cell 3
test_filenames = [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
test_image_paths = [os.path.join(TEST_IMG_DIR, f) for f in test_filenames]

TEST_CACHE_PATH = "test_features.npy"
if os.path.exists(TEST_CACHE_PATH):
    X_test = np.load(TEST_CACHE_PATH, mmap_mode="r")
else:
    X_test = parallel_preprocess(test_image_paths)
    np.save(TEST_CACHE_PATH, X_test)

rf_pred = rf_clf.predict(X_test)
gb_pred = gb_clf.predict(X_test)

combined = np.vstack([rf_pred, gb_pred]).T
mode_result = mode(combined, axis=1).mode.ravel()
test_pred = mode_result.astype(int)




## === cell 4
df_test = pd.DataFrame({"image_id": test_filenames, "label": test_pred})

sample_sub = pd.read_csv(TEST_CSV, nrows=5)
assert list(sample_sub.columns) == ["image_id", "label"], "Column mismatch!"

df_test.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv")
