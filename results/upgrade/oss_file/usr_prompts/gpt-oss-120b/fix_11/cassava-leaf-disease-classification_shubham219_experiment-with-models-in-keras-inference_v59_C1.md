# 5. Code solution

## === cell 0
import pandas as pd
import numpy as np
import tensorflow as tf
import glob
from tensorflow.keras.models import Sequential
from tensorflow.keras.applications import EfficientNetB0
from sklearn.model_selection import train_test_split

SEED = 42
BATCH_SIZE = 256  # larger batch reduces steps per epoch
IMG_SIZE = (224, 224)  # image size
EPOCHS = 3
AUTOTUNE = tf.data.AUTOTUNE

tf.random.set_seed(SEED)
tf.keras.utils.set_random_seed(SEED)  # ensure reproducibility

tf.config.optimizer.set_jit(True)





## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification/"
TRAIN_IMG_DIR = f"{BASE_PATH}train_images/"
TEST_IMG_DIR = f"{BASE_PATH}test_images/"
TRAIN_CSV = f"{BASE_PATH}train.csv"

df_train = pd.read_csv(TRAIN_CSV)
df_train["full_path"] = TRAIN_IMG_DIR + df_train["image_id"]

df_tr, df_val = train_test_split(
    df_train,
    test_size=0.1,
    random_state=SEED,
    stratify=df_train["label"],
)




## === cell 2
def _load_image(path, label):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    return img, label


def _load_image_no_label(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    return img


train_dataset = tf.data.Dataset.from_tensor_slices(
    (df_tr["full_path"].values, df_tr["label"].values.astype(np.int32))
)
train_dataset = (
    train_dataset.map(_load_image, num_parallel_calls=AUTOTUNE, deterministic=False)
    .shuffle(buffer_size=1000, seed=SEED)  # shuffle before batching
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)  # no full‑dataset cache
)

val_dataset = tf.data.Dataset.from_tensor_slices(
    (df_val["full_path"].values, df_val["label"].values.astype(np.int32))
)
val_dataset = (
    val_dataset.map(_load_image, num_parallel_calls=AUTOTUNE, deterministic=False)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)  # no full‑dataset cache
)




## === cell 3
data_augmentation = tf.keras.Sequential(
    [
        tf.keras.layers.RandomFlip(mode="horizontal", seed=SEED),
        tf.keras.layers.RandomRotation(
            factor=20 / 360, seed=SEED
        ),  # 20‑degree rotation
        tf.keras.layers.RandomTranslation(
            height_factor=0.1, width_factor=0.1, seed=SEED
        ),
        tf.keras.layers.RandomZoom(height_factor=0.1, width_factor=0.1, seed=SEED),
    ]
)

base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=IMG_SIZE + (3,)
)
base_model.trainable = False  # freeze backbone initially

model = Sequential(
    [
        data_augmentation,
        tf.keras.layers.Rescaling(1.0 / 255),  # match original rescaling
        base_model,
        tf.keras.layers.GlobalAveragePooling2D(),
        tf.keras.layers.Dense(5, activation="softmax"),
    ]
)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)




## === cell 4
model.fit(train_dataset, epochs=EPOCHS, validation_data=val_dataset, verbose=1)




## === cell 5
test_images = sorted(glob.glob(TEST_IMG_DIR + "*.jpg"))  # ensure deterministic order
df_test = pd.DataFrame({"path": test_images})
df_test["image_id"] = df_test["path"].str.split("/").str[-1]

test_dataset = tf.data.Dataset.from_tensor_slices(df_test["path"].values)
test_dataset = (
    test_dataset.map(
        _load_image_no_label, num_parallel_calls=AUTOTUNE, deterministic=False
    )
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)  # removed cache to avoid loading all test images at once
)

pred_probs = model.predict(test_dataset, verbose=1)
pred_labels = np.argmax(pred_probs, axis=1)




## === cell 6
submission = pd.DataFrame({"image_id": df_test["image_id"], "label": pred_labels})
submission = submission[["image_id", "label"]]
submission.to_csv("submission.csv", index=False)




## === cell 7
print(submission.head())
