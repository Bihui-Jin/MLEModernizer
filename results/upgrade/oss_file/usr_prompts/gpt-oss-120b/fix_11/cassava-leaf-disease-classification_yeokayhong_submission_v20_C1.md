class TrainImageDataset(Dataset):
    """
    Optimized dataset: store image ids and labels in plain lists to avoid pandas .loc look‑ups
    inside every __getitem__, and cache transformed tensors to prevent repeated work.
    """

    def __init__(self, df):
        self.image_ids = df["image_id"].tolist()
        self.labels = df["label"].astype(int).tolist()
        self.root = train_images_dir
        self.transform = val_transforms
        self._cache = [None] * len(self.image_ids)  # cache for transformed tensors

    def __len__(self):
        return len(self.image_ids)

    def __getitem__(self, idx):
        cached = self._cache[idx]
        if cached is not None:
            return cached, self.labels[idx]

        img_id = self.image_ids[idx]
        label = self.labels[idx]
        img_path = os.path.join(self.root, img_id)
        image = Image.open(img_path).convert("RGB")
        tensor = self.transform(image)
        self._cache[idx] = tensor  # store for future epochs
        return tensor, label

def resize_max_side(img, size):
    width, height = img.size  # correct order
    if max(width, height) > size:
        return img

    if width > height:
        new_width = size
        new_height = int(size * height / width)
    else:
        new_height = size
        new_width = int(size * width / height)

    return transforms.functional.resize(img, (new_width, new_height))
