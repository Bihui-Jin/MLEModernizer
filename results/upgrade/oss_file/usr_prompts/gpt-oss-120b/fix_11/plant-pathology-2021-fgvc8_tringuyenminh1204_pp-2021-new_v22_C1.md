## === cell 0
import os, numpy as np, pandas as pd

BASE = "../input/plant-pathology-2021-fgvc8/"
train_df = pd.read_csv(os.path.join(BASE, "train.csv"))
test_df = pd.read_csv(os.path.join(BASE, "sample_submission.csv"))

assert "image" in train_df.columns and "labels" in train_df.columns
assert "image" in test_df.columns



## === cell 1
DISEASES = [
    "scab",
    "frog_eye_leaf_spot",
    "complex",
    "rust",
    "powdery_mildew",
    "healthy",
]
num_classes = len(DISEASES)

DISEASE_TO_IDX = {d: i for i, d in enumerate(DISEASES)}


def multi_hot(label_str):
    """Convert space‑delimited label string to a multi‑hot vector."""
    arr = np.zeros(num_classes, dtype=np.float32)
    for token in label_str.split():
        idx = DISEASE_TO_IDX.get(token)
        if idx is not None:
            arr[idx] = 1.0
    return arr


train_labels_matrix = np.stack(train_df["labels"].apply(multi_hot).values)
class_counts = train_labels_matrix.sum(axis=0)
most_common_idx = int(class_counts.argmax())
most_common_label = DISEASES[most_common_idx]



## === cell 2
pred_strings = [most_common_label] * len(test_df)



## === cell 3
submission = pd.DataFrame({"image": test_df["image"], "labels": pred_strings})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"{submission_path} written, preview:")
print(submission.head())
