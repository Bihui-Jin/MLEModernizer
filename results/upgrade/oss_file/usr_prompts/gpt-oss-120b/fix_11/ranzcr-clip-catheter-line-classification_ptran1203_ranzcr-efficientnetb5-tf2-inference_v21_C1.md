## Dataset
`train.csv` contains image IDs, binary labels, and patient IDs.

TFRecords are available for both train and test.

### Columns
- `StudyInstanceUID` - unique ID for each image
- `ETT - Abnormal` - endotracheal tube placement abnormal
- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
os.environ["PROTOCOL_BUFFERS_PYTHON_DISABLE_C_EXTENSION"] = "1"

import tensorflow as tf
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split

from tensorflow.keras import mixed_precision

mixed_precision.set_global_policy("mixed_float16")

tf.random.set_seed(42)
np.random.seed(42)

W = H = 338
N_CLASSES = 11
AUTOTUNE = tf.data.experimental.AUTOTUNE
BATCH_SIZE = 256  # larger batch reduces steps per epoch, speeding up training

BASE_PATH = "../input/ranzcr-clip-catheter-line-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test")
TRAIN_CSV_PATH = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUBMISSION_PATH = os.path.join(BASE_PATH, "sample_submission.csv")

mean = np.array([0.485, 0.456, 0.406], dtype=np.float32)
std = np.array([0.229, 0.224, 0.225], dtype=np.float32)


def load_image(file_path):
    """Read, decode (as 3‑channel), resize and normalise an image."""
    img = tf.io.read_file(file_path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, (H, W))
    img = tf.cast(img, tf.float32) / 255.0
    img = (img - mean) / std
    return img


def parse_path_label(path, label):
    return load_image(path), label


def extract_id_from_path(path):
    return tf.strings.regex_replace(tf.strings.split(path, os.sep)[-1], ".jpg$", "")




## === cell 1
train_df = pd.read_csv(TRAIN_CSV_PATH)
target_cols = [
    c for c in train_df.columns if c not in ["StudyInstanceUID", "PatientID"]
]

train_paths = [
    os.path.join(TRAIN_IMG_DIR, f"{uid}.jpg") for uid in train_df["StudyInstanceUID"]
]
train_labels = train_df[target_cols].values.astype(np.float32)

train_paths, val_paths, train_labels, val_labels = train_test_split(
    train_paths, train_labels, test_size=0.1, random_state=42
)

train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(
    parse_path_label,
    num_parallel_calls=AUTOTUNE,
    deterministic=False,
)
train_ds = train_ds.cache("/tmp/train_cache")
train_ds = train_ds.shuffle(2048).batch(BATCH_SIZE).prefetch(AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
val_ds = val_ds.map(
    parse_path_label,
    num_parallel_calls=AUTOTUNE,
    deterministic=False,
)
val_ds = val_ds.cache("/tmp/val_cache")
val_ds = val_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)




## === cell 2
def get_model(base_model_cls, baseline_weight="imagenet", lr=1e-4):
    """Create EfficientNet model with a sigmoid head."""
    base_model = base_model_cls(
        include_top=False,
        input_shape=(H, W, 3),
        pooling="avg",
        weights=baseline_weight,
    )
    x = tf.keras.layers.Dropout(0.3)(base_model.output)
    outputs = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base_model.input, outputs=outputs)

    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )
    return model


base_model_cls = tf.keras.applications.EfficientNetB0
model = get_model(base_model_cls)




## === cell 3
model.fit(
    train_ds,
    epochs=12,
    validation_data=val_ds,
    verbose=1,
)




## === cell 4
test_files = [
    os.path.join(TEST_IMG_DIR, f)
    for f in os.listdir(TEST_IMG_DIR)
    if f.lower().endswith(".jpg")
]
test_ids = [os.path.splitext(os.path.basename(p))[0] for p in test_files]

test_ds = tf.data.Dataset.from_tensor_slices(test_files)
test_ds = test_ds.map(load_image, num_parallel_calls=AUTOTUNE, deterministic=False)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)

preds = model.predict(test_ds, verbose=1)

if preds.shape[1] > len(target_cols):
    preds = preds[:, : len(target_cols)]

submission = pd.DataFrame(preds, columns=target_cols)
submission.insert(0, "StudyInstanceUID", test_ids)

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
