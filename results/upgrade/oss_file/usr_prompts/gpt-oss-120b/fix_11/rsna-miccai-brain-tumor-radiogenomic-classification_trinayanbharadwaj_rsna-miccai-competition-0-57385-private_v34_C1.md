## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

## Submission Format
For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value
00001,0.5
00013,0.5
00015,0.5
etc.
```

def get_test_ids_from_dir(test_dir: str):
    """
    Scan the test directory and return a list of BraTS21ID strings
    sorted alphabetically. This is a fallback if the sample_submission
    file cannot be located.
    """
    if not os.path.isdir(test_dir):
        raise FileNotFoundError(f"Test directory not found: {test_dir}")
    case_paths = sorted(
        [entry.path for entry in os.scandir(test_dir) if entry.is_dir()]
    )
    ids = [os.path.basename(p) for p in case_paths]
    return ids


def find_test_dir(start_path: str = "."):
    """
    Recursively search for a directory named 'test' that appears to contain
    subject sub‑folders with DICOM files. Returns the first matching path.
    """
    for root, dirs, files in os.walk(start_path):
        if os.path.basename(root) == "test":
            for sub in dirs:
                sub_path = os.path.join(root, sub)
                if any(
                    fname.lower().endswith(".dcm") for fname in os.listdir(sub_path)
                ):
                    return root
    return None


def locate_file(possible_paths):
    """Return the first existing file path from a list, or None."""
    for p in possible_paths:
        if os.path.isfile(p):
            return p
    return None


def locate_dir(possible_paths):
    """Return the first existing directory path from a list, or None."""
    for p in possible_paths:
        if os.path.isdir(p):
            return p
    return None
