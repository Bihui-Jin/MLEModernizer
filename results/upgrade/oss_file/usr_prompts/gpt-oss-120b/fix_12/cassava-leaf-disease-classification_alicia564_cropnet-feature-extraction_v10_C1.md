# 5. Code solution

## === cell 0
import os
import math
import numpy as np
import pandas as pd
import tensorflow as tf
from sklearn.model_selection import train_test_split
from tensorflow.keras.applications.efficientnet import preprocess_input, EfficientNetB0
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D
from tensorflow.keras.models import Model
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau, Callback

tf.keras.mixed_precision.set_global_policy("mixed_float16")
tf.keras.utils.set_random_seed(42)

label_to_disease = pd.read_json(
    "/kaggle/input/cassava-leaf-disease-classification/label_num_to_disease_map.json",
    typ="series",
)
train_csv = pd.read_csv("/kaggle/input/cassava-leaf-disease-classification/train.csv")

train_csv["disease"] = train_csv["label"].map(label_to_disease).astype(str)
train_csv["path"] = (
    "/kaggle/input/cassava-leaf-disease-classification/train_images/"
    + train_csv["image_id"]
)
train_csv["label"] = train_csv["label"].astype(str)

train_df, valid_df = train_test_split(
    train_csv, test_size=0.2, stratify=train_csv["label"], random_state=42
)

BATCH_SIZE = 256
IMG_SIZE = (224, 224)
NUM_CLASSES = 5
worker_count = min(os.cpu_count() or 4, 8)

data_augmentation = tf.keras.Sequential(
    [
        tf.keras.layers.RandomFlip("horizontal_and_vertical"),
        tf.keras.layers.RandomRotation(0.2),
        tf.keras.layers.RandomZoom(0.2),
        tf.keras.layers.RandomTranslation(0.2, 0.2),
    ]
)


def decode_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    return img


def process_train(path, label):
    img = decode_image(path)
    img = data_augmentation(img)
    img = preprocess_input(img)
    label = tf.cast(label, tf.int32)
    label = tf.one_hot(label, NUM_CLASSES)
    return img, label


def process_valid(path, label):
    img = decode_image(path)
    img = preprocess_input(img)
    label = tf.cast(label, tf.int32)
    label = tf.one_hot(label, NUM_CLASSES)
    return img, label


train_paths = train_df["path"].values
train_labels = train_df["label"].astype(int).values
valid_paths = valid_df["path"].values
valid_labels = valid_df["label"].astype(int).values

train_options = tf.data.Options()
train_options.experimental_deterministic = False
valid_options = tf.data.Options()
valid_options.experimental_deterministic = False

train_ds = (
    tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
    .shuffle(len(train_paths), seed=42, reshuffle_each_iteration=True)
    .map(process_train, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
    .with_options(train_options)
)

valid_ds = (
    tf.data.Dataset.from_tensor_slices((valid_paths, valid_labels))
    .map(process_valid, num_parallel_calls=tf.data.AUTOTUNE)
    .cache()  # cache decoded & pre‑processed validation data
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
    .with_options(valid_options)
)

steps_per_epoch = math.ceil(len(train_paths) / BATCH_SIZE)
validation_steps = math.ceil(len(valid_paths) / BATCH_SIZE)




## === cell 1
class EarlyStoppingCallback(Callback):
    def on_epoch_end(self, epoch, logs=None):
        if self.model.stop_training:
            print(f"Early stopping triggered at epoch {epoch + 1}.")


early_stopping = EarlyStopping(
    monitor="val_loss", patience=3, restore_best_weights=True, verbose=1
)
learning_rate_reduction = ReduceLROnPlateau(
    monitor="val_loss", patience=2, factor=0.5, min_lr=1e-6, verbose=1
)

base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3)
)
x = GlobalAveragePooling2D()(base_model.output)
output = Dense(NUM_CLASSES, activation="softmax", dtype="float32")(
    x
)  # keep output in fp32
model = Model(inputs=base_model.input, outputs=output)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(
    train_ds,
    validation_data=valid_ds,
    epochs=5,
    steps_per_epoch=steps_per_epoch,
    validation_steps=validation_steps,
    callbacks=[early_stopping, learning_rate_reduction],
    verbose=2,
)




## === cell 2
disease_to_index = {
    str(i): i for i in range(NUM_CLASSES)
}  # same order as training labels
index_to_disease = {v: k for k, v in disease_to_index.items()}
disease_to_label = train_df.set_index("disease")["label"].to_dict()
idx_to_label = {
    idx: disease_to_label[disease] for idx, disease in index_to_disease.items()
}

image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
filenames = sorted([f for f in os.listdir(image_dir) if f.lower().endswith(".jpg")])
test_paths = [os.path.join(image_dir, f) for f in filenames]


def process_test(path):
    img = decode_image(path)
    img = preprocess_input(img)
    return img


test_ds = (
    tf.data.Dataset.from_tensor_slices(test_paths)
    .map(process_test, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

probs_batch = model.predict(test_ds, verbose=0)  # shape (N, 5)
pred_indices = np.argmax(probs_batch, axis=1)
predictions = [int(idx_to_label[idx]) for idx in pred_indices]

submission_df = pd.DataFrame({"image_id": filenames, "label": predictions})
submission_path = "/kaggle/working/submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission file created: {submission_path}")
print(submission_df.head())
