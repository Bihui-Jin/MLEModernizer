## === cell 0
import os
import json
import numpy as np
import pandas as pd
import cv2
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from joblib import Parallel, delayed

SEED = 42
np.random.seed(SEED)

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_PATH = os.path.join(BASE_PATH, "train_images")
TEST_IMG_PATH = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUBMIT = os.path.join(BASE_PATH, "sample_submission.csv")

IMG_SIZE = (128, 128)  # image resize dimensions
N_ESTIMATORS = 1200  # number of trees in the forest (unchanged)
HIST_BINS = 16  # bins per HSV channel for colour histogram


def load_and_preprocess_path(image_path):
    """Read an image, resize, flatten pixels and append HSV colour histogram."""
    img = cv2.imread(image_path, cv2.IMREAD_COLOR)
    if img is None:
        flat = np.zeros(IMG_SIZE[0] * IMG_SIZE[1] * 3, dtype=np.float32)
        hist = np.zeros(HIST_BINS * 3, dtype=np.float32)
        return np.concatenate([flat, hist])

    img_resized = cv2.resize(img, IMG_SIZE, interpolation=cv2.INTER_AREA)
    flat = img_resized.reshape(-1).astype(np.float32)

    hsv = cv2.cvtColor(img_resized, cv2.COLOR_BGR2HSV)
    hist_channels = [
        np.histogram(hsv[:, :, c], bins=HIST_BINS, range=(0, 256), density=True)[
            0
        ].astype(np.float32)
        for c in range(3)
    ]
    hist = np.concatenate(hist_channels)

    return np.concatenate([flat, hist])


train_df = pd.read_csv(TRAIN_CSV)

_feature_len = IMG_SIZE[0] * IMG_SIZE[1] * 3 + HIST_BINS * 3
num_samples = len(train_df)

train_img_paths = [
    os.path.join(TRAIN_IMG_PATH, img_id) for img_id in train_df["image_id"]
]

X_list = Parallel(n_jobs=-1, backend="threading")(
    delayed(load_and_preprocess_path)(p) for p in train_img_paths
)
X = np.stack(X_list).astype(np.float32)

y = train_df["label"].astype(np.int32).values

X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, random_state=SEED, stratify=y
)

rf_clf = RandomForestClassifier(
    n_estimators=N_ESTIMATORS,
    random_state=SEED,
    n_jobs=-1,
    class_weight="balanced",
)
rf_clf.fit(X_train, y_train)

val_pred = rf_clf.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy: {val_acc:.4f}")



## === cell 1
sample_sub = pd.read_csv(SAMPLE_SUBMIT)
test_ids = sample_sub["image_id"].tolist()
num_test = len(test_ids)

test_img_paths = [os.path.join(TEST_IMG_PATH, img_id) for img_id in test_ids]
test_X_list = Parallel(n_jobs=-1, backend="threading")(
    delayed(load_and_preprocess_path)(p) for p in test_img_paths
)
test_X = np.stack(test_X_list).astype(np.float32)

test_preds = rf_clf.predict(test_X)

submission = pd.DataFrame({"image_id": test_ids, "label": test_preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
