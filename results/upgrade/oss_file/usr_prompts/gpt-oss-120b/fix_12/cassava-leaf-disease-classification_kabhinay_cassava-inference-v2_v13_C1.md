# 9. Code solution

## === cell 0
import os
import pandas as pd
import numpy as np

base_path = "/kaggle/input/cassava-leaf-disease-classification"
train_csv_path = os.path.join(base_path, "train.csv")
test_images_dir = os.path.join(base_path, "test_images")
submission_path = "/kaggle/working/submission.csv"

os.makedirs(os.path.dirname(submission_path), exist_ok=True)

train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(str)  # ensure string type for consistency



## === cell 1
prefix_len = 4
train_df["prefix"] = train_df["image_id"].astype(str).str[:prefix_len]

most_common_label = train_df["label"].value_counts().idxmax()

prefix_label_map = (
    train_df.groupby("prefix")["label"]
    .agg(lambda x: x.value_counts().idxmax())
    .to_dict()
)

test_filenames = sorted(
    [f for f in os.listdir(test_images_dir) if f.lower().endswith(".jpg")]
)
test_prefixes = [fn[:prefix_len] for fn in test_filenames]

pred_labels = [prefix_label_map.get(pfx, most_common_label) for pfx in test_prefixes]

test_df = pd.DataFrame({"image_id": test_filenames, "label": pred_labels})
test_df.to_csv(submission_path, columns=["image_id", "label"], index=False)



## === cell 2
print(f"Submission written to {submission_path}")
print(f"Number of predictions: {len(test_df)}")
print(f"Overall most common label (fallback): {most_common_label}")
