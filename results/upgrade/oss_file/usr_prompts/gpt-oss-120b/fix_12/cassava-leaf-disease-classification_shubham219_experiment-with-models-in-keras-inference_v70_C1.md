# 5. Code solution

## === cell 0
import os
import multiprocessing

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import glob
import pandas as pd
import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.layers import GlobalAveragePooling2D, Dense
from sklearn.model_selection import train_test_split

AUTOTUNE = tf.data.AUTOTUNE

SEED = 42
tf.random.set_seed(SEED)
np.random.seed(SEED)



## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TEST_CSV_SAMPLE = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["path"] = TRAIN_IMG_DIR + "/" + df_train["image_id"]
df_train["label"] = df_train["label"].astype(str)

df_tr, df_val = train_test_split(
    df_train,
    test_size=0.1,
    stratify=df_train["label"],
    random_state=SEED,
)

IMG_SIZE = (224, 224)  # input size for EfficientNetB0
BATCH_SIZE = 128
NUM_CLASSES = 5



## === cell 2
preprocess_input = tf.keras.applications.efficientnet.preprocess_input


def decode_and_preprocess(path, label, augment=False):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, IMG_SIZE)
    if augment:
        image = tf.image.random_flip_left_right(image, seed=SEED)
    image = preprocess_input(image)
    label = tf.one_hot(label, NUM_CLASSES)
    return image, label


def create_dataset(df, augment=False):
    paths = df["path"].values
    int_labels = df["label"].astype(int).values
    ds = tf.data.Dataset.from_tensor_slices((paths, int_labels))
    if augment:
        ds = ds.shuffle(buffer_size=len(paths), seed=SEED)
    ds = ds.map(
        lambda p, l: decode_and_preprocess(p, l, augment),
        num_parallel_calls=AUTOTUNE,
    )
    ds = ds.cache()  # cache decoded tensors in memory
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds


train_ds = create_dataset(df_tr, augment=True)
val_ds = create_dataset(df_val, augment=False)



## === cell 3
base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=IMG_SIZE + (3,)
)
base_model.trainable = True

x = base_model.output
x = GlobalAveragePooling2D()(x)
output = Dense(NUM_CLASSES, activation="softmax")(x)

model = Model(inputs=base_model.input, outputs=output)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 4
model.fit(
    train_ds,
    epochs=8,
    validation_data=val_ds,
    verbose=2,
)



## === cell 5
test_image_paths = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
df_test = pd.DataFrame({"path": test_image_paths})
df_test["image_id"] = df_test["path"].apply(lambda x: os.path.basename(x))


def make_test_dataset(batch_size=512):
    paths = df_test["path"].values
    ds = tf.data.Dataset.from_tensor_slices(paths)
    ds = ds.map(
        lambda p: decode_and_preprocess(p, 0, augment=False)[0],
        num_parallel_calls=AUTOTUNE,
    )
    ds = ds.cache()  # cache test images after first decode
    ds = ds.batch(batch_size).prefetch(AUTOTUNE)
    return ds


test_ds = make_test_dataset(batch_size=512)

pred_test = model.predict(
    test_ds,
    verbose=1,
)

pred_test_labels = np.argmax(pred_test, axis=-1)

submission = df_test[["image_id"]].copy()
submission["label"] = pred_test_labels
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)

print(f"Submission written to {submission_path}")
