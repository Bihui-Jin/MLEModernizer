**[train/test]_tfrecords** the image files in tfrecord format.

**label_num_to_disease_map.json** The mapping between each disease code and the real disease name.

import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import pandas as pd
import tensorflow as tf
from tensorflow.keras.preprocessing.image import ImageDataGenerator

batch_size = 64
img_height = 256
img_width = 256
num_classes = 5

base_input = "/kaggle/input/cassava-leaf-disease-classification"
base_dir = "/kaggle/working"

## === cell 1
train_dir = os.path.join(base_input, "train_images")
test_dir = os.path.join(base_input, "test_images")
train_csv_path = os.path.join(base_input, "train.csv")
sample_sub_path = os.path.join(base_input, "sample_submission.csv")

labels = pd.read_csv(train_csv_path)
labels["label"] = labels["label"].astype(str)

sample_sub = pd.read_csv(sample_sub_path)

train_datagen = ImageDataGenerator(
    rescale=1.0 / 255,
    rotation_range=40,
    width_shift_range=0.2,
    height_shift_range=0.2,
    shear_range=0.2,
    zoom_range=0.2,
    horizontal_flip=True,
    validation_split=0.2,
)

val_datagen = ImageDataGenerator(rescale=1.0 / 255, validation_split=0.2)

train_ds = train_datagen.flow_from_dataframe(
    dataframe=labels,
    directory=train_dir,
    x_col="image_id",
    y_col="label",
    target_size=(img_height, img_width),
    batch_size=batch_size,
    class_mode="categorical",
    subset="training",
    shuffle=True,
    seed=42,
)

val_ds = val_datagen.flow_from_dataframe(
    dataframe=labels,
    directory=train_dir,
    x_col="image_id",
    y_col="label",
    target_size=(img_height, img_width),
    batch_size=batch_size,
    class_mode="categorical",
    subset="validation",
    shuffle=False,
    seed=42,
)



## === cell 2
base_model = tf.keras.applications.EfficientNetB0(
    include_top=False,
    input_shape=(img_height, img_width, 3),
    weights="imagenet",
)

x = tf.keras.layers.GlobalAveragePooling2D()(base_model.output)
x = tf.keras.layers.Dropout(0.2)(x)
outputs = tf.keras.layers.Dense(num_classes, activation="softmax", dtype="float32")(x)

model = tf.keras.Model(inputs=base_model.input, outputs=outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 3
epochs = 20  # increased epochs for better learning
model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=epochs,
    workers=4,
    use_multiprocessing=True,
    verbose=2,
)
