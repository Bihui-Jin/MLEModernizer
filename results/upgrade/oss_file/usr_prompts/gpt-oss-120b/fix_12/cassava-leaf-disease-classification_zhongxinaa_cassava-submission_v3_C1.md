def build_model(num_classes=5):
    """Construct EfficientNetV2B0‑based classifier (unchanged)."""
    base = efficientnet_v2.EfficientNetV2B0(
        include_top=False, weights="imagenet", input_shape=(224, 224, 3)
    )
    base.trainable = False
    x = GlobalAveragePooling2D()(base.output)
    output = Dense(num_classes, activation="softmax")(x)
    model = Model(inputs=base.input, outputs=output)
    model.compile(
        optimizer=Adam(learning_rate=1e-4),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
    )
    return model




## === cell 3
train_df = pd.read_csv(os.path.join(WORK_DIR, "train.csv"))
train_df["label"] = train_df["label"].astype(str)

val_df = train_df.sample(frac=0.2, random_state=42)
train_df = train_df.drop(val_df.index)

BATCH_SIZE = 128
IMG_SIZE = (224, 224)


def _load_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    return img


def _augment(img):
    img = tf.image.random_flip_left_right(img, seed=42)
    img = tf.keras.layers.RandomRotation(0.111, fill_mode="nearest", seed=42)(img)
    img = tf.keras.layers.RandomZoom(
        height_factor=0.2, width_factor=0.2, fill_mode="nearest", seed=42
    )(img)
    img = tf.keras.layers.RandomTranslation(
        height_factor=0.1, width_factor=0.1, fill_mode="nearest", seed=42
    )(img)
    return img


def _process_path_label(path, label, training):
    img = _load_image(path)
    if training:
        img = _augment(img)
    label = tf.cast(tf.strings.to_number(label, out_type=tf.int32), tf.int32)
    label = tf.one_hot(label, depth=5)
    return img, label


def make_dataset(df, training=True):
    paths = [
        os.path.join(WORK_DIR, "train_images", fname) for fname in df["image_id"].values
    ]
    labels = df["label"].values
    ds = tf.data.Dataset.from_tensor_slices((paths, labels))
    ds = ds.shuffle(buffer_size=len(df), seed=42) if training else ds
    ds = ds.map(
        lambda p, l: _process_path_label(p, l, training),
        num_parallel_calls=tf.data.AUTOTUNE,
    )
    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
    return ds


train_ds = make_dataset(train_df, training=True)
val_ds = make_dataset(val_df, training=False)

model = build_model(num_classes=5)

callbacks = [
    EarlyStopping(patience=5, restore_best_weights=True, monitor="val_accuracy"),
    ReduceLROnPlateau(patience=3, factor=0.5, monitor="val_accuracy"),
]

model.fit(
    train_ds,
    epochs=15,
    validation_data=val_ds,
    callbacks=callbacks,
    verbose=2,
)
