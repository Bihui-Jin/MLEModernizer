# 1. Kaggle task description

## Task
Identify hotels from images.

## Metric
Mean Average Precision @ 5 (MAP@5)

## Submission Format
For each image in the test set, you must predict a space-delimited list of hotel IDs that could match that image. The first ID should be the most relevant one and the last the least relevant one. The file should contain a header and have the following format:

```
image,hotel_id
99e91ad5f2870678.jpg,36363 53586 18807 64314 60181
b5cc62ab665591a9.jpg,36363 53586 18807 64314 60181
d5664a972d5a644b.jpg,36363 53586 18807 64314 60181
```

## Dataset
**train.csv** - The training set metadata.

- `image` - The image ID.

- `chain` - An ID code for the hotel chain. A `chain` of zero (0) indicates that the hotel is either not part of a chain or the chain is not known. This field is not available for the test set. The number of hotels per chain varies widely.

- `hotel_id` - The hotel ID. The target class.

- `timestamp` - When the image was taken. Provided for the training set only.

**sample_submission.csv** - A sample submission file in the correct format.

- `image` The image ID

- `hotel_id` The hotel ID. The target class.

**train_images** - The training set contains 97000+ images from around 7700 hotels from across the globe. All of the images for each hotel chain are in a dedicated subfolder for that chain.

**test_images** - The test set images. This competition has a hidden test set: only three images are provided here as samples while the remaining 13,000 images will be available to your notebook once it is submitted.

# 5. Code solution

## === cell 0
import pandas as pd
from pathlib import Path
from fastai.vision.all import *
from functools import partial
from fastai.metrics import top_k_accuracy


def first_existing_path(*candidates):
    """Return the first Path that exists among the given candidates."""
    for p in candidates:
        path = Path(p)
        if path.exists():
            return path
    raise FileNotFoundError(f"None of the candidate paths exist: {candidates}")




## === cell 1
import fastai, torch

print("fastai version:", fastai.__version__, "torch version:", torch.__version__)



## === cell 2
train_path = first_existing_path(
    "input/hotel-id-2021-fgvc8/train.csv",
    "/kaggle/input/hotel-id-2021-fgvc8/train.csv",
    "data/hotel-id-2021-fgvc8/train.csv",
    "train.csv",
)
train_df = pd.read_csv(train_path)
print("train rows:", len(train_df))



## === cell 3
train_images_path = first_existing_path(
    "train_images",
    "data/hotel-id-2021-fgvc8/train_images",
    "hotel-id-2021-fgvc8/train_images",
    "input/hotel-id-2021-fgvc8/train_images",
)

test_images_path = first_existing_path(
    "test_images",
    "data/hotel-id-2021-fgvc8/test_images",
    "hotel-id-2021-fgvc8/test_images",
    "input/hotel-id-2021-fgvc8/test_images",
)



## === cell 4
train_df["filepath"] = train_df["chain"].astype(str) + "/" + train_df["image"]

dls = ImageDataLoaders.from_df(
    train_df,
    path=train_images_path,
    fn_col="filepath",  # full relative path to image file
    label_col="hotel_id",
    valid_pct=0.1,
    seed=42,
    item_tfms=Resize(224),
    batch_tfms=aug_transforms(),
    bs=32,
    num_workers=0,
)
print("Classes:", len(dls.vocab))



## === cell 5
metric = partial(top_k_accuracy, k=5)
learn = vision_learner(dls, resnet34, metrics=metric, pretrained=True)
learn.fine_tune(5, base_lr=3e-3)  # a few more epochs for better MAP@5



## === cell 6
sample_sub_path = first_existing_path(
    "input/hotel-id-2021-fgvc8/sample_submission.csv",
    "/kaggle/input/hotel-id-2021-fgvc8/sample_submission.csv",
    "data/hotel-id-2021-fgvc8/sample_submission.csv",
    "sample_submission.csv",
)
submission = pd.read_csv(sample_sub_path)

orig_path = dls.path
dls.path = test_images_path
test_dl = dls.test_dl(submission["image"])
dls.path = orig_path  # restore original path

preds, _ = learn.get_preds(dl=test_dl)  # (num_test, num_classes)
top5_idx = torch.topk(preds, k=5, dim=1).indices
vocab = learn.dls.vocab
top5_str = [" ".join([vocab[i] for i in row]) for row in top5_idx.cpu().numpy()]

submission["hotel_id"] = top5_str

out_path = Path("submission.csv")
submission.to_csv(out_path, index=False)
print(f"Submission saved to {out_path}")
print(submission.head())
