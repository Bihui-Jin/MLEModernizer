- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging

- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

def auto_select_accelerator():
    """
    Reference:
        * https://www.kaggle.com/mgornergoogle/getting-started-with-100-flowers-on-tpu
        * https://www.kaggle.com/xhlulu/ranzcr-efficientnet-tpu-training
    """
    try:
        tpu = tf.distribute.cluster_resolver.TPUClusterResolver()
        tf.config.experimental_connect_to_cluster(tpu)
        tf.tpu.experimental.initialize_tpu_system(tpu)
        strategy = tf.distribute.experimental.TPUStrategy(tpu)
        print("Running on TPU:", tpu.master())
    except ValueError:
        strategy = tf.distribute.get_strategy()
    print(f"Running on {strategy.num_replicas_in_sync} replicas")
    return strategy


def build_decoder(with_labels=True, target_size=(IMAGE_SIZE, IMAGE_SIZE), ext="jpg"):
    preprocess_fn = tf.keras.applications.efficientnet.preprocess_input

    def decode(path):
        file_bytes = tf.io.read_file(path)
        if ext == "png":
            img = tf.image.decode_png(file_bytes, channels=3)
        elif ext in ["jpg", "jpeg"]:
            img = tf.image.decode_jpeg(file_bytes, channels=3)
        else:
            raise ValueError("Image extension not supported")
        img = tf.image.resize(tf.cast(img, tf.float32), target_size)
        img = preprocess_fn(img)  # EfficientNet proper scaling
        return img

    def decode_with_labels(path, label):
        return decode(path), label

    return decode_with_labels if with_labels else decode


def build_augmenter(with_labels=True):
    def augment(img):
        img = tf.image.random_flip_left_right(img)
        img = tf.image.random_flip_up_down(img)
        return img

    def augment_with_labels(img, label):
        return augment(img), label

    return augment_with_labels if with_labels else augment


def build_dataset(
    paths,
    labels=None,
    bsize=32,
    cache=True,
    decode_fn=None,
    augment_fn=None,
    augment=True,
    repeat=True,
    shuffle=1024,
    cache_dir="",
):
    if cache_dir != "" and cache is True:
        os.makedirs(cache_dir, exist_ok=True)

    if decode_fn is None:
        decode_fn = build_decoder(labels is not None)

    if augment_fn is None:
        augment_fn = build_augmenter(labels is not None)

    AUTO = tf.data.experimental.AUTOTUNE
    slices = paths if labels is None else (paths, labels)

    dset = tf.data.Dataset.from_tensor_slices(slices)
    dset = dset.map(decode_fn, num_parallel_calls=AUTO)
    dset = dset.cache(cache_dir) if cache else dset
    dset = dset.map(augment_fn, num_parallel_calls=AUTO) if augment else dset
    dset = dset.repeat() if repeat else dset
    dset = dset.shuffle(shuffle) if shuffle else dset
    dset = dset.batch(bsize).prefetch(AUTO)

    return dset

## === cell 5
with strategy.scope():
    model_path = "../input/inception-fold-0/model_fold_0.h5"
    if os.path.exists(model_path):
        model = tf.keras.models.load_model(model_path)
    else:
        base = tf.keras.applications.EfficientNetB0(
            weights="imagenet",
            include_top=False,
            input_shape=(IMAGE_SIZE, IMAGE_SIZE, 3),
        )
        for layer in base.layers[:-40]:
            layer.trainable = False
        for layer in base.layers[-40:]:
            layer.trainable = True

        x = tf.keras.layers.GlobalAveragePooling2D()(base.output)
        outputs = tf.keras.layers.Dense(len(label_cols), activation="sigmoid")(x)
        model = tf.keras.Model(inputs=base.input, outputs=outputs)

        optimizer = tf.keras.optimizers.Adam(learning_rate=1e-4)
        model.compile(optimizer=optimizer, loss="binary_crossentropy")

    train_csv_path = os.path.join(load_dir, "train.csv")
    train_df = pd.read_csv(train_csv_path)

    train_labels = train_df[label_cols].values.astype(np.float32)

    train_paths = list(load_dir + "train/" + train_df["StudyInstanceUID"] + ".jpg")

    X_train, X_val, y_train, y_val = train_test_split(
        train_paths,
        train_labels,
        test_size=0.1,
        random_state=42,
    )

    train_ds = build_dataset(
        X_train,
        labels=y_train,
        bsize=BATCH_SIZE,
        repeat=True,
        shuffle=256,  # smaller shuffle buffer reduces overhead
        augment=True,
        cache=True,  # keep caching enabled
        cache_dir="",  # empty string => in‑memory cache
    )
    val_ds = build_dataset(
        X_val,
        labels=y_val,
        bsize=BATCH_SIZE,
        repeat=False,
        shuffle=False,
        augment=False,
        cache=True,
        cache_dir="",  # in‑memory cache for validation as well
    )

    steps_per_epoch = max(1, len(X_train) // BATCH_SIZE)
    validation_steps = max(1, len(X_val) // BATCH_SIZE)

    callbacks = [
        tf.keras.callbacks.EarlyStopping(
            monitor="val_loss",
            patience=4,
            restore_best_weights=True,
            verbose=1,
        ),
        tf.keras.callbacks.ReduceLROnPlateau(
            monitor="val_loss",
            factor=0.5,
            patience=2,
            verbose=1,
            min_lr=1e-6,
        ),
    ]

    model.fit(
        train_ds,
        epochs=20,
        steps_per_epoch=steps_per_epoch,
        validation_data=val_ds,
        validation_steps=validation_steps,
        callbacks=callbacks,
        verbose=2,
    )

preds = model.predict(dtest, verbose=1)
