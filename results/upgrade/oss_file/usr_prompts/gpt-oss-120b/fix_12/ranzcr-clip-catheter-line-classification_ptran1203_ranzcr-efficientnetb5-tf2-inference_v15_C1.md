## Dataset
`train.csv` contains image IDs, binary labels, and patient IDs.

TFRecords are available for both train and test.

We've also included `train_annotations.csv`. These are segmentation annotations for training samples that have them. They are included solely as additional information for competitors.

- train.csv - contains image IDs, binary labels, and patient IDs.
- sample_submission.csv - a sample submission file in the correct format
- test - test images
- train - training images

### Columns
- `StudyInstanceUID` - unique ID for each image
- `ETT - Abnormal` - endotracheal tube placement abnormal
- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

import os, sys

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

for mod in list(sys.modules):

from sklearnex import patch_sklearn

import pandas as pd
import numpy as np
import cv2
from sklearn.ensemble import RandomForestRegressor
from sklearn.multioutput import MultiOutputRegressor
from joblib import Parallel, delayed  # parallel image loading

base_path = "../input/ranzcr-clip-catheter-line-classification"
train_dir = os.path.join(base_path, "train")
test_dir = os.path.join(base_path, "test")
train_csv_path = os.path.join(base_path, "train.csv")
sample_submission_path = os.path.join(base_path, "sample_submission.csv")

sample_sub = pd.read_csv(sample_submission_path, nrows=0)
target_cols = list(sample_sub.columns[1:])
N_CLASSES = len(target_cols)

IMG_H, IMG_W = 64, 64  # small enough for fast training




## === cell 1
def load_and_preprocess_image(path: str) -> np.ndarray:
    """
    Load an image from `path`, resize to (IMG_H, IMG_W) and normalize to [0, 1].
    Returns a flattened vector (1‑D) suitable for scikit‑learn models.
    """
    img = cv2.imread(path, cv2.IMREAD_COLOR)
    if img is None:
        raise FileNotFoundError(f"Image not found: {path}")
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = cv2.resize(img, (IMG_W, IMG_H))
    img = img.astype(np.float32) / 255.0
    return img.flatten()




## === cell 2
train_df = pd.read_csv(train_csv_path)
train_df["image_path"] = train_df["StudyInstanceUID"].apply(
    lambda uid: os.path.join(train_dir, f"{uid}.jpg")
)

existing_mask = train_df["image_path"].apply(os.path.exists)
if not existing_mask.all():
    missing = train_df.loc[~existing_mask, "StudyInstanceUID"].tolist()
    print(f"Warning: {len(missing)} training images missing and will be skipped.")
    train_df = train_df.loc[existing_mask].reset_index(drop=True)

train_paths = train_df["image_path"].values
X_train = np.stack(
    Parallel(n_jobs=-1, backend="loky", batch_size=100)(
        delayed(load_and_preprocess_image)(p) for p in train_paths
    )
)
y_train = train_df[target_cols].values.astype(np.float32)




## === cell 3
rf = RandomForestRegressor(
    n_estimators=200,
    max_depth=None,
    n_jobs=-1,
    random_state=42,
    min_samples_leaf=1,
)
model = MultiOutputRegressor(rf)

print("Training RandomForest model...")
model.fit(X_train, y_train)
print("Training completed.")




## === cell 4
test_fnames = [f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")]
test_paths = [os.path.join(test_dir, fname) for fname in test_fnames]
test_ids = [os.path.splitext(fname)[0] for fname in test_fnames]

X_test = np.stack(
    Parallel(n_jobs=-1, backend="loky", batch_size=100)(
        delayed(load_and_preprocess_image)(p) for p in test_paths
    )
)

print("Running inference on test set...")
test_preds = model.predict(X_test)
test_preds = np.clip(test_preds, 0.0, 1.0)

sub_df = pd.DataFrame(test_preds, columns=target_cols)
sub_df["StudyInstanceUID"] = test_ids
sub_df = sub_df[["StudyInstanceUID"] + target_cols]

submission_path = "submission.csv"
sub_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
