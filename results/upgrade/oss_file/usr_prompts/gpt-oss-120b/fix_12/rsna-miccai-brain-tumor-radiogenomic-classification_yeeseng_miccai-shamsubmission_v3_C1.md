# 1. Kaggle task description

## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

## Submission Format
For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value
00001,0.5
00013,0.5
00015,0.5
etc.
```

## Dataset
- **train/** - folder containing the training files, with each top-level folder representing a subject. **NOTE:** There are some unexpected issues with the following three cases in the training dataset, participants can exclude the cases during training: `[00109, 00123, 00709]`. We have checked and confirmed that the testing dataset is free from such issues.
- **train_labels.csv** - file containing the target `MGMT_value` for each subject in the training data (e.g. the presence of MGMT promoter methylation)
- **test/** - the test files, which use the same structure as `train/`; your task is to predict the `MGMT_value` for each subject in the test data. **NOTE**: the total size of the rerun test set (Public and Private) is ~5x the size of the Public test set
- **sample_submission.csv** - a sample submission file in the correct format

import numpy as np  # linear algebra
import pandas as pd  # data processing, CSV file I/O (e.g. pd.read_csv)
import glob
import os
from pathlib import Path


def first_existing_file(candidates):
    """Return the first Path in candidates that exists as a file, or None."""
    for p in candidates:
        if p.is_file():
            return p
    return None


def first_existing_dir(candidates):
    """Return the first Path in candidates that exists as a directory, or None."""
    for p in candidates:
        if p.is_dir():
            return p
    return None




## === cell 1
submission_candidates = [
    Path("..") / "input" / "miccai-testsubmissions" / "submission.csv",
    Path("/kaggle/input/miccai-testsubmissions/submission.csv"),
    Path("data") / "miccai-testsubmissions" / "submission.csv",
    Path("input") / "miccai-testsubmissions" / "submission.csv",
]
submission_path = first_existing_file(submission_candidates)

if submission_path:
    submissionDF = pd.read_csv(submission_path, dtype=str).set_index("BraTS21ID")
    scoreDict = submissionDF["MGMT_value"].astype(float).to_dict()
    print(
        f"Loaded previous submission with {len(scoreDict)} entries from {submission_path}."
    )
else:
    scoreDict = {}
    print("No previous submission found; proceeding with default predictions.")

print(
    f"Global mean MGMT_value: {global_mean:.5f} -> default prediction set to {default_pred:.5f}"
)




## === cell 3
test_dir_candidates = [
    Path("..")
    / "input"
    / "rsna-miccai-brain-tumor-radiogenomic-classification"
    / "test",
    Path("/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/test"),
    Path("data") / "rsna-miccai-brain-tumor-radiogenomic-classification" / "test",
    Path("input") / "rsna-miccai-brain-tumor-radiogenomic-classification" / "test",
]
test_dir = first_existing_dir(test_dir_candidates)

if not test_dir:
    raise FileNotFoundError(
        "Test directory not found in any of the expected locations."
    )
print(f"Using test directory: {test_dir}")

listOfStudyPaths = [p for p in test_dir.iterdir() if p.is_dir()]
listOfStudies = sorted([p.name for p in listOfStudyPaths])

if not listOfStudies:
    raise ValueError(
        "No study folders found in the test directory; cannot create submission."
    )

predList = []
for eachStudy in listOfStudies:
    if eachStudy in scoreDict:
        pred = float(scoreDict[eachStudy])
        pred = float(np.clip(pred, 0.0, 1.0))
        predList.append(pred)
    else:
        predList.append(default_pred)

predList = [float(np.clip(p, 0.0, 1.0)) for p in predList]

submissionDF = pd.DataFrame(
    {"BraTS21ID": listOfStudies, "MGMT_value": predList},
    columns=["BraTS21ID", "MGMT_value"],
)
submissionDF["MGMT_value"] = submissionDF["MGMT_value"].astype(float)




## === cell 4
output_path = Path("submission.csv")
