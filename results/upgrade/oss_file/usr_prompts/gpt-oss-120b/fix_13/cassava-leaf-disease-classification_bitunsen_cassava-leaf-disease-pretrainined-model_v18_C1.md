# 5. Code solution

## === cell 0
import os
import json
import numpy as np
import pandas as pd
import random
from PIL import Image

import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers

import multiprocessing

MAX_WORKERS = 4  # a safe small number that works well on most CPUs
NUM_WORKERS = min(MAX_WORKERS, multiprocessing.cpu_count())

try:
    tf.config.threading.set_inter_op_parallelism_threads(MAX_WORKERS)
    tf.config.threading.set_intra_op_parallelism_threads(MAX_WORKERS)
    tf.config.optimizer.set_jit(True)
except Exception:
    pass

gpus = tf.config.list_physical_devices("GPU")
if gpus:
    try:
        tf.config.experimental.set_memory_growth(gpus[0], True)
    except Exception:
        pass

random.seed(42)
np.random.seed(42)
tf.random.set_seed(42)



## === cell 1
BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")
TRAIN_CSV = os.path.join(BASE_DIR, "train.csv")
SAMPLE_SUBMISSION = os.path.join(BASE_DIR, "sample_submission.csv")

with open(os.path.join(BASE_DIR, "label_num_to_disease_map.json")) as f:
    label_map = json.load(f)


train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(str)
file_paths = train_df["image_id"].apply(lambda x: os.path.join(TRAIN_DIR, x)).values
labels = train_df["label"].astype(int).values

num_samples = len(file_paths)
indices = np.arange(num_samples)
np.random.shuffle(indices)

split = int(0.8 * num_samples)
train_idx, val_idx = indices[:split], indices[split:]

train_paths, train_labels = file_paths[train_idx], labels[train_idx]
val_paths, val_labels = file_paths[val_idx], labels[val_idx]

data_augmentation = keras.Sequential(
    [
        layers.RandomFlip(mode="horizontal_and_vertical"),
        layers.RandomRotation(factor=0.125),  # 45 degrees ≈ 0.125 of a full turn
        layers.RandomTranslation(height_factor=0.1, width_factor=0.1),
        layers.RandomZoom(height_factor=0.4, width_factor=0.4),
    ],
    name="data_augmentation",
)


def preprocess_image(path, label, augment=False):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [224, 224])
    image = image / 255.0  # rescale

    if augment:
        image = data_augmentation(image, training=True)
    return image, label


train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(
    lambda p, l: preprocess_image(p, l, augment=True),
    num_parallel_calls=tf.data.AUTOTUNE,
)
train_ds = train_ds.shuffle(buffer_size=10000, seed=42)
train_ds = train_ds.batch(256).prefetch(tf.data.AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
val_ds = val_ds.map(
    lambda p, l: preprocess_image(p, l, augment=False),
    num_parallel_calls=tf.data.AUTOTUNE,
)
val_ds = val_ds.batch(256).prefetch(tf.data.AUTOTUNE)



## === cell 2
base_model = keras.applications.MobileNetV2(
    input_shape=(224, 224, 3), include_top=False, weights="imagenet"
)

base_model.trainable = True
for layer in base_model.layers[:-20]:
    layer.trainable = False

model = keras.Sequential(
    [
        base_model,
        layers.GlobalAveragePooling2D(),
        layers.Dropout(0.2),
        layers.Dense(5, activation="softmax"),
    ]
)

model.compile(
    optimizer=keras.optimizers.Adam(),
    loss=keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)

model.summary()



## === cell 3
EPOCHS = 12  # increased epochs for better learning

model.fit(
    train_ds,
    epochs=EPOCHS,
    validation_data=val_ds,
    verbose=2,
)



## === cell 4
test_files = sorted([f for f in os.listdir(TEST_DIR) if f.lower().endswith(".jpg")])
test_paths = [os.path.join(TEST_DIR, f) for f in test_files]

test_ds = tf.data.Dataset.from_tensor_slices(test_paths)


def preprocess_test(path):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [224, 224])
    image = image / 255.0
    return image


test_ds = test_ds.map(preprocess_test, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(256).prefetch(tf.data.AUTOTUNE)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_files, "label": pred_labels.astype(int)})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
