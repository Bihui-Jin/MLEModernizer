train_df = pd.read_csv(TRAIN_CSV)

IMG_SIZE = (224, 224)
BATCH_SIZE = 128  # reduced batch size to lower memory pressure and prevent swapping
AUTOTUNE = tf.data.AUTOTUNE


def preprocess_path(path, label):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    return img, label


def make_dataset(df, training=True, cache_name="train_cache"):
    """
    Build a tf.data pipeline.
    Uses a file‑based cache (stored under ./cache) to avoid huge in‑memory caches that cause swapping.
    The rest of the pipeline (map, shuffle, batch, prefetch) is unchanged, preserving exact behavior.
    """
    img_paths = df["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_LOC, x)).values
    labels = df["label"].values
    ds = tf.data.Dataset.from_tensor_slices((img_paths, labels))
    ds = ds.map(preprocess_path, num_parallel_calls=AUTOTUNE, deterministic=False)
    cache_path = str(CACHE_DIR / f"{cache_name}.tfdata")
    ds = ds.cache(cache_path)  # file‑based cache instead of in‑memory
    if training:
        ds = ds.shuffle(1000, seed=42)
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds


train_ds = make_dataset(train_df, training=True, cache_name="train_cache")
