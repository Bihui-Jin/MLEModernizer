## === cell 0
import os, random, numpy as np, pandas as pd, tempfile

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
try:
    from google.protobuf import message_factory

    if not hasattr(message_factory.MessageFactory, "GetPrototype"):

        def _GetPrototype(self, descriptor):
            return self.GetMessageClass(descriptor)

        message_factory.MessageFactory.GetPrototype = _GetPrototype
except Exception:
    pass

from PIL import Image
import tensorflow as tf
from tensorflow.keras import layers, models, applications

tf.config.optimizer.set_jit(True)

random.seed(42)
np.random.seed(42)
tf.random.set_seed(42)

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification/"
if not os.path.isdir(BASE_PATH):
    BASE_PATH = "../input/cassava-leaf-disease-classification/"

TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")



## === cell 1
IMG_SIZE = 224
BATCH_SIZE = 128  # reduced batch size to lower peak memory usage
AUTOTUNE = tf.data.experimental.AUTOTUNE

train_df = pd.read_csv(TRAIN_CSV)
train_df["image_path"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)
image_paths = train_df["image_path"].values
labels = train_df["label"].values.astype(np.int32)


def _load_image(path, label):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = tf.cast(img, tf.float32) / 255.0
    return img, label


train_ds = tf.data.Dataset.from_tensor_slices((image_paths, labels))
train_ds = (
    train_ds.map(_load_image, num_parallel_calls=AUTOTUNE)
    .shuffle(buffer_size=len(train_df), seed=42, reshuffle_each_iteration=True)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)



## === cell 2
base_model = applications.EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(IMG_SIZE, IMG_SIZE, 3)
)
base_model.trainable = False  # freeze base

inputs = layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = base_model(inputs, training=False)
x = layers.GlobalAveragePooling2D()(x)
outputs = layers.Dense(5, activation="softmax")(x)
model = models.Model(inputs, outputs)

model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)



## === cell 3
EPOCHS = 3
model.fit(train_ds, epochs=EPOCHS, verbose=2)



## === cell 4
test_images = sorted(
    [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
)


def load_test_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0
    return img


test_paths = [os.path.join(TEST_IMG_DIR, img) for img in test_images]
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(load_test_image, num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)



## === cell 5
batch_preds = model.predict(test_ds, verbose=0)
preds = np.argmax(batch_preds, axis=1).tolist()
assert len(preds) == len(test_images), "Prediction length mismatch"



## === cell 6
submission = pd.DataFrame({"image_id": test_images, "label": preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
