# 5. Code solution

## === cell 0
import os, json, random

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, applications

gpus = tf.config.experimental.list_physical_devices("GPU")
if gpus:
    try:
        for gpu in gpus:
            tf.config.experimental.set_memory_growth(gpu, True)
    except Exception as e:
        print("GPU memory growth set failed:", e)

seed = 42
tf.random.set_seed(seed)
np.random.seed(seed)
random.seed(seed)

BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_DIR, "train_images")
TEST_IMG_DIR = os.path.join(BASE_DIR, "test_images")
TRAIN_CSV = os.path.join(BASE_DIR, "train.csv")
SAMPLE_SUB = os.path.join(BASE_DIR, "sample_submission.csv")
TRAIN_TFRECORD_DIR = os.path.join(BASE_DIR, "train_tfrecords")

CACHE_DIR = "/kaggle/working/tf_cache"
os.makedirs(CACHE_DIR, exist_ok=True)

print("Data directories verified:")
print("  Train images:", TRAIN_IMG_DIR)
print("  Test images :", TEST_IMG_DIR)



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)

IMG_SIZE = 224  # MobileNetV2 default input size
BATCH_SIZE = 256
AUTOTUNE = tf.data.AUTOTUNE

train_tfrecord_paths = tf.io.gfile.glob(os.path.join(TRAIN_TFRECORD_DIR, "*.tfrec"))
print(f"Found {len(train_tfrecord_paths)} TFRecord files for training.")


def _parse_tfrecord(example_proto):
    feature_description = {
        "image": tf.io.FixedLenFeature([], tf.string),
        "label": tf.io.FixedLenFeature([], tf.int64, default_value=0),
    }
    parsed_example = tf.io.parse_single_example(example_proto, feature_description)
    img = tf.image.decode_jpeg(parsed_example["image"], channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0
    label = tf.cast(parsed_example["label"], tf.int32)
    return img, label


train_ds = tf.data.TFRecordDataset(train_tfrecord_paths, num_parallel_reads=AUTOTUNE)
train_ds = (
    train_ds.map(_parse_tfrecord, num_parallel_calls=AUTOTUNE)
    .shuffle(buffer_size=10000, seed=seed)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)



## === cell 2
base_model = applications.MobileNetV2(
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
    include_top=False,
    weights="imagenet",
    pooling="avg",
)
base_model.trainable = True

model = models.Sequential(
    [
        base_model,
        layers.Dense(128, activation="relu"),
        layers.Dropout(0.2),
        layers.Dense(5, activation="softmax"),
    ]
)

model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()

EPOCHS = 6
history = model.fit(train_ds, epochs=EPOCHS, verbose=1)



## === cell 3
test_filenames = sorted(
    [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
)
print(f"Number of test images: {len(test_filenames)}")


def _parse_test(filename):
    img_path = tf.strings.join([TEST_IMG_DIR, "/", filename])
    img = tf.io.read_file(img_path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0
    return img


test_ds = tf.data.Dataset.from_tensor_slices(test_filenames)
test_ds = (
    test_ds.map(_parse_test, num_parallel_calls=AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)



## === cell 4
pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

assert len(pred_labels) == len(test_filenames), "Prediction length mismatch"



## === cell 5
submission = pd.DataFrame({"image_id": test_filenames, "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
