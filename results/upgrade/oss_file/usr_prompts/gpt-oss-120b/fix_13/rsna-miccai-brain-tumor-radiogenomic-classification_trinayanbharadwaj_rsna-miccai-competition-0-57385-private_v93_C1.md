import os
import pandas as pd
import numpy as np


class ConstantModel:
    """Simple model that returns a fixed probability of 0.5 for the positive class.
    The output shape matches the typical (n_samples, 2) format: [prob_neg, prob_pos].
    """

    def predict(self, X):
        n = X.shape[0] if isinstance(X, np.ndarray) else len(X)
        prob_pos = np.full((n, 1), 0.5, dtype=np.float32)
        prob_neg = 1.0 - prob_pos
        return np.hstack([prob_neg, prob_pos])
