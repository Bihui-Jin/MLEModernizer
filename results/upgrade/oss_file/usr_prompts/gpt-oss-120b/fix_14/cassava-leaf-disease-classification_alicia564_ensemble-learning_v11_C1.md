# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras.applications.efficientnet import EfficientNetB0, preprocess_input
from tensorflow.keras.layers import GlobalAveragePooling2D, Dense
from tensorflow.keras.models import Model
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau
from tensorflow.keras import mixed_precision
from sklearn.model_selection import train_test_split

mixed_precision.set_global_policy("mixed_float16")

BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_DIR, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_DIR, "train_images")
TEST_IMG_DIR = os.path.join(BASE_DIR, "test_images")

train_meta = pd.read_csv(TRAIN_CSV)
train_meta["path"] = train_meta["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)
train_meta["label_encoded"] = train_meta["label"].astype(int)

train_df, valid_df = train_test_split(
    train_meta,
    test_size=0.1,
    stratify=train_meta["label_encoded"],
    random_state=42,
)



## === cell 1
BATCH_SIZE = 128
IMG_SIZE = (224, 224)

data_augmentation = tf.keras.Sequential(
    [
        tf.keras.layers.RandomFlip(mode="horizontal_and_vertical"),
        tf.keras.layers.RandomRotation(factor=0.125),  # 45 degrees = 0.125 of 2π
        tf.keras.layers.RandomTranslation(height_factor=0.2, width_factor=0.2),
        tf.keras.layers.RandomZoom(height_factor=0.2, width_factor=0.2),
    ]
)


def _load_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    return img


def _prepare_train(img, label):
    img = tf.cast(img, tf.float32)
    img = data_augmentation(img)
    img = preprocess_input(img)  # EfficientNet scaling
    return img, label


def _prepare_valid(path, label):
    img = _load_image(path)
    img = tf.cast(img, tf.float32)
    img = preprocess_input(img)
    return img, label


train_paths = train_df["path"].values
train_labels = train_df["label_encoded"].values
train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(
    lambda p, l: (_load_image(p), l), num_parallel_calls=tf.data.AUTOTUNE
)
train_ds = train_ds.cache()
train_ds = train_ds.shuffle(
    buffer_size=len(train_paths), seed=42, reshuffle_each_iteration=True
)
train_ds = train_ds.map(_prepare_train, num_parallel_calls=tf.data.AUTOTUNE)
train_ds = train_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

valid_paths = valid_df["path"].values
valid_labels = valid_df["label_encoded"].values
valid_ds = tf.data.Dataset.from_tensor_slices((valid_paths, valid_labels))
valid_ds = valid_ds.map(_prepare_valid, num_parallel_calls=tf.data.AUTOTUNE)
valid_ds = valid_ds.batch(BATCH_SIZE).cache().prefetch(tf.data.AUTOTUNE)



## === cell 2
tf.keras.backend.clear_session()
base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3)
)
x = GlobalAveragePooling2D()(base_model.output)
output = Dense(5, activation="softmax")(x)
model = Model(inputs=base_model.input, outputs=output)

base_model.trainable = False
model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)

early_stopping = EarlyStopping(
    monitor="val_loss", patience=3, restore_best_weights=True, verbose=1
)
lr_reduction = ReduceLROnPlateau(
    monitor="val_loss", factor=0.5, patience=2, min_lr=1e-6, verbose=1
)

model.fit(
    train_ds,
    epochs=5,
    validation_data=valid_ds,
    callbacks=[early_stopping, lr_reduction],
    verbose=2,
)

base_model.trainable = True
for layer in base_model.layers[:-20]:
    layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(1e-4),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(
    train_ds,
    epochs=3,
    validation_data=valid_ds,
    callbacks=[early_stopping, lr_reduction],
    verbose=2,
)

cropnet_model = model



## === cell 3
img_size = (224, 224)
batch_size = 32

test_files = [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
test_paths = [os.path.join(TEST_IMG_DIR, f) for f in test_files]


def _load_test_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, img_size)
    img = tf.cast(img, tf.float32)
    img = preprocess_input(img)  # same preprocessing as training
    return img


test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(_load_test_image, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(batch_size).prefetch(tf.data.AUTOTUNE)

preds = cropnet_model.predict(test_ds, verbose=0)
predicted_classes = np.argmax(preds, axis=1).astype(int)

submission_df = pd.DataFrame({"image_id": test_files, "label": predicted_classes})
submission_path = "/kaggle/working/submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission file created: {submission_path}")
print(submission_df.head())
