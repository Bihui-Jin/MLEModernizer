# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import random
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.preprocessing import image as kp_image
import tempfile  # new import for disk caching

tf.config.optimizer.set_jit(True)

tf.config.threading.set_intra_op_parallelism_threads(8)
tf.config.threading.set_inter_op_parallelism_threads(8)

if tf.config.list_physical_devices("GPU"):
    from tensorflow.keras import mixed_precision

    mixed_precision.set_global_policy("mixed_float16")

SEED = 42
tf.random.set_seed(SEED)
np.random.seed(SEED)
random.seed(SEED)



## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TEST_CSV = os.path.join(BASE_PATH, "sample_submission.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")

train_df = pd.read_csv(TRAIN_CSV)
test_df = pd.read_csv(TEST_CSV)



## === cell 2
IMG_SIZE = 128
BATCH_SIZE = 512  # larger batch reduces number of steps per epoch
AUTOTUNE = tf.data.AUTOTUNE


def preprocess_image(file_path, label=None):
    """Load an image file, decode, resize and scale to [0,1]."""
    img = tf.io.read_file(file_path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, (IMG_SIZE, IMG_SIZE))
    img = img / 255.0
    if label is None:
        return img
    else:
        return img, label


train_file_paths = [
    os.path.join(TRAIN_IMG_DIR, fname) for fname in train_df["image_id"]
]
train_labels = train_df["label"].values.astype(np.int32)

indices = np.arange(len(train_file_paths))
np.random.shuffle(indices)
split = int(0.9 * len(indices))
train_idx, val_idx = indices[:split], indices[split:]

options = tf.data.Options()
options.experimental_deterministic = False

train_cache_file = os.path.join(tempfile.gettempdir(), "train_cache")
val_cache_file = os.path.join(tempfile.gettempdir(), "val_cache")

train_ds = tf.data.Dataset.from_tensor_slices(
    (np.array(train_file_paths)[train_idx], train_labels[train_idx])
)
train_ds = (
    train_ds.shuffle(buffer_size=1024, seed=SEED, reshuffle_each_iteration=False)
    .map(
        lambda x, y: preprocess_image(x, y),
        num_parallel_calls=AUTOTUNE,
        deterministic=False,
    )
    .cache(train_cache_file)  # cache to disk instead of RAM
    .batch(BATCH_SIZE, drop_remainder=True)  # fixed shape helps performance
    .prefetch(AUTOTUNE)
    .with_options(options)
)

val_ds = tf.data.Dataset.from_tensor_slices(
    (np.array(train_file_paths)[val_idx], train_labels[val_idx])
)
val_ds = (
    val_ds.map(
        lambda x, y: preprocess_image(x, y),
        num_parallel_calls=AUTOTUNE,
        deterministic=False,
    )
    .cache(val_cache_file)  # cache to disk
    .batch(BATCH_SIZE, drop_remainder=False)
    .prefetch(AUTOTUNE)
    .with_options(options)
)



## === cell 3
base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(IMG_SIZE, IMG_SIZE, 3)
)
base_model.trainable = True

inputs = layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = base_model(inputs, training=False)
x = layers.GlobalAveragePooling2D()(x)
x = layers.Dropout(0.2)(x)
outputs = layers.Dense(5, activation="softmax")(x)  # 5 classes
model = keras.Model(inputs, outputs)

model.compile(
    optimizer=keras.optimizers.Adam(learning_rate=1e-4),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 4
EPOCHS = 6  # unchanged training epochs
model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS, verbose=2)



## === cell 5
test_file_paths = [os.path.join(TEST_IMG_DIR, fname) for fname in test_df["image_id"]]
test_ds = tf.data.Dataset.from_tensor_slices(test_file_paths)
test_ds = (
    test_ds.map(
        lambda x: preprocess_image(x), num_parallel_calls=AUTOTUNE, deterministic=False
    )
    .cache()  # cache test images once (fits easily in memory)
    .batch(BATCH_SIZE, drop_remainder=False)
    .prefetch(AUTOTUNE)
    .with_options(options)
)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_df["image_id"], "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
