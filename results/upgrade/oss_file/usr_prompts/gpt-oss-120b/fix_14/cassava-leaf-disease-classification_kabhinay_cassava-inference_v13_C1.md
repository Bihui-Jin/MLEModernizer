# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf
import math

tf.random.set_seed(42)
np.random.seed(42)




## === cell 1
base_path = "/kaggle/input/cassava-leaf-disease-classification"
train_csv_path = os.path.join(base_path, "train.csv")
train_img_dir = os.path.join(base_path, "train_images")
test_img_dir = os.path.join(base_path, "test_images")

train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(str)  # keep original string type

train_paths = (
    train_df["image_id"].apply(lambda x: os.path.join(train_img_dir, x)).values
)
train_labels_int = train_df["label"].astype(int).values  # 0‑4

IMG_SIZE = (299, 299)
BATCH_SIZE = 512  # larger batch to reduce iteration overhead
NUM_CLASSES = 5


def _decode_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = tf.cast(img, tf.float32) / 255.0
    img = tf.keras.applications.xception.preprocess_input(img)
    return img


base_model = tf.keras.applications.Xception(
    weights="imagenet",
    include_top=False,
    input_shape=(299, 299, 3),
    pooling="avg",
)
base_model.trainable = False
FEATURE_DIM = base_model.output_shape[-1]  # 2048

train_path_ds = tf.data.Dataset.from_tensor_slices(train_paths)
train_img_ds = (
    train_path_ds.map(_decode_and_preprocess, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

features = base_model.predict(train_img_ds, verbose=0)  # (N, FEATURE_DIM)

labels_onehot = tf.keras.utils.to_categorical(train_labels_int, NUM_CLASSES).astype(
    np.float32
)

N = features.shape[0]
rng = np.random.RandomState(42)
perm = rng.permutation(N)
features = features[perm]
labels_onehot = labels_onehot[perm]

val_examples = int(0.1 * N)
features_val = features[:val_examples]
labels_val = labels_onehot[:val_examples]
features_train = features[val_examples:]
labels_train = labels_onehot[val_examples:]




## === cell 2
inputs = tf.keras.Input(shape=(FEATURE_DIM,))
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(inputs)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)




## === cell 3
model.fit(
    x=features_train,
    y=labels_train,
    batch_size=BATCH_SIZE,
    epochs=3,
    validation_data=(features_val, labels_val),
    verbose=1,
)




## === cell 4
test_files = sorted(os.listdir(test_img_dir))  # keep deterministic order
test_paths = [os.path.join(test_img_dir, f) for f in test_files]

test_path_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_img_ds = (
    test_path_ds.map(_decode_and_preprocess, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

test_features = base_model.predict(test_img_ds, verbose=0)  # (num_test, FEATURE_DIM)

preds = model.predict(test_features, batch_size=BATCH_SIZE, verbose=0)
pred_labels = np.argmax(preds, axis=1)

submission = pd.DataFrame({"image_id": test_files, "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
