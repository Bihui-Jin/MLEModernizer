# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, optimizers, callbacks, mixed_precision
from tensorflow.keras.applications import EfficientNetB0
import warnings

warnings.filterwarnings("ignore")
SEED = 42
tf.random.set_seed(SEED)
np.random.seed(SEED)

mixed_precision.set_global_policy("mixed_float16")



## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_DIR = os.path.join(BASE_PATH, "train_images")
TEST_DIR = os.path.join(BASE_PATH, "test_images")

IMG_SIZE = 224
BATCH_SIZE = 256  # reduced to lower memory usage and avoid swapping
NUM_CLASSES = 5
EPOCHS = 5
VALID_SPLIT = 0.1
AUTOTUNE = tf.data.AUTOTUNE

options = tf.data.Options()
options.experimental_deterministic = False

train_df = pd.read_csv(TRAIN_CSV)

train_paths = tf.constant(
    [os.path.join(TRAIN_DIR, fname) for fname in train_df["image_id"].values]
)
train_labels = tf.constant(train_df["label"].values, dtype=tf.int32)

full_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
full_ds = full_ds.with_options(options)

full_ds = full_ds.shuffle(buffer_size=1000, seed=SEED, reshuffle_each_iteration=False)

val_size = int(VALID_SPLIT * len(train_df))
train_size = len(train_df) - val_size

train_ds = full_ds.take(train_size)
val_ds = full_ds.skip(train_size)


def _decode_resize(path, label):
    """Read image file, decode JPEG, resize and scale to [0,1]."""
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0
    return img, label


train_ds = (
    train_ds.map(_decode_resize, num_parallel_calls=AUTOTUNE, deterministic=False)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

val_ds = (
    val_ds.map(_decode_resize, num_parallel_calls=AUTOTUNE, deterministic=False)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(IMG_SIZE, IMG_SIZE, 3)
)
base_model.trainable = False  # freeze for quick training

model = models.Sequential(
    [
        layers.RandomFlip(mode="horizontal", seed=SEED),
        layers.RandomRotation(factor=20 / 360, seed=SEED),
        layers.RandomTranslation(height_factor=0.1, width_factor=0.1, seed=SEED),
        base_model,
        layers.GlobalAveragePooling2D(),
        layers.Dropout(0.2, seed=SEED),
        layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

model.compile(
    optimizer=optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

early_stop = callbacks.EarlyStopping(
    patience=3, restore_best_weights=True, monitor="val_accuracy"
)
reduce_lr = callbacks.ReduceLROnPlateau(patience=2, factor=0.5, monitor="val_accuracy")

model.fit(
    train_ds,
    epochs=EPOCHS,
    validation_data=val_ds,
    callbacks=[early_stop, reduce_lr],
    verbose=2,
)



## === cell 2
test_files = sorted([f for f in os.listdir(TEST_DIR) if f.lower().endswith(".jpg")])
test_paths = tf.constant([os.path.join(TEST_DIR, f) for f in test_files])


def _decode_resize_test(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0
    return img


test_ds = (
    tf.data.Dataset.from_tensor_slices(test_paths)
    .map(_decode_resize_test, num_parallel_calls=AUTOTUNE, deterministic=False)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_files, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
