# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers, models
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.applications.efficientnet import preprocess_input

tf.config.optimizer.set_jit(True)  # enable XLA JIT compilation
tf.config.threading.set_intra_op_parallelism_threads(8)
tf.config.threading.set_inter_op_parallelism_threads(8)

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

BATCH_SIZE = 128
IMG_SIZE = (224, 224)

train_df = pd.read_csv(TRAIN_CSV)
train_paths = [os.path.join(TRAIN_IMG_DIR, fname) for fname in train_df["image_id"]]
train_labels = train_df["label"].astype(np.int32).values

rng = np.random.default_rng(seed=42)
indices = rng.permutation(len(train_paths))
split = int(0.1 * len(train_paths))
val_idx, train_idx = indices[:split], indices[split:]

val_paths = [train_paths[i] for i in val_idx]
val_labels = train_labels[val_idx]
train_paths = [train_paths[i] for i in train_idx]
train_labels = train_labels[train_idx]


def _load_image(path, label=None):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = preprocess_input(img)
    if label is None:
        return img
    return img, label


options = tf.data.Options()
options.experimental_deterministic = False

train_ds = (
    tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
    .map(_load_image, num_parallel_calls=tf.data.AUTOTUNE)
    .cache()  # cache decoded images to avoid re‑decoding each epoch
    .shuffle(buffer_size=len(train_paths), seed=42, reshuffle_each_iteration=False)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
    .with_options(options)
)

val_ds = (
    tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
    .map(_load_image, num_parallel_calls=tf.data.AUTOTUNE)
    .cache()  # validation set is small; caching speeds up repeated evaluation
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
    .with_options(options)
)



## === cell 1
test_files = sorted(
    [
        f
        for f in os.listdir(TEST_IMG_DIR)
        if f.lower().endswith(".jpg") and os.path.isfile(os.path.join(TEST_IMG_DIR, f))
    ]
)
test_paths = [os.path.join(TEST_IMG_DIR, f) for f in test_files]

test_ds = (
    tf.data.Dataset.from_tensor_slices(test_paths)
    .map(_load_image, num_parallel_calls=tf.data.AUTOTUNE)
    .cache()  # cache test images for fast inference
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
    .with_options(options)
)



## === cell 2
base = EfficientNetB0(include_top=False, input_shape=(224, 224, 3), weights="imagenet")
base.trainable = False  # freeze base for quick training

inputs = layers.Input(shape=(224, 224, 3))
x = base(inputs, training=False)
x = layers.GlobalAveragePooling2D()(x)
outputs = layers.Dense(5, activation="softmax")(x)
model = models.Model(inputs, outputs)

model.compile(
    optimizer=keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 3
model_path = "../input/keras-available-models-part-1-training/model.h5"
if os.path.exists(model_path):
    model = keras.models.load_model(model_path)
else:
    model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=2)



## === cell 4
pred_probs = model.predict(test_ds, verbose=0)
y_preds = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_files, "label": y_preds})
assert len(submission) == len(test_files)

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}, shape: {submission.shape}")
