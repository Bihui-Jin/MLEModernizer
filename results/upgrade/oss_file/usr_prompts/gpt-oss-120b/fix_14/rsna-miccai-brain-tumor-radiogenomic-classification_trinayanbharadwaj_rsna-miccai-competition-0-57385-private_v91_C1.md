## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

## Submission Format
For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value
00001,0.5
00013,0.5
00015,0.5
etc.
```

- **train_labels.csv** - file containing the target `MGMT_value` for each subject in the training data (e.g. the presence of MGMT promoter methylation)
- **test/** - the test files, which use the same structure as `train/`; your task is to predict the `MGMT_value` for each subject in the test data. **NOTE**: the total size of the rerun test set (Public and Private) is ~5x the size of the Public test set
- **sample_submission.csv** - a sample submission file in the correct format

## === cell 0
import os
import pandas as pd
import numpy as np  # retained for potential future use

TEST_DIR = "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/test"
SAMPLE_SUBMISSION = "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/sample_submission.csv"
TRAIN_LABELS = (
    "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/train_labels.csv"
)




## === cell 1
def get_test_ids(test_dir: str, fallback_csv: str) -> list:
    """
    Return a list of BraTS21ID strings for the test set.

    1. Try to read the sub‑folder names under ``test_dir``.
    2. If the directory does not exist or contains no IDs, fall back to the
       ``fallback_csv`` which holds the official list of IDs.
    """
    if os.path.isdir(test_dir):
        ids = [
            entry.name
            for entry in sorted(os.scandir(test_dir), key=lambda e: e.name)
            if entry.is_dir()
        ]
        if ids:
            return ids

    if os.path.isfile(fallback_csv):
        df = pd.read_csv(fallback_csv, dtype=str)
        if "BraTS21ID" in df.columns:
            return df["BraTS21ID"].tolist()

    raise FileNotFoundError(
        f"Could not locate test IDs in '{test_dir}' and fallback file '{fallback_csv}' is missing or malformed."
    )


test_ids = get_test_ids(TEST_DIR, SAMPLE_SUBMISSION)



## === cell 2
PROBLEMATIC_IDS = {"00109", "00123", "00709"}

if os.path.isfile(TRAIN_LABELS):
    train_df = pd.read_csv(TRAIN_LABELS, dtype={"BraTS21ID": str, "MGMT_value": float})
    if "BraTS21ID" in train_df.columns:
        mask = ~train_df["BraTS21ID"].isin(PROBLEMATIC_IDS)
        filtered = train_df[mask]
    else:
        filtered = train_df

    if "MGMT_value" in filtered.columns and not filtered.empty:
        global_mean = filtered["MGMT_value"].mean()
    else:
        global_mean = 0.5  # fallback to neutral probability
else:
    global_mean = 0.5  # fallback if training labels are unavailable

predictions = [global_mean] * len(test_ids)
predictions = np.clip(predictions, 0.0, 1.0).tolist()




## === cell 3
def create_submission(ids, preds):
    """
    Build the submission DataFrame with the required column names.
    """
    return pd.DataFrame({"BraTS21ID": ids, "MGMT_value": preds})


submission_df = create_submission(test_ids, predictions)



## === cell 4
submission_path = "/kaggle/working/submission.csv"
os.makedirs(os.path.dirname(submission_path), exist_ok=True)
submission_df.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path} with {len(submission_df)} rows.")
