## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
import tensorflow as tf

tf.config.threading.set_inter_op_parallelism_threads(4)
tf.config.threading.set_intra_op_parallelism_threads(4)

import json
import numpy as np
import pandas as pd
from tensorflow.keras import layers, models, applications, optimizers
from sklearn.model_selection import train_test_split

BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")
IMG_HEIGHT = 224
IMG_WIDTH = 224
BATCH_SIZE = 256  # larger batch → fewer steps per epoch, same gradient effect
PRE_TRAINED_MODEL = "../input/xceptionv1/Cassava_Best_Model_XceptionV3_V01.hdf5"

tf.random.set_seed(42)
np.random.seed(42)




## === cell 1
with open(os.path.join(BASE_DIR, "label_num_to_disease_map.json")) as file:
    label_map = json.load(file)

train_df = pd.read_csv(os.path.join(BASE_DIR, "train.csv"))
train_df["label"] = train_df["label"].astype(np.int32)

train_df, val_df = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=42
)




## === cell 2
def _process_path(path):
    """Read an image file, decode, resize to target size, and normalize."""
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(
        img, [IMG_HEIGHT, IMG_WIDTH], method=tf.image.ResizeMethod.BILINEAR
    )
    img = tf.cast(img, tf.float32) / 255.0
    return img


def make_dataset(df, data_dir, shuffle=True):
    """
    Create a tf.data.Dataset from a dataframe.
    Caches are kept only for non‑training datasets (validation / test) to
    avoid holding the entire training set in RAM while preserving exact
    semantics for evaluation and inference.
    """
    paths = df["image_id"].apply(lambda x: os.path.join(data_dir, x)).values

    if "label" in df.columns:
        labels = df["label"].astype(np.int32).values
        ds = tf.data.Dataset.from_tensor_slices((paths, labels))
        ds = ds.map(
            lambda p, l: (_process_path(p), tf.cast(l, tf.int32)),
            num_parallel_calls=tf.data.AUTOTUNE,
        )
        if not shuffle:
            ds = ds.cache()
        if shuffle:
            ds = ds.shuffle(buffer_size=1024, seed=42, reshuffle_each_iteration=True)
    else:
        ds = tf.data.Dataset.from_tensor_slices(paths)
        ds = ds.map(_process_path, num_parallel_calls=tf.data.AUTOTUNE)
        ds = ds.cache()  # test set is small enough for cache

    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
    return ds


train_ds = make_dataset(train_df, TRAIN_DIR, shuffle=True)
val_ds = make_dataset(val_df, TRAIN_DIR, shuffle=False)




## === cell 3
try:
    model = tf.keras.models.load_model(PRE_TRAINED_MODEL)
    print("Loaded pretrained model.")
except Exception as e:
    print(f"Pretrained model not found or failed to load ({e}), building a new model.")
    base_model = applications.MobileNetV2(
        input_shape=(IMG_HEIGHT, IMG_WIDTH, 3), include_top=False, weights="imagenet"
    )
    base_model.trainable = False  # freeze backbone
    inputs = layers.Input(shape=(IMG_HEIGHT, IMG_WIDTH, 3))
    x = base_model(inputs, training=False)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.Dropout(0.2)(x)
    outputs = layers.Dense(5, activation="softmax")(x)
    model = models.Model(inputs, outputs)
    model.compile(
        optimizer=optimizers.Adam(),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )
    model.fit(train_ds, validation_data=val_ds, epochs=5, verbose=0)




## === cell 4
test_filenames = sorted(os.listdir(TEST_DIR))
test_df = pd.DataFrame({"image_id": test_filenames})
test_ds = make_dataset(test_df, TEST_DIR, shuffle=False)




## === cell 5
pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_filenames, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
