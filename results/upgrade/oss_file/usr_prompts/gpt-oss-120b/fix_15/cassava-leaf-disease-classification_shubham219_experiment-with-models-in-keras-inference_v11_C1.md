## === cell 0
import os
import glob
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
from sklearn.neural_network import MLPClassifier
import concurrent.futures  # will use ThreadPoolExecutor instead of ProcessPoolExecutor
import random
import gc

SEED = 42
np.random.seed(SEED)
random.seed(SEED)  # deterministic for any random ops

IMG_SIZE = (48, 48)  # width, height (was 64x64)




## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TEST_CSV_SAMPLE = os.path.join(BASE_PATH, "sample_submission.csv")

train_df = pd.read_csv(TRAIN_CSV)


def load_and_preprocess(img_path, size=IMG_SIZE):
    """Open an image, resize, normalize to [0,1], and return a flattened float32 array."""
    img = Image.open(img_path).convert("RGB")
    img = img.resize(size)
    arr = np.asarray(img, dtype=np.float32) / 255.0  # normalize
    return arr.flatten()


def _loader_worker(args):
    """Worker that receives (path, size) and returns the pre‑processed array."""
    path, size = args
    return load_and_preprocess(path, size)


def parallel_load(paths, size=IMG_SIZE):
    """
    Load images in parallel using a thread pool (avoids inter‑process pickling overhead).
    - Uses up to the number of CPU cores, capped at 16.
    - A larger chunksize reduces scheduling overhead.
    """
    n = len(paths)
    dim = size[0] * size[1] * 3
    X = np.empty((n, dim), dtype=np.float32)

    cpu_cnt = os.cpu_count() or 1
    max_workers = min(cpu_cnt, 16)  # limit workers to avoid overload

    args_iter = ((p, size) for p in paths)

    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        for i, arr in enumerate(executor.map(_loader_worker, args_iter, chunksize=256)):
            X[i] = arr
    return X


train_images = [
    os.path.join(TRAIN_IMG_DIR, img_id) for img_id in train_df["image_id"].values
]

X = parallel_load(train_images)
y = train_df["label"].values
X_scaled = X  # already normalized
del X
gc.collect()




## === cell 2
X_tr, X_val, y_tr, y_val = train_test_split(
    X_scaled, y, test_size=0.1, random_state=SEED, stratify=y
)

clf = MLPClassifier(
    hidden_layer_sizes=(512, 256, 128),  # unchanged architecture
    activation="relu",
    solver="adam",
    alpha=1e-4,
    batch_size=128,
    learning_rate="adaptive",
    max_iter=200,  # unchanged number of epochs
    early_stopping=False,
    random_state=SEED,
)
clf.fit(X_tr, y_tr)

val_pred = clf.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy: {val_acc:.4f}")




## === cell 3
test_image_paths = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
test_df = pd.DataFrame(test_image_paths, columns=["path"])
test_df["image_id"] = test_df["path"].apply(lambda p: os.path.basename(p))

X_test = parallel_load(test_df["path"].values)
X_test_scaled = X_test  # already normalized
del X_test
gc.collect()

test_pred = clf.predict(X_test_scaled)

submission = pd.DataFrame({"image_id": test_df["image_id"], "label": test_pred})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
