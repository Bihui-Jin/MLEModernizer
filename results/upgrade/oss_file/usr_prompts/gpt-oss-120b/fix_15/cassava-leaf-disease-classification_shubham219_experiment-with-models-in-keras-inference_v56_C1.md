# 5. Code solution

## === cell 0
import os
import glob
import numpy as np
import pandas as pd
import multiprocessing

try:
    from google.protobuf import message_factory  # noqa: F401
except Exception:
    pass

import tensorflow as tf
from tensorflow.keras.models import load_model
from tensorflow.keras.applications import EfficientNetB3
from tensorflow.keras.applications.efficientnet import preprocess_input
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras import layers, models, optimizers

SEED = 42
DEBUG = False
tf.random.set_seed(SEED)
np.random.seed(SEED)

NUM_WORKERS = min(4, multiprocessing.cpu_count())



## === cell 1
weight_path = "../input/cassava-leaf-disease-classification/model-v17/model_v0.38.h5"
model_loaded = False

if not os.path.exists(weight_path):
    for root, _, files in os.walk("../input"):
        for f in files:
            if f.lower().endswith(".h5"):
                weight_path = os.path.join(root, f)
                if DEBUG:
                    print(f"Found alternative weight file: {weight_path}")
                break
        if os.path.exists(weight_path):
            break

if os.path.exists(weight_path):
    try:
        my_model = load_model(
            weight_path,
            custom_objects={"EfficientNetB3": EfficientNetB3},
            compile=False,
        )
        model_loaded = True
        if DEBUG:
            print("Loaded model from", weight_path)
    except Exception as e:
        if DEBUG:
            print("Failed to load model:", e)

if not model_loaded:
    if DEBUG:
        print("Pretrained model not found – building and briefly training a new model.")
    base = EfficientNetB3(weights="imagenet", include_top=False, pooling="avg")
    base.trainable = False

    inputs = layers.Input(shape=(512, 512, 3))
    x = preprocess_input(inputs)
    x = base(x, training=False)
    outputs = layers.Dense(5, activation="softmax")(x)
    my_model = models.Model(inputs, outputs)

    my_model.compile(
        optimizer=optimizers.Adam(),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )

    train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
    train_df = pd.read_csv(train_csv_path)
    train_df["label"] = train_df["label"].astype(str)
    train_df["path"] = train_df["image_id"].apply(
        lambda img: os.path.join(
            "../input/cassava-leaf-disease-classification/train_images", img
        )
    )

    train_idg = ImageDataGenerator(
        preprocessing_function=preprocess_input,
        horizontal_flip=True,
        rotation_range=20,
        width_shift_range=0.1,
        height_shift_range=0.1,
        zoom_range=0.1,
    )
    train_gen = train_idg.flow_from_dataframe(
        dataframe=train_df,
        x_col="path",
        y_col="label",
        target_size=(512, 512),
        batch_size=32,
        class_mode="sparse",
        shuffle=True,
        seed=SEED,
        workers=NUM_WORKERS,
        use_multiprocessing=True,
    )

    my_model.fit(
        train_gen,
        epochs=3,
        steps_per_epoch=train_gen.n // train_gen.batch_size,
        verbose=1,
    )
    if DEBUG:
        print("Fallback model training complete.")



## === cell 2
test_images = glob.glob(
    "../input/cassava-leaf-disease-classification/test_images/*.jpg"
)
df_test = pd.DataFrame(test_images, columns=["path"])


def make_test_gen(batch_size=32, flip=False):
    """Create a test ImageDataGenerator; set flip=True for horizontal flip."""
    test_idg = ImageDataGenerator(
        preprocessing_function=preprocess_input, horizontal_flip=flip
    )
    test_gen = test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=batch_size,
        seed=SEED,
        shuffle=False,
        class_mode=None,
        target_size=(512, 512),
        workers=NUM_WORKERS,
        use_multiprocessing=True,
    )
    return test_gen




## === cell 3
test_gen_orig = make_test_gen(batch_size=32, flip=False)
pred_orig = my_model.predict(test_gen_orig, verbose=1)

test_gen_flip = make_test_gen(batch_size=32, flip=True)
pred_flip = my_model.predict(test_gen_flip, verbose=1)

pred_avg = (pred_orig + pred_flip) / 2.0
pred_labels = np.argmax(pred_avg, axis=-1)

final_submission = df_test.copy()
final_submission["image_id"] = final_submission["path"].apply(os.path.basename)
final_submission["label"] = pred_labels.astype(int)

final_csv = final_submission[["image_id", "label"]]
final_csv.to_csv("submission.csv", index=False)
