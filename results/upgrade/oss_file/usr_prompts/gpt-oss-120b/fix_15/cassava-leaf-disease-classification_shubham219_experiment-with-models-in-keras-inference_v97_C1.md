# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import tensorflow as tf

tf.config.optimizer.set_jit(True)

tf.config.threading.set_intra_op_parallelism_threads(os.cpu_count())
tf.config.threading.set_inter_op_parallelism_threads(os.cpu_count())

import glob
import pandas as pd
import numpy as np
import tensorflow.keras.layers as tfl
from tensorflow.keras.models import Model
from sklearn.model_selection import train_test_split

policy = tf.keras.mixed_precision.Policy("mixed_float16")
tf.keras.mixed_precision.set_global_policy(policy)

SEED = 42
tf.random.set_seed(SEED)
np.random.seed(SEED)
os.environ["PYTHONHASHSEED"] = str(SEED)

BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["label"] = df_train["label"].astype(str)
df_train["path"] = df_train["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))
df_train = df_train[df_train["path"].apply(os.path.isfile)].reset_index(drop=True)




## === cell 1
train_df, val_df = train_test_split(
    df_train,
    test_size=0.1,
    stratify=df_train["label"],
    random_state=SEED,
)

IMG_SIZE = (256, 256)
BATCH_SIZE = 256  # reduced to limit RAM usage while keeping same epoch semantics
NUM_CLASSES = 5

preprocess = tf.keras.applications.resnet50.preprocess_input


def load_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = preprocess(img)
    return img


augment = tf.keras.Sequential(
    [
        tfl.RandomFlip("horizontal", seed=SEED),
        tfl.RandomRotation(0.04, seed=SEED),  # ~15 degrees
    ]
)


def prepare_dataset(df, training=False):
    """
    Build a tf.data pipeline.
    Switch to in‑memory caching (ds.cache()) instead of file‑based caching.
    This removes costly temporary file I/O while preserving the exact same
    preprocessing, augmentation, and batching logic.
    """
    paths = df["path"].values
    label_ints = df["label"].astype(int).values

    ds = tf.data.Dataset.from_tensor_slices((paths, label_ints))
    ds = ds.shuffle(buffer_size=len(paths), seed=SEED, reshuffle_each_iteration=False)

    def _load(path, label):
        img = load_and_preprocess(path)
        label_onehot = tf.one_hot(label, NUM_CLASSES)
        return img, label_onehot

    ds = ds.map(_load, num_parallel_calls=tf.data.AUTOTUNE)

    ds = ds.cache()

    if training:

        def _aug(img, label_onehot):
            img = augment(img, training=True)
            return img, label_onehot

        ds = ds.map(_aug, num_parallel_calls=tf.data.AUTOTUNE)

    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
    return ds


train_ds = prepare_dataset(train_df, training=True)
val_ds = prepare_dataset(val_df, training=False)




## === cell 2
try:
    weight_path = "/kaggle/input/resnet50-trasnfer-learning-on-tpu/cassava_base.h5"
    my_model = tf.keras.models.load_model(weight_path)
except Exception:
    base = tf.keras.applications.ResNet50(
        include_top=False,
        weights="imagenet",
        input_shape=IMG_SIZE + (3,),
        pooling="avg",
    )
    base.trainable = False

    x = base.output
    x = tfl.Dense(256, activation="relu")(x)
    outputs = tfl.Dense(NUM_CLASSES, activation="softmax")(x)
    my_model = Model(inputs=base.input, outputs=outputs)
    my_model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
    )

my_model.fit(train_ds, epochs=8, validation_data=val_ds, verbose=1)




## === cell 3
test_images = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
df_test = pd.DataFrame(test_images, columns=["path"])


def make_test_dataset(df, batch_size=BATCH_SIZE):
    paths = df["path"].values
    ds = tf.data.Dataset.from_tensor_slices(paths)

    def _load(path):
        img = load_and_preprocess(path)
        return img

    ds = ds.map(_load, num_parallel_calls=tf.data.AUTOTUNE)
    ds = ds.cache()
    ds = ds.batch(batch_size).prefetch(tf.data.AUTOTUNE)
    return ds


test_ds = make_test_dataset(df_test)
pred_test = my_model.predict(test_ds, verbose=1)
pred_test_labels = np.argmax(pred_test, axis=-1)

final_submission = pd.DataFrame(
    {
        "image_id": df_test["path"].apply(lambda x: os.path.basename(x)),
        "label": pred_test_labels,
    }
)

final_csv_path = "submission.csv"
final_submission.to_csv(final_csv_path, index=False)
print(f"Submission written to {final_csv_path}")
