labels_df = pd.read_csv(label_path)

num_classes = len(label_names)

labels_df["label_idx"] = labels_df["breed"].map(label_to_index)

train_image_paths = [
    os.path.join(train_dir, f"{img_id}.jpg") for img_id in labels_df["id"]
