# 1. Kaggle task description

## Task
Given a dataset of images of dogs, predict the breed of each image.

## Metric
Multi Class Log Loss.

## Submission Format
For each image in the test set, you must predict a probability for each of the different breeds. The file should contain a header and have the following format:
```
id,affenpinscher,afghan_hound,..,yorkshire_terrier
000621fb3cbb32d8935728e48679680e,0.0083,0.0,...,0.0083
etc.
```

## Dataset Description
- `train.zip` - the training set, you are provided the breed for these dogs
- `test.zip` - the test set, you must predict the probability of each breed for each image
- `sample_submission.csv` - a sample submission file in the correct format
- `labels.csv` - the breeds for the images in the train set

# 5. Code solution

## === cell 0
import pandas as pd
import torch
from pathlib import Path
from fastai.vision.all import *

set_seed(42, reproducible=True)
torch.backends.cudnn.benchmark = True  # selects fastest convolution algorithms

BASE_INPUT = Path("/kaggle/input/dog-breed-identification")
TRAIN_PATH = BASE_INPUT / "train"
TEST_PATH = BASE_INPUT / "test"
LABEL_CSV = BASE_INPUT / "labels.csv"

bs = 64  # larger batch size reduces number of iterations per epoch
sz = 224
arch = resnet152  # keep the original strong backbone




## === cell 1
labels_df = pd.read_csv(LABEL_CSV)
label_dict = dict(zip(labels_df["id"], labels_df["breed"]))

breed_block = DataBlock(
    blocks=(ImageBlock, CategoryBlock),
    get_items=get_image_files,
    splitter=RandomSplitter(valid_pct=0.2, seed=42),
    get_y=lambda o: label_dict[o.stem],
    item_tfms=Resize(sz),
    batch_tfms=aug_transforms(),
)

dls = breed_block.dataloaders(
    TRAIN_PATH, bs=bs, shuffle=True, num_workers=8, pin_memory=True
)




## === cell 2
learn = vision_learner(
    dls, arch, loss_func=LabelSmoothingCrossEntropy(), metrics=accuracy, cbs=MixUp()
).to_fp16()  # mixed‑precision training speeds up GPU work with negligible FP differences

learn.fine_tune(30, base_lr=1e-3)  # same number of epochs as original




## === cell 3
test_files = get_image_files(TEST_PATH)
test_dl = learn.dls.test_dl(test_files)
logits, _ = learn.tta(dl=test_dl)  # TTA kept as in original logic
probs = torch.nn.functional.softmax(logits, dim=1).cpu().numpy()

sample_sub_path = BASE_INPUT / "sample_submission.csv"
sample_sub = pd.read_csv(sample_sub_path, nrows=0)  # header only
submission_cols = sample_sub.columns.tolist()
breed_cols = [c for c in submission_cols if c != "id"]

pred_df = pd.DataFrame(probs, columns=learn.dls.vocab)
pred_df = pred_df[breed_cols]  # reorder to match sample submission
pred_df.insert(0, "id", [p.stem for p in test_files])

submission_path = "/kaggle/working/submission.csv"
pred_df.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
