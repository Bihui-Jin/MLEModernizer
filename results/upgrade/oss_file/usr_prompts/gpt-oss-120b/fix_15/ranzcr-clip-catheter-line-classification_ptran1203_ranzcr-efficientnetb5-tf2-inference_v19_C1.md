def load_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3, dct_method="INTEGER_FAST")
    img = tf.image.resize(img, (H, W))
    img = tf.cast(img, tf.float32) / 255.0
    img = (img - mean) / std
    return img


def make_dataset(file_paths, batch_size=64, shuffle=False, cache_name=None):
    ds = tf.data.Dataset.from_tensor_slices(file_paths)
    ds = ds.map(
        load_and_preprocess,
        num_parallel_calls=AUTOTUNE,
        deterministic=False,
    )
    if cache_name:
        ds = ds.cache(os.path.join(CACHE_DIR, cache_name))  # disk cache
    else:
        ds = ds.cache()  # in‑memory cache
    if shuffle:
        ds = ds.shuffle(10000)
    ds = ds.batch(batch_size).prefetch(AUTOTUNE)
    return ds


def make_dataset_with_labels(
    file_paths, labels, batch_size=64, shuffle=False, cache_name=None
):
    ds = tf.data.Dataset.from_tensor_slices((file_paths, labels))
    ds = ds.map(
        lambda p, l: (load_and_preprocess(p), l),
        num_parallel_calls=AUTOTUNE,
        deterministic=False,
    )
    if cache_name:
        ds = ds.cache(os.path.join(CACHE_DIR, cache_name))
    else:
        ds = ds.cache()
    if shuffle:
        ds = ds.shuffle(10000)
    ds = ds.batch(batch_size).prefetch(AUTOTUNE)
    return ds

## === cell 3
def get_model(base_model_fn, baseline_weight="imagenet", lr=1e-4):
    base_model = base_model_fn(
        include_top=False,
        input_shape=(H, W, 3),
        pooling="avg",
        weights=baseline_weight,
    )
    x = tf.keras.layers.Dropout(0.3)(base_model.output)
    outputs = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.models.Model(inputs=base_model.input, outputs=outputs)

    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )
    return model
