## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value

00013,0.5
00015,0.5
etc.
```

import os
import pandas as pd
import numpy as np


def locate_path(*relative_parts):
    """
    Resolve a file or directory path that may be referenced relative to the notebook
    or directly under the Kaggle input folder.
    - Strips any leading ".." components.
    - Tries the path relative to the current working directory.
    - Falls back to the absolute Kaggle input location (/kaggle/input/...).
    Raises FileNotFoundError if none of the attempts succeed.
    """
    cleaned_parts = [p for p in relative_parts if p != ".."]
    rel_path = os.path.join(*cleaned_parts)
    if os.path.exists(rel_path):
        return rel_path
    abs_path = os.path.join("/kaggle/input", *cleaned_parts)
    if os.path.exists(abs_path):
        return abs_path
    cwd_path = os.path.abspath(rel_path)
    if os.path.exists(cwd_path):
        return cwd_path
    raise FileNotFoundError(
        f"Unable to locate path: tried {rel_path}, {abs_path}, and {cwd_path}"
    )

    test_zip_path = locate_path(

    with zipfile.ZipFile(test_zip_path, "r") as zf:
        zf.extractall(temp_dir)
    extracted_candidate = os.path.join(temp_dir, "test")
    test_dir = extracted_candidate if os.path.isdir(extracted_candidate) else temp_dir

if not test_ids:
    raise ValueError("No test cases found in the test directory.")
print("Number of test cases:", len(test_ids))
test_ids = [str(t) for t in test_ids]
