def resolve_path(*parts):
    """
    Return a valid path to a data file/folder.
    Checks:
    1. Absolute path.
    2. Relative to current working directory.
    3. Kaggle input location ("/kaggle/input/...").
    """
    p = os.path.abspath(os.path.join(*parts))
    if os.path.exists(p):
        return p
    p = os.path.abspath(os.path.join(".", *parts))
    if os.path.exists(p):
        return p
    kaggle_root = "/kaggle/input"
    for i, part in enumerate(parts):
        if "rsna-miccai" in part:
            sub_path = os.path.join(kaggle_root, *parts[i:])
            if os.path.exists(sub_path):
                return sub_path
    return p

def compute_intensity_based_predictions(case_ids, base_path):
    """
    Simple baseline: read the first FLAIR DICOM slice of each case,
    compute its mean intensity, scale to [0, 1] and use as a probability.
    Missing data fall back to the global training mean.
    """
    means = []
    for cid in case_ids:
        try:
            if dicom is None:
                raise ImportError("pydicom not available")
            flair_dir = os.path.join(base_path, cid, "FLAIR")
            first_file = None
            for entry in os.scandir(flair_dir):
                if entry.is_file() and entry.name.lower().endswith(".dcm"):
                    first_file = entry.path
                    break
            if first_file is None:
                raise FileNotFoundError("No DICOM found")
            ds = dicom.dcmread(first_file)
            img = ds.pixel_array.astype(np.float32)
            means.append(img.mean())
        except Exception:
            means.append(np.nan)

    arr = np.array(means, dtype=np.float32)
    if np.isnan(arr).any():
        if np.isfinite(arr).any():
            valid_mean = np.nanmean(arr)
        else:
            valid_mean = global_mean
        arr = np.where(np.isnan(arr), valid_mean, arr)

    min_val, max_val = arr.min(), arr.max()
    if max_val > min_val:
        scaled = (arr - min_val) / (max_val - min_val)
    else:
        scaled = np.full_like(arr, global_mean, dtype=np.float32)

    scaled = 1.0 - scaled

    return scaled


predictions = compute_intensity_based_predictions(case_ids, test_path)
