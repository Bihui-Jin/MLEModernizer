# 5. Code solution

## === cell 0
import os


import tensorflow as tf
from tensorflow.keras import (
    layers,
    models,
    applications,
    optimizers,
    losses,
    metrics,
    mixed_precision,
)
import numpy as np
import pandas as pd
import random

seed = 42
tf.random.set_seed(seed)
np.random.seed(seed)
random.seed(seed)

mixed_precision.set_global_policy("mixed_float16")
tf.config.optimizer.set_jit(True)

cpu_devices = tf.config.experimental.list_physical_devices("CPU")
cpu_count = len(cpu_devices) if cpu_devices else 1
tf.config.threading.set_intra_op_parallelism_threads(cpu_count)
tf.config.threading.set_inter_op_parallelism_threads(cpu_count)




## === cell 1
BASE_PATH = os.path.abspath("../input/cassava-leaf-disease-classification")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")

train_df = pd.read_csv(TRAIN_CSV)
test_images = sorted(
    [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
)

train_image_paths = [os.path.join(TRAIN_IMG_DIR, img) for img in train_df["image_id"]]
train_labels = train_df["label"].values.astype("int32")




## === cell 2
IMAGE_SIZE = 224
BATCH_SIZE = 64  # reduced to lower per‑step memory pressure
AUTOTUNE = tf.data.experimental.AUTOTUNE
NUM_CLASSES = 5


def decode_image(img_path, label=None):
    img = tf.io.read_file(img_path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.convert_image_dtype(img, tf.float32)
    img = tf.image.resize(img, (IMAGE_SIZE, IMAGE_SIZE))
    if label is None:
        return img
    else:
        return img, label


total_cnt = len(train_image_paths)
indices = np.arange(total_cnt)
np.random.shuffle(indices)
split_idx = int(0.9 * total_cnt)

shuffled_paths = np.array(train_image_paths)[indices]
shuffled_labels = train_labels[indices]

train_paths = shuffled_paths[:split_idx]
train_lbls = shuffled_labels[:split_idx]
val_paths = shuffled_paths[split_idx:]
val_lbls = shuffled_labels[split_idx:]

train_ds = (
    tf.data.Dataset.from_tensor_slices((train_paths, train_lbls))
    .map(decode_image, num_parallel_calls=AUTOTUNE, deterministic=False)
    .cache()  # <-- added cache
    .shuffle(buffer_size=1000, seed=seed, reshuffle_each_iteration=False)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

val_ds = (
    tf.data.Dataset.from_tensor_slices((val_paths, val_lbls))
    .map(decode_image, num_parallel_calls=AUTOTUNE, deterministic=False)
    .cache()  # in‑memory cache is fine for the ~10% validation split.
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)




## === cell 3
base_model = applications.EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(IMAGE_SIZE, IMAGE_SIZE, 3)
)
base_model.trainable = False  # freeze pretrained weights

model = models.Sequential(
    [
        base_model,
        layers.GlobalAveragePooling2D(),
        layers.Dropout(0.2),
        layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

model.compile(
    optimizer=optimizers.Adam(),
    loss=losses.SparseCategoricalCrossentropy(),
    metrics=[metrics.SparseCategoricalAccuracy()],
)

model.summary()




## === cell 4
EPOCHS = 5
history = model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS, verbose=2)




## === cell 5
test_image_paths = [os.path.join(TEST_IMG_DIR, img) for img in test_images]
test_ds = tf.data.Dataset.from_tensor_slices(test_image_paths)
test_ds = (
    test_ds.map(
        lambda x: decode_image(x), num_parallel_calls=AUTOTUNE, deterministic=False
    )
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)




## === cell 6
pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

assert len(pred_labels) == len(test_images), "Prediction length mismatch."




## === cell 7
submission = pd.DataFrame({"image_id": test_images, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
