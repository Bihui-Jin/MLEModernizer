# 5. Code solution

## === cell 0
import os
import json
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler

SEED = 42
np.random.seed(SEED)




## === cell 1
BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")




## === cell 2
map_path = os.path.join(BASE_DIR, "label_num_to_disease_map.json")
if os.path.exists(map_path):
    with open(map_path) as f:
        MAP_CLASSES = json.load(f)
else:
    MAP_CLASSES = {str(i): f"class_{i}" for i in range(5)}
print("Label map:", MAP_CLASSES)




## === cell 3
if not os.path.isdir(TRAIN_DIR):
    raise FileNotFoundError(f"Train image directory not found: {TRAIN_DIR}")
if not os.path.isdir(TEST_DIR):
    raise FileNotFoundError(f"Test image directory not found: {TEST_DIR}")

train_files = os.listdir(TRAIN_DIR)
test_files = os.listdir(TEST_DIR)
print(f"Found {len(train_files)} train images and {len(test_files)} test images.")




## === cell 4
train_csv_path = os.path.join(BASE_DIR, "train.csv")
if not os.path.isfile(train_csv_path):
    raise FileNotFoundError(f"train.csv not found at {train_csv_path}")
train_df = pd.read_csv(train_csv_path)
print("Train dataframe shape:", train_df.shape)




## === cell 5
def image_stats(path, size=(64, 64)):
    """Load image, resize, normalize and return a flattened RGB vector."""
    try:
        img = Image.open(path).convert("RGB")
        img = img.resize(size)
        arr = np.asarray(img, dtype=np.float32) / 255.0  # normalised pixels
        return arr.flatten()  # shape (size[0]*size[1]*3,)
    except Exception:
        return np.zeros(size[0] * size[1] * 3, dtype=np.float32)


X_train = []
y_train = []
for _, row in train_df.iterrows():
    img_path = os.path.join(TRAIN_DIR, row["image_id"])
    X_train.append(image_stats(img_path))
    y_train.append(int(row["label"]))
X_train = np.stack(X_train)
y_train = np.array(y_train)

scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)

print("Training LogisticRegression on", X_train.shape[0], "samples...")
model = LogisticRegression(
    multi_class="multinomial",
    solver="lbfgs",
    C=5.0,
    max_iter=1000,
    n_jobs=-1,
    class_weight="balanced",
    random_state=SEED,
)
model.fit(X_train, y_train)
print("Model training completed.")




## === cell 6
def predict_label(img_path):
    vec = image_stats(img_path)
    vec = scaler.transform(vec.reshape(1, -1))  # apply same scaling
    return int(model.predict(vec)[0])


predictions = []
for img_id in test_files:
    img_path = os.path.join(TEST_DIR, img_id)
    pred = predict_label(img_path)
    predictions.append(pred)

test_df = pd.DataFrame({"image_id": test_files, "label": predictions})

sample_sub_path = os.path.join(BASE_DIR, "sample_submission.csv")
if os.path.isfile(sample_sub_path):
    sample_sub = pd.read_csv(sample_sub_path)
    submission = (
        test_df.set_index("image_id").reindex(sample_sub["image_id"]).reset_index()
    )
else:
    submission = test_df

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
