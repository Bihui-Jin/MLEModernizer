## === cell 0
import os, json, random, tempfile
import numpy as np, pandas as pd
from PIL import Image, ImageOps, ImageFilter
import tensorflow as tf

tf.config.optimizer.set_jit(True)  # XLA
policy = tf.keras.mixed_precision.Policy("mixed_float16")
tf.keras.mixed_precision.set_global_policy(policy)

tf.random.set_seed(42)
random.seed(42)
np.random.seed(42)

tf.config.threading.set_inter_op_parallelism_threads(4)
tf.config.threading.set_intra_op_parallelism_threads(4)

gpus = tf.config.experimental.list_physical_devices("GPU")
for gpu in gpus:
    tf.config.experimental.set_memory_growth(gpu, True)

BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")



## === cell 1
with open(os.path.join(BASE_DIR, "label_num_to_disease_map.json")) as f:
    LABEL_MAP = json.load(f)

print("Label map:", LABEL_MAP)



## === cell 2
train_df = pd.read_csv(os.path.join(BASE_DIR, "train.csv"))
print(f"Training samples: {len(train_df)}")



## === cell 3
IMG_HEIGHT = 400
IMG_WIDTH = 400
BATCH_SIZE = 64  # keep batch size
AUTOTUNE = tf.data.experimental.AUTOTUNE
EPOCHS = 5  # original epochs


def _load_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_HEIGHT, IMG_WIDTH])
    img = tf.cast(img, tf.float16) / 255.0  # keep float16 for mixed precision
    return img


def build_dataset(df, data_dir, shuffle=True):
    """
    Creates a tf.data.Dataset from a dataframe.
    Removed the full in‑memory cache and reduced the shuffle buffer size
    to avoid excessive RAM usage and swapping, which dramatically speeds up
    training while preserving data order randomness.
    """
    image_paths = df["image_id"].apply(lambda x: os.path.join(data_dir, x)).values
    labels = df["label"].values
    ds = tf.data.Dataset.from_tensor_slices((image_paths, labels))

    ds = ds.map(
        lambda p, l: (_load_image(p), l),
        num_parallel_calls=AUTOTUNE,
        deterministic=False,
    )

    if shuffle:
        ds = ds.shuffle(buffer_size=2048, seed=42, reshuffle_each_iteration=True)

    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds


val_frac = 0.1
val_size = int(len(train_df) * val_frac)
train_subset = train_df.iloc[:-val_size]
val_subset = train_df.iloc[-val_size:]

train_ds = build_dataset(train_subset, TRAIN_DIR, shuffle=True)
val_ds = build_dataset(val_subset, TRAIN_DIR, shuffle=False)



## === cell 4
base_model = tf.keras.applications.Xception(
    weights="imagenet", include_top=False, input_shape=(IMG_HEIGHT, IMG_WIDTH, 3)
)
base_model.trainable = False  # freeze backbone initially

inputs = tf.keras.Input(shape=(IMG_HEIGHT, IMG_WIDTH, 3))
x = base_model(inputs, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()



## === cell 5
history = model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS, verbose=2)

base_model.trainable = True
model.compile(
    optimizer=tf.keras.optimizers.Adam(1e-5),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)
fine_tune_epochs = 2
model.fit(train_ds, validation_data=val_ds, epochs=fine_tune_epochs, verbose=2)



## === cell 6
test_filenames = os.listdir(TEST_DIR)
test_df = pd.DataFrame({"image_id": test_filenames})


def build_test_dataset(df, data_dir):
    image_paths = df["image_id"].apply(lambda x: os.path.join(data_dir, x)).values
    ds = tf.data.Dataset.from_tensor_slices(image_paths)
    ds = ds.map(
        lambda p: _load_image(p),
        num_parallel_calls=AUTOTUNE,
        deterministic=False,
    )
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)  # no caching to save RAM
    return ds


test_ds = build_test_dataset(test_df, TEST_DIR)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_df["image_id"], "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
