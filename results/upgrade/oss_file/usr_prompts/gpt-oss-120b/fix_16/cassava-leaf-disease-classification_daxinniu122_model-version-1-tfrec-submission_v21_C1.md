import os
import pandas as pd
import numpy as np
import tensorflow as tf
from tensorflow import keras

tf.random.set_seed(42)
np.random.seed(42)

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"


def build_transfer_model(
    base_model_fn, input_shape=(224, 224, 3), num_classes=5, pretrained=True
):
    """
    Build a transfer‑learning model.
    If pretrained=True, ImageNet weights are loaded (provides strong baseline accuracy).
    """
    weights = "imagenet" if pretrained else None
    base = base_model_fn(weights=weights, include_top=False, input_shape=input_shape)
    base.trainable = False
    inputs = keras.Input(shape=input_shape)
    x = tf.keras.applications.imagenet_utils.preprocess_input(inputs)
    x = base(x, training=False)
    x = keras.layers.GlobalAveragePooling2D()(x)
    outputs = keras.layers.Dense(num_classes, activation="softmax")(x)
    model = keras.Model(inputs, outputs)
    model.compile(
        optimizer=keras.optimizers.Adam(),
        loss=keras.losses.SparseCategoricalCrossentropy(),
        metrics=["accuracy"],
    )
    return model


def build_datasets(train_csv, train_img_dir, batch_size=256):
    """Create shuffled training and validation datasets with efficient caching."""
    df = pd.read_csv(train_csv)
    img_paths = df["image_id"].apply(lambda x: os.path.join(train_img_dir, x)).values
    labels = df["label"].values.astype(np.int32)

    def _load(path, label):
        img = tf.io.read_file(path)
        img = tf.image.decode_jpeg(img, channels=3)
        img = tf.image.resize(img, (224, 224))
        return img, label

    ds = tf.data.Dataset.from_tensor_slices((img_paths, labels))
    ds = ds.shuffle(buffer_size=len(df), seed=42, reshuffle_each_iteration=True)

    val_count = int(0.1 * len(df))
    val_ds = ds.take(val_count)
    train_ds = ds.skip(val_count)

    train_ds = (
        train_ds.map(_load, num_parallel_calls=tf.data.AUTOTUNE, deterministic=False)
        .cache()
        .batch(batch_size)
        .prefetch(tf.data.AUTOTUNE)
    )
    val_ds = (
        val_ds.map(_load, num_parallel_calls=tf.data.AUTOTUNE, deterministic=False)
        .cache()
        .batch(batch_size)
        .prefetch(tf.data.AUTOTUNE)
    )

    return train_ds, val_ds


def load_or_train_model(model_path, base_model_fn, train_ds=None, val_ds=None):
    """Load a saved model if present; otherwise train a new one."""
    if os.path.exists(model_path):
        try:
            return tf.keras.models.load_model(model_path)
        except Exception:
            pass  # fallback to training if loading fails

    if train_ds is None or val_ds is None:
        raise ValueError(
            "Training datasets must be supplied when model file is missing."
        )
    model = build_transfer_model(base_model_fn, pretrained=True)
    model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=1)
    return model


train_csv = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_img_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images/"
sample_sub_path = (
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)

train_ds, val_ds = build_datasets(train_csv, train_img_dir, batch_size=256)

model_path = "/kaggle/working/ResNet50_f.h5"
model = load_or_train_model(
    model_path, tf.keras.applications.ResNet50, train_ds, val_ds
)

sample_sub = pd.read_csv(sample_sub_path)
test_images_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"

test_paths = [os.path.join(test_images_dir, fname) for fname in sample_sub["image_id"]]


def _load_test(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, (224, 224))
    return img


test_ds = (
    tf.data.Dataset.from_tensor_slices(test_paths)
    .map(_load_test, num_parallel_calls=tf.data.AUTOTUNE, deterministic=False)
    .cache()
    .batch(64)
    .prefetch(tf.data.AUTOTUNE)
)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1).astype(int)

submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
