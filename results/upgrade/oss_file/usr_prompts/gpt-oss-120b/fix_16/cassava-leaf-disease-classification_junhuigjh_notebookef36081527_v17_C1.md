# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from joblib import Parallel, delayed  # parallel image preprocessing

SEED = 42
np.random.seed(SEED)

DATA_ROOT = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
TRAIN_IMG_DIR = os.path.join(DATA_ROOT, "train_images")
TEST_IMG_DIR = os.path.join(DATA_ROOT, "test_images")
SAMPLE_SUB = os.path.join(DATA_ROOT, "sample_submission.csv")



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(str)

label_to_index = {str(i): i for i in range(5)}
train_df["label_idx"] = train_df["label"].map(label_to_index)

train_split, val_split = train_test_split(
    train_df,
    test_size=0.1,
    random_state=SEED,
    stratify=train_df["label_idx"],
)

IMG_SIZE = 96


def load_and_preprocess(img_path):
    """Load an image, resize, flatten, and append simple colour statistics."""
    img = Image.open(img_path).convert("RGB")
    img = img.resize((IMG_SIZE, IMG_SIZE))
    arr = np.asarray(img, dtype=np.uint8)  # shape (H,W,3)

    flat = arr.flatten().astype(np.float32)

    means = arr.mean(axis=(0, 1))
    stds = arr.std(axis=(0, 1))
    stats = np.concatenate([means, stds]).astype(np.float32)

    return np.concatenate([flat, stats])


def build_feature_matrix(df, img_dir):
    """Create feature matrix X and label vector y using parallel preprocessing."""
    paths = df["image_id"].apply(lambda fname: os.path.join(img_dir, fname)).values
    features_list = Parallel(n_jobs=-1, backend="loky")(
        delayed(load_and_preprocess)(p) for p in paths
    )
    features = np.stack(features_list)
    labels = df["label_idx"].values.astype(np.int32)
    return features, labels


X_train, y_train = build_feature_matrix(train_split, TRAIN_IMG_DIR)

rf_clf = RandomForestClassifier(
    n_estimators=800,
    max_depth=None,
    max_features=None,
    class_weight="balanced",
    random_state=SEED,
    n_jobs=-1,
)
rf_clf.fit(X_train, y_train)

X_val, y_val = build_feature_matrix(val_split, TRAIN_IMG_DIR)
val_preds = rf_clf.predict(X_val)
val_acc = accuracy_score(y_val, val_preds)
print(f"Validation accuracy (enhanced): {val_acc:.4f}")



## === cell 2
test_filenames = sorted(
    [
        f
        for f in os.listdir(TEST_IMG_DIR)
        if f.lower().endswith((".png", ".jpg", ".jpeg"))
    ]
)

test_paths = [os.path.join(TEST_IMG_DIR, fname) for fname in test_filenames]
test_features_list = Parallel(n_jobs=-1, backend="loky")(
    delayed(load_and_preprocess)(p) for p in test_paths
)
X_test = np.stack(test_features_list)

test_pred_idxs = rf_clf.predict(X_test)



## === cell 3
submission = pd.DataFrame(
    {
        "image_id": test_filenames,
        "label": test_pred_idxs,
    }
)

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
