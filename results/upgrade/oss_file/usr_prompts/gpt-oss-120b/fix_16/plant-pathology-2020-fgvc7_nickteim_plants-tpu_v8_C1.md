## === cell 0
import os
import numpy as np
import pandas as pd

BASE_PATH = "/kaggle/input/plant-pathology-2020-fgvc7/"

train = pd.read_csv(os.path.join(BASE_PATH, "train.csv"))
test = pd.read_csv(os.path.join(BASE_PATH, "test.csv"))
sub = pd.read_csv(os.path.join(BASE_PATH, "sample_submission.csv"))

print("train shape:", train.shape)
print("test shape:", test.shape)




## === cell 1
train_labels = train.loc[:, "healthy":].values.astype(
    np.float32
)  # shape (n_samples, 4)
class_means = train_labels.mean(axis=0)  # shape (4,)

num_test = test.shape[0]
probs = np.tile(class_means, (num_test, 1))

print("Baseline prediction shape:", probs.shape)




## === cell 2
sub.loc[:, "healthy":] = probs
output_path = "submission.csv"
sub.to_csv(output_path, index=False)

print(f"Submission saved to {output_path}")
print(sub.head())
