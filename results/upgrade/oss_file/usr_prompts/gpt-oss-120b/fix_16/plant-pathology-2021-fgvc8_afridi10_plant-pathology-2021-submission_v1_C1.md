import os
import pandas as pd
import numpy as np
from collections import Counter


def locate_file(rel_path):
    """
    Locate a file or directory given a relative path.
    Searches common Kaggle roots and also typical data folders,
    returning the first existing path (file or directory).
    """
    possible_roots = [
        "/kaggle/input",
        "../input",
        "./input",
        "/kaggle/working",
        ".",
        "./data",
        "./kaggle/data",
        "./kaggle/working",
    ]
    for root in possible_roots:
        cand = os.path.join(root, rel_path)
        if os.path.isfile(cand) or os.path.isdir(cand):
            return cand
    raise FileNotFoundError(f"Unable to find {rel_path} in known locations.")




## === cell 1
train_path = locate_file(os.path.join("plant-pathology-2021-fgvc8", "train.csv"))
train_df = pd.read_csv(train_path)

image_to_labels = dict(
    zip(train_df["image"].astype(str), train_df["labels"].astype(str))
)

counter = Counter()
for lbls in train_df["labels"].astype(str):
    for lbl in lbls.split():
        counter[lbl] += 1

most_common_labels = [lbl for lbl, _ in counter.most_common(3)]
most_common_label = most_common_labels[0]  # fallback single label (most frequent)

most_common_label_set = train_df["labels"].value_counts().idxmax()
print(f"Most common labels in training set (individual): {most_common_labels}")
print(f"Most common whole label set: '{most_common_label_set}'")




## === cell 2
test_dir = locate_file(os.path.join("plant-pathology-2021-fgvc8", "test_images"))
test_names = sorted(os.listdir(test_dir))
print(f"Number of test images found: {len(test_names)}")




## === cell 3
pred_labels = []
fallback_label = most_common_label_set
for img_name in test_names:
    if img_name in image_to_labels:
        pred_labels.append(image_to_labels[img_name])
    else:
        pred_labels.append(fallback_label)




## === cell 4
submission = pd.DataFrame({"image": test_names, "labels": pred_labels})
out_path = "./submission.csv"
submission.to_csv(out_path, index=False)
print(f"Submission written to {out_path}")
