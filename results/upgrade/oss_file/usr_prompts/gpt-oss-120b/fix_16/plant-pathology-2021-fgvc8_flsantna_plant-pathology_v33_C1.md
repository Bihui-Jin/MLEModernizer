## === cell 0
import os
import pandas as pd
import numpy as np

output_dir = "./"
test_dir = "../input/plant-pathology-2021-fgvc8/test_images/"

train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
train_df = pd.read_csv(train_path)

all_labels = train_df["labels"].str.split(" ")

label_counts = {}
for lbls in all_labels:
    for lbl in lbls:
        label_counts[lbl] = label_counts.get(lbl, 0) + 1

avg_labels_per_image = int(round(train_df["labels"].str.split(" ").apply(len).mean()))
if avg_labels_per_image < 1:
    avg_labels_per_image = 1

sorted_labels = sorted(label_counts.items(), key=lambda x: x[1], reverse=True)
baseline_labels = [lbl for lbl, _ in sorted_labels[:avg_labels_per_image]]

if not baseline_labels:
    most_common = max(label_counts.items(), key=lambda x: x[1])[0]
    baseline_labels = [most_common]

print(f"Baseline will predict {len(baseline_labels)} label(s): {baseline_labels}")

if __name__ == "__main__":
