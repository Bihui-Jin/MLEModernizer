## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.model_selection import train_test_split
from sklearn.multiclass import OneVsRestClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import roc_auc_score
import concurrent.futures

np.random.seed(42)

DEFAULT_INPUT_DIR = "/kaggle/input"
if not os.path.isdir(DEFAULT_INPUT_DIR):
    DEFAULT_INPUT_DIR = "./input"
WORK_DIR = os.path.join(DEFAULT_INPUT_DIR, "ranzcr-clip-catheter-line-classification")
print("WORK_DIR set to:", WORK_DIR)

TRAIN_CSV_PATH = os.path.join(WORK_DIR, "train.csv")
SAMPLE_SUB_PATH = os.path.join(WORK_DIR, "sample_submission.csv")
TRAIN_IMG_DIR = os.path.join(WORK_DIR, "train")
TEST_IMG_DIR = os.path.join(WORK_DIR, "test")

IMAGE_SIZE = (96, 96)  # width, height

train_df = pd.read_csv(TRAIN_CSV_PATH)
label_cols = [
    "ETT - Abnormal",
    "ETT - Borderline",
    "ETT - Normal",
    "NGT - Abnormal",
    "NGT - Borderline",
    "NGT - Incompletely Imaged",
    "NGT - Normal",
    "CVC - Abnormal",
    "CVC - Borderline",
    "CVC - Normal",
    "Swan Ganz Catheter Present",
]
y = train_df[label_cols].values.astype(np.float32)


def build_image_list(df, img_dir):
    """Create a list of file paths matching the StudyInstanceUIDs."""
    paths = []
    missing = 0
    for uid in df["StudyInstanceUID"]:
        fname = f"{uid}.jpg"
        fpath = os.path.join(img_dir, fname)
        if os.path.isfile(fpath):
            paths.append(fpath)
        else:
            missing += 1
            paths.append(None)
    if missing:
        print(f"Warning: {missing} images not found in {img_dir}")
    return paths


train_images = build_image_list(train_df, TRAIN_IMG_DIR)


def _process_path(p):
    """Load a single image, resize, grayscale, normalize and flatten."""
    if p is None:
        return None
    try:
        with Image.open(p) as img:
            img = img.convert("L").resize(IMAGE_SIZE, Image.BILINEAR)
            arr = np.asarray(img, dtype=np.float32) / 255.0
        return arr.ravel()
    except Exception:
        return None


def load_features(image_paths, size=IMAGE_SIZE):
    """
    Load images in parallel using a limited thread pool,
    resize, grayscale, normalize and flatten. Returns a NumPy array
    of shape (n_images, size[0]*size[1]) with dtype float32.
    """
    n = len(image_paths)
    flat_dim = size[0] * size[1]
    feats = np.zeros((n, flat_dim), dtype=np.float32)

    max_workers = max(1, (os.cpu_count() or 1) // 2)
    chunk_size = 1024  # reduces thread‑dispatch overhead

    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        for idx, result in enumerate(
            executor.map(_process_path, image_paths, chunksize=chunk_size)
        ):
            if result is not None:
                feats[idx] = result

    return feats


print("Loading training images ...")
X = load_features(train_images)

X_tr, X_val, y_tr, y_val = train_test_split(
    X, y, test_size=0.15, random_state=42, shuffle=True
)

clf = OneVsRestClassifier(
    LogisticRegression(
        C=4.0,
        max_iter=500,
        n_jobs=1,  # single core per binary classifier
        solver="saga",
        penalty="l2",
        class_weight="balanced",
        random_state=42,
    ),
    n_jobs=-1,  # parallelize the OneVsRest fitting across labels
)


print("Training model ...")
clf.fit(X_tr, y_tr)




## === cell 1
val_pred = clf.predict_proba(X_val)
auc_scores = [roc_auc_score(y_val[:, i], val_pred[:, i]) for i in range(y_val.shape[1])]
print("Validation AUC per label:", np.round(auc_scores, 4))
print("Mean Validation AUC:", np.mean(auc_scores))




## === cell 2
sample_sub = pd.read_csv(SAMPLE_SUB_PATH)
test_uids = sample_sub["StudyInstanceUID"].tolist()
test_images = [os.path.join(TEST_IMG_DIR, f"{uid}.jpg") for uid in test_uids]

print("Loading test images ...")
X_test = load_features(test_images)

print("Generating predictions for test set ...")
test_pred = clf.predict_proba(X_test)  # shape (n_test, n_labels)

submission_cols = sample_sub.columns.tolist()
present_labels = [col for col in label_cols if col in submission_cols]

ss = sample_sub.copy()
ss[present_labels] = test_pred[:, : len(present_labels)]

output_path = "submission.csv"
ss.to_csv(output_path, index=False)
print(f"Submission file written to {output_path}")
