# 1. Kaggle task description

## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

## Submission Format
For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value
00001,0.5
00013,0.5
00015,0.5
etc.
```

## Dataset
- **train/** - folder containing the training files, with each top-level folder representing a subject. **NOTE:** There are some unexpected issues with the following three cases in the training dataset, participants can exclude the cases during training: `[00109, 00123, 00709]`. We have checked and confirmed that the testing dataset is free from such issues.
- **train_labels.csv** - file containing the target `MGMT_value` for each subject in the training data (e.g. the presence of MGMT promoter methylation)
- **test/** - the test files, which use the same structure as `train/`; your task is to predict the `MGMT_value` for each subject in the test data. **NOTE**: the total size of the rerun test set (Public and Private) is ~5x the size of the Public test set
- **sample_submission.csv** - a sample submission file in the correct format

import os

## === cell 1
def locate_test_dir():
    """
    Return an absolute path to the test folder.
    Searches common Kaggle locations and falls back to a generic relative path.
    Added extra fallback under ./data/... for local environments.
    """
    possible_paths = [
        "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/test",
        "./input/rsna-miccai-brain-tumor-radiogenomic-classification/test",
        "../input/rsna-miccai-brain-tumor-radiogenomic-classification/test",
        "./rsna-miccai-brain-tumor-radiogenomic-classification/test",
        "../rsna-miccai-brain-tumor-radiogenomic-classification/test",
        "./test",  # generic fallback for local runs
        "./data/rsna-miccai-brain-tumor-radiogenomic-classification/test",  # added fallback
    ]
    for p in possible_paths:
        if os.path.isdir(p):
            return os.path.abspath(p)
    fallback = os.path.abspath("test")
    if os.path.isdir(fallback):
        return fallback
    raise FileNotFoundError(
        "Test directory not found in any of the expected locations."
    )


try:
    test_path = locate_test_dir()
except FileNotFoundError:
    test_path = None

def get_case_ids(path_test=None):
    """
    Return a sorted list of BraTS21ID strings (folder names) in the test set.
    If the test directory is missing or empty, fall back to the IDs listed in the
    sample_submission file.
    """
    if path_test and os.path.isdir(path_test):
        case_dirs = [d for d in os.scandir(path_test) if d.is_dir()]
        if case_dirs:
            case_dirs = sorted(case_dirs, key=lambda x: x.path)
            return [os.path.basename(d.path) for d in case_dirs]

    possible_samples = [
        "./sample_submission.csv",
        "./input/sample_submission.csv",
        "../input/sample_submission.csv",
        "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/sample_submission.csv",
    ]
    for p in possible_samples:
        if os.path.isfile(p):
            sample_df = pd.read_csv(p, usecols=["BraTS21ID"])
            return sorted(sample_df["BraTS21ID"].astype(str).unique())
    raise FileNotFoundError("No test IDs found and sample_submission.csv is missing.")


case_ids = get_case_ids(test_path)

def locate_train_labels():
    """
    Find the training labels CSV in common locations.
    """
    possible_paths = [
        "./train_labels.csv",
        "./input/train_labels.csv",
        "../input/train_labels.csv",
        "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/train_labels.csv",
    ]
    for p in possible_paths:
        if os.path.isfile(p):
            return os.path.abspath(p)
    raise FileNotFoundError("train_labels.csv not found.")


train_labels_path = locate_train_labels()
train_df = pd.read_csv(train_labels_path)

problematic_ids = {"00109", "00123", "00709"}
clean_train_df = train_df[~train_df["BraTS21ID"].astype(str).isin(problematic_ids)]

baseline_prob = clean_train_df["MGMT_value"].mean()
baseline_prob = np.clip(baseline_prob, 0.0, 1.0)

rng = np.random.default_rng(seed=42)
noise = rng.uniform(-1e-6, 1e-6, size=len(case_ids)).astype(np.float32)
predictions = np.full(len(case_ids), baseline_prob, dtype=np.float32) + noise
predictions = np.clip(predictions, 0.0, 1.0)




## === cell 4
def get_output_path(filename="submission.csv"):
    """
    Determine a safe location to write the submission file.
    Prefer /kaggle/working (Kaggle kernels) but fall back to the current
    working directory if that location is unavailable or not writable.
    """
    kaggle_work = "/kaggle/working"
    if os.path.isdir(kaggle_work) and os.access(kaggle_work, os.W_OK):
        out_dir = kaggle_work
    else:
        out_dir = os.getcwd()
    os.makedirs(out_dir, exist_ok=True)
    return os.path.join(out_dir, filename)


output_path = get_output_path()

assert len(case_ids) == len(predictions), "Mismatch between IDs and predictions"

sub_df = pd.DataFrame({"BraTS21ID": case_ids, "MGMT_value": predictions})
sub_df.to_csv(output_path, index=False)
print(f"Submission written to {output_path}")
