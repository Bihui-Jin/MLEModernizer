## === cell 0
import os
import numpy as np
import pandas as pd
import cv2
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import roc_auc_score

print("Input directory contents:", os.listdir("../input"))



## === cell 1
train_csv = (
    pd.read_csv("../input/train.csv")
    .sample(frac=1, random_state=42)
    .reset_index(drop=True)
)
image_ids = train_csv["id"].values.tolist()
labels = train_csv["has_cactus"].values.tolist()

train_ids, val_ids, train_y, val_y = train_test_split(
    image_ids, labels, test_size=0.1, random_state=42, stratify=labels
)




## === cell 2
def load_image(path):
    """Read an image, resize to 32x32, normalize to [0,1].
    Returns a 32x32x3 array; if reading fails, returns an array of zeros."""
    img = cv2.imread(path, cv2.IMREAD_COLOR)
    if img is None:
        return np.zeros((32, 32, 3), dtype=np.float32)
    img = cv2.resize(img, (32, 32))
    return img.astype(np.float32) / 255.0


def get_image_from_train(imname):
    return load_image(os.path.join("../input/train/train", imname))


def get_image_from_test(imname):
    return load_image(os.path.join("../input/test/test", imname))




## === cell 3
train_imgs = [np.reshape(get_image_from_train(i), -1) for i in train_ids]
val_imgs = [np.reshape(get_image_from_train(i), -1) for i in val_ids]

X_train = np.array(train_imgs)
X_val = np.array(val_imgs)
y_train = np.array(train_y)
y_val = np.array(val_y)



## === cell 4
rf = RandomForestClassifier(
    n_estimators=300,
    max_depth=None,
    min_samples_split=2,
    n_jobs=-1,
    random_state=42,
    class_weight="balanced",
)
rf.fit(X_train, y_train)

val_pred = rf.predict_proba(X_val)[:, 1]
auc = roc_auc_score(y_val, val_pred)
print("Validation AUC:", auc)



## === cell 5
test_list = os.listdir("../input/test/test")
test_imgs = [np.reshape(get_image_from_test(name), -1) for name in test_list]
X_test = np.array(test_imgs)

test_pred = rf.predict_proba(X_test)[:, 1]



## === cell 6
submission = pd.DataFrame({"id": test_list, "has_cactus": test_pred})
submission.to_csv("result.csv", index=False)
print("Submission saved to result.csv with", len(submission), "rows.")
