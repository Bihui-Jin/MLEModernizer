# 1. Kaggle task description

## Task
Create a classifier to predict whether an image contains a cactus.

## Metric
Area under the ROC curve.

## Submission Format
For each ID in the test set, you must predict a probability for the `has_cactus` variable. The file should contain a header and have the following format:

```
id,has_cactus
000940378805c44108d287872b2f04ce.jpg,0.5
0017242f54ececa4512b4d7937d1e21e.jpg,0.5
001ee6d8564003107853118ab87df407.jpg,0.5
etc.
```

## Dataset
This dataset contains a large number of 32 x 32 thumbnail images containing aerial photos of a cactus. The file name of an image corresponds to its `id`.

- **train/** - the training set images
- **test/** - the test set images (you must predict the labels of these)
- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

## === cell 0
import numpy as np, pandas as pd, zipfile
from pathlib import Path
from fastai.vision.all import *
import torch

BASE = Path("../input/aerial-cactus-identification")
TMP = Path("../kaggle/temp")
TMP.mkdir(parents=True, exist_ok=True)

with zipfile.ZipFile(BASE / "train.zip", "r") as z:
    z.extractall(TMP)
with zipfile.ZipFile(BASE / "test.zip", "r") as z:
    z.extractall(TMP)

train_df = pd.read_csv(BASE / "train.csv")
test_df = pd.read_csv(BASE / "sample_submission.csv")


## === cell 1
trfm = aug_transforms(
    size=224,
    do_flip=True,
    flip_vert=True,
    max_rotate=10.0,
    max_zoom=1.1,
    max_lighting=0.2,
    max_warp=0.2,
    p_affine=0.75,
    p_lighting=0.75,
)

dls = ImageDataLoaders.from_df(
    train_df,
    path=TMP,
    fn_col="id",  # column with image file names
    label_col="has_cactus",  # column with 0/1 labels
    valid_pct=0.2,
    seed=42,
    item_tfms=Resize(224),
    batch_tfms=trfm,
    bs=64,
)


## === cell 2
learn = cnn_learner(
    dls, resnet34, loss_func=CrossEntropyLossFlat(), metrics=[error_rate, accuracy]
)


## === cell 3
learn.fine_tune(5)


## === cell 4
pred_probs = []
ids = []
for img_path in get_image_files(TMP / "test"):
    ids.append(img_path.name)  # filename as required by submission
    prob = learn.predict(img_path)[2][1].item()  # probability of class 1 (has_cactus)
    pred_probs.append(prob)


## === cell 5
submission = pd.DataFrame({"id": ids, "has_cactus": pred_probs})
submission_path = Path("/kaggle/working/submission.csv")
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
