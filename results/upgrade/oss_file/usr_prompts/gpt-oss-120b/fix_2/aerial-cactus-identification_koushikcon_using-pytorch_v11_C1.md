import os
import numpy as np
import pandas as pd
import cv2
import torch
import torchvision
import torch.nn as nn
import torchvision.transforms as transforms
from PIL import ImageFile

ImageFile.LOAD_TRUNCATED_IMAGES = True

use_cuda = torch.cuda.is_available()
device = torch.device("cuda" if use_cuda else "cpu")
if not use_cuda:
    print("No GPU found. Running on CPU.")



## === cell 1
train_labels = pd.read_csv("../input/aerial-cactus-identification/train.csv")
test_frame = pd.read_csv("../input/aerial-cactus-identification/sample_submission.csv")



## === cell 2
model = torchvision.models.vgg16(pretrained=True)
for param in model.features.parameters():
    param.requires_grad = False

model.classifier = nn.Sequential(
    nn.Linear(25088, 1), nn.Sigmoid()  # VGG‑16 feature size -> 1
)
model = model.to(device)
model.eval()  # set to evaluation mode

test_transform = transforms.Compose(
    [
        transforms.ToPILImage(),
        transforms.Resize(224),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)




## === cell 3
def predict(image_path: str):
    """
    Reads an image, applies the test_transform and returns the model's probability.
    """
    img = cv2.imread(image_path)
    if img is None:
        return 0.5
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = test_transform(img)  # shape: C x H x W
    img = img.unsqueeze(0).to(device)  # add batch dimension
    with torch.no_grad():
        prob = model(img)  # shape: (1, 1)
    return prob.item()  # return scalar probability




## === cell 4
submission = pd.DataFrame(columns=["id", "has_cactus"])
for idx, row in test_frame.iterrows():
    img_path = os.path.join(
        "../input/aerial-cactus-identification/test/test", row["id"]
    )
    prob = predict(img_path)
    submission = submission.append(
        {"id": row["id"], "has_cactus": prob}, ignore_index=True
    )

submission = submission[["id", "has_cactus"]]

def create_download_link(df, title="Download CSV file", filename="submission.csv"):
