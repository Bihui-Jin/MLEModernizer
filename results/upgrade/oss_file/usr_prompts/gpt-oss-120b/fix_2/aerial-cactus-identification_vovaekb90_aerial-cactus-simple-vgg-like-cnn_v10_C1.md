# 1. Kaggle task description

## Task
Create a classifier to predict whether an image contains a cactus.

## Metric
Area under the ROC curve.

## Submission Format
For each ID in the test set, you must predict a probability for the `has_cactus` variable. The file should contain a header and have the following format:

```
id,has_cactus
000940378805c44108d287872b2f04ce.jpg,0.5
0017242f54ececa4512b4d7937d1e21e.jpg,0.5
001ee6d8564003107853118ab87df407.jpg,0.5
etc.
```

## Dataset
This dataset contains a large number of 32 x 32 thumbnail images containing aerial photos of a cactus. The file name of an image corresponds to its `id`.

- **train/** - the training set images
- **test/** - the test set images (you must predict the labels of these)
- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

def load_images(glob_path):
    imgs, names = [], []
    for p in glob.glob(glob_path):
        names.append(os.path.basename(p))
        img = cv2.imread(p, cv2.IMREAD_COLOR)  # 32x32x3
        imgs.append(img)
    return np.array(imgs), np.array(names)


train_imgs, train_names = load_images(
    "../input/aerial-cactus-identification/train/*.jpg"
)
print("train images shape:", train_imgs.shape)

test_imgs, test_names = load_images("../input/aerial-cactus-identification/test/*.jpg")
print("test images shape:", test_imgs.shape)




## === cell 2
train_meta = pd.read_csv("../input/aerial-cactus-identification/train.csv")
lookup = dict(zip(train_meta["id"], train_meta["has_cactus"]))

train_labels = np.array([lookup[name] for name in train_names], dtype=np.float32)
print("train labels distribution:", np.bincount(train_labels.astype(int)))




## === cell 3
train_imgs = train_imgs.astype(np.float32) / 255.0
test_imgs = test_imgs.astype(np.float32) / 255.0

X_tr, X_val, y_tr, y_val = train_test_split(
    train_imgs, train_labels, test_size=0.2, random_state=7, stratify=train_labels
)
print("split shapes:", X_tr.shape, X_val.shape)




## === cell 4
datagen = ImageDataGenerator(
    rotation_range=60, zoom_range=0.2, horizontal_flip=True, vertical_flip=True
)
datagen.fit(X_tr)




## === cell 5
input_shape = X_tr.shape[1:]  # (32, 32, 3)
model = Sequential(
    [
        Conv2D(32, (3, 3), activation="relu", input_shape=input_shape),
        MaxPooling2D(2, 2),
        Conv2D(64, (3, 3), activation="relu"),
        MaxPooling2D(2, 2),
        Flatten(),
        Dropout(0.5),
        Dense(1, activation="sigmoid"),
    ]
)
model.compile(optimizer="nadam", loss="binary_crossentropy", metrics=["accuracy"])
model.summary()




## === cell 6
