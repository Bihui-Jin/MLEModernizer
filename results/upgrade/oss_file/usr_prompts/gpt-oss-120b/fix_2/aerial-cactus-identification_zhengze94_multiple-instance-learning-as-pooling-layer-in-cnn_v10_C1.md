## Task
Create a classifier to predict whether an image contains a cactus.

## Metric
Area under the ROC curve.

## Submission Format
For each ID in the test set, you must predict a probability for the `has_cactus` variable. The file should contain a header and have the following format:

```
id,has_cactus

0017242f54ececa4512b4d7937d1e21e.jpg,0.5
001ee6d8564003107853118ab87df407.jpg,0.5
etc.
```

## Dataset
This dataset contains a large number of 32 x 32 thumbnail images containing aerial photos of a cactus. The file name of an image corresponds to its `id`.

- **train/** - the training set images
- **test/** - the test set images (you must predict the labels of these)
- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

## === cell 0
import os, numpy as np, pandas as pd, cv2

print("Input directory contents:", os.listdir("../input"))



## === cell 1
from tqdm import tqdm
from PIL import Image


def load_image(path):
    """Load an image, convert to RGB, resize to 32x32 if needed, return as float32 array normalized to [0,1]."""
    img = Image.open(path).convert("RGB")
    img = img.resize((32, 32))
    return np.asarray(img, dtype=np.float32) / 255.0


def load_images_from_folder(folder):
    """Return a dict {filename: image_array} for all files in the folder."""
    images = {}
    for fname in os.listdir(folder):
        fpath = os.path.join(folder, fname)
        try:
            images[fname] = load_image(fpath)
        except Exception as e:
            print(f"Warning: could not load {fpath}: {e}")
    return images




## === cell 2
train_img_folder = "../input/train/train/"
test_img_folder = "../input/test/test/"

train_imgs = load_images_from_folder(train_img_folder)
test_imgs = load_images_from_folder(test_img_folder)

print(f"Loaded {len(train_imgs)} training images, {len(test_imgs)} test images")



## === cell 3
train_csv = pd.read_csv("../input/train.csv")
X_list = []
y_list = []

for _, row in train_csv.iterrows():
    img_id = row["id"]
    if img_id in train_imgs:
        X_list.append(train_imgs[img_id])
        y_list.append(int(row["has_cactus"]))
    else:
        pass

X = np.stack(X_list)  # shape (n_samples, 32, 32, 3)
y = np.array(y_list)

submission_df = pd.read_csv("../input/sample_submission.csv")
X_test = np.stack([test_imgs[fid] for fid in submission_df["id"] if fid in test_imgs])

print("Training data shape:", X.shape, "Labels shape:", y.shape)
print("Test data shape:", X_test.shape)



## === cell 4
from sklearn.model_selection import train_test_split
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import roc_auc_score

X_flat = X.reshape((X.shape[0], -1))
X_test_flat = X_test.reshape((X_test.shape[0], -1))

X_tr, X_val, y_tr, y_val = train_test_split(
    X_flat, y, test_size=0.2, random_state=42, stratify=y
)

model = GradientBoostingClassifier(
    n_estimators=200, learning_rate=0.1, max_depth=3, random_state=42
)
model.fit(X_tr, y_tr)

val_pred = model.predict_proba(X_val)[:, 1]
val_auc = roc_auc_score(y_val, val_pred)
print(f"Validation AUC: {val_auc:.4f}")



## === cell 5
test_pred = model.predict_proba(X_test_flat)[:, 1]

submission_df["has_cactus"] = test_pred
submission_path = "sample_submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
