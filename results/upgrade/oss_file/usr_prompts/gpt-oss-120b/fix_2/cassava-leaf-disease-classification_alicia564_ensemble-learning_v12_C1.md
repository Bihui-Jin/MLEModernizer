    weights="imagenet", include_top=False, input_shape=(224, 224, 3)

model = Model(inputs=base_model.input, outputs=output)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

## === cell 4
img_size = (224, 224)
pred_labels = []
img_ids = []

for fname in sorted(os.listdir(test_images_dir)):
    if not fname.lower().endswith(".jpg"):
        continue
    img_path = os.path.join(test_images_dir, fname)
    img = load_img(img_path, target_size=img_size)
    arr = img_to_array(img) / 255.0
    arr = preprocess_input(arr)
    arr = np.expand_dims(arr, axis=0)

    probs = model.predict(arr, verbose=0)
    pred_class = int(np.argmax(probs, axis=1)[0])
    pred_labels.append(pred_class)
    img_ids.append(fname)

submission = pd.DataFrame({"image_id": img_ids, "label": pred_labels})

submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file created: {submission_path}")
