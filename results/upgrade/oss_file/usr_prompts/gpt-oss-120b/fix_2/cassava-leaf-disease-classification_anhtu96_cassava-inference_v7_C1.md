ROOT_DIR = "/kaggle/input/"
AUTOTUNE = tf.data.experimental.AUTOTUNE



## === cell 2
train_csv_path = os.path.join(
    ROOT_DIR, "cassava-leaf-disease-classification", "train.csv"
)
train_df = pd.read_csv(train_csv_path)

train_image_dir = os.path.join(
    ROOT_DIR, "cassava-leaf-disease-classification", "train_images"
)
train_paths = (
    train_df["image_id"].apply(lambda x: os.path.join(train_image_dir, x)).values
)
train_labels = train_df["label"].values.astype(np.int32)


def _preprocess(path, label):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, (224, 224))
    image = tf.keras.applications.resnet50.preprocess_input(tf.cast(image, tf.float32))
    return image, label


full_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
full_ds = full_ds.map(_preprocess, num_parallel_calls=AUTOTUNE)

total_cnt = len(train_paths)
train_cnt = int(0.9 * total_cnt)

train_ds = full_ds.take(train_cnt).shuffle(1000).batch(32).prefetch(AUTOTUNE)
val_ds = full_ds.skip(train_cnt).batch(32).prefetch(AUTOTUNE)



## === cell 3
base_model = tf.keras.applications.ResNet50(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3)
)
base_model.trainable = False  # freeze base

inputs = tf.keras.Input(shape=(224, 224, 3))
x = base_model(inputs, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)

model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=2)




## === cell 4
def _parse_function(example, feature_description):
    parsed_example = tf.io.parse_single_example(example, feature_description)
    image = tf.io.decode_jpeg(parsed_example["image"], channels=3)
    image = tf.cast(image, tf.float32)
    image = tf.image.resize(image, (224, 224))
    image = tf.keras.applications.resnet50.preprocess_input(image)
    return image, parsed_example["image_name"]


def load_data(path, batch_size=32, train=True):
    filenames = tf.io.gfile.glob(path)
    dataset = tf.data.TFRecordDataset(filenames, num_parallel_reads=AUTOTUNE)
    if train:
        feature_description = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
            "target": tf.io.FixedLenFeature([], tf.int64),
        }
    else:
        feature_description = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
        }
    parsed_dataset = dataset.map(
        lambda x: _parse_function(x, feature_description), num_parallel_calls=AUTOTUNE
    )
    return parsed_dataset.batch(batch_size).prefetch(AUTOTUNE)




## === cell 5
test_set = load_data(
    ROOT_DIR + "cassava-leaf-disease-classification/test_tfrecords/ld_test*.tfrec",
    batch_size=1,
    train=False,
)

test_IDs = []
predictions = []

for images, names in test_set:
    img_id = names.numpy()[0].decode()
    probs = model.predict(images, verbose=0)
    pred_label = int(np.argmax(probs, axis=1)[0])
    test_IDs.append(img_id)
    predictions.append(pred_label)

assert len(test_IDs) == len(predictions), "Mismatch between IDs and predictions"

submission_df = pd.DataFrame({"image_id": test_IDs, "label": predictions})
submission_df.to_csv("submission.csv", index=False)
