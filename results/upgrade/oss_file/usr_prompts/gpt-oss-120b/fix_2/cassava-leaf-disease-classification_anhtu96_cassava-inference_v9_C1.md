def _parse_function(example, feature_description):
    """Parse a single TFRecord example."""
    parsed_example = tf.io.parse_single_example(example, feature_description)
    image = tf.io.decode_jpeg(parsed_example["image"], channels=3)
    image = tf.cast(image, tf.float32)
    image = tf.image.resize(image, (224, 224))
    image = tf.keras.applications.resnet50.preprocess_input(image)
    if "target" in feature_description:
        target = parsed_example["target"]
        return image, target
    return image, parsed_example["image_name"]


def load_data(path, batch_size=32, train=True):
    """Create a tf.data.Dataset from TFRecord files."""
    filenames = tf.io.gfile.glob(path)
    dataset = tf.data.TFRecordDataset(filenames, num_parallel_reads=tf.data.AUTOTUNE)
    if train:
        feature_description = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
            "target": tf.io.FixedLenFeature([], tf.int64),
        }
    else:
        feature_description = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
        }
    parsed_dataset = dataset.map(
        lambda x: _parse_function(x, feature_description),
        num_parallel_calls=tf.data.AUTOTUNE,
    )
    return parsed_dataset.batch(batch_size).prefetch(AUTOTUNE)

def get_model():
    """Load the pretrained model if available; otherwise build a lightweight fallback."""
    model_path = ROOT_DIR + "cassava-training/effnetb1_finetuned.h5"
    try:
        return tf.keras.models.load_model(model_path)
    except Exception as e:
        base = tf.keras.applications.EfficientNetB1(
            weights="imagenet", include_top=False, input_shape=(224, 224, 3)
        )
        x = tf.keras.layers.GlobalAveragePooling2D()(base.output)
        outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
        model = tf.keras.Model(inputs=base.input, outputs=outputs)
        model.compile(
            optimizer="adam",
            loss="sparse_categorical_crossentropy",
            metrics=["accuracy"],
        )
        return model


model = get_model()

    predictions.append(pred_label)
