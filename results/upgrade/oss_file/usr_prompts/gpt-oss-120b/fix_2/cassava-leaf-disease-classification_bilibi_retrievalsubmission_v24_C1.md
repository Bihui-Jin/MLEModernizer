class CassavaDataset(Dataset):
    def __init__(self, df, img_dir, transform=None):
        """
        df       : DataFrame with columns ['image_id', 'label'] (label may be absent for test)
        img_dir  : root directory containing the images
        transform: torchvision transforms to apply
        """
        self.df = df.reset_index(drop=True)
        self.img_dir = pathlib.Path(img_dir)
        self.transform = transform

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = self.img_dir / row["image_id"]
        img = torchvision.io.read_image(str(img_path)).float() / 255.0  # shape C,H,W
        if self.transform:
            img = self.transform(img)
        if "label" in row:
            label = int(row["label"])
            return img, label
        else:
            return img, row["image_id"]

model = resnet50(pretrained=True)
num_ftrs = model.fc.in_features
model.fc = nn.Linear(num_ftrs, 5)  # 5 disease categories (0‑4)
model = model.to(device)

criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.AdamW(model.parameters(), lr=3e-4, weight_decay=1e-4)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=5)




## === cell 4
def train_one_epoch(epoch):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    prog_bar = tqdm(train_loader, desc=f"Epoch {epoch+1} [train]")
    for imgs, labels in prog_bar:
        imgs = imgs.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        running_loss += loss.item() * imgs.size(0)
        _, preds = torch.max(outputs, 1)
        correct += (preds == labels).sum().item()
        total += imgs.size(0)
        prog_bar.set_postfix(loss=running_loss / total, acc=correct / total)

    epoch_loss = running_loss / total
    epoch_acc = correct / total
    return epoch_loss, epoch_acc


def validate(epoch):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0
    with torch.no_grad():
        prog_bar = tqdm(val_loader, desc=f"Epoch {epoch+1} [val]")
        for imgs, labels in prog_bar:
            imgs = imgs.to(device)
            labels = labels.to(device)

            outputs = model(imgs)
            loss = criterion(outputs, labels)

            running_loss += loss.item() * imgs.size(0)
            _, preds = torch.max(outputs, 1)
            correct += (preds == labels).sum().item()
            total += imgs.size(0)
            prog_bar.set_postfix(loss=running_loss / total, acc=correct / total)

    val_loss = running_loss / total
    val_acc = correct / total
    return val_loss, val_acc


EPOCHS = 5
for epoch in range(EPOCHS):
    train_loss, train_acc = train_one_epoch(epoch)
    val_loss, val_acc = validate(epoch)
    scheduler.step()
    print(

        f"Val loss: {val_loss:.4f}, acc: {val_acc:.4f}"
    )
