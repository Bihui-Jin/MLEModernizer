## === cell 1
class InferDataset(Dataset):
    def __init__(self, img_dir, transform=None):
        self.img_dir = img_dir
        self.filenames = [f for f in os.listdir(img_dir) if f.lower().endswith(".jpg")]
        self.transform = transform

    def __len__(self):
        return len(self.filenames)

    def __getitem__(self, idx):
        fn = self.filenames[idx]
        img_path = os.path.join(self.img_dir, fn)
        img = cv2.imread(img_path)
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        if self.transform:
            img = self.transform(img)
        return img, fn


val_transform = T.Compose(
    [
        T.ToPILImage(),
        T.Resize(640),
        T.CenterCrop(640),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)

test_img_dir = os.path.join(
    "input", "cassava-leaf-disease-classification", "test_images"
)
infer_dataset = InferDataset(test_img_dir, transform=val_transform)
infer_loader = DataLoader(infer_dataset, batch_size=32, shuffle=False, num_workers=2)
