class CassavaDataset(Dataset):
    def __init__(self, img_dir, df=None, transform=None, inference=False):
        """
        img_dir: folder containing images
        df: pandas DataFrame with columns ['image_id','label'] (None for inference)
        transform: torchvision transforms
        inference: if True, dataset returns (image_tensor, image_id) only
        """
        self.img_dir = img_dir
        self.transform = transform
        self.inference = inference
        if df is not None:
            self.img_ids = df["image_id"].values
            self.labels = df["label"].values.astype(int)
        else:
            self.img_ids = [
                f for f in os.listdir(img_dir) if f.lower().endswith(".jpg")
            ]
            self.labels = None

    def __len__(self):
        return len(self.img_ids)

    def __getitem__(self, idx):
        img_id = self.img_ids[idx]
        img_path = os.path.join(self.img_dir, img_id)
        image = cv2.imread(img_path)  # BGR
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)  # convert to RGB
        if self.transform:
            image = self.transform(image)
        if self.inference:
            return image, img_id
        else:
            label = self.labels[idx]
            return image, label


train_df = pd.read_csv(TRAIN_CSV)

val_frac = 0.1
val_size = int(len(train_df) * val_frac)

val_subset = train_df.iloc[-val_size:].reset_index(drop=True)

train_dataset = CassavaDataset(
    TRAIN_IMG_DIR, df=train_subset, transform=train_transform
)

    train_dataset, batch_size=64, shuffle=True, num_workers=2, pin_memory=True
)
val_loader = DataLoader(
    val_subset, batch_size=64, shuffle=False, num_workers=2, pin_memory=True
)



## === cell 3
model = models.resnet50(pretrained=True)
num_ftrs = model.fc.in_features
model.fc = nn.Linear(num_ftrs, 5)
model = model.to(device)

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=1e-4)



## === cell 4
epochs = 2
for epoch in range(epochs):
    model.train()
    running_loss = 0.0
    for imgs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{epochs} [train]"):
        imgs = imgs.to(device, dtype=torch.float)
        labels = labels.to(device)
        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        running_loss += loss.item() * imgs.size(0)
    epoch_loss = running_loss / len(train_loader.dataset)
    print(f"Train loss: {epoch_loss:.4f}")

    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for imgs, labels in tqdm(val_loader, desc=f"Epoch {epoch+1}/{epochs} [val]"):
            imgs = imgs.to(device, dtype=torch.float)
            labels = labels.to(device)
            outputs = model(imgs)
            _, preds = torch.max(outputs, 1)
            correct += (preds == labels).sum().item()
            total += labels.size(0)
    acc = correct / total
    print(f"Validation accuracy: {acc:.4f}")
