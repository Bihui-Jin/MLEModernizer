## === cell 0
import os
import pandas as pd
import torch
import cv2
from torch.utils.data import Dataset, DataLoader
import torchvision.transforms as T
import torchvision.models as models



## === cell 1
TEST_IMG_DIR = "../input/cassava-leaf-disease-classification/test_images"
SAMPLE_SUBMIT_PATH = (
    "../input/cassava-leaf-disease-classification/sample_submission.csv"
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = models.resnet50(weights=models.ResNet50_Weights.IMAGENET1K_V2).to(device)
model.eval()




## === cell 2
class TestImageDataset(Dataset):
    def __init__(self, img_dir, transforms):
        self.img_dir = img_dir
        self.transforms = transforms
        self.img_names = sorted(
            [f for f in os.listdir(img_dir) if f.lower().endswith(".jpg")]
        )

    def __len__(self):
        return len(self.img_names)

    def __getitem__(self, idx):
        img_path = os.path.join(self.img_dir, self.img_names[idx])
        img = cv2.imread(img_path)
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        if self.transforms:
            img = self.transforms(img)
        return img, self.img_names[idx]


test_transforms = T.Compose(
    [
        T.ToPILImage(),
        T.Resize((224, 224)),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)

test_dataset = TestImageDataset(TEST_IMG_DIR, transforms=test_transforms)
test_loader = DataLoader(test_dataset, batch_size=32, shuffle=False, num_workers=0)



## === cell 3
submission_path = "./submission.csv"
with open(submission_path, "w") as f:
    f.write("image_id,label\n")

    with torch.no_grad():
        for imgs, img_names in test_loader:
            imgs = imgs.to(device)
            logits = model(imgs)
            preds = torch.argmax(logits, dim=1).cpu().numpy()
            for img_name, pred in zip(img_names, preds):
                f.write(f"{img_name},{int(pred)}\n")



## === cell 4
sub_df = pd.read_csv(submission_path)
sample_df = pd.read_csv(SAMPLE_SUBMIT_PATH)

assert len(sub_df) == len(
    sample_df
), f"Submission length {len(sub_df)} does not match expected {len(sample_df)}"
print(f"Submission written to {submission_path} with {len(sub_df)} rows.")
