## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMAGES_PATH = os.path.join(BASE_PATH, "train_images")
TEST_TFRECORD_PATH = os.path.join(BASE_PATH, "test_tfrecords")
TRAIN_CSV_PATH = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUBMISSION_PATH = os.path.join(BASE_PATH, "sample_submission.csv")

AUTOTUNE = tf.data.experimental.AUTOTUNE
BATCH_SIZE = 32
IMAGE_SIZE = (224, 224)  # EfficientNetB0 default
NUM_CLASSES = 5




## === cell 2
def decode_and_resize(img_bytes):
    img = tf.io.decode_jpeg(img_bytes, channels=3)
    img = tf.image.resize(img, IMAGE_SIZE)
    img = img / 255.0
    return img


def load_image(path, label):
    img_bytes = tf.io.read_file(path)
    img = decode_and_resize(img_bytes)
    return img, label




## === cell 3
train_df = pd.read_csv(TRAIN_CSV_PATH)
train_df["label"] = train_df["label"].astype(np.int32)

train_df["filepath"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMAGES_PATH, x)
)

train_paths, val_paths, train_labels, val_labels = train_test_split(
    train_df["filepath"].values,
    train_df["label"].values,
    test_size=0.1,
    stratify=train_df["label"],
    random_state=42,
)

train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(load_image, num_parallel_calls=AUTOTUNE)
train_ds = train_ds.shuffle(1024).batch(BATCH_SIZE).prefetch(AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
val_ds = val_ds.map(load_image, num_parallel_calls=AUTOTUNE)
val_ds = val_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)



## === cell 4
base_model = tf.keras.applications.EfficientNetB0(
    include_top=False, weights="imagenet", input_shape=IMAGE_SIZE + (3,), pooling="avg"
)

inputs = tf.keras.Input(shape=IMAGE_SIZE + (3,))
x = base_model(inputs, training=False)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)

model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=2)




## === cell 5
def decode_img(img):
    img = tf.io.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMAGE_SIZE)
    img = img / 255.0
    return img


def read_test_tfrecord(example):
    TFREC_FORMAT = {
        "image": tf.io.FixedLenFeature([], tf.string),
        "image_name": tf.io.FixedLenFeature([], tf.string),
    }
    example = tf.io.parse_single_example(example, TFREC_FORMAT)
    img = decode_img(example["image"])
    name = example["image_name"]
    return img, name


TEST_FILENAMES = tf.io.gfile.glob(os.path.join(TEST_TFRECORD_PATH, "ld_test*.tfrec"))


def get_test_dataset():
    ds = tf.data.TFRecordDataset(TEST_FILENAMES, num_parallel_reads=AUTOTUNE)
    ds = ds.map(read_test_tfrecord, num_parallel_calls=AUTOTUNE)
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds


test_ds = get_test_dataset()

all_preds = []
all_ids = []
for images, ids in test_ds:
    probs = model.predict(images, verbose=0)
    preds = np.argmax(probs, axis=-1)
    all_preds.extend(preds.tolist())
    all_ids.extend([id.decode("utf-8") for id in ids.numpy()])



## === cell 6
submission = pd.DataFrame({"image_id": all_ids, "label": all_preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
