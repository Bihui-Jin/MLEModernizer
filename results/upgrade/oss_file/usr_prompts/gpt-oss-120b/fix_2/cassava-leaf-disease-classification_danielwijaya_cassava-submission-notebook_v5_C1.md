GCS_PATH = "../input/cassava-leaf-disease-classification"
AUTOTUNE = tf.data.experimental.AUTOTUNE
BATCH_SIZE = 128  # 16*8 as in original code
IMAGE_SIZE = [512, 512]
NUM_CLASSES = 5


def dataset_sizes(filenames):
    nums = [int(re.compile(r"-([0-9]*)\.").search(f).group(1)) for f in filenames]
    return np.sum(nums)




## === cell 2
def decode_img(img_bytes):
    img = tf.io.decode_jpeg(img_bytes, channels=3)
    img = tf.cast(img, tf.float32) / 255.0
    img = tf.reshape(img, [*IMAGE_SIZE, 3])
    return img


def read_tfrecord(example, labeled):
    if labeled:
        fmt = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "target": tf.io.FixedLenFeature([], tf.int64),
        }
    else:
        fmt = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
        }
    example = tf.io.parse_single_example(example, fmt)
    img = decode_img(example["image"])
    if labeled:
        label = tf.cast(example["target"], tf.int32)
        return img, label
    else:
        return img, example["image_name"]


def load_dataset(filenames, labeled=True, ordered=False):
    opts = tf.data.Options()
    if not ordered:
        opts.experimental_deterministic = False
    ds = tf.data.TFRecordDataset(filenames, num_parallel_reads=AUTOTUNE)
    ds = ds.with_options(opts)
    ds = ds.map(partial(read_tfrecord, labeled=labeled), num_parallel_calls=AUTOTUNE)
    return ds


def to_float32(image, label):
    return tf.cast(image, tf.float32), label




## === cell 3
base_model = tf.keras.applications.EfficientNetB0(
    input_shape=(*IMAGE_SIZE, 3), include_top=False, weights="imagenet"
)
base_model.trainable = False  # freeze backbone for quick training

inputs = tf.keras.Input(shape=(*IMAGE_SIZE, 3))
x = base_model(inputs, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
x = tf.keras.layers.Dropout(0.2)(x)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)



## === cell 4
train_filenames = tf.io.gfile.glob(
    os.path.join(GCS_PATH, "train_tfrecords", "ld_train*.tfrec")
)
train_files, val_files = train_test_split(
    train_filenames, test_size=0.1, random_state=42
)

train_ds = load_dataset(train_files, labeled=True, ordered=False)
train_ds = train_ds.map(to_float32).shuffle(2048).batch(BATCH_SIZE).prefetch(AUTOTUNE)

val_ds = load_dataset(val_files, labeled=True, ordered=False)
val_ds = val_ds.map(to_float32).batch(BATCH_SIZE).prefetch(AUTOTUNE)

model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=2)
