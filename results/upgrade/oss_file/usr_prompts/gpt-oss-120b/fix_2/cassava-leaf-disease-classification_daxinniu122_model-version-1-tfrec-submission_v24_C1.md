test_images_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"

for image_name in sample_sub.image_id:
    img_path = os.path.join(test_images_dir, image_name)
    if not os.path.exists(img_path):
        raise FileNotFoundError(f"Test image not found: {img_path}")
    img = keras.preprocessing.image.load_img(img_path)
    img = img_to_array(img)
    img = tf.reshape(tf.image.resize(img, (512, 512)), (-1, 512, 512, 3))
    prediction1 = model1.predict(img, verbose=0) * alpha_1
    prediction2 = model2.predict(img, verbose=0) * alpha_2
    ensemble_pred = prediction1 + prediction2
    preds.append(np.argmax(ensemble_pred, axis=1)[0])

my_submission = pd.DataFrame({"image_id": sample_sub.image_id, "label": preds})
output_path = "/kaggle/working/submission.csv"
my_submission.to_csv(output_path, index=False)
print(f"Submission saved to {output_path}")
