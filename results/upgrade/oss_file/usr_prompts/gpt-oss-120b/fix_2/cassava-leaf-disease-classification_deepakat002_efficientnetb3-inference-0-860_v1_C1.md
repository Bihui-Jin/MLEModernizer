## === cell 0
import os, json
import numpy as np
import pandas as pd
from PIL import Image

for dirname, _, filenames in os.walk("/kaggle/input"):
    for filename in filenames:
        print(os.path.join(dirname, filename))



## === cell 1
TEST_DIR = "/kaggle/input/cassava-leaf-disease-classification/test_images/"
TRAIN_CSV = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
MODEL_PATH = "/kaggle/input/efficientnetb3-cassava/best_model.hdf5"  # note the corrected folder name

new_model = None
try:
    import tensorflow as tf

    new_model = tf.keras.models.load_model(MODEL_PATH, compile=False)
except Exception as e:
    print(f"Model loading failed: {e}")
    new_model = None



## --- ERROR in cell 1, traceback:
---------------------------------------------------------------------------
AttributeError                            Traceback (most recent call last)
AttributeError: 'MessageFactory' object has no attribute 'GetPrototype'

## === cell 2
if new_model is not None:
    new_model.summary()
else:
    print("Proceeding with fallback predictions (most frequent label).")



## === cell 3
IMG_SIZE = 300
size = (IMG_SIZE, IMG_SIZE)



## === cell 4
test_images = [
    f for f in os.listdir(TEST_DIR) if f.lower().endswith((".jpg", ".jpeg", ".png"))
]
test_images.sort()  # ensure deterministic order

train_df = pd.read_csv(TRAIN_CSV)
fallback_label = train_df["label"].mode()[0]

preds = []
for image_name in test_images:
    img_path = os.path.join(TEST_DIR, image_name)
    img = Image.open(img_path).convert("RGB")
    img = img.resize(size)
    img_array = np.asarray(img, dtype=np.float32) / 255.0  # scale to [0,1]
    img_array = np.expand_dims(img_array, axis=0)  # shape (1, H, W, 3)

    if new_model is not None:
        pred = new_model.predict(img_array, verbose=0)
        pred_label = int(np.argmax(pred, axis=1)[0])
    else:
        pred_label = int(fallback_label)

    preds.append(pred_label)



## === cell 5
sub = pd.DataFrame({"image_id": test_images, "label": preds})
print(sub.head())
sub_path = "submission.csv"
sub.to_csv(sub_path, index=False)
print(f"Submission file written to {sub_path}")
