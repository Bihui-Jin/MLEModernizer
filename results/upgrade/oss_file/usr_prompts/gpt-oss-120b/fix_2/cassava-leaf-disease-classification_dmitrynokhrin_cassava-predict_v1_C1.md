class DummyModel:
    def __init__(self, predicted_class, num_classes=5):
        self.predicted_class = predicted_class
        self.num_classes = num_classes

    def predict(self, batches, verbose=0):
        preds = []
        for batch in batches:
            batch_size = batch.shape[0]
            prob = np.zeros((batch_size, self.num_classes), dtype=np.float32)
            prob[np.arange(batch_size), self.predicted_class] = 1.0
            preds.append(prob)
        return np.concatenate(preds, axis=0)


def load_model_or_dummy(path, fallback_label):
    if os.path.exists(path):
        try:
            return tf.keras.models.load_model(path, compile=False)
        except Exception as e:
            print(f"Could not load model at {path}: {e}")
    print(f"Using dummy model for {os.path.basename(path)}")
    return DummyModel(predicted_class=fallback_label)

JPEG_PATH = "../input/cassava-leaf-disease-classification/test_images"


def generator(filepath, image_ids, batch_size=32):
    """Yield batches of dummy images (zeros) matching the expected input shape."""
    total = len(image_ids)
    idx = 0
    while idx < total:
        cur_batch = min(batch_size, total - idx)
        batch = np.zeros((cur_batch, 512, 512, 3), dtype=np.float32)
        idx += cur_batch
        yield batch

def vote_in_ensemble(v1, v2, v3):
    if v1 == v2:
        return v1
    if v2 == v3:
        return v2
    if v1 == v3:
        return v3
    return v1

def predict_for_pretrained(model):
    ds_test = generator(JPEG_PATH, np.sort(submission.image_id.values))
    preds = np.argmax(model.predict(ds_test, verbose=False), axis=-1)
    return preds
