## === cell 2
def gem(x, p=3, eps=1e-5):
    return F.avg_pool2d(x.clamp(min=eps).pow(p), (x.size(-2), x.size(-1))).pow(1.0 / p)


class GeM(nn.Module):
    def __init__(self, p=3, eps=1e-5):
        super(GeM, self).__init__()
        self.p = nn.Parameter(torch.ones(1) * p)
        self.eps = eps

    def forward(self, x):
        return gem(x, p=self.p, eps=self.eps)

    def __repr__(self):
        return f"{self.__class__.__name__}(p={self.p.item():.4f}, eps={self.eps})"




## === cell 3
class Net(nn.Module):
    def __init__(self, num_classes=5):
        super().__init__()
        self.model = timm.create_model("seresnext50_32x4d", pretrained=True)
        self._avg_pooling = nn.AdaptiveAvgPool2d(1)
        self.dropout = nn.Dropout(0.5)
        self._fc = nn.Linear(2048, num_classes, bias=True)

    def forward(self, x):
        x = x / 255.0  # normalise to [0,1]
        bs = x.size(0)
        feats = self.model.forward_features(x)
        pooled = self._avg_pooling(feats).view(bs, -1)
        dropped = self.dropout(pooled)
        out = self._fc(dropped)
        return out




## === cell 4
class TrainDataset(Dataset):
    def __init__(self, csv_path, img_dir, img_size=640):
        self.df = pd.read_csv(csv_path)
        self.img_dir = img_dir
        self.img_size = img_size

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = os.path.join(self.img_dir, row["image_id"])
        img = cv2.imread(img_path, cv2.IMREAD_COLOR)
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = cv2.resize(img, (self.img_size, self.img_size))
        img = np.transpose(img, (2, 0, 1)).astype(np.float32)  # C,H,W
        label = int(row["label"])
        return img, label




## === cell 5
class TestDatasetTTA:
    """Returns original + 7 TTA versions (rotations + h‑flip) for a single image."""

    def __init__(self, img_dir, img_size=640):
        self.root_dir = img_dir
        self.ds = os.listdir(img_dir)
        self.img_size = img_size

    def __len__(self):
        return len(self.ds)

    def _preprocess(self, image):
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = cv2.resize(image, (self.img_size, self.img_size))

        rots = [np.rot90(image, k) for k in range(4)]
        flips = [np.fliplr(image)]
        flips_rots = [np.rot90(flips[0], k) for k in range(1, 4)]

        batch = np.stack(rots + [image] + flips + flips_rots, axis=0)
        batch = np.transpose(batch, (0, 3, 1, 2)).astype(np.float32)
        return batch

    def __getitem__(self, idx):
        fname = self.ds[idx]
        img_path = os.path.join(self.root_dir, fname)
        img = cv2.imread(img_path, cv2.IMREAD_COLOR)
        batch = self._preprocess(img)
        return fname, batch

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
torch.backends.cudnn.benchmark = True

train_dataset = TrainDataset(train_csv_path, train_img_dir, img_size=640)
train_loader = DataLoader(
    train_dataset, batch_size=64, shuffle=True, num_workers=4, pin_memory=True
)

model = Net().to(device)
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

epochs = 3
model.train()
for epoch in range(epochs):
    total_loss = 0.0
    for imgs, lbls in train_loader:
        imgs = imgs.to(device)
        lbls = lbls.to(device)

        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, lbls)
        loss.backward()
        optimizer.step()
        total_loss += loss.item() * imgs.size(0)
    avg_loss = total_loss / len(train_loader.dataset)
    print(f"Epoch [{epoch+1}/{epochs}] - Loss: {avg_loss:.4f}")

test_dataset = TestDatasetTTA(test_img_dir, img_size=640)
model.eval()
merge_res_dict = {}
image_ids = []

for idx in range(len(test_dataset)):
    fname, batch = test_dataset[idx]
    image_ids.append(fname)
    batch_tensor = torch.from_numpy(batch).to(device)  # shape (8,3,640,640)
    with torch.no_grad():
        logits = model(batch_tensor)  # (8,5)
        probs = torch.nn.functional.softmax(logits, dim=1)
        probs = probs.cpu().numpy()
        mean_prob = probs.mean(axis=0)  # average over TTA
    merge_res_dict[fname] = mean_prob

submission = pd.DataFrame(
    {
        "image_id": image_ids,
        "label": [int(np.argmax(merge_res_dict[f])) for f in image_ids],
    }
)

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
