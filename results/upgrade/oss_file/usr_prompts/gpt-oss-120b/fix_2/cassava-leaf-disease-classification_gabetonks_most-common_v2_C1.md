train_df.head()




## === cell 2
def extract_mean_rgb(image_path):
    """Read image with matplotlib and return mean RGB as a length‑3 vector."""
    img = plt.imread(image_path)
    if img.ndim == 2:  # grayscale
        img = np.stack([img] * 3, axis=-1)
    elif img.shape[2] == 4:  # RGBA
        img = img[:, :, :3]
    img_small = img[::4, ::4, :]
    return img_small.mean(axis=(0, 1))


train_features = []
for img_id in train_df["image_id"]:
    img_path = os.path.join(train_images_dir, img_id)
    train_features.append(extract_mean_rgb(img_path))
X_train = np.stack(train_features)
y_train = train_df["label"].values

scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)



## === cell 3
model = LogisticRegression(
    multi_class="multinomial", solver="lbfgs", max_iter=200, n_jobs=5, random_state=42
)
model.fit(X_train_scaled, y_train)
