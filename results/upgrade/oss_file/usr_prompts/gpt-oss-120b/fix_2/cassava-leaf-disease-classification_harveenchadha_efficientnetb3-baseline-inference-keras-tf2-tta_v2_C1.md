BASE_PATH = "./input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["path"] = df_train["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))

train_df, val_df = train_test_split(
    df_train, test_size=0.1, stratify=df_train["label"], random_state=SEED
)

train_idg = ImageDataGenerator(
    rescale=1.0 / 255.0,
    rotation_range=15,
    width_shift_range=0.1,
    height_shift_range=0.1,
    shear_range=0.1,
    zoom_range=0.1,
    horizontal_flip=True,
    fill_mode="nearest",
)

val_idg = ImageDataGenerator(rescale=1.0 / 255.0)


def make_generator(dataframe, batch_size=32, shuffle=True, augment=False):
    idg = train_idg if augment else val_idg
    return idg.flow_from_dataframe(
        dataframe=dataframe,
        x_col="path",
        y_col="label",
        target_size=(300, 300),
        color_mode="rgb",
        class_mode="categorical",
        batch_size=batch_size,
        shuffle=shuffle,
        seed=SEED,
    )


train_gen = make_generator(train_df, batch_size=64, shuffle=True, augment=True)
val_gen = make_generator(val_df, batch_size=64, shuffle=False, augment=False)

num_classes = 5
base = EfficientNetB0(weights="imagenet", include_top=False, input_shape=(300, 300, 3))
x = GlobalAveragePooling2D()(base.output)
x = Dropout(0.2)(x)
outputs = Dense(num_classes, activation="softmax")(x)
model = Model(inputs=base.input, outputs=outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

ckpt_path = "best_model.h5"
callbacks = [
    ModelCheckpoint(
        ckpt_path, monitor="val_accuracy", save_best_only=True, mode="max", verbose=1
    ),
    EarlyStopping(
        monitor="val_accuracy",
        patience=5,
        mode="max",
        restore_best_weights=True,
        verbose=1,
    ),
    ReduceLROnPlateau(
        monitor="val_accuracy", factor=0.5, patience=3, mode="max", verbose=1
    ),
]

if not os.path.exists(ckpt_path):
    model.fit(
        train_gen, epochs=10, validation_data=val_gen, callbacks=callbacks, verbose=2
    )
else:
    model = load_model(ckpt_path)

test_images = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
df_test = pd.DataFrame(test_images, columns=["path"])
test_gen = ImageDataGenerator(rescale=1.0 / 255.0).flow_from_dataframe(
    dataframe=df_test,
    x_col="path",
    y_col=None,
    target_size=(300, 300),
    color_mode="rgb",
    class_mode=None,
    batch_size=128,
    shuffle=False,
    seed=SEED,
)

pred_test = model.predict(test_gen, verbose=1)
pred_labels = np.argmax(pred_test, axis=-1)

submission = pd.DataFrame(
    {
        "image_id": df_test["path"].apply(lambda p: os.path.basename(p)),
        "label": pred_labels,
    }
)
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
