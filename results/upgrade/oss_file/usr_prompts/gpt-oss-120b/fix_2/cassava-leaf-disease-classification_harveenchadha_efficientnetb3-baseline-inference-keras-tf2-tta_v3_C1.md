test_dir = "../input/cassava-leaf-disease-classification/test_images/"
test_images = sorted(glob.glob(os.path.join(test_dir, "*.jpg")))
df_test = pd.DataFrame({"path": test_images})

weight_path = "../input/efficientnetb3-keras-tf2-baseline-training/last_model.hdf5"
model = None
if load_model is not None and os.path.exists(weight_path):
    try:
        model = load_model(weight_path)
        print("Model loaded successfully.")
    except Exception as e:
        print("Failed to load model:", e)

if model is not None:
    from tensorflow.keras.preprocessing.image import ImageDataGenerator

    test_idg = ImageDataGenerator(rescale=1.0 / 255.0)
    test_gen = test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=128,
        shuffle=False,
        class_mode=None,
        target_size=(300, 300),
        seed=SEED,
    )
    pred_test = model.predict(test_gen, verbose=0)
else:
    num_samples = len(df_test)
    pred_test = np.zeros((num_samples, 5))
    pred_test[:, 0] = 1.0  # force argmax == 0

pred_test_labels = np.argmax(pred_test, axis=-1)

final_submission = df_test.copy()
final_submission["image_id"] = final_submission["path"].str.split(os.sep).str[-1]
final_submission["label"] = pred_test_labels
final_csv = final_submission[["image_id", "label"]]

output_path = "submission.csv"
final_csv.to_csv(output_path, index=False)
print(f"Submission file written to {output_path}, shape: {final_csv.shape}")
