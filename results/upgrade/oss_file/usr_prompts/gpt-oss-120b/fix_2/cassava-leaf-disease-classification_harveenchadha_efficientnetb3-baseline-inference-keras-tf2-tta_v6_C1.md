# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import pandas as pd
import numpy as np
import tensorflow as tf
import glob
from tensorflow.keras.models import Model, load_model, Sequential
from tensorflow.keras.applications import EfficientNetB3
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D
from tensorflow.keras.optimizers import Adam
from sklearn.model_selection import train_test_split

SEED = 42
BATCH_SIZE = 32
IMG_SIZE = (300, 300)
NUM_CLASSES = 5
tf.random.set_seed(SEED)
np.random.seed(SEED)



## === cell 1
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
train_img_dir = "../input/cassava-leaf-disease-classification/train_images/"
train_df = pd.read_csv(train_csv_path)
train_df["path"] = train_img_dir + train_df["image_id"].astype(str)

train_df_split, val_df_split = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=SEED
)



## === cell 2
train_idg = ImageDataGenerator(
    rescale=1.0 / 255.0,
    horizontal_flip=True,
    rotation_range=20,
    width_shift_range=0.1,
    height_shift_range=0.1,
    zoom_range=0.2,
)

val_idg = ImageDataGenerator(rescale=1.0 / 255.0)

train_gen = train_idg.flow_from_dataframe(
    dataframe=train_df_split,
    x_col="path",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="raw",
    shuffle=True,
    seed=SEED,
)

val_gen = val_idg.flow_from_dataframe(
    dataframe=val_df_split,
    x_col="path",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="raw",
    shuffle=False,
)




## === cell 3
def build_model():
    base = EfficientNetB3(
        include_top=False, weights="imagenet", input_shape=IMG_SIZE + (3,)
    )
    x = GlobalAveragePooling2D()(base.output)
    output = Dense(NUM_CLASSES, activation="softmax")(x)
    model = Model(inputs=base.input, outputs=output)
    model.compile(
        optimizer=Adam(learning_rate=1e-4),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )
    return model


model_path = "../input/myeffnet/best_effnetb3.hdf5"
try:
    my_model = load_model(model_path)
except Exception as e:
    my_model = build_model()
    my_model.fit(
        train_gen,
        steps_per_epoch=len(train_gen),
        validation_data=val_gen,
        validation_steps=len(val_gen),
        epochs=2,
        verbose=1,
    )



## === cell 4
test_img_dir = "../input/cassava-leaf-disease-classification/test_images/"
test_images = glob.glob(test_img_dir + "*.jpg")
df_test = pd.DataFrame(test_images, columns=["path"])


def make_test_gen(batch_size=64):
    test_idg = ImageDataGenerator(rescale=1.0 / 255.0)
    test_gen = test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=batch_size,
        seed=SEED,
        shuffle=False,
        class_mode=None,
        target_size=IMG_SIZE,
    )
    return test_gen


test_gen = make_test_gen(batch_size=128)



## === cell 5
pred_test = my_model.predict(test_gen, verbose=1)
pred_test_labels = np.argmax(pred_test, axis=-1)



## === cell 6
final_submission = df_test.copy()
final_submission["image_id"] = final_submission["path"].str.split("/").str[-1]
final_submission["label"] = pred_test_labels.astype(int)

final_csv = final_submission[["image_id", "label"]]
final_csv.head()



## === cell 7
final_csv.to_csv("submission.csv", index=False)
