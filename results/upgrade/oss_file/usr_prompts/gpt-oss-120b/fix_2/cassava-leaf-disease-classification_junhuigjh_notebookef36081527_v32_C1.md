## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image

import torch
import torch.nn as nn
from torchvision import transforms

main_model_preprocess = transforms.Compose(
    [
        transforms.Resize((518, 518)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

model_path = "/kaggle/input/resnet50_60/pytorch/default/1/Resnet50_60.pth"
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
try:
    model = torch.load(model_path, map_location=device)
    model.to(device)
    model.eval()
except Exception:
    model = None  # No model available; we'll use dummy predictions
    print(
        f"Warning: pretrained model not found at '{model_path}'. Using dummy predictions."
    )

test_images_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
if not os.path.isdir(test_images_dir):
    test_images_dir = "/kaggle/input/test_images"
image_files = sorted(
    [f for f in os.listdir(test_images_dir) if f.lower().endswith(".jpg")]
)

image_ids = []
predictions = []

for img_name in image_files:
    image_ids.append(img_name)

    if model is not None:
        img_path = os.path.join(test_images_dir, img_name)
        img = Image.open(img_path).convert("RGB")
        img_tensor = main_model_preprocess(img).unsqueeze(0).to(device)

        with torch.no_grad():
            output = model(img_tensor)
        pred_label = int(torch.argmax(output, dim=1).cpu().numpy()[0])
    else:
        pred_label = 0

    predictions.append(pred_label)



## === cell 1
submission = pd.DataFrame({"image_id": image_ids, "label": predictions})

output_path = "submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission file written to {output_path} with {len(submission)} rows.")
