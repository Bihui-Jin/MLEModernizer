def preprocess_for_keras(pil_img):
    """
    Resize, scale to [0,1] and normalize to mean=0.5, std=0.5
    (equivalent to (x/255 - 0.5) / 0.5 = x/127.5 - 1).
    Returns a NumPy array of shape (1, 512, 512, 3).
    """
    img = pil_img.resize((512, 512))
    arr = np.asarray(img).astype(np.float32) / 127.5 - 1.0
    return np.expand_dims(arr, axis=0)


torch_transforms = transforms.Compose(
    [
        transforms.Resize((512, 512)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

model1 = load_model(
    "/kaggle/input/densenet_70_512x512/keras/default/2/Densenet_70_512x512_weights (1).keras"
)

model2 = torch.load(
    "/kaggle/input/resnet50_70_512x512/pytorch/default/1/Resnet50_70_512x512.pth"
)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model2.to(device)
model2.eval()

test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
test_images = sorted(os.listdir(test_dir))  # deterministic order

combined_probs = []
image_ids = []

for idx, img_name in enumerate(test_images, start=1):
    image_ids.append(img_name)
    pil_img = Image.open(os.path.join(test_dir, img_name)).convert("RGB")

    img_keras = preprocess_for_keras(pil_img)  # (1, 512, 512, 3)

    img_torch = torch_transforms(pil_img).unsqueeze(0).to(device)  # (1, 3, 512, 512)

    pred1 = model1.predict(img_keras, verbose=False)[0]  # (5,)
    with torch.no_grad():
        pred2 = model2(img_torch).cpu().numpy()[0]  # (5,)

    combined_probs.append(np.concatenate([pred1, pred2]))

    if idx % 100 == 0 or idx == len(test_images):
        print(f"Processed {idx}/{len(test_images)} images", end="\r")

combined_probs_array = np.array(combined_probs)  # shape (N, 10)
