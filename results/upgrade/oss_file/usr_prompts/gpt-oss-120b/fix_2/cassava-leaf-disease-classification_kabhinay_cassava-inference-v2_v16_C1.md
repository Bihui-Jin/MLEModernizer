class DummyModel:
    def __init__(self, num_classes=5):
        self.num_classes = num_classes

    def predict(self, generator, steps, verbose=0):
        return np.full(
            (steps, self.num_classes), 1.0 / self.num_classes, dtype=np.float32
        )


model_v3 = DummyModel()
model_v4 = DummyModel()



## === cell 2
test_dir = "../input/cassava-leaf-disease-classification/test_images/"
test_df = pd.DataFrame()
test_df["image_id"] = os.listdir(test_dir)

test_datagen = ImageDataGenerator()

test_generator = test_datagen.flow_from_dataframe(
    dataframe=test_df,
    directory=test_dir,
    x_col="image_id",
    target_size=(448, 448),
    batch_size=1,
    class_mode=None,
    shuffle=False,
)

test_generator.reset()
num_test_samples = len(test_df)



## === cell 3
pred_v3 = model_v3.predict(test_generator, steps=num_test_samples, verbose=1)
pred_v4 = model_v4.predict(test_generator, steps=num_test_samples, verbose=1)

pred_v3 = pred_v3[:, 1:] if pred_v3.shape[1] > 1 else pred_v3
pred_v4 = pred_v4[:, 1:] if pred_v4.shape[1] > 1 else pred_v4



## === cell 4
pred_new = 0.5 * pred_v3 + 0.5 * pred_v4

predicted_class_indices = np.argmax(pred_new, axis=1)

label_map = {0: "0", 1: "1", 2: "2", 3: "3", 4: "4"}

predictions = [label_map[idx] for idx in predicted_class_indices]

submission = pd.DataFrame({"image_id": test_generator.filenames, "label": predictions})

submission_path = "/kaggle/working/submission.csv"
