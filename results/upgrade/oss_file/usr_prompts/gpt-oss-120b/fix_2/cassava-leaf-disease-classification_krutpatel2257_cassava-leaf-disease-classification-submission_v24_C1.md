import os

import torch

## === cell 1
model_path = "../input/en-b4-tta-calr-clahe/model(14).pth"
sample_sub_path = "../input/cassava-leaf-disease-classification/sample_submission.csv"

## === cell 2
model = efficientnet_b4(weights=EfficientNet_B4_Weights.IMAGENET1K_V1)
num_ftrs = model.classifier[1].in_features
model.classifier[1] = nn.Linear(num_ftrs, 5)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = model.to(device)
model.eval()

if os.path.exists(model_path):
    state_dict = torch.load(model_path, map_location=device)
    model.load_state_dict(state_dict, strict=False)

## === cell 4
sample_sub = pd.read_csv(sample_sub_path)
tta_count = 5

predictions = []
for _, row in sample_sub.iterrows():
    img_path = os.path.join(test_images_path, row.image_id)
    image = np.array(Image.open(img_path).convert("RGB"))
    img_pred_sum = 0
    for _ in range(tta_count):
        aug_image = sub_aug(image=image)["image"]
        tensor_img = torch.from_numpy(aug_image).permute(2, 0, 1).float()
        tensor_img = tensor_img.to(device)
        with torch.no_grad():
            outputs = model(tensor_img.unsqueeze(0))
        img_pred_sum += outputs
    img_pred_avg = img_pred_sum / tta_count
    _, pred_label = torch.max(img_pred_avg, 1)
    predictions.append([row.image_id, int(pred_label.item())])

sub_df = pd.DataFrame(predictions, columns=["image_id", "label"])
sub_df.to_csv("submission.csv", index=False)
print(sub_df.head())
