## === cell 2
class CassavaDataset(Dataset):
    """Dataset for Cassava images.

    For training it returns (image_tensor, label).
    For test it returns (image_tensor, image_id).
    """

    def __init__(self, img_dir, df=None, transform=None, is_test=False):
        self.img_dir = img_dir
        self.transform = transform
        self.is_test = is_test
        if df is not None:
            self.filenames = df["image_id"].tolist()
            self.labels = df["label"].tolist()
        else:
            self.filenames = sorted(os.listdir(img_dir))
            self.labels = None

    def __len__(self):
        return len(self.filenames)

    def __getitem__(self, idx):
        fname = self.filenames[idx]
        path = os.path.join(self.img_dir, fname)
        img = Image.open(path).convert("RGB")
        if self.transform:
            img = self.transform(img)
        if self.is_test:
            return img, fname
        else:
            return img, self.labels[idx]

train_img_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images/"

    train_df, [train_len, val_len], generator=torch.Generator().manual_seed(seed)

## === cell 4
train_transform = T.Compose(
    [
        T.RandomResizedCrop(224, scale=(0.8, 1.0)),
        T.RandomHorizontalFlip(),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)

test_transform = T.Compose(
    [
        T.Resize((224, 224)),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)

batch_size = 64
num_workers = 2

train_dataset = CassavaDataset(
    train_img_dir, df=train_set_df.dataset, transform=train_transform, is_test=False
)
val_dataset = CassavaDataset(
    train_img_dir, df=val_set_df.dataset, transform=test_transform, is_test=False
)

train_loader = DataLoader(
    train_dataset,
    batch_size=batch_size,
    shuffle=True,
    num_workers=num_workers,
    pin_memory=True,
)
val_loader = DataLoader(
    val_dataset,
    batch_size=batch_size,
    shuffle=False,
    num_workers=num_workers,
    pin_memory=True,
)



## === cell 5
model = models.vit_b_16(weights=models.ViT_B_16_Weights.IMAGENET1K_V1)
model.heads = nn.Linear(model.heads.in_features, 5)  # 5 classes
model = model.to(device)

criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.AdamW(model.parameters(), lr=5e-4, weight_decay=1e-4)


def train_one_epoch(loader):
    model.train()
    total_loss = 0.0
    for imgs, labels in loader:
        imgs = imgs.to(device)
        labels = labels.to(device)
        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        total_loss += loss.item() * imgs.size(0)
    return total_loss / len(loader.dataset)


def evaluate(loader):
    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for imgs, labels in loader:
            imgs = imgs.to(device)
            labels = labels.to(device)
            outputs = model(imgs)
            preds = outputs.argmax(dim=1)
            correct += (preds == labels).sum().item()
            total += labels.size(0)
    return correct / total


epochs = 2
for epoch in range(1, epochs + 1):
    train_loss = train_one_epoch(train_loader)
    val_acc = evaluate(val_loader)
    print(f"Epoch {epoch}: Train loss {train_loss:.4f} | Val Acc {val_acc:.4f}")
