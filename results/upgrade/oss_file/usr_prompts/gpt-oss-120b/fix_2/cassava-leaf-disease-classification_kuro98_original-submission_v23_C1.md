import os

torch.manual_seed(3407)
torch.cuda.manual_seed(3407)
cudnn.deterministic = True
cudnn.benchmark = False

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Device:", device)

test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"
img_size = 384
batch_size = 16
num_workers = 4
num_classes = 5
tta = False  # disabled to keep inference simple and reliable

model_path = "/kaggle/input/efficient-net/vit_cont_3.pt"
if os.path.exists(model_path):
    try:
        model = torch.load(model_path, map_location=device)
        print("Loaded model from:", model_path)
    except Exception as e:
        print(
            f"Failed to load checkpoint ({e}), falling back to pretrained EfficientNet‑B4."
        )
        model = None
else:
    model = None
    print("Checkpoint not found, using pretrained EfficientNet‑B4.")

if model is None:
    eff = models.efficientnet_b4(weights=models.EfficientNet_B4_Weights.IMAGENET1K_V1)
    in_features = eff.classifier[1].in_features
    eff.classifier[1] = nn.Linear(in_features, num_classes)
    model = eff

model.to(device)
model.eval()




## === cell 1
class CassavaDataset(VisionDataset):
    """Custom dataset for the Cassava test images."""

    def __init__(self, data_dir, transform=None, ttas=None, img_size=384):
        super().__init__(root=data_dir)
        self.transform = transform
        self.ttas = ttas
        self.images = sorted(os.listdir(data_dir))  # deterministic order
        self.cc = v2.CenterCrop((400, 400))

    def __getitem__(self, idx):
        filename = self.images[idx]
        img_path = os.path.join(self.root, filename)
        img = Image.open(img_path).convert("RGB")
        img = self.cc(img)

        if self.ttas is not None and self.transform is not None:
            img = [self.transform(t(img)) for t in self.ttas]
        elif self.transform:
            img = self.transform(img)

        return img, filename

    def __len__(self):
        return len(self.images)




## === cell 2
test_transforms = v2.Compose(
    [
        v2.Resize((img_size, img_size), interpolation=InterpolationMode.BICUBIC),
        v2.ToImage(),
        v2.ToDtype(torch.float32, scale=True),
        v2.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)

ttas = None

test_dataset = CassavaDataset(test_dir, transform=test_transforms, ttas=ttas)

test_loader = DataLoader(
    test_dataset,
    batch_size=batch_size,
    shuffle=False,  # keep order to match filenames with predictions
    num_workers=num_workers,
    pin_memory=True,
)

normalizer = torch.nn.Softmax(dim=1)



## === cell 3
all_names = []
all_preds = []

with torch.no_grad():
    for inputs, filenames in test_loader:
        if isinstance(inputs, list):
            inputs = torch.cat(inputs, dim=0)
        inputs = inputs.to(device)

        logits = normalizer(model(inputs))
        if tta:
            batch_preds = torch.stack(torch.split(logits, len(filenames)), dim=0)
            logits = torch.mean(batch_preds, dim=0)

        pred_labels = torch.argmax(logits, dim=1).cpu().tolist()
        all_names.extend(filenames)
        all_preds.extend(pred_labels)
