## === cell 1
ROOT_DIR = "../input/cassava-leaf-disease-classification/"
TRAIN_CSV = os.path.join(ROOT_DIR, "train.csv")
TEST_DIR = os.path.join(ROOT_DIR, "test_images/")

train_df = pd.read_csv(TRAIN_CSV)



## === cell 2
IMG_SIZE = 64  # Small size for quick training
BATCH_SIZE = 32
EPOCHS = 5
NUM_CLASSES = 5


def load_and_preprocess(image_path):
    """Load an image file, resize, and convert to an array."""
    img = Image.open(image_path).convert("RGB")
    img = img.resize((IMG_SIZE, IMG_SIZE))
    return np.array(img) / 255.0  # normalize


train_images = []
train_labels = []

for _, row in train_df.iterrows():
    img_path = os.path.join(ROOT_DIR, "train_images", row["image_id"])
    if os.path.exists(img_path):
        train_images.append(load_and_preprocess(img_path))
        train_labels.append(row["label"])
    else:
        continue

X = np.stack(train_images, axis=0).astype("float32")
y = tf.keras.utils.to_categorical(train_labels, NUM_CLASSES)

X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, random_state=42, stratify=y
)



## === cell 3
model = models.Sequential(
    [
        layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3)),
        layers.Conv2D(32, (3, 3), activation="relu"),
        layers.MaxPooling2D(),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D(),
        layers.Conv2D(128, (3, 3), activation="relu"),
        layers.GlobalAveragePooling2D(),
        layers.Dense(64, activation="relu"),
        layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

model.compile(optimizer="adam", loss="categorical_crossentropy", metrics=["accuracy"])

model.fit(
    X_train,
    y_train,
    validation_data=(X_val, y_val),
    epochs=EPOCHS,
    batch_size=BATCH_SIZE,
    verbose=2,
)



## === cell 4
test_images = sorted(os.listdir(TEST_DIR))  # ensure deterministic order
test_preds = []

for img_name in test_images:
    img_path = os.path.join(TEST_DIR, img_name)
    img_arr = load_and_preprocess(img_path)
    img_arr = np.expand_dims(img_arr, axis=0)  # batch dimension
    pred = model.predict(img_arr, verbose=0)
    label = int(np.argmax(pred, axis=1)[0])
    test_preds.append(label)
