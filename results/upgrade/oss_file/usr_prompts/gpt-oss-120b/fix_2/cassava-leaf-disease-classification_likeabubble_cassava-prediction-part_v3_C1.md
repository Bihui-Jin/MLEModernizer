def parse_train_example(example):
    """Parse a training TFRecord example."""
    schema = {
        "image": tf.io.FixedLenFeature([], tf.string),
        "image_name": tf.io.FixedLenFeature([], tf.string),
        "label": tf.io.FixedLenFeature([], tf.int64),
    }
    parsed = tf.io.parse_single_example(example, schema)
    image = tf.image.decode_jpeg(parsed["image"], channels=3)
    image = tf.cast(image, tf.float32) / 255.0
    image = tf.image.resize(image, (224, 224))  # MobileNetV2 size
    label = tf.one_hot(parsed["label"], depth=5)  # 5 classes
    return image, label


def parse_test_example(example):
    """Parse a test TFRecord example (no label)."""
    schema = {
        "image": tf.io.FixedLenFeature([], tf.string),
        "image_name": tf.io.FixedLenFeature([], tf.string),
    }
    parsed = tf.io.parse_single_example(example, schema)
    image = tf.image.decode_jpeg(parsed["image"], channels=3)
    image = tf.cast(image, tf.float32) / 255.0
    image = tf.image.resize(image, (224, 224))
    name = parsed["image_name"]
    return image, name
