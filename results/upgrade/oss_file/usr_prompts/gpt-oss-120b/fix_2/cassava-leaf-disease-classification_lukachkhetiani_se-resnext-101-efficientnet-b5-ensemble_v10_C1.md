## === cell 0
import os, glob, cv2, tqdm
import torch
import pandas as pd
import numpy as np
import timm

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))

hrnet = timm.create_model("seresnext101_32x4d", num_classes=5, pretrained=False)
hrnet_path = "../input/ensemblev3/seresnext_model_last.pth"
if os.path.exists(hrnet_path):
    hrnet.load_state_dict(torch.load(hrnet_path, map_location=device))
else:
    raise FileNotFoundError(f"HRNet weights not found at {hrnet_path}")
hrnet.to(device).eval()
use_hrnet = True

use_efficient = False
try:
    from efficientnet_pytorch import EfficientNet

    efficient = EfficientNet.from_name("efficientnet-b5", num_classes=5)
    eff_path = "../input/ensemblev3/eff_model_last.pth"
    if os.path.exists(eff_path):
        efficient.load_state_dict(torch.load(eff_path, map_location=device))
        efficient.to(device).eval()
        use_efficient = True
    else:
        print("EfficientNet weights not found; skipping EfficientNet.")
except Exception as e:
    print(f"EfficientNet could not be loaded ({e}); continuing without it.")

print(f"Models loaded – HRNet: {use_hrnet}, EfficientNet: {use_efficient}")

## === cell 1
test_dir = "../input/cassava-leaf-disease-classification/test_images"
files = glob.glob(os.path.join(test_dir, "*"))
names, preds = [], []


def preprocess(img_bgr):
    """Resize, apply CLAHE on the L channel, and convert to a torch tensor."""
    img = cv2.resize(img_bgr, (299, 299))
    lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)
    l = clahe.apply(l)
    lab = cv2.merge((l, a, b))
    img = cv2.cvtColor(lab, cv2.COLOR_LAB2BGR)
    tensor = (
        torch.tensor(img.transpose(2, 0, 1), dtype=torch.float).unsqueeze(0).to(device)
    )
    return tensor




## === cell 2
for file_path in tqdm.tqdm(files, desc="Predicting"):
    img = cv2.imread(file_path)
    if img is None:
        continue  # skip unreadable files
    input_tensor = preprocess(img)

    with torch.no_grad():
        hr_out = hrnet(input_tensor)
        if use_efficient:
            eff_out = efficient(input_tensor)
            total = (hr_out + eff_out) / 2.0
        else:
            total = hr_out

        prob = torch.nn.functional.softmax(total, dim=1)
        pred_label = int(torch.argmax(prob, dim=1).cpu().item())

    names.append(os.path.basename(file_path))
    preds.append(pred_label)
