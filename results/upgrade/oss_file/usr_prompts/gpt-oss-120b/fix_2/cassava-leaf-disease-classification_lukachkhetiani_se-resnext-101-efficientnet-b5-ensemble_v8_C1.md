## === cell 1
test_path = "../input/cassava-leaf-disease-classification/test_images/*"
files = glob.glob(test_path)
names, labels = [], []


def process(image):
    img = cv2.resize(image, (384, 384))
    lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)
    l = clahe.apply(l)
    lab = cv2.merge((l, a, b))
    img = cv2.cvtColor(lab, cv2.COLOR_LAB2BGR)
    tensor = torch.tensor(img.transpose(2, 0, 1), dtype=torch.float).unsqueeze(0).cuda()
    return tensor


with torch.no_grad():
    for file in tqdm.tqdm(files):
        img = cv2.imread(file)
        if img is None:
            continue
        topred = process(img)
        hr_out = hrnet(topred)
        eff_out = efficient(topred)
        total = (hr_out + eff_out) / 2
        prob = F.softmax(total, dim=1)
        pred_label = int(torch.argmax(prob, dim=1).cpu().item())
        names.append(os.path.basename(file))
        labels.append(pred_label)
