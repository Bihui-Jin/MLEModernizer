## === cell 0
from pathlib import Path
import pandas as pd
import os
from fastai.vision.all import *

base_path = Path("/kaggle/input/cassava-leaf-disease-classification")
train_csv_path = base_path / "train.csv"
train_images_path = base_path / "train_images"
test_images_path = base_path / "test_images"



## === cell 1
df = pd.read_csv(train_csv_path)

df["label"] = df["label"].astype(str)

cassava_block = DataBlock(
    blocks=(ImageBlock, CategoryBlock),
    get_x=lambda row: train_images_path / row["image_id"],
    get_y=ColReader("label"),
    splitter=RandomSplitter(valid_pct=0.2, seed=42),
    item_tfms=Resize(224),
    batch_tfms=aug_transforms(mult=1.0),
)

dls = cassava_block.dataloaders(df, bs=64)



## === cell 2
learn = vision_learner(dls, resnet34, metrics=accuracy)

learn.fine_tune(3)



## === cell 3
submission_path = Path("submission.csv")
pred_df = pd.DataFrame(columns=["image_id", "label"])

for root, _, files in os.walk(test_images_path):
    for file_name in files:
        img_path = Path(root) / file_name
        pred, _, _ = learn.predict(img_path)
        pred_df = pred_df.append(
            {"image_id": file_name, "label": int(pred)}, ignore_index=True
        )


pred_df.to_csv(submission_path, index=False)
