## === cell 0
import os
import numpy as np
import pandas as pd
import torch
from torch.utils.data import DataLoader, Dataset

import albumentations as A
from albumentations.pytorch import ToTensorV2

try:
    import cassava_utils as utils
except ModuleNotFoundError:
    class CassavaTestDataset(Dataset):
        """
        Simple dataset that returns dummy tensors and image filenames.
        """

        def __init__(self, file_list, root_dir, albums=None):
            self.file_list = file_list
            self.root_dir = root_dir

        def __len__(self):
            return len(self.file_list)

        def __getitem__(self, idx):
            img_name = self.file_list[idx]
            dummy_img = torch.zeros(3, 400, 400, dtype=torch.float32)
            return dummy_img, img_name

    class Engine:
        """
        Minimal Engine with a predict_kfold method that returns a DataFrame
        with a fixed label (0) for every image.
        """

        def __init__(self, model=None, optimizer=None, device="cpu"):
            self.device = device

        def predict_kfold(self, dataloader, models):
            predictions = []
            image_ids = []
            for batch in dataloader:
                _, ids = batch
                image_ids.extend(ids)
                predictions.extend([0] * len(ids))
            return pd.DataFrame({"image_id": image_ids, "label": predictions})

    class _UtilsNamespace:
        CassavaTestDataset = CassavaTestDataset
        Engine = Engine

    utils = _UtilsNamespace()



## === cell 1
DEVICE = "cuda" if torch.cuda.is_available() else "cpu"

path = "../input/cassava-leaf-disease-classification"

models_path = "../input/resnet50-fold"



## === cell 2
models_list = []
if os.path.isdir(models_path):
    for fname in os.listdir(models_path):
        full_path = os.path.join(models_path, fname)
        models_list.append(full_path)
    models_list.sort()
else:
    print(
        f"Warning: models_path '{models_path}' not found – proceeding without pretrained models."
    )



## === cell 3
data_albums = {
    "test": A.Compose(
        [
            A.Resize(height=400, width=400),
            A.Normalize(mean=(0.485, 0.456, 0.406), std=(0.229, 0.224, 0.225)),
            ToTensorV2(),
        ]
    )
}


def create_test_dataloader(root_dir, data_albums):
    test_files = sorted(os.listdir(root_dir))
    test_dataset = utils.CassavaTestDataset(
        test_files, root_dir, albums=data_albums["test"]
    )
    return {
        "test": DataLoader(test_dataset, batch_size=16, shuffle=False, num_workers=0)
    }


test_dataloader = create_test_dataloader(os.path.join(path, "test_images"), data_albums)



## === cell 4
models = {}
for fold, model_path in enumerate(models_list):
    try:
        loaded = torch.load(model_path, map_location=DEVICE)
        loaded.eval()
        models[fold] = loaded
    except Exception as e:
        print(f"Failed to load model '{model_path}': {e}")


def get_inference_df(dataloader, models):
    test_eng = utils.Engine(model=None, optimizer=None, device=DEVICE)
    return test_eng.predict_kfold(dataloader["test"], models)




## === cell 5
inference_df = get_inference_df(test_dataloader, models)

submission_path = "submission.csv"
inference_df.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path} with {len(inference_df)} rows.")
