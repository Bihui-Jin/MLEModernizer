## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image




## === cell 1
work_dir = "../input/cassava-leaf-disease-classification/"
os.listdir(work_dir)
train_path = "/kaggle/input/cassava-leaf-disease-classification/train_images"




## === cell 2
data = pd.read_csv(work_dir + "train.csv")




## === cell 3
train_dir = os.path.join(work_dir, "train_images")
max_per_class = 200  # limit samples per class for speed
class_sums = {}
class_counts = {}
samples_per_class = {c: 0 for c in range(5)}

for _, row in data.iterrows():
    label = int(row["label"])
    if samples_per_class[label] >= max_per_class:
        continue
    img_path = os.path.join(train_dir, row["image_id"])
    try:
        img = Image.open(img_path).convert("RGB")
        img = img.resize((32, 32))  # small size for quick stats
        mean_rgb = np.asarray(img).mean(axis=(0, 1))  # average R,G,B
        class_sums[label] = class_sums.get(label, 0) + mean_rgb
        class_counts[label] = class_counts.get(label, 0) + 1
        samples_per_class[label] += 1
    except Exception:
        continue

centroids = {label: class_sums[label] / class_counts[label] for label in class_sums}




## === cell 4
TEST_DIR = os.path.join(work_dir, "test_images/")
test_images = os.listdir(TEST_DIR)
predictions = []

for img_name in test_images:
    img_path = os.path.join(TEST_DIR, img_name)
    try:
        img = Image.open(img_path).convert("RGB")
        img = img.resize((32, 32))
        mean_rgb = np.asarray(img).mean(axis=(0, 1))
        distances = {
            lbl: np.linalg.norm(mean_rgb - centroids[lbl]) for lbl in centroids
        }
        pred_label = min(distances, key=distances.get)
    except Exception:
        pred_label = 0  # fallback to a default class
    predictions.append(int(pred_label))

sub = pd.DataFrame({"image_id": test_images, "label": predictions})
sub.to_csv("submission.csv", index=False)
