## === cell 1
import os
import pandas as pd
import cv2 as cv
from pathlib import Path

import torch
import torch.nn as nn
from torch.utils.data import DataLoader, Dataset

from torchvision import models

from albumentations import Compose, Normalize
from albumentations.pytorch import ToTensorV2



## === cell 2
device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")




## === cell 3
class Config:
    cfg = {
        "batch_size": 32,
        "num_workers": 4,
        "image_size": (512, 512),
        "num_classes": 5,
        "model_path": "/kaggle/input/effnetb0f3/3_fold_model_effnet_b0_best.torch",
    }




## === cell 4
base_dir = Path("/kaggle/input/cassava-leaf-disease-classification")
test_img_dir = base_dir / "test_images"
test_df = pd.read_csv(base_dir / "sample_submission.csv", index_col=0)




## === cell 5
class CassavaDataset(Dataset):
    def __init__(self, df, image_size, augments=None):
        self.filenames = df.index.tolist()
        self.image_size = image_size
        self.augments = augments

    def __getitem__(self, idx):
        img_path = os.path.join(test_img_dir, self.filenames[idx])
        image = cv.imread(img_path)
        if image is None:
            raise FileNotFoundError(f"Image not found: {img_path}")
        image = cv.resize(image, self.image_size)
        image = cv.cvtColor(image, cv.COLOR_BGR2RGB)

        if self.augments:
            image = self.augments(image=image)["image"]

        return {"X": image}

    def __len__(self):
        return len(self.filenames)




## === cell 6
class Augments:
    test_augments = Compose(
        [
            Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225], p=1.0),
            ToTensorV2(p=1.0),
        ],
        p=1.0,
    )




## === cell 7
test_dataset = CassavaDataset(
    df=test_df, image_size=Config.cfg["image_size"], augments=Augments.test_augments
)

test_dataloader = DataLoader(
    test_dataset,
    batch_size=Config.cfg["batch_size"],
    shuffle=False,
    num_workers=Config.cfg["num_workers"],
)




## === cell 8
def efficientnet_b0(num_classes):
    model = models.efficientnet_b0(weights=models.EfficientNet_B0_Weights.IMAGENET1K_V1)
    in_features = model.classifier[1].in_features
    model.classifier[1] = nn.Linear(in_features, num_classes)
    return model




## === cell 9
model = efficientnet_b0(Config.cfg["num_classes"]).to(device)

ckpt_path = Config.cfg["model_path"]
if os.path.isfile(ckpt_path):
    checkpoint = torch.load(ckpt_path, map_location=device)
    model.load_state_dict(checkpoint["model_state_dict"])
    best_score = checkpoint.get("best_valid_score", None)
    epoch_num = checkpoint.get("num_epoch", None)
    print(f"Loaded checkpoint – best score: {best_score}, epoch: {epoch_num}")
else:
    print("Checkpoint not found – using ImageNet pretrained weights.")

model.eval()



## === cell 10
y_prediction = []
with torch.no_grad():
    for batch in test_dataloader:
        X_test = batch["X"].to(device)
        logits = model(X_test)
        preds = logits.argmax(dim=-1).cpu().numpy()
        y_prediction.extend(preds)

print(f"Generated predictions for {len(y_prediction)} images.")



## === cell 11
test_df["label"] = y_prediction
submission_path = "submission.csv"
test_df.to_csv(submission_path)
print(f"Submission file written to {submission_path}")
