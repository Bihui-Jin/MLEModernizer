import os
import pandas as pd
import cv2 as cv
from pathlib import Path

import torch
import torch.nn as nn
from torch.utils.data import DataLoader, Dataset
import torchvision.models as models

from albumentations import Compose, Normalize
from albumentations.pytorch import ToTensorV2




## === cell 1
class Config:
    cfg = {
        "batch_size": 32,
        "num_workers": 4,
        "image_size": (512, 512),
        "num_classes": 5,
        "epochs": 1,  # quick fine‑tune
        "lr": 1e-4,
        "model_path": None,  # not used; we train on‑the‑fly
    }


base_dir = Path("/kaggle/input/cassava-leaf-disease-classification")
train_csv = base_dir / "train.csv"
test_csv = base_dir / "sample_submission.csv"
train_img_dir = base_dir / "train_images"
test_img_dir = base_dir / "test_images"

device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")




## === cell 2
class CassavaDataset(Dataset):
    def __init__(self, df, img_dir, image_size, mode="test", augments=None):
        """
        mode='train' returns (image, label)
        mode='test' returns (image, image_id)
        """
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.image_size = image_size
        self.mode = mode
        self.augments = augments

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = os.path.join(self.img_dir, row["image_id"])
        image = cv.imread(img_path)
        image = cv.resize(image, self.image_size)
        image = cv.cvtColor(image, cv.COLOR_BGR2RGB)

        if self.augments:
            image = self.augments(image=image)["image"]

        if self.mode == "train":
            label = int(row["label"])
            return {"X": image, "y": label}
        else:
            return {"X": image, "image_id": row["image_id"]}




## === cell 3
class Augments:
    train_aug = Compose(
        [
            Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225], p=1.0),
            ToTensorV2(p=1.0),
        ],
        p=1.0,
    )

    test_aug = Compose(
        [
            Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225], p=1.0),
            ToTensorV2(p=1.0),
        ],
        p=1.0,
    )




## === cell 4
train_df = pd.read_csv(train_csv)
train_dataset = CassavaDataset(
    df=train_df,
    img_dir=train_img_dir,
    image_size=Config.cfg["image_size"],
    mode="train",
    augments=Augments.train_aug,
)

train_loader = DataLoader(
    train_dataset,
    batch_size=Config.cfg["batch_size"],
    shuffle=True,
    num_workers=Config.cfg["num_workers"],
    pin_memory=True,
)




## === cell 5
def efficientnet_b0(num_classes):
    model = models.efficientnet_b0(pretrained=True)
    model.classifier[1] = nn.Linear(
        in_features=model.classifier[1].in_features, out_features=num_classes, bias=True
    )
    return model




## === cell 6
model = efficientnet_b0(Config.cfg["num_classes"]).to(device)
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=Config.cfg["lr"])



## === cell 7
model.train()
for epoch in range(Config.cfg["epochs"]):
    epoch_loss = 0.0
    for batch in train_loader:
        images = batch["X"].to(device)
        labels = batch["y"].to(device)

        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        epoch_loss += loss.item() * images.size(0)

    avg_loss = epoch_loss / len(train_loader.dataset)
    print(f'Epoch [{epoch+1}/{Config.cfg["epochs"]}] - Loss: {avg_loss:.4f}')



## === cell 8
test_df = pd.read_csv(test_csv, index_col=0)
test_dataset = CassavaDataset(
    df=test_df,
    img_dir=test_img_dir,
    image_size=Config.cfg["image_size"],
    mode="test",
    augments=Augments.test_aug,
)

test_loader = DataLoader(
    test_dataset,
    batch_size=Config.cfg["batch_size"],
    shuffle=False,
    num_workers=Config.cfg["num_workers"],
    pin_memory=True,
)



## === cell 9
model.eval()
predictions = []

with torch.no_grad():
    for batch in test_loader:

        preds = outputs.argmax(dim=1).cpu().numpy()
        predictions.extend(preds)

test_df["label"] = predictions
submission_path = "submission.csv"
test_df.to_csv(submission_path)
print(f"Submission written to {submission_path}")
