## === cell 6
class CassavaDataset(Dataset):
    def __init__(self, df, image_size, augments=None):
        self.image_ids = df.index.tolist()
        self.image_size = image_size
        self.augments = augments

    def __len__(self):
        return len(self.image_ids)

    def __getitem__(self, idx):
        img_name = self.image_ids[idx]
        img_path = test_img_dir / img_name
        image = cv.imread(str(img_path))
        if image is None:
            image = np.zeros((*self.image_size, 3), dtype=np.uint8)
        else:
            image = cv.resize(image, self.image_size)
            image = cv.cvtColor(image, cv.COLOR_BGR2RGB)

        if self.augments:
            image = self.augments(image=image)["image"]

        return {"X": image}




## === cell 7
test_dataset = CassavaDataset(
    df=test_df, image_size=Config.cfg["image_size"], augments=Augments.test_augments
)

test_dataloader = DataLoader(
    test_dataset,
    batch_size=Config.cfg["batch_size"],
    shuffle=False,
    num_workers=Config.cfg["num_workers"],
    pin_memory=True,
)




## === cell 8
def build_efficientnet_b0(num_classes: int):
    weights = EfficientNet_B0_Weights.DEFAULT
    model = efficientnet_b0(weights=weights)
    in_features = model.classifier[1].in_features
    model.classifier[1] = nn.Linear(in_features, num_classes)
    return model




## === cell 9
model = build_efficientnet_b0(Config.cfg["num_classes"]).to(device)
model.eval()



## === cell 10
y_prediction = []

with torch.no_grad():
    for batch in test_dataloader:
        X_test = batch["X"].to(device)
        logits = model(X_test)
        preds = logits.argmax(dim=-1).cpu().numpy()
        y_prediction.extend(preds)

print(f"Generated predictions for {len(y_prediction)} images.")



## === cell 11
submission = test_df.copy()
submission["label"] = y_prediction
submission.reset_index(inplace=True)  # turn index into a column named 'image_id'
