def predict_image(img_path):
    """Return a predicted label for a single image."""
    img = cv2.cvtColor(cv2.imread(img_path), cv2.COLOR_BGR2RGB)
    img_resized = cv2.resize(img, (512, 512))
    img_batch = np.expand_dims(img_resized, axis=0)

    if model is not None or model2 is not None:
        preds = []
        if model is not None:
            preds.append(model.predict(img_batch))
        if model2 is not None:
            preds.append(model2.predict(img_batch))
        avg_pred = np.mean(preds, axis=0)
        return int(np.argmax(avg_pred, axis=1)[0])
    else:
        return int(most_common_label)




## === cell 7
predicted_labels = []
for img_name in test_images:
    img_path = os.path.join(TEST_DIR, img_name)
    pred_label = predict_image(img_path)
    predicted_labels.append(pred_label)



## === cell 8
submission_df = pd.DataFrame({"image_id": test_images, "label": predicted_labels})
submission_df.to_csv(SUBMISSION_PATH, index=False)
