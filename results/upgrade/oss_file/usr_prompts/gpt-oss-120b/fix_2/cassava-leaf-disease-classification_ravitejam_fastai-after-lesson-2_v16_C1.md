## === cell 0
import os
import shutil
from pathlib import Path

import numpy as np
import pandas as pd
import torch
from fastai.vision.all import *
from IPython.core.interactiveshell import InteractiveShell

InteractiveShell.ast_node_interactivity = "all"

train = True



## === cell 1
data_path = Path("../input/cassava-leaf-disease-classification")
assert data_path.exists(), f"Data path {data_path} not found"



## === cell 2
labels = pd.read_csv(data_path / "train.csv")
labels["image_id"] = labels["image_id"].apply(lambda x: f"train_images/{x}")



## === cell 3
dls = ImageDataLoaders.from_df(
    df=labels,
    path=data_path,
    fn_col="image_id",
    label_col="label",
    valid_pct=0.2,
    seed=42,
    bs=64,
    item_tfms=RandomResizedCrop(512, min_scale=0.75, ratio=(1.0, 1.0)),
    batch_tfms=aug_transforms(),
    num_workers=0,
)



## === cell 4
learn = cnn_learner(dls, resnet50, metrics=[error_rate, accuracy]).to_fp16()



## === cell 5
if train:
    learn.fine_tune(1, cbs=[MixUp(0.5)])  # one epoch is enough to produce predictions



## === cell 6
test_df = pd.read_csv(data_path / "sample_submission.csv")
test_df["image_id"] = test_df["image_id"].apply(lambda x: f"test_images/{x}")

test_dl = dls.test_dl(test_df, with_labels=False)



## === cell 7
preds, _ = learn.get_preds(dl=test_dl)
pred_classes = preds.argmax(dim=1).cpu().numpy()



## === cell 8
submission = pd.read_csv(data_path / "sample_submission.csv")
submission["label"] = pred_classes
submission.to_csv("submission.csv", index=False)
