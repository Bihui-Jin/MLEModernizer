## === cell 0
import pandas as pd
import numpy as np
import glob
import os
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from PIL import Image
import warnings

warnings.filterwarnings("ignore")

SEED = 42
np.random.seed(SEED)



## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TEST_CSV = os.path.join(BASE_PATH, "sample_submission.csv")  # only for column template

df_train = pd.read_csv(TRAIN_CSV)
df_train["filepath"] = df_train["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)




## === cell 2
def load_and_preprocess(image_path, size=(64, 64)):
    """Load an image, resize, and flatten to a 1‑D numpy array."""
    try:
        img = Image.open(image_path).convert("RGB")
        img = img.resize(size)
        return np.asarray(img).flatten()
    except Exception:
        return np.zeros(size[0] * size[1] * 3, dtype=np.uint8)


SUBSET_SIZE = 5000  # adjust if needed; must be <= total rows
df_subset = df_train.sample(n=SUBSET_SIZE, random_state=SEED)

X_train = np.stack(df_subset["filepath"].apply(load_and_preprocess).values)
y_train = df_subset["label"].values



## === cell 3
rf_clf = RandomForestClassifier(
    n_estimators=150,
    max_depth=None,
    random_state=SEED,
    n_jobs=-1,
    class_weight="balanced",
)
rf_clf.fit(X_train, y_train)



## === cell 4
test_image_paths = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
df_test = pd.DataFrame(test_image_paths, columns=["path"])
df_test["image_id"] = df_test["path"].apply(lambda p: os.path.basename(p))

X_test = np.stack(df_test["path"].apply(load_and_preprocess).values)

test_pred = rf_clf.predict(X_test)



## === cell 5
submission = df_test[["image_id"]].copy()
submission["label"] = test_pred.astype(int)

sample_sub = pd.read_csv(TEST_CSV, nrows=5)
assert list(sample_sub.columns) == ["image_id", "label"], "Column mismatch!"

submission.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv")
