# 5. Code solution

## === cell 0
import pandas as pd
import numpy as np
import tensorflow as tf
import glob
from tensorflow.keras.models import Sequential
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from sklearn.model_selection import train_test_split

SEED = 42
BATCH_SIZE = 32
IMG_SIZE = (224, 224)  # smaller size for faster training
EPOCHS = 3
tf.random.set_seed(SEED)



## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification/"
TRAIN_IMG_DIR = f"{BASE_PATH}train_images/"
TEST_IMG_DIR = f"{BASE_PATH}test_images/"
TRAIN_CSV = f"{BASE_PATH}train.csv"

df_train = pd.read_csv(TRAIN_CSV)

df_train["full_path"] = TRAIN_IMG_DIR + df_train["image_id"]

df_tr, df_val = train_test_split(
    df_train, test_size=0.1, random_state=SEED, stratify=df_train["label"]
)



## === cell 2
train_idg = ImageDataGenerator(
    rescale=1.0 / 255,
    horizontal_flip=True,
    rotation_range=20,
    width_shift_range=0.1,
    height_shift_range=0.1,
    zoom_range=0.1,
)

val_idg = ImageDataGenerator(rescale=1.0 / 255)

train_gen = train_idg.flow_from_dataframe(
    dataframe=df_tr,
    x_col="full_path",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="raw",  # yields labels as scalars
    shuffle=True,
    seed=SEED,
)

val_gen = val_idg.flow_from_dataframe(
    dataframe=df_val,
    x_col="full_path",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="raw",
    shuffle=False,
)



## === cell 3
base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=IMG_SIZE + (3,)
)
base_model.trainable = False  # freeze backbone initially

model = Sequential(
    [
        base_model,
        tf.keras.layers.GlobalAveragePooling2D(),
        tf.keras.layers.Dense(5, activation="softmax"),
    ]
)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 4
model.fit(train_gen, epochs=EPOCHS, validation_data=val_gen, verbose=1)



## === cell 5
test_images = glob.glob(TEST_IMG_DIR + "*.jpg")
df_test = pd.DataFrame({"path": test_images})
df_test["image_id"] = df_test["path"].str.split("/").str[-1]

test_idg = ImageDataGenerator(rescale=1.0 / 255)
test_gen = test_idg.flow_from_dataframe(
    dataframe=df_test,
    x_col="path",
    y_col=None,
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode=None,
    shuffle=False,
)

pred_probs = model.predict(test_gen, verbose=1)
pred_labels = np.argmax(pred_probs, axis=1)



## === cell 6
submission = pd.DataFrame({"image_id": df_test["image_id"], "label": pred_labels})

submission = submission[["image_id", "label"]]

submission.to_csv("submission.csv", index=False)



## === cell 7
print(submission.head())
