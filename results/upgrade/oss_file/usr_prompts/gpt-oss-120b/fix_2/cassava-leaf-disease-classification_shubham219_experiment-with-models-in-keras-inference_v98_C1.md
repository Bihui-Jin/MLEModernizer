## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUBMISSION = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["path"] = df_train["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))

df_train, df_val = train_test_split(
    df_train, test_size=0.1, random_state=SEED, stratify=df_train["label"]
)

preprocess = tf.keras.applications.resnet50.preprocess_input
train_idg = ImageDataGenerator(
    preprocessing_function=preprocess,
    horizontal_flip=True,
    rotation_range=20,
    width_shift_range=0.1,
    height_shift_range=0.1,
    zoom_range=0.2,
)

val_idg = ImageDataGenerator(preprocessing_function=preprocess)

train_gen = train_idg.flow_from_dataframe(
    dataframe=df_train,
    x_col="path",
    y_col="label",
    target_size=(256, 256),
    batch_size=32,
    class_mode="categorical",
    shuffle=True,
    seed=SEED,
)

val_gen = val_idg.flow_from_dataframe(
    dataframe=df_val,
    x_col="path",
    y_col="label",
    target_size=(256, 256),
    batch_size=32,
    class_mode="categorical",
    shuffle=False,
)

weight_path = "/kaggle/input/resnet50-trasnfer-learning-on-tpu/cassava_base.h5"
if os.path.exists(weight_path):
    my_model = tf.keras.models.load_model(weight_path)
else:
    base = tf.keras.applications.ResNet50(
        weights="imagenet", include_top=False, input_shape=(256, 256, 3)
    )
    x = tf.keras.layers.GlobalAveragePooling2D()(base.output)
    output = tf.keras.layers.Dense(5, activation="softmax")(x)
    my_model = tf.keras.Model(inputs=base.input, outputs=output)

    my_model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
    )

    my_model.fit(
        train_gen,
        epochs=2,
        validation_data=val_gen,
        verbose=2,
    )
