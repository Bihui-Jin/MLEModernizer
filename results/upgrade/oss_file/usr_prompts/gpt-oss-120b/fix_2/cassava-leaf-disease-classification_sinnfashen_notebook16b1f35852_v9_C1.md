## === cell 0
import os
import glob
import numpy as np
import pandas as pd
from PIL import Image
import torch
import torchvision
from torch.utils.data import DataLoader, Dataset, random_split
from torchvision import transforms



## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"

TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")


def collect_jpg_files(root_dir):
    return [
        os.path.basename(p)
        for p in glob.glob(os.path.join(root_dir, "**", "*.jpg"), recursive=True)
    ]


train_files = pd.read_csv(TRAIN_CSV)["image_id"].tolist()
test_files = collect_jpg_files(TEST_IMG_DIR)



## === cell 2
normalize = transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
preprocess = transforms.Compose(
    [transforms.Resize((512, 512)), transforms.ToTensor(), normalize]
)


def default_loader(path):
    img = Image.open(path).convert("RGB")
    return preprocess(img)




## === cell 3
class CassavaDataset(Dataset):
    def __init__(
        self, files, labels=None, img_dir=TRAIN_IMG_DIR, loader=default_loader
    ):
        self.files = files
        self.labels = labels
        self.img_dir = img_dir
        self.loader = loader

    def __len__(self):
        return len(self.files)

    def __getitem__(self, idx):
        img_name = self.files[idx]
        img_path = os.path.join(self.img_dir, img_name)
        if not os.path.isfile(img_path):
            candidates = glob.glob(
                os.path.join(self.img_dir, "**", img_name), recursive=True
            )
            if candidates:
                img_path = candidates[0]
            else:
                raise FileNotFoundError(f"Image {img_name} not found.")
        img = self.loader(img_path)
        if self.labels is not None:
            return img, self.labels[idx]
        return img_name, img




## === cell 4
train_df = pd.read_csv(TRAIN_CSV)
label_map = train_df["label"].astype(int).values
train_dataset = CassavaDataset(train_files, label_map, img_dir=TRAIN_IMG_DIR)

train_len = int(0.9 * len(train_dataset))
val_len = len(train_dataset) - train_len
train_ds, val_ds = random_split(train_dataset, [train_len, val_len])

batch_size = 64
train_loader = DataLoader(
    train_ds, batch_size=batch_size, shuffle=True, num_workers=2, pin_memory=True
)
val_loader = DataLoader(
    val_ds, batch_size=batch_size, shuffle=False, num_workers=2, pin_memory=True
)



## === cell 5
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
MODEL_PATH = "../input/cassavaleaffsolmodels/resnet2.mdl"


def build_model():
    model = torchvision.models.resnet50(pretrained=True)
    model.fc = torch.nn.Linear(model.fc.in_features, 5)
    return model


model = build_model()
if os.path.isfile(MODEL_PATH):
    try:
        model.load_state_dict(torch.load(MODEL_PATH, map_location=device))
    except Exception as e:
        print(f"Failed to load custom checkpoint ({e}), using pretrained model.")
else:
    print("Custom model file not found; using pretrained ImageNet weights.")

model = model.to(device)



## === cell 6
criterion = torch.nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)


def train_one_epoch(loader):
    model.train()
    running_loss = 0.0
    for imgs, targets in loader:
        imgs = imgs.to(device)
        targets = targets.to(device)
        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, targets)
        loss.backward()
        optimizer.step()
        running_loss += loss.item() * imgs.size(0)
    return running_loss / len(loader.dataset)


def evaluate(loader):
    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for imgs, targets in loader:
            imgs = imgs.to(device)
            targets = targets.to(device)
            outputs = model(imgs)
            _, preds = torch.max(outputs, 1)
            correct += (preds == targets).sum().item()
            total += targets.size(0)
    return correct / total if total > 0 else 0.0


EPOCHS = 2
for epoch in range(EPOCHS):
    loss = train_one_epoch(train_loader)
    acc = evaluate(val_loader)
    print(f"Epoch {epoch+1}/{EPOCHS} - Loss: {loss:.4f} - Val Acc: {acc:.4f}")



## === cell 7
test_dataset = CassavaDataset(test_files, labels=None, img_dir=TEST_IMG_DIR)
test_loader = DataLoader(
    test_dataset, batch_size=32, shuffle=False, num_workers=2, pin_memory=True
)

model.eval()
pred_labels = []
pred_files = []
with torch.no_grad():
    for fnames, imgs in test_loader:
        imgs = imgs.to(device)
        logits = model(imgs)
        _, preds = torch.max(logits, 1)
        pred_labels.extend(preds.cpu().numpy().tolist())
        pred_files.extend(fnames)

submission = pd.DataFrame({"image_id": pred_files, "label": pred_labels})
submission = submission.sort_values("image_id").reset_index(drop=True)



## === cell 8
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}, shape: {submission.shape}")
