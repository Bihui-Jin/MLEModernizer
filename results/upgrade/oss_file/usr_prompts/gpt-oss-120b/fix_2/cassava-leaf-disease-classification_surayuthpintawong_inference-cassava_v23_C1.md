## === cell 1
class TestDataset(Dataset):
    def __init__(self, root_dir, transform=None):
        """
        Args:
            root_dir (string): Directory with all the images.
            transform (callable, optional): Optional transform to be applied
                on a sample.
        """
        self.root_dir = root_dir
        self.transform = transform
        self.images = sorted(os.listdir(root_dir))

    def __len__(self):
        return len(self.images)

    def __getitem__(self, idx):
        if torch.is_tensor(idx):
            idx = idx.tolist()
        img_name = self.images[idx]
        img_path = os.path.join(self.root_dir, img_name)
        image = cv2.imread(img_path)  # BGR
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)

        if self.transform:
            transformed = self.transform(image=image)
            image = transformed["image"]
        return img_name, image




## === cell 2
transform = A.Compose(
    [
        A.CenterCrop(width=512, height=512),
        A.Normalize(
            mean=(0.485, 0.456, 0.406),
            std=(0.229, 0.224, 0.225),
            max_pixel_value=255.0,
            p=1.0,
        ),
        ToTensorV2(),
    ]
)

test_dataset = TestDataset(
    root_dir="../input/cassava-leaf-disease-classification/test_images/",
    transform=transform,
)
testloader = DataLoader(test_dataset, batch_size=4, shuffle=False, num_workers=1)

try:
    from efficientnet_pytorch import EfficientNet

    model = EfficientNet.from_name(model_name, num_classes=5).to(device)
    checkpoint_path = os.path.join("..", "input", folder_name, f"{model_full_name}.pt")
    model.load_state_dict(torch.load(checkpoint_path, map_location=device))
except Exception:
    backbone = models.efficientnet_b4(
        weights=models.EfficientNet_B4_Weights.IMAGENET1K_V1
    ).to(device)
    backbone.classifier[1] = nn.Linear(backbone.classifier[1].in_features, 5).to(device)
    model = backbone

model.eval()



## === cell 3
names = []
predicted = []
with torch.no_grad():
    for names_batch, images_batch in testloader:
        images_batch = images_batch.to(device).float()
        outputs = model(images_batch)
        preds = torch.argmax(outputs, dim=1).cpu().numpy()
        names.extend(list(names_batch))
        predicted.extend(preds.tolist())
