df



## === cell 4
num_classes = len(df.label.unique())
num_classes




## === cell 5
class Model(nn.Module):
    def __init__(self):
        super(Model, self).__init__()
        try:
            base_model = models.resnet101(
                weights=models.ResNet101_Weights.IMAGENET1K_V1
            )
        except Exception:
            base_model = models.resnet101(pretrained=True)
        self.base = nn.Sequential(*list(base_model.children())[:-2])
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.dense = nn.Linear(2048, num_classes)

    def forward(self, x):
        x = self.base(x)
        x = self.pool(x)
        x = x.reshape(x.shape[0], -1)
        return self.dense(x)




## === cell 6
device = "cuda:0" if torch.cuda.is_available() else "cpu"



## === cell 7
model = Model().to(device)



## === cell 8
checkpoint_path = "/kaggle/input/resnet-cassava-model/model_101_20.pth"
if os.path.exists(checkpoint_path):
    try:
        state_dict = torch.load(checkpoint_path, map_location=device)
        model.load_state_dict(state_dict)
        print("Custom checkpoint loaded.")
    except Exception as e:
        print(f"Failed to load custom checkpoint: {e}")
else:
    print("Custom checkpoint not found; using ImageNet pretrained weights.")

submission_df




## === cell 11
class TestDataset(data.Dataset):
    def __init__(self, df, transforms):
        super(TestDataset, self).__init__()
        self.df = df
        self.transforms = transforms

    def __getitem__(self, index):
        row = self.df.iloc[index]
        img = Image.open(f"{input_path}/test_images/{row.image_id}")
        return self.transforms(img)

    def __len__(self):
        return len(self.df)




## === cell 12
test_transforms = T.Compose(
    [T.Resize((512, 512)), T.ToTensor(), T.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))]
)



## === cell 13
test_dataset = TestDataset(df=submission_df, transforms=test_transforms)
test_dataloader = data.DataLoader(dataset=test_dataset, shuffle=False, batch_size=12)



## === cell 14
model.eval()
test_preds = []
with torch.no_grad():
    for x in test_dataloader:
        x = x.to(device)  # respect CPU/GPU availability
        y_pred = model(x)
        test_preds += y_pred.cpu().numpy().argmax(axis=1).tolist()
test_preds = np.array(test_preds)



## === cell 15
submission_df["label"] = test_preds



## === cell 16
submission_df.to_csv("submission.csv", index=False)
