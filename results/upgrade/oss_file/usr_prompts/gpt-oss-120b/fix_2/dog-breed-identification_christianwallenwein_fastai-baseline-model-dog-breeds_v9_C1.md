## === cell 0
from fastai.vision.all import *
import pandas as pd
from pathlib import Path
import os

base_path = Path("../input")
train_path = base_path / "train"
test_path = base_path / "test"



## === cell 1
labels_path = base_path / "labels.csv"
labels_df = pd.read_csv(labels_path)
label_dict = dict(zip(labels_df["id"], labels_df["breed"]))


def label_func(fname: Path):
    """Return breed for a given image filename."""
    return label_dict.get(fname.stem, "unknown")




## === cell 2
dblock = DataBlock(
    blocks=(ImageBlock, CategoryBlock),
    get_items=get_image_files,
    splitter=RandomSplitter(valid_pct=0.2, seed=42),
    get_y=label_func,
    item_tfms=Resize(224),
    batch_tfms=aug_transforms(),
)

dls = dblock.dataloaders(train_path, bs=16, num_workers=0)



## === cell 3
learn = vision_learner(dls, resnet18, metrics=accuracy, model_dir="/tmp/models")
learn.fine_tune(1)  # one epoch as in the original script



## === cell 4
test_files = get_image_files(test_path)
test_dl = learn.dls.test_dl(test_files)
preds, _ = learn.get_preds(dl=test_dl)

sample_sub = pd.read_csv(base_path / "sample_submission.csv")
breed_cols = list(sample_sub.columns[1:])  # columns after 'id'

preds_df = pd.DataFrame(preds.numpy(), columns=learn.dls.vocab)
preds_df = preds_df[breed_cols]

ids = [f.stem for f in test_files]
submission = pd.concat([pd.DataFrame({"id": ids}), preds_df], axis=1)



## === cell 5
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
print(submission.head())
