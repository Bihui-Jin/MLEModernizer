## === cell 4
class NeuralNet(nn.Module):
    def __init__(self, extractors, hidden_size, vocab_size, device):
        super().__init__()
        self.extractors = nn.ModuleList([e.to(device) for e in extractors])
        self.classifier = nn.Linear(hidden_size, vocab_size).to(device)

    def forward(self, x):
        feats = torch.cat([e(x) for e in self.extractors], dim=1)
        return self.classifier(feats)
