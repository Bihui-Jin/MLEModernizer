## === cell 0
import os
import pandas as pd

BASE_INPUT = "../input"
TEST_DIR = os.path.join(BASE_INPUT, "dog-breed-identification", "test")
SAMPLE_SUBMIT_PATH = os.path.join(
    BASE_INPUT, "dog-breed-identification", "sample_submission.csv"
)
OUTPUT_SUBMISSION = "submission.csv"



## === cell 1
sample_df = pd.read_csv(SAMPLE_SUBMIT_PATH)
breed_cols = [c for c in sample_df.columns if c != "id"]
num_breeds = len(breed_cols)
uniform_prob = 1.0 / num_breeds



## === cell 2
test_files = [
    f for f in os.listdir(TEST_DIR) if f.lower().endswith((".jpg", ".jpeg", ".png"))
]
test_ids = sorted([os.path.splitext(f)[0] for f in test_files])

submission_df = pd.DataFrame({"id": test_ids})
for breed in breed_cols:
    submission_df[breed] = uniform_prob



## === cell 3
submission_df.to_csv(OUTPUT_SUBMISSION, index=False)
print(
    f"Submission file written to {OUTPUT_SUBMISSION} with shape {submission_df.shape}"
)
