def get_img_label(fpath):
    category = fpath.split(".")[-3]
    if category == "dog":
        return [1, 0]
    elif category == "cat":
        return [0, 1]
    else:
        return [0, 0]


def load_image(f):
    img = Image.open(f).convert("RGB")
    img = img.resize((IMG_HEIGHT, IMG_WIDTH), Image.ANTIALIAS)
    return np.asarray(img, dtype=np.uint8)


def get_img_array_labels(fpaths):
    imgs = [load_image(f) for f in fpaths]
    labels = [get_img_label(basename(f)) for f in fpaths]
    return np.stack(imgs), np.array(labels)
