## === cell 0
import os
import glob
import numpy as np
import pandas as pd
from tqdm import tqdm



## === cell 1
possible_paths = [
    "../input/train/",
    "../input/dogs-vs-cats-redux-kernels-edition/train/",
    "/kaggle/input/dogs-vs-cats-redux-kernels-edition/train/",
    "train/",
]
train_dir = None
for p in possible_paths:
    if os.path.isdir(p):
        train_dir = p
        break
if train_dir is None:
    raise FileNotFoundError("Training directory not found in expected locations.")

dog_count = 0
total_count = 0
for root, _, files in os.walk(train_dir):
    for f in files:
        if f.lower().endswith(".jpg"):
            total_count += 1
            if "dog" in f.lower():
                dog_count += 1
baseline_prob = dog_count / total_count if total_count > 0 else 0.5
print(f"Baseline dog probability: {baseline_prob:.5f} ({dog_count}/{total_count})")



## === cell 2
sample_paths = [
    "../input/sample_submission.csv",
    "../input/dogs-vs-cats-redux-kernels-edition/sample_submission.csv",
    "/kaggle/input/dogs-vs-cats-redux-kernels-edition/sample_submission.csv",
    "sample_submission.csv",
]
sample_path = None
for p in sample_paths:
    if os.path.isfile(p):
        sample_path = p
        break
if sample_path is None:
    raise FileNotFoundError("sample_submission.csv not found.")

submission_df = pd.read_csv(sample_path)
print(f"Found {len(submission_df)} test entries.")



## === cell 3
submission_df["label"] = baseline_prob
submission_df = submission_df[["id", "label"]]



## === cell 4
output_path = "submission.csv"
submission_df.to_csv(output_path, index=False)
print(f"Submission saved to {output_path}")
