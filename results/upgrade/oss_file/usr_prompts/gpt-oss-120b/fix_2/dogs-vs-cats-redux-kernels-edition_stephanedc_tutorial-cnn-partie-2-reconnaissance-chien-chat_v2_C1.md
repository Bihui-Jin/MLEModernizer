import pandas as pd
from pathlib import Path

BASE_PATH = Path("/kaggle/input/dogs-vs-cats-redux-kernels-edition")

TEST_DIR = BASE_PATH / "test" / "unknown"  # test images are in the unknown sub‑folder



## === cell 1
test_files = sorted(
    [
        f.name
        for f in TEST_DIR.iterdir()
        if f.is_file() and f.suffix.lower() in {".jpg", ".jpeg", ".png"}
    ]
)

submission = pd.DataFrame(
    {
        "id": [int(Path(fname).stem) for fname in test_files],
        "label": 0.5,  # neutral probability
    }
)



## === cell 2
output_path = "results.csv"
submission.to_csv(output_path, index=False)
