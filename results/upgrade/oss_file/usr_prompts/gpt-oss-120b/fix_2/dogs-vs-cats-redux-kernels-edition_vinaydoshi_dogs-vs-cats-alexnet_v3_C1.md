import os
import pandas as pd
import numpy as np




## === cell 1
def list_images(base_path, exts=(".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff")):
    for root, _, files in os.walk(base_path):
        for f in files:
            if f.lower().endswith(exts):
                yield os.path.join(root, f)




## === cell 2
train_path = "/kaggle/input/dogs-vs-cats-redux-kernels-edition/train/"

test_img_paths = list(list_images(test_path))
test_ids = [os.path.splitext(os.path.basename(p))[0] for p in test_img_paths]

print(f"Found {len(test_ids)} test images.")
