## === cell 0
import os
import pandas as pd



## === cell 1
test_dir = "/kaggle/input/dogs-vs-cats-redux-kernels-edition/test/"
test_paths = []
for root, _, files in os.walk(test_dir):
    for f in files:
        if f.lower().endswith(".jpg"):
            test_paths.append(os.path.join(root, f))

test_ids = [os.path.splitext(os.path.basename(p))[0] for p in test_paths]



## === cell 2
baseline_prob = 0.5
submission = pd.DataFrame({"id": test_ids, "label": [baseline_prob] * len(test_ids)})



## === cell 3
output_path = "/kaggle/working/submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission written to {output_path} with {len(submission)} rows.")
