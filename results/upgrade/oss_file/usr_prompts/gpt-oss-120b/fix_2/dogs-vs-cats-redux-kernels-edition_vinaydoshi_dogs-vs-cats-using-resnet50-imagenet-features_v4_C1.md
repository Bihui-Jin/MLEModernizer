import os
import numpy as np

from pathlib import Path




## === cell 1
def get_test_ids(test_root: str):
    """
    Walk through test_root recursively and return a list of ids extracted from image filenames.
    Expected filenames like '123.jpg' or '456.png'. The numeric part before the first dot is used.
    """
    ids = []
    for path in Path(test_root).rglob("*"):
        if path.is_file() and path.suffix.lower() in {
            ".jpg",
            ".jpeg",
            ".png",
            ".bmp",
            ".tif",
            ".tiff",
        }:
            stem = path.stem
            ids.append(stem)
    return ids
