train_dir = os.path.join(input_root, "dogs-vs-cats-redux-kernels-edition", "train")

    input_root, "dogs-vs-cats-redux-kernels-edition", "test", "unknown"
)

assert os.path.isdir(train_dir), f"Train directory not found: {train_dir}"
assert os.path.isdir(test_dir), f"Test directory not found: {test_dir}"
print(f"Train dir: {train_dir}")
print(f"Test dir: {test_dir}")




## === cell 2
test_files = glob.glob(os.path.join(test_dir, "*.jpg"))
assert test_files, "No test images found."


def extract_id(filepath):
    name = os.path.basename(filepath)
    return int(name.split(".")[0])


test_files_sorted = sorted(test_files, key=extract_id)
test_ids = [extract_id(f) for f in test_files_sorted]

predictions = np.full(len(test_ids), 0.5, dtype=float)




## === cell 3
submission_df = pd.DataFrame({"id": test_ids, "label": predictions})
submission_path = "submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path} with {len(submission_df)} rows.")
