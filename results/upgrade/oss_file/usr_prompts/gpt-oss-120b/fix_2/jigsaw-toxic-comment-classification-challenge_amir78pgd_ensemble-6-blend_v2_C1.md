import os, glob
import pandas as pd
import numpy as np


def locate_file(rel_path):
    candidates = [
        os.path.abspath(rel_path),
        os.path.join(os.getcwd(), rel_path),
        os.path.join(os.getcwd(), "data", rel_path),
        os.path.join(os.getcwd(), "kaggle", "data", rel_path),
        os.path.join("..", "input", rel_path),
        os.path.join("input", rel_path),
    ]
    for cand in candidates:
        if os.path.isfile(cand):
            return cand
    raise FileNotFoundError(f"Unable to locate {rel_path}")


sample_sub_path = locate_file("sample_submission.csv")
train_path = locate_file("train.csv")


## === cell 1
submission_df = pd.read_csv(sample_sub_path)

label_cols = ["toxic", "severe_toxic", "obscene", "threat", "insult", "identity_hate"]

train_df = pd.read_csv(train_path, usecols=["id"] + label_cols)
label_means = train_df[label_cols].mean()

submission_df[label_cols] = label_means.values

output_path = "submission.csv"
