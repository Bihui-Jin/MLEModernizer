## === cell 0
import os
import numpy as np, pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression

TRAIN_PATH = "../input/train.csv"
TEST_PATH = "../input/test.csv"
SUBMIT_PATH = "submission.csv"



## === cell 1
train_df = pd.read_csv(TRAIN_PATH)
test_df = pd.read_csv(TEST_PATH)

label_cols = ["toxic", "severe_toxic", "obscene", "threat", "insult", "identity_hate"]
assert set(label_cols).issubset(train_df.columns), "Missing label columns in train data"



## === cell 2
tfidf = TfidfVectorizer(
    lowercase=True,
    stop_words="english",
    max_features=50000,  # limit memory usage
    ngram_range=(1, 2),
)
X_train = tfidf.fit_transform(train_df["comment_text"].astype(str))
X_test = tfidf.transform(test_df["comment_text"].astype(str))



## === cell 3
preds = {}
for col in label_cols:
    y = train_df[col].astype(int)
    model = LogisticRegression(
        solver="saga", max_iter=1000, class_weight="balanced", n_jobs=5, verbose=0
    )
    model.fit(X_train, y)
    preds[col] = model.predict_proba(X_test)[:, 1]  # probability of the positive class



## === cell 4
submission = pd.DataFrame({"id": test_df["id"]})
for col in label_cols:
    submission[col] = preds[col]

submission.to_csv(SUBMIT_PATH, index=False)
print(f"Submission written to {SUBMIT_PATH}, shape: {submission.shape}")
