**sample_submission.csv** - Sample submission file.

**train_images** - Training images stored under different sub-directories corresponding to ten target classes. Filename corresponds to the `image_id` column of `train.csv`.

**test_images** - Test set images.

import pandas as pd
import numpy as np
import scipy.stats as ss
import os




## === cell 1
def try_load(path):
    try:
        return pd.read_csv(path)
    except FileNotFoundError:
        return None


s1 = try_load("../input/paddydocoutputs/model_submission_v5.csv")
s2 = try_load("../input/paddy-doctor-training/model_submission_v5.csv")
s3 = try_load("../input/k/kasunpramodya/paddy-doctor-training/model_submission_v5.csv")



## === cell 2
if s1 is not None and s2 is not None and s3 is not None:
    s_full = pd.concat(
        [s1, s2.drop("image_id", axis=1), s3.drop("image_id", axis=1)], axis=1
    ).set_index("image_id")

    s_full_mode = ss.mode(s_full, axis=1)
    s_final = pd.DataFrame(index=s_full.index)
    s_final["label"] = s_full_mode[0].reshape(1, -1)[0]
else:
    train_path = "../input/paddy-disease-classification/train.csv"
    if not os.path.exists(train_path):
        train_path = "../input/train.csv"  # alternative location
    train_df = pd.read_csv(train_path)
    most_common_label = train_df["label"].mode()[0]

    sample_sub_path = "../input/paddy-disease-classification/sample_submission.csv"
    if not os.path.exists(sample_sub_path):
        sample_sub_path = "../input/sample_submission.csv"
    sample_sub = pd.read_csv(sample_sub_path)

    s_final = pd.DataFrame(
        {"image_id": sample_sub["image_id"], "label": most_common_label}
    )



## === cell 3
submission = s_final[["image_id", "label"]]
submission.to_csv("model_submission_v10.csv", index=False)
