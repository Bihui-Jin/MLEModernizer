BASE_DIR = "/kaggle/input/plant-pathology-2020-fgvc7"


def format_path(st):
    return os.path.join(BASE_DIR, "images", f"{st}.jpg")




## === cell 2
train_path = os.path.join(BASE_DIR, "train.csv")
test_path = os.path.join(BASE_DIR, "test.csv")
sample_sub_path = os.path.join(BASE_DIR, "sample_submission.csv")

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)
submission_df = pd.read_csv(sample_sub_path)

label_cols = ["healthy", "multiple_diseases", "rust", "scab"]
train_labels = train_df[label_cols].values.astype(np.float32)

baseline_probs = train_labels.mean(axis=0)  # shape (4,)

preds = np.tile(baseline_probs, (len(test_df), 1))

submission_df.loc[:, label_cols] = preds

submission_path = "submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
print(submission_df.head())



## === cell 3
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score

train_paths, val_paths, train_lbls, val_lbls = train_test_split(
    train_df["image_id"].values,
    train_labels,
    test_size=0.15,
    random_state=2020,
    stratify=train_labels,
)

val_preds = np.tile(baseline_probs, (len(val_lbls), 1))
auc = roc_auc_score(val_lbls, val_preds, average="macro", multi_class="ovr")
print(f"Validation macro ROC‑AUC (baseline): {auc:.5f}")
