import os
import pandas as pd




## === cell 1
def compute_baseline_predictions():
    train_path = "../input/plant-pathology-2020-fgvc7/train.csv"
    train_df = pd.read_csv(train_path)

    label_cols = ["healthy", "multiple_diseases", "rust", "scab"]
    mean_probs = train_df[label_cols].mean()

    test_path = "../input/plant-pathology-2020-fgvc7/test.csv"
    test_df = pd.read_csv(test_path)

    pred_df = pd.DataFrame({"image_id": test_df["image_id"]})
    for col in label_cols:
        pred_df[col] = mean_probs[col]

    return pred_df




## === cell 2
def write_submission(df, output_path="./submission.csv"):
    cols = ["image_id", "healthy", "multiple_diseases", "rust", "scab"]
    df = df[cols]
    df.to_csv(output_path, index=False)
    print(f"Submission written to {output_path}")

if __name__ == "__main__":
