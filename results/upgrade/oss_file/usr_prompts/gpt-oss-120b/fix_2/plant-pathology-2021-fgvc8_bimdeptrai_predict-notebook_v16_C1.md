label_split = train.labels.apply(lambda x: x.split())
mlb = MultiLabelBinarizer().fit(label_split)
labels_onehot = pd.DataFrame(mlb.transform(label_split), columns=mlb.classes_)



## === cell 5
num_classes = labels_onehot.shape[1]
preds = np.zeros((len(submissions), num_classes), dtype=np.float32)



## === cell 6
thresh = 0.4
for i in range(len(submissions)):
    if "healthy" in mlb.classes_ and preds[i][
        mlb.classes_.tolist().index("healthy")
    ] == np.max(preds[i]):
        submissions.at[i, "labels"] = "healthy"
    else:
        selected = labels_onehot.columns[preds[i] >= thresh]
        if len(selected) == 0 or "healthy" in selected:
            top_idx = np.argmax(preds[i])
            submissions.at[i, "labels"] = labels_onehot.columns[top_idx]
        else:
            submissions.at[i, "labels"] = " ".join(selected)



## === cell 7
submissions.to_csv("submission.csv", index=False)
