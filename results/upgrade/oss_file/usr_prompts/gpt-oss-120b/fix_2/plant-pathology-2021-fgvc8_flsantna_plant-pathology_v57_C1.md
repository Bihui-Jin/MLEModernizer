## === cell 0
import os
import pandas as pd

output_dir = "./"
test_dir = "../input/plant-pathology-2021-fgvc8/test_images/"

train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
train_df = pd.read_csv(train_path)

one_hot = train_df["labels"].str.get_dummies(sep=" ")
most_common_label = one_hot.sum().idxmax()
print(f"Most common label in training set: {most_common_label}")

test_images = sorted([f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")])
print(f"Found {len(test_images)} test images.")




## === cell 1
def predict_labels(image_filenames, label):
    """
    Returns a list of [image_name, predicted_labels] rows.
    The predictions are space‑delimited strings; here each image gets a single label.
    """
    predictions = []
    for img_name in image_filenames:
        predictions.append([img_name, label])
    return predictions




## === cell 2
if __name__ == "__main__":
    prediction_rows = predict_labels(test_images, most_common_label)

    submission_df = pd.DataFrame(prediction_rows, columns=["image", "labels"])

    submission_path = os.path.join(output_dir, "submission.csv")
    submission_df.to_csv(submission_path, index=False)
    print(f"Submission file written to {submission_path}")
