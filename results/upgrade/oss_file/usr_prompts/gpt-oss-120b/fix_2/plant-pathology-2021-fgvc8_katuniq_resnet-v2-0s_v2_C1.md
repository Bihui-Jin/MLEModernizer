num_classes = labels.shape[1]

## === cell 6
thresh = {
    "complex": 0.23,
    "frog_eye_leaf_spot": 0.23,
    "healthy": 0.23,
    "powdery_mildew": 0.23,
    "rust": 0.23,
    "scab": 0.23,
}



## === cell 7
pred_labels = []
for i in range(len(submissions)):
    if "healthy" in class_names and preds[i][class_names.index("healthy")] == np.max(
        preds[i]
    ):
        pred_labels.append("healthy")
        continue

    chosen = []
    for j, label in enumerate(class_names):
        if preds[i][j] > thresh.get(label, 0.23):
            chosen.append(label)

    if not chosen or "healthy" in chosen:
        max_prob = np.max(preds[i])
        chosen = [class_names[j] for j, p in enumerate(preds[i]) if p >= max_prob]

    pred_labels.append(" ".join(chosen))

submissions["labels"] = pred_labels

submissions.to_csv("submission.csv", index=False)
