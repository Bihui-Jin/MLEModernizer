def decode_image(filename, label=None, image_size=(512, 512)):
    bits = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(bits, channels=3)
    image = tf.cast(image, tf.float32) / 255.0
    image = tf.image.resize(image, image_size)
    if label is None:
        return image
    else:
        return image, label




## === cell 2
BATCH_SIZE = 32
AUTO = tf.data.experimental.AUTOTUNE



## === cell 3
source = "../input/plant-pathology-2021-fgvc8/test_images"
IMAGE_PATHS = [
    os.path.join(source, f)
    for f in os.listdir(source)
    if re.search(r"([a-zA-Z0-9\s_\\.\-\(\):])+(\.jpg|\.jpeg|\.png)$", f, re.IGNORECASE)
]



## === cell 4
test_dataset = (
    tf.data.Dataset.from_tensor_slices(IMAGE_PATHS)
    .map(decode_image, num_parallel_calls=AUTO)
    .batch(BATCH_SIZE)
)




## === cell 5
class FixedDropout(tf.keras.layers.Dropout):
    def _get_noise_shape(self, inputs):
        if self.noise_shape is None:
            return self.noise_shape
        symbolic_shape = K.shape(inputs)
        noise_shape = [
            symbolic_shape[axis] if shape is None else shape
            for axis, shape in enumerate(self.noise_shape)
        ]
        return tuple(noise_shape)




## === cell 6
model_path = "../input/3smresnet50/3SMResNet50.h5"
if os.path.exists(model_path):
    model = tf.keras.models.load_model(
        model_path, compile=False, custom_objects={"FixedDropout": FixedDropout}
    )
    probs = model.predict(test_dataset, verbose=0)
else:
    probs = np.zeros((len(IMAGE_PATHS), 5), dtype=np.float32)



## === cell 7
name = {
    0: "scab",
    1: "frog_eye_leaf_spot",
    2: "complex",
    3: "rust",
    4: "powdery_mildew",
    6: "healthy",  # index 6 is used for the default healthy label
}
threshold = {0: 0.25, 1: 0.35, 2: 0.25, 3: 0.35, 4: 0.35}



## === cell 8
pred_string = []
for line in probs:
    s = ""
    count = 0
    for i in range(5):
        if line[i] > threshold[i]:
            s += name[i] + " "
            count += 1
    if count >= 2:
        notComplex = True
        for i in range(5):
            if line[i] > threshold[i] and name[i] == "complex":
                notComplex = False
                break
        if notComplex:
            s += "complex" + " "
    if s.strip() == "":
        s = name[6]
    pred_string.append(s.strip())



## === cell 9
df = pd.DataFrame(
