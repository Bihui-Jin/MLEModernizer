def load_features(paths, img_size=64):
    """
    Read image files, resize to img_size×img_size, scale to [0,1],
    and flatten to 1‑D vectors.
    """
    features = []
    for p in paths:
        img = cv2.imread(p)
        if img is None:
            img = np.zeros((img_size, img_size, 3), dtype=np.uint8)
        else:
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
            img = cv2.resize(img, (img_size, img_size))
        img = img.astype(np.float32) / 255.0
        features.append(img.flatten())
    return np.stack(features)




## === cell 6
print("Loading training images ...")
X = load_features(train_images)
y = labels

X_tr, X_val, y_tr, y_val = train_test_split(
    X, y, test_size=0.15, random_state=42, stratify=y
)

clf = OneVsRestClassifier(LogisticRegression(max_iter=200, n_jobs=-1, solver="lbfgs"))
print("Training model ...")
clf.fit(X_tr, y_tr)



## === cell 7
val_pred = clf.predict_proba(X_val)
auc_scores = [roc_auc_score(y_val[:, i], val_pred[:, i]) for i in range(y_val.shape[1])]
print("Validation AUC per label:", np.round(auc_scores, 4))
print("Mean Validation AUC:", np.mean(auc_scores))



## === cell 8
