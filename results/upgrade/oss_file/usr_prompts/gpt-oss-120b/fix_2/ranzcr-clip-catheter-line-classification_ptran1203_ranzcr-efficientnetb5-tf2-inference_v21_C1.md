def triple_image(image):
    """Convert a single‑channel image to 3‑channel by replication."""
    return tf.concat([image] * 3, axis=-1)


def load_image(file_path):
    """Read, decode, resize and normalise an image."""
    img = tf.io.read_file(file_path)
    img = tf.image.decode_png(img, channels=1)  # original images are grayscale
    img = tf.image.resize(img, (H, W))
    img = triple_image(img)  # (H, W, 3)
    img = tf.cast(img, tf.float32) / 255.0
    img = (img - mean) / std
    return img


def extract_id(file_path):
    """Return the StudyInstanceUID extracted from the filename."""
    filename = tf.strings.regex_replace(tf.strings.basename(file_path), ".jpg", "")
    return filename

def get_model(
    base_model_cls,
    baseline_weight=None,
    init_weight=None,
    lr=0.001,
    optimizer=tf.optimizers.Adam,
):
    """Create EfficientNet model with a sigmoid head."""
    base_model = base_model_cls(
        include_top=False,
        input_shape=(H, W, 3),
        pooling="avg",
        weights=baseline_weight,
    )
    x = tf.keras.layers.Dropout(0.3)(base_model.output)
    outputs = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base_model.input, outputs=outputs)

    model.compile(
        optimizer=optimizer(learning_rate=lr),
        loss="binary_crossentropy",
