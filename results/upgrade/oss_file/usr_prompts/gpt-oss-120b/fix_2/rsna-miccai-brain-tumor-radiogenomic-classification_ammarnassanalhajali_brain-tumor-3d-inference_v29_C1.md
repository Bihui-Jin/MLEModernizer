Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric

```
etc.
```

def load_dicom_image(path, img_size=IMAGE_SIZE, voi_lut=True, rotate=0):
    """Load a single DICOM slice, apply VOI LUT, rotate and resize."""
    dicom = pydicom.dcmread(path)
    data = dicom.pixel_array
    if voi_lut:
        data = apply_voi_lut(dicom, dicom)
    if rotate > 0:
        rot_choices = [
            0,
            cv2.ROTATE_90_CLOCKWISE,
            cv2.ROTATE_90_COUNTERCLOCKWISE,
            cv2.ROTATE_180,
        ]
        data = cv2.rotate(data, rot_choices[rotate])
    data = cv2.resize(data, (img_size, img_size))
    return data


def load_dicom_images_3d(
    scan_id,
    num_imgs=NUM_IMAGES,
    img_size=IMAGE_SIZE,
    mri_type="FLAIR",
    split="test",
    rotate=0,
):
    """Load a stack of DICOM slices for a given scan and MRI modality."""
    pattern = f"{data_directory}/{split}/{scan_id}/{mri_type}/*.dcm"
    files = sorted(
        glob.glob(pattern),
        key=lambda var: [
            int(x) if x.isdigit() else x for x in re.findall(r"[^0-9]|[0-9]+", var)
        ],
    )
    if not files:
        return np.zeros((1, img_size, img_size, num_imgs), dtype=np.float32)

    middle = len(files) // 2
    num_imgs2 = num_imgs // 2
    p1 = max(0, middle - num_imgs2)
    p2 = min(len(files), middle + num_imgs2)
    selected = files[p1:p2]

    slices = [load_dicom_image(f, rotate=rotate) for f in selected]
    img3d = np.stack(slices, axis=-1)  # shape (H, W, N)

    if img3d.shape[-1] < num_imgs:
        pad_width = num_imgs - img3d.shape[-1]
        pad = np.zeros((img_size, img_size, pad_width), dtype=img3d.dtype)
        img3d = np.concatenate([img3d, pad], axis=-1)

    if np.max(img3d) > np.min(img3d):
        img3d = (img3d - np.min(img3d)) / (np.max(img3d) - np.min(img3d))

    return np.expand_dims(img3d, 0)  # shape (1, H, W, N)

from keras.utils import Sequence


class Dataset(Sequence):
    """Simple Keras Sequence yielding one 3‑D volume per batch."""

    def __init__(self, df, is_train=True, batch_size=1, shuffle=True):
        self.ids = df["BraTS21ID"].values
        self.paths = df["BraTS21ID5"].values
        self.y = df["MGMT_value"].values if "MGMT_value" in df.columns else None
        self.is_train = is_train
        self.batch_size = batch_size
        self.shuffle = shuffle
        self.on_epoch_end()

    def __len__(self):
        return math.ceil(len(self.ids) / self.batch_size)

    def __getitem__(self, idx):
        batch_start = idx * self.batch_size
        batch_end = min((idx + 1) * self.batch_size, len(self.ids))
        batch_paths = self.paths[batch_start:batch_end]

        batch_X = np.concatenate([load_dicom_images_3d(p) for p in batch_paths], axis=0)
        if self.is_train and self.y is not None:
            batch_y = self.y[batch_start:batch_end]
            return batch_X, batch_y
        else:
            return batch_X

    def on_epoch_end(self):
        if self.shuffle and self.is_train:
            combined = list(zip(self.ids, self.paths, self.y))
            random.shuffle(combined)
            self.ids, self.paths, self.y = zip(*combined)
