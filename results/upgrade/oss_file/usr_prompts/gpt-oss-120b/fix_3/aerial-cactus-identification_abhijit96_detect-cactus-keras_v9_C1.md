import os, sys, numpy as np, pandas as pd, cv2
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split


def load_image(base_dirs, filename):
    """Try to read an image from a list of possible base directories.
    Returns a (32,32,3) float32 array normalized to [0,1].
    If the image cannot be read, returns an array of zeros."""
    for base in base_dirs:
        path = os.path.join(base, filename)
        if os.path.exists(path):
            img = cv2.imread(path, cv2.IMREAD_COLOR)
            if img is not None:
                img = cv2.resize(img, (32, 32))
                return img.astype(np.float32) / 255.0
    return np.zeros((32, 32, 3), dtype=np.float32)


train_csv = (
    pd.read_csv("../input/train.csv")
    .sample(frac=1, random_state=42)
    .reset_index(drop=True)
)
train_ids = train_csv["id"].values
train_labels = train_csv["has_cactus"].values

train_dirs = [
    "../input/train/train",
    "../input/aerial-cactus-identification/train/train",
    "../input/train",
    "../input/aerial-cactus-identification/train",
]

images = []
for img_name in train_ids:
    images.append(load_image(train_dirs, img_name))
X = np.stack(images, axis=0)  # shape (N, 32, 32, 3)
X_flat = X.reshape(len(X), -1)  # flatten for sklearn
y = train_labels.astype(np.int32)

X_train, X_val, y_train, y_val = train_test_split(
    X_flat, y, test_size=0.1, random_state=42, stratify=y
)

rf = RandomForestClassifier(
    n_estimators=300,
    max_depth=None,
    min_samples_leaf=1,
    n_jobs=-1,
    random_state=42,
    class_weight="balanced",
)
rf.fit(X_train, y_train)




## === cell 1
test_dir = "../input/test"
alt_test_dirs = ["../input/aerial-cactus-identification/test", "../input/test"]

test_files = sorted(os.listdir(test_dir)) if os.path.isdir(test_dir) else []
if not test_files:
    for alt in alt_test_dirs:
        if os.path.isdir(alt):
            test_files = sorted(os.listdir(alt))
            test_dir = alt
            break

test_images = []
for fname in test_files:
    test_images.append(load_image([test_dir], fname))
test_X = np.stack(test_images, axis=0).reshape(len(test_images), -1)

test_pred_proba = rf.predict_proba(test_X)[:, 1]



## === cell 2
submission = pd.DataFrame({"id": test_files, "has_cactus": test_pred_proba})
