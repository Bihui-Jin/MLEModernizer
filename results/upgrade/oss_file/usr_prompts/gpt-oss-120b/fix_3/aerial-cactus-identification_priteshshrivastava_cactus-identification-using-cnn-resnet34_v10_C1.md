- **train.csv** - the training set labels, indicates whether the image has a cactus (`has_cactus = 1`)
- **sample_submission.csv** - a sample submission file in the correct format

## === cell 0
from fastai.vision.all import *
import pandas as pd
from pathlib import Path

root = Path("../input/aerial-cactus-identification")
train_path = root / "train"
test_path = root / "test"

train_df = pd.read_csv(root / "train.csv")
test_df = pd.read_csv(root / "sample_submission.csv")
print("train samples:", len(train_df), "test samples:", len(test_df))



## === cell 1
dls = ImageDataLoaders.from_df(
    df=train_df,
    path=train_path,
    fn_col="id",
    label_col="has_cactus",
    valid_pct=0.2,
    seed=42,
    item_tfms=Resize(224),
    batch_tfms=aug_transforms(do_flip=False),
)
print(dls)



## === cell 2
learn = cnn_learner(dls, resnet34, pretrained=False, metrics=accuracy)
learn.fine_tune(1)  # 1 epoch fine‑tuning (core logic unchanged)



## === cell 3
test_files = [test_path / f for f in test_df["id"]]
test_dl = learn.dls.test_dl(test_files)
preds, _ = learn.get_preds(dl=test_dl)

proba = torch.nn.functional.softmax(preds, dim=1)[:, 1].cpu().numpy()



## === cell 4
submission = pd.DataFrame({"id": test_df["id"], "has_cactus": proba})
submission.to_csv("submission.csv", index=False)
print("submission saved, first rows:")
print(submission.head())
