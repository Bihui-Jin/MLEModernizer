# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import random
import numpy as np
import pandas as pd
import cv2
import matplotlib.pyplot as plt
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.applications import EfficientNetB3
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense
from tensorflow.keras.optimizers import Adam
from sklearn.utils.class_weight import compute_class_weight



## === cell 1
train_dir = "/kaggle/input/aerial-cactus-identification/train"
test_dir = "/kaggle/input/aerial-cactus-identification/test"



## === cell 2
train_df = pd.read_csv("/kaggle/input/aerial-cactus-identification/train.csv")
train_df["has_cactus"] = train_df["has_cactus"].astype(str)
train_df.head()




## === cell 3
def count_files(directory):
    return len(
        [f for f in os.listdir(directory) if os.path.isfile(os.path.join(directory, f))]
    )


print(f"Train images found: {count_files(train_dir)}")
print(f"Test  images found: {count_files(test_dir)}")



## === cell 4
class_ratio = train_df["has_cactus"].value_counts(normalize=True) * 100
print(class_ratio)



## === cell 5
class_weights = compute_class_weight(
    class_weight="balanced",
    classes=np.unique(train_df["has_cactus"]),
    y=train_df["has_cactus"],
)
class_weights_dict = dict(enumerate(class_weights))
print("Class weights:", class_weights_dict)




## === cell 6
def custom_preprocessing(image):
    """Random rotations / flips and simple brightness scaling."""
    k = random.randint(0, 3)
    image = np.rot90(image, k)
    if random.random() > 0.5:
        image = np.fliplr(image)
    if random.random() > 0.5:
        image = np.flipud(image)
    factor = random.uniform(0.8, 1.2)
    image = np.clip(image * factor, 0, 255).astype(np.uint32) / 255.0
    return image


train_datagen = ImageDataGenerator(
    validation_split=0.10,
    preprocessing_function=custom_preprocessing,
    rescale=1 / 255.0,
)

val_datagen = ImageDataGenerator(validation_split=0.10, rescale=1 / 255.0)



## === cell 7
train_generator = train_datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=train_dir,
    x_col="id",
    y_col="has_cactus",
    target_size=(32, 32),
    color_mode="rgb",
    class_mode="binary",
    batch_size=256,
    shuffle=True,
    subset="training",
    seed=42,
)

val_generator = val_datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=train_dir,
    x_col="id",
    y_col="has_cactus",
    target_size=(32, 32),
    color_mode="rgb",
    class_mode="binary",
    batch_size=256,
    shuffle=False,
    subset="validation",
    seed=42,
)



## === cell 8
test_filenames = sorted(os.listdir(test_dir))
test_df = pd.DataFrame({"id": test_filenames})
test_datagen = ImageDataGenerator(rescale=1 / 255.0)

test_generator = test_datagen.flow_from_dataframe(
    dataframe=test_df,
    directory=test_dir,
    x_col="id",
    y_col=None,
    target_size=(32, 32),
    color_mode="rgb",
    class_mode=None,
    batch_size=32,
    shuffle=False,
)



## === cell 9
efficient_net = EfficientNetB3(
    weights="imagenet", input_shape=(32, 32, 3), include_top=False, pooling="max"
)

model = Sequential(
    [
        efficient_net,
        Dense(120, activation="relu"),
        Dense(120, activation="relu"),
        Dense(1, activation="sigmoid"),
    ]
)

model.summary()



## === cell 10
model.compile(
    optimizer=Adam(learning_rate=5e-5),
    loss="binary_crossentropy",
    metrics=["accuracy"],
)



## === cell 11
history = model.fit(
    train_generator,
    epochs=20,
    validation_data=val_generator,
    class_weight=class_weights_dict,
    verbose=1,
)



## === cell 12
preds = model.predict(test_generator, steps=len(test_generator), verbose=1)



## === cell 13
image_ids = test_generator.filenames  # e.g., 'image.jpg' paths
image_ids = [os.path.basename(p) for p in image_ids]
predictions = preds.ravel()
submission = pd.DataFrame({"id": image_ids, "has_cactus": predictions})
submission.head()



## === cell 14
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
print("Finished.")
