## === cell 0
import os
import cv2
import numpy as np
import pandas as pd

try:
    import tensorflow as tf
    from tensorflow.keras.applications import efficientnet

    tf_available = True
except Exception as e:
    tf_available = False
    tf = None
    efficientnet = None



## === cell 1
base_path = os.path.abspath(
    os.path.join("..", "input", "cassava-leaf-disease-classification")
)
test_img_dir = os.path.join(base_path, "test_images") + os.sep
train_img_dir = os.path.join(base_path, "train_images") + os.sep
train_csv_path = os.path.join(base_path, "train.csv")



## === cell 2
if not tf_available:
    from sklearn.ensemble import RandomForestClassifier

    train_df = pd.read_csv(train_csv_path)

    def extract_features(image_path: str) -> np.ndarray:
        img = cv2.imread(image_path)
        if img is None:
            return np.zeros(3, dtype=np.float32)
        img_resized = cv2.resize(img, (64, 64))  # small size for speed
        mean_vals = img_resized.mean(axis=(0, 1))
        return mean_vals.astype(np.float32)

    feature_list = []
    label_list = []
    for idx, row in train_df.iterrows():
        img_path = os.path.join(train_img_dir, row["image_id"])
        feats = extract_features(img_path)
        feature_list.append(feats)
        label_list.append(row["label"])
    X_train = np.stack(feature_list)  # shape (n_samples, 3)
    y_train = np.array(label_list, dtype=np.int32)

    rf_model = RandomForestClassifier(
        n_estimators=200,
        max_depth=None,
        random_state=42,
        n_jobs=-1,
        min_samples_split=2,
        min_samples_leaf=1,
    )
    rf_model.fit(X_train, y_train)




## === cell 3
def predict_label(image_path: str) -> int:
    """
    Predict a label for a single image.

    • If TensorFlow/EfficientNet is available, use the pretrained model.
    • Otherwise, use the lightweight RandomForest trained on mean‑color features.
    """
    img = cv2.imread(image_path)
    if img is None:
        return 0

    img_resized = cv2.resize(img, (300, 300))

    if tf_available and efficientnet is not None:
        img_preproc = efficientnet.preprocess_input(img_resized.astype(np.float32))
        img_batch = np.expand_dims(img_preproc, axis=0)  # (1, 300, 300, 3)
        preds = efficientnet.EfficientNetB3(weights="imagenet").predict(
            img_batch, verbose=0
        )
        label = int(np.argmax(preds) % 5)
    else:
        small_img = cv2.resize(img, (64, 64))
        mean_vals = (
            small_img.mean(axis=(0, 1)).astype(np.float32).reshape(1, -1)
        )  # (1,3)
        label = int(rf_model.predict(mean_vals)[0])
    return label




## === cell 4
image_ids = []
pred_labels = []

for filename in os.listdir(test_img_dir):
    if filename.lower().endswith((".jpg", ".jpeg", ".png")):
        img_path = os.path.join(test_img_dir, filename)
        label = predict_label(img_path)
        image_ids.append(filename)
        pred_labels.append(label)

submission_df = pd.DataFrame({"image_id": image_ids, "label": pred_labels})

submission_path = os.path.abspath("submission.csv")
submission_df.to_csv(submission_path, index=False)
print(f"Submission file written to: {submission_path}")
