## === cell 1
ROOT_DIR = "/kaggle/input/"
DATA_DIR = os.path.join(ROOT_DIR, "cassava-leaf-disease-classification")
TRAIN_IMG_DIR = os.path.join(DATA_DIR, "train_images")
TEST_IMG_DIR = os.path.join(DATA_DIR, "test_images")
TRAIN_CSV = os.path.join(DATA_DIR, "train.csv")
SAMPLE_SUBMIT = os.path.join(DATA_DIR, "sample_submission.csv")

HEIGHT, WIDTH = 240, 240  # EfficientNetB0 input size
BATCH_SIZE = 32
AUTOTUNE = tf.data.experimental.AUTOTUNE
NUM_CLASSES = 5
SEED = 42
tf.random.set_seed(SEED)
np.random.seed(SEED)



## === cell 2
train_df = pd.read_csv(TRAIN_CSV)


def _load_and_preprocess(path, label=None):
    """Read an image file, decode, resize, optionally augment, and return (image, label)."""
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [HEIGHT, WIDTH])
    if label is not None:
        image = tf.image.random_flip_left_right(image, seed=SEED)
        image = tf.image.random_flip_up_down(image, seed=SEED)
    image = tf.cast(image, tf.float32) / 255.0  # normalise to [0,1]
    if label is None:
        return image
    return image, label


train_paths = [os.path.join(TRAIN_IMG_DIR, img_id) for img_id in train_df["image_id"]]
train_labels = train_df["label"].values.astype(np.int32)

train_dataset = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_dataset = train_dataset.map(
    lambda p, l: _load_and_preprocess(p, l), num_parallel_calls=AUTOTUNE
)
train_dataset = (
    train_dataset.shuffle(10000, seed=SEED).batch(BATCH_SIZE).prefetch(AUTOTUNE)
)



## === cell 3
base_model = tf.keras.applications.EfficientNetB0(
    include_top=False, weights="imagenet", input_shape=(HEIGHT, WIDTH, 3), pooling="avg"
)

base_model.trainable = False

inputs = tf.keras.Input(shape=(HEIGHT, WIDTH, 3))
x = base_model(inputs, training=False)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)



## === cell 4
EPOCHS_FROZEN = 5
model.fit(train_dataset, epochs=EPOCHS_FROZEN, verbose=2)

base_model.trainable = True
model.compile(
    optimizer=tf.keras.optimizers.Adam(1e-4),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)

EPOCHS_FINE = 5
model.fit(train_dataset, epochs=EPOCHS_FINE, verbose=2)



## === cell 5
sample_sub = pd.read_csv(SAMPLE_SUBMIT)
test_ids = sample_sub["image_id"].tolist()
test_paths = [os.path.join(TEST_IMG_DIR, img_id) for img_id in test_ids]

test_dataset = tf.data.Dataset.from_tensor_slices(test_paths)
test_dataset = test_dataset.map(
    lambda p: _load_and_preprocess(p), num_parallel_calls=AUTOTUNE
)
test_dataset = test_dataset.batch(BATCH_SIZE).prefetch(AUTOTUNE)

pred_probs = model.predict(test_dataset, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission_df = pd.DataFrame({"image_id": test_ids, "label": pred_labels})
submission_path = "submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}, rows: {len(submission_df)}")
