import os

## === cell 1
model = torchvision.models.resnet50(pretrained=False)
num_ftrs = model.fc.in_features
model.fc = torch.nn.Linear(num_ftrs, 5)

weights_path = "../input/baseline-weights/epoch_18.pth"
if os.path.isfile(weights_path):
    state = torch.load(weights_path, map_location="cpu")
    if isinstance(state, dict) and "model" in state:
        state = state["model"]
    model.load_state_dict(state, strict=False)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = model.to(device)
model.eval()




## === cell 2
val_transforms = T.Compose(
    [
        T.ToPILImage(),
        T.Resize(256),
        T.CenterCrop(224),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)




## === cell 3
class InferDataset(Dataset):
    def __init__(self, img_dir, transform):
        self.img_dir = img_dir
        self.transform = transform
        self.filenames = [f for f in os.listdir(img_dir) if f.lower().endswith(".jpg")]
        self.filenames.sort()  # deterministic order

    def __len__(self):
        return len(self.filenames)

    def __getitem__(self, idx):
        fn = self.filenames[idx]
        img_path = os.path.join(self.img_dir, fn)
        img = cv2.imread(img_path)  # BGR
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        if self.transform:
            img = self.transform(img)
        return img, fn


def _find_test_dir():
    """
    Search common Kaggle locations for the folder containing test images.
    Falls back to a recursive walk if none of the heuristics match.
    """
    candidates = [
        os.path.join("data", "cassava-leaf-disease-classification", "test_images"),
        os.path.join("input", "cassava-leaf-disease-classification", "test_images"),
        os.path.join(
            "kaggle", "input", "cassava-leaf-disease-classification", "test_images"
        ),
        os.path.join(
            "kaggle", "working", "cassava-leaf-disease-classification", "test_images"
        ),
        "test_images",
    ]
    for d in candidates:
        if os.path.isdir(d):
            return d
    for root, dirs, _ in os.walk("."):
        if "test_images" in dirs:
            return os.path.join(root, "test_images")
    raise FileNotFoundError("Test image directory not found.")


test_dir = _find_test_dir()
infer_dataset = InferDataset(test_dir, val_transforms)
infer_loader = DataLoader(infer_dataset, batch_size=32, shuffle=False, num_workers=0)




## === cell 4
submission_path = "./submission.csv"
with open(submission_path, "w") as f:
    f.write("image_id,label\n")
    with torch.no_grad():
        for images, fnames in infer_loader:
            images = images.to(device)
            logits = model(images)
            preds = torch.argmax(logits, dim=1).cpu().numpy()
            for pred, fn in zip(preds, fnames):
                f.write(f"{fn},{int(pred)}\n")




## === cell 5
sub_df = pd.read_csv(submission_path)
assert len(sub_df) == len(
