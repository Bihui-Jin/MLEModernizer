# 5. Code solution

## === cell 0
import os, copy, random, json, pathlib
from pathlib import Path

import pandas as pd
import numpy as np
import cv2  # added for image reading

import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader
import torchvision.transforms as T
import torchvision.models as models

possible_roots = [
    Path("/kaggle/input/cassava-leaf-disease-classification"),
    Path("./input/cassava-leaf-disease-classification"),
    Path("./data/cassava-leaf-disease-classification"),
]
DATA_ROOT = next((p for p in possible_roots if p.exists()), None)
if DATA_ROOT is None:
    raise FileNotFoundError(
        "Could not locate cassava-leaf-disease-classification data directory."
    )

TRAIN_IMG_DIR = DATA_ROOT / "train_images"
TEST_IMG_DIR = DATA_ROOT / "test_images"
TRAIN_CSV = DATA_ROOT / "train.csv"
SUBMISSION_PATH = Path("./submission.csv")



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(int)


class CassavaTrainDataset(Dataset):
    def __init__(self, df, img_dir, transforms):
        self.df = df.reset_index(drop=True)
        self.img_dir = Path(img_dir)
        self.transforms = transforms

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = self.img_dir / row["image_id"]
        img = cv2.imread(str(img_path))[:, :, ::-1]  # BGR to RGB
        if self.transforms:
            img = self.transforms(img)
        label = row["label"]
        return img, label


class CassavaTestDataset(Dataset):
    def __init__(self, img_dir, transforms):
        self.img_dir = Path(img_dir)
        self.transforms = transforms
        self.fns = sorted(
            [p.name for p in self.img_dir.iterdir() if p.suffix.lower() == ".jpg"]
        )

    def __len__(self):
        return len(self.fns)

    def __getitem__(self, idx):
        fn = self.fns[idx]
        img_path = self.img_dir / fn
        img = cv2.imread(str(img_path))[:, :, ::-1]
        if self.transforms:
            img = self.transforms(img)
        return img, fn


def get_transforms(train: bool):
    if train:
        return T.Compose(
            [
                T.ToPILImage(),
                T.Resize(224),
                T.RandomHorizontalFlip(),
                T.RandomVerticalFlip(),
                T.ToTensor(),
                T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
            ]
        )
    else:
        return T.Compose(
            [
                T.ToPILImage(),
                T.Resize(224),
                T.ToTensor(),
                T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
            ]
        )




## === cell 2
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

model = models.resnet18(pretrained=True)
model.fc = nn.Linear(model.fc.in_features, 5)
model = model.to(device)

criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)

train_dataset = CassavaTrainDataset(train_df, TRAIN_IMG_DIR, get_transforms(train=True))
train_loader = DataLoader(
    train_dataset, batch_size=64, shuffle=True, num_workers=0, pin_memory=True
)

model.train()
for epoch in range(5):  # increased epochs to boost performance modestly
    running_loss = 0.0
    for imgs, labels in train_loader:
        imgs = imgs.to(device, non_blocking=True)
        labels = labels.to(device)

        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        running_loss += loss.item() * imgs.size(0)
    epoch_loss = running_loss / len(train_loader.dataset)
    print(f"Epoch {epoch+1}/5 - Loss: {epoch_loss:.4f}")



## === cell 3
test_dataset = CassavaTestDataset(TEST_IMG_DIR, get_transforms(train=False))
test_loader = DataLoader(
    test_dataset, batch_size=64, shuffle=False, num_workers=0, pin_memory=True
)

model.eval()
predictions = []
filenames = []

with torch.no_grad():
    for imgs, fns in test_loader:
        imgs = imgs.to(device)
        logits = model(imgs)
        preds = torch.argmax(logits, dim=1).cpu().numpy()
        predictions.extend(preds.tolist())
        filenames.extend(fns)

assert (
    len(predictions) == len(filenames) == len(test_dataset)
), "Length mismatch between predictions and test set"



## === cell 4
with open(SUBMISSION_PATH, "w") as f:
    f.write("image_id,label\n")
    for fn, pred in zip(filenames, predictions):
        f.write(f"{fn},{pred}\n")

print(f"Submission written to {SUBMISSION_PATH} (rows: {len(predictions)})")
