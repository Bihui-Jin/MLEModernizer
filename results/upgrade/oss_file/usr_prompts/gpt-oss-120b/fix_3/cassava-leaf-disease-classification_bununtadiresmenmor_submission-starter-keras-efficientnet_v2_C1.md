## === cell 0
import os
import json
import numpy as np
import pandas as pd
import cv2
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

SEED = 42
np.random.seed(SEED)

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_PATH = os.path.join(BASE_PATH, "train_images")
TEST_IMG_PATH = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUBMIT = os.path.join(BASE_PATH, "sample_submission.csv")

IMG_SIZE = (64, 64)  # formerly (32, 32)
N_ESTIMATORS = 300  # formerly 150




## === cell 1
train_df = pd.read_csv(TRAIN_CSV)


def load_and_preprocess(image_path):
    """Read an image, resize, and flatten to a 1‑D uint8 vector."""
    img = cv2.imread(image_path, cv2.IMREAD_COLOR)
    if img is None:
        return np.zeros(IMG_SIZE[0] * IMG_SIZE[1] * 3, dtype=np.uint8)
    img = cv2.resize(img, IMG_SIZE, interpolation=cv2.INTER_AREA)
    return img.flatten()


X = []
y = []

for _, row in train_df.iterrows():
    img_file = os.path.join(TRAIN_IMG_PATH, row["image_id"])
    X.append(load_and_preprocess(img_file))
    y.append(row["label"])

X = np.array(X, dtype=np.uint8)
y = np.array(y, dtype=np.int32)

X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.2, random_state=SEED, stratify=y
)

rf_clf = RandomForestClassifier(
    n_estimators=N_ESTIMATORS,
    random_state=SEED,
    n_jobs=-1,
    max_depth=None,
    min_samples_split=2,
    min_samples_leaf=1,
    class_weight="balanced",  # help with any class imbalance
)
rf_clf.fit(X_train, y_train)

val_pred = rf_clf.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy: {val_acc:.4f}")




## === cell 2
sample_sub = pd.read_csv(SAMPLE_SUBMIT)

test_features = []
test_ids = []

for img_id in sample_sub["image_id"]:
    img_path = os.path.join(TEST_IMG_PATH, img_id)
    test_features.append(load_and_preprocess(img_path))
    test_ids.append(img_id)

test_X = np.array(test_features, dtype=np.uint8)

test_preds = rf_clf.predict(test_X)

submission = pd.DataFrame({"image_id": test_ids, "label": test_preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
