## === cell 0
import os
import cv2
import json
import numpy as np
import pandas as pd
from pathlib import Path
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score




## === cell 1
BASE_INPUT = Path("../input/cassava-leaf-disease-classification")
TRAIN_CSV = BASE_INPUT / "train.csv"
TEST_SAMPLE_SUB = BASE_INPUT / "sample_submission.csv"
TRAIN_IMG_DIR = BASE_INPUT / "train_images"
TEST_IMG_DIR = BASE_INPUT / "test_images"

IMG_SIZE = (64, 64)  # larger size for richer features
SEED = 42
HIST_BINS = 16  # per channel histogram bins


def load_and_preprocess(img_path: Path) -> np.ndarray:
    """Read an image, resize, convert to RGB, normalize, flatten and append color histograms."""
    img = cv2.imread(str(img_path))
    if img is None:
        flat_len = IMG_SIZE[0] * IMG_SIZE[1] * 3
        hist_len = HIST_BINS * 3
        return np.zeros(flat_len + hist_len, dtype=np.float32)

    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img_resized = cv2.resize(img, IMG_SIZE, interpolation=cv2.INTER_AREA)
    img_norm = img_resized.astype(np.float32) / 255.0
    flat = img_norm.flatten()

    hist_features = []
    for channel in range(3):
        hist = cv2.calcHist([img_resized], [channel], None, [HIST_BINS], [0, 256])
        hist = hist.flatten()
        hist_sum = hist.sum()
        if hist_sum > 0:
            hist = hist / hist_sum
        hist_features.append(hist)
    hist_vec = np.concatenate(hist_features)

    return np.concatenate([flat, hist_vec])


train_df = pd.read_csv(TRAIN_CSV)

X_list = []
y_list = []
for idx, row in train_df.iterrows():
    img_path = TRAIN_IMG_DIR / row["image_id"]
    X_list.append(load_and_preprocess(img_path))
    y_list.append(row["label"])
X = np.stack(X_list, axis=0)
y = np.array(y_list)

X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, random_state=SEED, stratify=y
)




## === cell 2
rf_clf = RandomForestClassifier(
    n_estimators=500,  # more trees for better performance
    max_depth=None,
    class_weight="balanced",  # address any class imbalance
    n_jobs=-1,
    random_state=SEED,
    verbose=0,
)
rf_clf.fit(X_train, y_train)

val_preds = rf_clf.predict(X_val)
val_acc = accuracy_score(y_val, val_preds)
print(f"Validation accuracy: {val_acc:.5f}")




## === cell 3
test_sub = pd.read_csv(TEST_SAMPLE_SUB)
test_images = test_sub["image_id"].tolist()

X_test = []
for image_id in test_images:
    img_path = TEST_IMG_DIR / image_id
    X_test.append(load_and_preprocess(img_path))
X_test = np.stack(X_test, axis=0)

test_preds = rf_clf.predict(X_test)

submission = pd.DataFrame({"image_id": test_images, "label": test_preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
