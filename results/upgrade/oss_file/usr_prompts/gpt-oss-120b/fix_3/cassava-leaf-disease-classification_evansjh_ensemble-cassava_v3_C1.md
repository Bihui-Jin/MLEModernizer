## === cell 0
import os
import pandas as pd
import numpy as np
from collections import Counter
from PIL import Image

train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_image_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
sample_submission_path = (
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)



## === cell 1
train_df = pd.read_csv(train_csv_path)

most_common_label = train_df["label"].mode().iloc[0]


def extract_feature(img_path, size=(32, 32)):
    """
    Load an image, resize to a small fixed size, and return the mean RGB values.
    Returns a 3‑element numpy array.
    """
    try:
        with Image.open(img_path) as im:
            im = im.convert("RGB")
            im = im.resize(size)
            arr = np.asarray(im, dtype=np.float32) / 255.0
            return arr.mean(axis=(0, 1))  # mean over height and width
    except Exception:
        return None


centroids = {}
for label in train_df["label"].unique():
    features = []
    class_images = train_df[train_df["label"] == label]["image_id"]
    for img_name in class_images:
        img_path = os.path.join(train_image_dir, img_name)
        feat = extract_feature(img_path)
        if feat is not None:
            features.append(feat)
    if features:
        centroids[label] = np.mean(features, axis=0)
    else:
        centroids[label] = np.zeros(3, dtype=np.float32)



## === cell 2
test_images = [
    f
    for f in os.listdir(test_image_dir)
    if f.lower().endswith((".jpg", ".jpeg", ".png"))
]




## === cell 3
def predict_label(feature, centroids_dict):
    """Assign the label of the nearest centroid (Euclidean distance)."""
    if feature is None:
        return most_common_label
    best_label = None
    best_dist = float("inf")
    for lbl, ctr in centroids_dict.items():
        dist = np.linalg.norm(feature - ctr)
        if dist < best_dist:
            best_dist = dist
            best_label = lbl
    return best_label if best_label is not None else most_common_label


predictions = []
for img_name in test_images:
    img_path = os.path.join(test_image_dir, img_name)
    feat = extract_feature(img_path)
    pred = predict_label(feat, centroids)
    predictions.append(pred)

submission_df = pd.DataFrame({"image_id": test_images, "label": predictions})



## === cell 4
submission_path = "/kaggle/working/submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission file saved at: {submission_path}")
