## === cell 0
import os
import random
import pandas as pd
import numpy as np
from collections import Counter
from PIL import Image



## === cell 1
train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_image_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
submission_path = "/kaggle/working/submission.csv"
random.seed(42)



## === cell 2
train_df = pd.read_csv(train_csv_path)
most_common_label = train_df["label"].mode()[0]
print(f"Most common training label (fallback): {most_common_label}")




## === cell 3
def mean_intensity(image_path):
    """Return the average grayscale intensity of an image."""
    try:
        im = Image.open(image_path).convert("L")
        return np.array(im).mean()
    except Exception:
        return np.nan


label_intensity = {}
samples_per_label = 200  # limit work to keep runtime low

for label in sorted(train_df["label"].unique()):
    ids = train_df[train_df["label"] == label]["image_id"].tolist()
    sampled_ids = random.sample(ids, min(samples_per_label, len(ids)))
    intensities = []
    for img_id in sampled_ids:
        img_path = os.path.join(train_image_dir, img_id)
        intensities.append(mean_intensity(img_path))
    label_intensity[int(label)] = np.nanmean(intensities)

print("Average grayscale intensity per label:")
for lbl, val in label_intensity.items():
    print(f"  label {lbl}: {val:.2f}")



## === cell 4
test_image_ids = [
    f
    for f in os.listdir(test_image_dir)
    if f.lower().endswith((".jpg", ".jpeg", ".png"))
]
print(f"Found {len(test_image_ids)} test images.")




## === cell 5
def predict_label(mean_val, label_means):
    """Assign the label whose mean intensity is closest to `mean_val`."""
    if np.isnan(mean_val):
        return most_common_label
    best_label = min(label_means.items(), key=lambda kv: abs(mean_val - kv[1]))[0]
    return int(best_label)


predicted_labels = []
for img_id in test_image_ids:
    img_path = os.path.join(test_image_dir, img_id)
    img_mean = mean_intensity(img_path)
    pred = predict_label(img_mean, label_intensity)
    predicted_labels.append(pred)



## === cell 6
submission_df = pd.DataFrame({"image_id": test_image_ids, "label": predicted_labels})
submission_df.to_csv(submission_path, index=False)
print(f"Submission file saved at: {submission_path}")
