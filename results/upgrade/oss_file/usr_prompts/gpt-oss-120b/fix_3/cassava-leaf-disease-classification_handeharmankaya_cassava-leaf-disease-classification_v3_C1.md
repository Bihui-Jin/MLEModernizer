## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras.models import load_model, Model, save_model
from tensorflow.keras.layers import Input, Conv2D, MaxPooling2D, Flatten, Dense

MODEL_PATH = "/kaggle/input/leaf/keras/default/1/leaf_model.keras"
TEST_DIR = "/kaggle/input/cassava-leaf-disease-classification/test_images"
TRAIN_DIR = "/kaggle/input/cassava-leaf-disease-classification/train_images"
TRAIN_CSV = "/kaggle/input/cassava-leaf-disease-classification/train.csv"

IMAGE_SIZE = [128, 128]
BATCH_SIZE = 32

## === cell 1
try:
    model = load_model(MODEL_PATH)
except Exception as e:
    print("Fallback triggered:", e)

    inputs = Input(shape=(*IMAGE_SIZE, 3))
    x = Conv2D(16, (3, 3), activation="relu")(inputs)
    x = MaxPooling2D()(x)
    x = Conv2D(32, (3, 3), activation="relu")(x)
    x = MaxPooling2D()(x)
    x = Flatten()(x)
    outputs = Dense(5, activation="softmax")(x)  # 5 classes
    model = Model(inputs, outputs)

    train_df = pd.read_csv(TRAIN_CSV)
    shuffled = train_df.sample(frac=1, random_state=42).reset_index(drop=True)
    split_idx = int(0.9 * len(shuffled))
    train_split = shuffled[:split_idx]
    val_split = shuffled[split_idx:]

    def make_dataset(df):
        paths = df["image_id"].apply(lambda x: os.path.join(TRAIN_DIR, x)).values
        labels = df["label"].values.astype(np.int32)
        ds = tf.data.Dataset.from_tensor_slices((paths, labels))

        def _load_image(path, label):
            img_bytes = tf.io.read_file(path)
            img = tf.image.decode_jpeg(img_bytes, channels=3)
            img = tf.image.resize(img, IMAGE_SIZE)
            img = tf.cast(img, tf.float32) / 255.0
            return img, label

        ds = ds.map(_load_image, num_parallel_calls=tf.data.AUTOTUNE)
        ds = ds.shuffle(1000).batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
        return ds

    train_dataset = make_dataset(train_split)
    val_dataset = make_dataset(val_split)

    model.compile(
        optimizer="adam",
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )
    model.fit(
        train_dataset,
        validation_data=val_dataset,
        epochs=5,
        verbose=1,
    )

    try:
        os.makedirs(os.path.dirname(MODEL_PATH), exist_ok=True)
        save_model(model, MODEL_PATH)
        print(f"Trained model saved to {MODEL_PATH}")
    except Exception:
        pass

sample_sub = pd.read_csv(

def process_image(image_path):
    bits = tf.io.read_file(image_path)
    image = tf.image.decode_jpeg(bits, channels=3)
    image = tf.cast(image, tf.float32) / 255.0
    image = tf.image.resize(image, IMAGE_SIZE)
    return image


test_dataset = tf.data.Dataset.from_tensor_slices(sample_sub["path"].values)
test_dataset = test_dataset.map(process_image, num_parallel_calls=tf.data.AUTOTUNE)
test_dataset = test_dataset.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)



## === cell 3
predictions = model.predict(test_dataset, verbose=1)
final_preds = np.argmax(predictions, axis=1)

sample_sub["label"] = final_preds
submission = sample_sub[["image_id", "label"]]
submission.to_csv("submission.csv", index=False)
