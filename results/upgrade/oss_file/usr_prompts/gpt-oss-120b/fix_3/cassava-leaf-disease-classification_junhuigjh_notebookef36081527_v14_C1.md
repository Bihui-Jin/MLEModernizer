# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras.preprocessing import image as kp_image
from tensorflow.keras.applications.resnet50 import ResNet50, preprocess_input
from sklearn.tree import DecisionTreeClassifier

BASE_INPUT = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_INPUT, "train_images")
TEST_IMG_DIR = os.path.join(BASE_INPUT, "test_images")
TRAIN_CSV = os.path.join(BASE_INPUT, "train.csv")
SAMPLE_SUB = os.path.join(BASE_INPUT, "sample_submission.csv")



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(int)  # ensure integer labels
train_image_paths = [
    os.path.join(TRAIN_IMG_DIR, fname) for fname in train_df["image_id"]
]




## === cell 2
def load_and_preprocess_tf(img_path):
    """TensorFlow‑based image loading and preprocessing."""
    img = tf.io.read_file(img_path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [224, 224])
    img = preprocess_input(img)  # same preprocessing as ImageNet
    return img




## === cell 3
feature_extractor = ResNet50(weights="imagenet", include_top=False, pooling="avg")



## === cell 4
BATCH_SIZE = 256  # larger batch reduces number of predict calls

train_ds = tf.data.Dataset.from_tensor_slices(train_image_paths)
train_ds = train_ds.map(
    lambda x: load_and_preprocess_tf(x), num_parallel_calls=tf.data.AUTOTUNE
)
train_ds = train_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

train_features = feature_extractor.predict(train_ds, verbose=0)
train_labels = train_df["label"].values



## === cell 5
tree_clf = DecisionTreeClassifier(criterion="gini", max_depth=6, random_state=42)
tree_clf.fit(train_features, train_labels)



## === cell 6
sample_sub = pd.read_csv(SAMPLE_SUB)  # provides ordered test image ids
test_image_ids = sample_sub["image_id"].tolist()
test_image_paths = [os.path.join(TEST_IMG_DIR, fname) for fname in test_image_ids]



## === cell 7
test_ds = tf.data.Dataset.from_tensor_slices(test_image_paths)
test_ds = test_ds.map(
    lambda x: load_and_preprocess_tf(x), num_parallel_calls=tf.data.AUTOTUNE
)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

test_features = feature_extractor.predict(test_ds, verbose=0)
test_predictions = tree_clf.predict(test_features)



## === cell 8
submission = pd.DataFrame({"image_id": test_image_ids, "label": test_predictions})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}, shape: {submission.shape}")
