## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.tree import DecisionTreeClassifier




## === cell 1
def extract_features(pil_img):
    """
    Compute simple color statistics:
    - mean of each RGB channel
    - standard deviation of each RGB channel
    Returns a 6‑element numpy array.
    """
    img_arr = np.asarray(pil_img).astype(np.float32) / 255.0  # shape (H, W, 3)
    flat = img_arr.reshape(-1, 3)
    means = flat.mean(axis=0)  # (3,)
    stds = flat.std(axis=0)  # (3,)
    return np.concatenate([means, stds])  # (6,)




## === cell 2
base_path = "/kaggle/input/cassava-leaf-disease-classification"
train_csv_path = os.path.join(base_path, "train.csv")
train_img_dir = os.path.join(base_path, "train_images")

train_df = pd.read_csv(train_csv_path)
train_image_ids = train_df["image_id"].tolist()
train_labels = train_df["label"].values

train_features = []
for img_id in train_image_ids:
    img_path = os.path.join(train_img_dir, img_id)
    pil_img = Image.open(img_path).convert("RGB")
    feats = extract_features(pil_img)
    train_features.append(feats)
train_features = np.stack(train_features, axis=0)  # (N_train, 6)

decision_tree = DecisionTreeClassifier(
    criterion="gini", max_depth=8, min_samples_split=12
)
decision_tree.fit(train_features, train_labels)



## === cell 3
