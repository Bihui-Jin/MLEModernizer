## === cell 0
import os
import numpy as np
import pandas as pd



## === cell 1
train_path = "../input/cassava-leaf-disease-classification/train.csv"
train_df = pd.read_csv(train_path)
train_df["label"] = train_df["label"].astype(int)
majority_label = train_df["label"].value_counts().idxmax()



## === cell 2
try:
    from PIL import Image
    from sklearn.ensemble import RandomForestClassifier
    from sklearn.model_selection import train_test_split

    libs_available = True
except Exception as e:
    print("Required ML libraries not available:", e)
    libs_available = False



## === cell 3
if libs_available:
    IMAGE_SIZE = (64, 64)  # small resolution to keep training quick
    MAX_TRAIN_SAMPLES = 5000  # limit number of training images processed

    train_images_dir = "../input/cassava-leaf-disease-classification/train_images/"

    X = []
    y = []
    sampled = 0
    for idx, row in train_df.iterrows():
        if sampled >= MAX_TRAIN_SAMPLES:
            break
        img_path = os.path.join(train_images_dir, row["image_id"])
        if not os.path.isfile(img_path):
            continue
        try:
            img = Image.open(img_path).convert("RGB")
            img = img.resize(IMAGE_SIZE)
            X.append(np.asarray(img).flatten())
            y.append(row["label"])
            sampled += 1
        except Exception as e:
            continue

    X = np.array(X)
    y = np.array(y)

    if len(X) == 0:
        print("No training images could be loaded – falling back to majority label.")
        libs_available = False
    else:
        X_train, X_val, y_train, y_val = train_test_split(
            X, y, test_size=0.1, random_state=42, stratify=y
        )

        rf = RandomForestClassifier(
            n_estimators=200,
            max_depth=None,
            n_jobs=-1,
            random_state=42,
        )
        rf.fit(X_train, y_train)
        val_acc = (rf.predict(X_val) == y_val).mean()
        print(f"Validation accuracy of simple RF model: {val_acc:.4f}")



## === cell 4
test_dir = "../input/cassava-leaf-disease-classification/test_images/"
test_filenames = sorted(
    [f for f in os.listdir(test_dir) if os.path.isfile(os.path.join(test_dir, f))]
)

if libs_available:
    test_X = []
    for fname in test_filenames:
        img_path = os.path.join(test_dir, fname)
        try:
            img = Image.open(img_path).convert("RGB")
            img = img.resize(IMAGE_SIZE)
            test_X.append(np.asarray(img).flatten())
        except Exception as e:
            test_X.append(None)

    valid_idx = [i for i, arr in enumerate(test_X) if arr is not None]
    if valid_idx:
        valid_arrays = np.stack([test_X[i] for i in valid_idx])
        preds = rf.predict(valid_arrays)
    else:
        preds = np.array([])

    final_preds = []
    pred_iter = iter(preds)
    for i in range(len(test_filenames)):
        if i in valid_idx:
            final_preds.append(next(pred_iter))
        else:
            final_preds.append(majority_label)
else:
    final_preds = [majority_label] * len(test_filenames)



## === cell 5
output_path = "/kaggle/working/submission.csv"
submission_df = pd.DataFrame({"image_id": test_filenames, "label": final_preds})
submission_df.to_csv(output_path, index=False)
print(f"Submission written to {output_path} with {len(submission_df)} rows.")
