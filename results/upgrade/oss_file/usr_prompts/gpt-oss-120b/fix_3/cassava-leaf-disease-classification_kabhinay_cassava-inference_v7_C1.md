base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=IMG_SIZE + (3,)
)
base_model.trainable = False  # freeze base

model = Sequential(
    [base_model, GlobalAveragePooling2D(), Dense(5, activation="softmax")]
)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()



## === cell 4
EPOCHS = 3

test_datagen = ImageDataGenerator(rescale=1.0 / 255)

test_df = pd.DataFrame({"image_id": sorted(os.listdir(TEST_DIR))})
test_generator = test_datagen.flow_from_dataframe(
    dataframe=test_df,
    directory=TEST_DIR,

    target_size=IMG_SIZE,
    batch_size=32,
    class_mode=None,
    shuffle=False,
)

preds = model.predict(test_generator, verbose=1)
predicted_class_indices = np.argmax(preds, axis=1)

label_map = {0: "0", 1: "1", 2: "2", 3: "3", 4: "4"}
predicted_labels = [label_map[idx] for idx in predicted_class_indices]
