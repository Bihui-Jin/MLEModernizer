## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf

IMG_SIZE = 512  # image dimensions
BATCH_SIZE = 32
NUM_CLASSES = 5

base = tf.keras.applications.EfficientNetB0(
    include_top=False,
    weights=None,  # start from random weights
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
)

model = tf.keras.Sequential(
    [
        base,
        tf.keras.layers.GlobalAveragePooling2D(),
        tf.keras.layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

model.compile(
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    optimizer="adam",
    metrics=["accuracy"],
)

## === cell 2
test_dir = os.path.join(
    "..", "input", "cassava-leaf-disease-classification", "test_images"
)
test_files = sorted([f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")])
file_paths = [os.path.join(test_dir, f) for f in test_files]


def _load_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0  # normalize to [0,1]
    return img


dataset = (
    tf.data.Dataset.from_tensor_slices(file_paths)
    .map(_load_and_preprocess, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

probabilities = model.predict(dataset, verbose=0)
pred_labels = np.argmax(probabilities, axis=1).astype(int)

submission = pd.DataFrame({"image_id": test_files, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
