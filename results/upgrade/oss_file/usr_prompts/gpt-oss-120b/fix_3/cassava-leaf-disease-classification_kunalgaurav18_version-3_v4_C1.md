test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
test_files = sorted(os.listdir(test_dir))

image_ids = []
pred_labels = []

for start in range(0, len(test_files), BATCH_SIZE):
    batch_files = test_files[start : start + BATCH_SIZE]
    batch_paths = [os.path.join(test_dir, f) for f in batch_files]

    batch_imgs = [
        tf.keras.preprocessing.image.load_img(p, target_size=(IMG_SIZE, IMG_SIZE))
        for p in batch_paths
    ]
    batch_arr = [tf.keras.preprocessing.image.img_to_array(img) for img in batch_imgs]
    batch_np = np.stack(batch_arr).astype(np.float32) / 255.0

    preds = cnn.predict(batch_np, batch_size=len(batch_np), verbose=0)
    batch_preds = np.argmax(preds, axis=1).astype(int)

    image_ids.extend(batch_files)
    pred_labels.extend(batch_preds.tolist())

submission = pd.DataFrame({"image_id": image_ids, "label": pred_labels})
submission.to_csv("submission.csv", index=False)
