## === cell 1
TRAIN_IMG_DIR = "/kaggle/input/cassava-leaf-disease-classification/train_images/"
TEST_IMG_DIR = "/kaggle/input/cassava-leaf-disease-classification/test_images/"


def _load_image(path):
    """Read, decode and resize an image."""
    img = tf.io.read_file(tf.strings.join([path]))
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, (224, 224))
    img = tf.cast(img, tf.float32) / 255.0
    return img


def _load_train(image_name, label):
    img_path = tf.strings.join([TRAIN_IMG_DIR, image_name])
    return _load_image(img_path), tf.one_hot(label, depth=5)


def _load_test(image_name):
    img_path = tf.strings.join([TEST_IMG_DIR, image_name])
    return _load_image(img_path), image_name




## === cell 2
train_df = pd.read_csv("/kaggle/input/cassava-leaf-disease-classification/train.csv")
train_images = train_df["image_id"].values
train_labels = train_df["label"].values.astype(np.int64)

train_ds = tf.data.Dataset.from_tensor_slices((train_images, train_labels))
train_ds = train_ds.map(_load_train, num_parallel_calls=tf.data.AUTOTUNE)
train_ds = train_ds.shuffle(2048).batch(32).prefetch(tf.data.AUTOTUNE)



## === cell 3
test_files = tf.io.gfile.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
test_filenames = [os.path.basename(p) for p in test_files]

test_ds = tf.data.Dataset.from_tensor_slices(test_filenames)
test_ds = test_ds.map(_load_test, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(32).prefetch(tf.data.AUTOTUNE)



## === cell 4
base = tf.keras.applications.MobileNetV2(
    input_shape=(224, 224, 3), include_top=False, weights="imagenet", pooling="avg"
)

inputs = keras.Input(shape=(224, 224, 3))
x = base(inputs, training=False)  # freeze base during the first epoch
outputs = keras.layers.Dense(5, activation="softmax")(x)
model = keras.Model(inputs, outputs)

model.compile(
    optimizer=keras.optimizers.Adam(learning_rate=1e-3),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()



## === cell 5
EPOCHS = 2
history = model.fit(train_ds, epochs=EPOCHS, verbose=2)



## === cell 6
pred_probs = []
filenames = []

for batch_images, batch_names in test_ds:
    probs = model.predict(batch_images, verbose=0)
    pred_probs.append(probs)
    filenames.append(batch_names.numpy())

pred_probs = np.concatenate(pred_probs, axis=0)
filenames = np.concatenate(filenames, axis=0).astype("U")  # ensure string dtype
pred_labels = np.argmax(pred_probs, axis=-1).astype(np.int64)



## === cell 7
submission_path = "submission.csv"
np.savetxt(
    submission_path,
    np.rec.fromarrays([filenames, pred_labels]),
    fmt=["%s", "%d"],
    delimiter=",",
    header="image_id,label",
    comments="",
)

print("First 5 rows of submission:")
with open(submission_path, "r") as f:
    for _ in range(6):
        print(f.readline().strip())
