## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image



## === cell 1
work_dir = "/kaggle/input/cassava-leaf-disease-classification/"
train_dir = os.path.join(work_dir, "train_images")
test_dir = os.path.join(work_dir, "test_images")
train_df = pd.read_csv(os.path.join(work_dir, "train.csv"))



## === cell 2
image_size = (64, 64)  # larger than 32×32 for richer info
class_sums = {}
class_counts = {}

for _, row in train_df.iterrows():
    label = int(row["label"])
    img_path = os.path.join(train_dir, row["image_id"])
    try:
        img = Image.open(img_path).convert("RGB")
        img = img.resize(image_size)
        arr = np.asarray(img).astype(np.float32)
        mean_rgb = arr.mean(axis=(0, 1))
        std_rgb = arr.std(axis=(0, 1))
        feat = np.concatenate([mean_rgb, std_rgb])  # shape (6,)
        class_sums[label] = class_sums.get(label, np.zeros(6)) + feat
        class_counts[label] = class_counts.get(label, 0) + 1
    except Exception:
        continue

centroids = {lbl: class_sums[lbl] / class_counts[lbl] for lbl in class_sums}



## === cell 3
test_images = sorted(os.listdir(test_dir))  # deterministic order
predictions = []

for img_name in test_images:
    img_path = os.path.join(test_dir, img_name)
    try:
        img = Image.open(img_path).convert("RGB")
        img = img.resize(image_size)
        arr = np.asarray(img).astype(np.float32)
        mean_rgb = arr.mean(axis=(0, 1))
        std_rgb = arr.std(axis=(0, 1))
        feat = np.concatenate([mean_rgb, std_rgb])
        distances = {lbl: np.linalg.norm(feat - centroids[lbl]) for lbl in centroids}
        pred_label = min(distances, key=distances.get)
    except Exception:
        pred_label = 0  # fallback class if image cannot be processed
    predictions.append(int(pred_label))

submission = pd.DataFrame({"image_id": test_images, "label": predictions})
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
