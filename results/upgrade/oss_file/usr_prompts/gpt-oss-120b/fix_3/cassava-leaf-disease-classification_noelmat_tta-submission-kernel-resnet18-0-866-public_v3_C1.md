from fastai.vision.all import *
import pandas as pd, json
from functools import partial
import torch



## === cell 1
path = Path("/kaggle/input/cassava-leaf-disease-classification/")
assert path.exists(), f"{path} not found"



## === cell 2
train_images_path = path / "train_images"
_cached_train_files = get_image_files(train_images_path)


def get_label(lbls, x):
    return lbls[Path(x).name]


def get_data(lbls, bs=64, presize=500, resize=384, num_workers=4, pin_memory=True):
    tfms = [
        Rotate(90),
        Warp(magnitude=0.4, p=1.0),
        Zoom(min_zoom=0.9, max_zoom=1.3),
        Brightness(max_lighting=0.5),
        Flip(),
        Contrast(),
        Resize(resize),
    ]
    comp = setup_aug_tfms(tfms)
    return DataBlock(
        blocks=(ImageBlock, CategoryBlock),
        get_items=lambda p: _cached_train_files,  # use cached file list
        get_y=partial(get_label, lbls),
        splitter=RandomSplitter(),
        item_tfms=[Resize(presize)],
        batch_tfms=[*comp, Normalize.from_stats(*imagenet_stats)],
    ).dataloaders(
        train_images_path, bs=bs, num_workers=num_workers, pin_memory=pin_memory
    )




## === cell 3
dls = get_data(labels, bs=128, presize=384, num_workers=4)
learn = cnn_learner(dls, models.resnet18, metrics=accuracy, pretrained=True)

learn.fine_tune(1, base_lr=1e-3)

test_files = get_image_files(path / "test_images")
test_dl = dls.test_dl(test_files, bs=256)

preds, _ = learn.tta(dl=test_dl, n=6)
preds = torch.argmax(preds, dim=1)

test_fn = [f.name for f in test_files]
