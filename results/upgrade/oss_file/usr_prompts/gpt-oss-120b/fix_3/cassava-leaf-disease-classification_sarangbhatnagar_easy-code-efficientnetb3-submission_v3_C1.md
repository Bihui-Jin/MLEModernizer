import tensorflow as tf

)

train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images/"

train_df = pd.read_csv(train_csv_path)  # columns: image_id, label
train_paths = (
    train_df["image_id"].apply(lambda x: os.path.join(train_images_dir, x)).values
)
train_labels = train_df["label"].values

AUTOTUNE = tf.data.AUTOTUNE
BATCH_SIZE = 32


def _parse_image(filename, label):
    img = tf.io.read_file(filename)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [128, 128])
    img = tf.keras.applications.efficientnet.preprocess_input(img)
    return img, label


train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(_parse_image, num_parallel_calls=AUTOTUNE)
train_ds = (
    train_ds.shuffle(buffer_size=len(train_df)).batch(BATCH_SIZE).prefetch(AUTOTUNE)
)

val_size = int(0.1 * len(train_df))
val_ds = train_ds.take(val_size // BATCH_SIZE)
train_ds = train_ds.skip(val_size // BATCH_SIZE)

EPOCHS = 5
model.fit(train_ds, epochs=EPOCHS, validation_data=val_ds, verbose=2)



## === cell 3
sample_sub_path = (
    "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
)
test_images_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"

ss = pd.read_csv(sample_sub_path)
preds = []

preprocess_fn = tf.keras.applications.efficientnet.preprocess_input

for img_name in ss["image_id"]:
    img_path = os.path.join(test_images_dir, img_name)
    img = tf.keras.preprocessing.image.load_img(img_path, target_size=(128, 128))
    img_array = tf.keras.preprocessing.image.img_to_array(img)
    img_array = preprocess_fn(img_array)
    img_batch = np.expand_dims(img_array, axis=0)  # shape (1, 128, 128, 3)

    probs = model.predict(img_batch, verbose=0)
    pred_label = int(np.argmax(probs, axis=1)[0])
    preds.append(pred_label)

submission = pd.DataFrame({"image_id": ss["image_id"], "label": preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
print(submission.head())
