## === cell 0
import pandas as pd
import numpy as np
import glob
import os
from PIL import Image

SEED = 42
DEBUG = False
np.random.seed(SEED)



## === cell 1
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
if not os.path.exists(train_csv_path):
    train_csv_path = "data/cassava-leaf-disease-classification/train.csv"
train_df = pd.read_csv(train_csv_path)

id_to_label = dict(zip(train_df["image_id"], train_df["label"]))



## === cell 2
train_images = glob.glob(
    "../input/cassava-leaf-disease-classification/train_images/*.jpg"
)
if not train_images:
    train_images = glob.glob(
        "data/cassava-leaf-disease-classification/train_images/*.jpg"
    )

train_ids = []
train_features = []

for path in train_images:
    img_id = os.path.basename(path)
    if img_id not in id_to_label:
        continue  # safety: skip any stray files
    try:
        img = Image.open(path).convert("RGB")
        arr = np.array(img)
        mean_rgb = arr.mean(axis=(0, 1))  # shape (3,)
        train_features.append(mean_rgb)
        train_ids.append(img_id)
    except Exception:
        continue

train_features = np.vstack(train_features)  # (N_train, 3)
train_labels = np.array([id_to_label[i] for i in train_ids])



## === cell 3
test_images = glob.glob(
    "../input/cassava-leaf-disease-classification/test_images/*.jpg"
)
if not test_images:
    test_images = glob.glob(
        "data/cassava-leaf-disease-classification/test_images/*.jpg"
    )

test_ids = []
test_features = []

for path in test_images:
    img_id = os.path.basename(path)
    test_ids.append(img_id)
    try:
        img = Image.open(path).convert("RGB")
        arr = np.array(img)
        mean_rgb = arr.mean(axis=(0, 1))
        test_features.append(mean_rgb)
    except Exception:
        test_features.append(np.full(3, np.nan))

test_features = np.vstack(test_features)  # (N_test, 3)



## === cell 4
most_common_label = train_df["label"].mode()[0]

pred_labels = []
for tf in test_features:
    if np.isnan(tf).any():
        pred_labels.append(most_common_label)
        continue
    dists = np.linalg.norm(train_features - tf, axis=1)
    nearest_idx = np.argmin(dists)
    pred_labels.append(train_labels[nearest_idx])



## === cell 5
submission = pd.DataFrame({"image_id": test_ids, "label": pred_labels})
submission.to_csv("submission.csv", index=False)
