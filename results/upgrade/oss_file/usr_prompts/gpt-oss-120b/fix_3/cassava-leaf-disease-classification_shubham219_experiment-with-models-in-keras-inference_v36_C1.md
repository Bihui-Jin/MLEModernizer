## === cell 0
import os
import glob
import pandas as pd
import numpy as np

SEED = 42
np.random.seed(SEED)



## === cell 1
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
test_images_dir = "../input/cassava-leaf-disease-classification/test_images/"

train_df = pd.read_csv(train_csv_path)
most_common_label = train_df["label"].mode()[0]

test_image_paths = glob.glob(os.path.join(test_images_dir, "*.jpg"))
df_test = pd.DataFrame(test_image_paths, columns=["path"])

submission = df_test.copy()
submission["image_id"] = submission["path"].str.split(os.sep).str[-1]
submission["label"] = most_common_label
final_csv = submission[["image_id", "label"]]



## === cell 2
output_path = "submission.csv"
final_csv.to_csv(output_path, index=False)
print(f"Submission file written to {output_path} with {len(final_csv)} rows.")
