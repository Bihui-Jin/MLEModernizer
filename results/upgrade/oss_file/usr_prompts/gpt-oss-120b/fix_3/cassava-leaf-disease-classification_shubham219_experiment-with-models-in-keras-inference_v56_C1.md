import pandas as pd
import numpy as np
import glob
import os

try:
    from google.protobuf import message_factory

    if not hasattr(message_factory.MessageFactory, "GetPrototype"):

        def _GetPrototype(self, descriptor):
            return self.GetMessageClass(descriptor)

        message_factory.MessageFactory.GetPrototype = _GetPrototype
except Exception:
    pass

import tensorflow as tf
from tensorflow.keras.models import load_model
from tensorflow.keras.applications import EfficientNetB3
from tensorflow.keras.preprocessing.image import ImageDataGenerator

SEED = 42
DEBUG = False
tf.random.set_seed(SEED)



## === cell 1
weight_path = "../input/model-v17/model_v0.38.h5"

try:
    my_model = load_model(
        weight_path, custom_objects={"EfficientNetB3": EfficientNetB3}
    )
    if DEBUG:
        print("Loaded model from", weight_path)
except Exception as e:
    if DEBUG:
        print("Failed to load model:", e)
        print("Creating a fresh EfficientNetB3 model as fallback.")
    base_model = EfficientNetB3(
        weights=None, include_top=False, input_shape=(512, 512, 3)
    )
    x = tf.keras.layers.GlobalAveragePooling2D()(base_model.output)
    outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
    my_model = tf.keras.Model(inputs=base_model.input, outputs=outputs)
    my_model.compile(optimizer="adam", loss="sparse_categorical_crossentropy")



## === cell 2
test_images = glob.glob(
    "../input/cassava-leaf-disease-classification/test_images/*.jpg"
)
df_test = pd.DataFrame(test_images, columns=["path"])


def make_test_gen(batch_size=64):
    test_idg = ImageDataGenerator()
    test_gen = test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=batch_size,
        seed=SEED,
        shuffle=False,
        class_mode=None,
        target_size=(512, 512),
    )
    return test_gen




## === cell 3
pred_list = []
test_gen = make_test_gen(batch_size=128)
pred_test = my_model.predict(test_gen, verbose=1)
pred_test_labels = np.argmax(pred_test, axis=-1)

final_submission = df_test.copy()
final_submission["image_id"] = final_submission["path"].str.split("/").str[-1]
final_submission["label"] = pred_test_labels.astype(int)

final_csv = final_submission[["image_id", "label"]]
final_csv.to_csv("submission.csv", index=False)
