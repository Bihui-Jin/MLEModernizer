import os
import glob
import pandas as pd
from collections import Counter


def locate_path(*relative_parts):
    """
    Return the first existing path among typical Kaggle locations.
    Looks in:
    - /kaggle/input/<...>
    - /kaggle/working/<...>
    - ./<...>
    """
    candidates = [
        os.path.join("/kaggle", "input", *relative_parts),
        os.path.join("/kaggle", "working", *relative_parts),
        os.path.join(*relative_parts),
    ]
    for path in candidates:
        if os.path.exists(path):
            return path
    raise FileNotFoundError(f"Unable to locate {'/'.join(relative_parts)}")
