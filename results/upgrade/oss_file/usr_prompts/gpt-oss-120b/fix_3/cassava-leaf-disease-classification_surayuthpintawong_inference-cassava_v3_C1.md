## === cell 1
class Rescale(object):
    """Rescale the image in a sample to a given size."""

    def __init__(self, output_size):
        assert isinstance(output_size, (int, tuple))
        self.output_size = output_size

    def __call__(self, sample):
        if isinstance(sample, dict):
            image, label = sample["image"], sample["label"]
        else:
            image = sample

        h, w = image.shape[:2]
        if isinstance(self.output_size, int):
            if h > w:
                new_h, new_w = self.output_size * h / w, self.output_size
            else:
                new_h, new_w = self.output_size, self.output_size * w / h
        else:
            new_h, new_w = self.output_size

        new_h, new_w = int(new_h), int(new_w)
        img = transform.resize(image, (new_h, new_w))

        if isinstance(sample, dict):
            return {"image": img, "label": label}
        else:
            return img


class RandomCrop(object):
    """Randomly crop the image in a sample."""

    def __init__(self, output_size):
        assert isinstance(output_size, (int, tuple))
        if isinstance(output_size, int):
            self.output_size = (output_size, output_size)
        else:
            assert len(output_size) == 2
            self.output_size = output_size

    def __call__(self, sample):
        if isinstance(sample, dict):
            image, label = sample["image"], sample["label"]
        else:
            image = sample

        h, w = image.shape[:2]
        new_h, new_w = self.output_size
        top = np.random.randint(0, h - new_h + 1)
        left = np.random.randint(0, w - new_w + 1)
        image = image[top : top + new_h, left : left + new_w]

        if isinstance(sample, dict):
            return {"image": image, "label": label}
        else:
            return image


class ToTensor(object):
    """Convert ndarrays in sample to Tensors."""

    def __call__(self, sample):
        if isinstance(sample, dict):
            image, label = sample["image"], sample["label"]
        else:
            image = sample

        image = image.transpose((2, 0, 1))  # HWC to CHW
        tensor = torch.from_numpy(image).float()
        if isinstance(sample, dict):
            return {"image": tensor, "label": label}
        else:
            return tensor

## === cell 3
class TrainDataset(Dataset):
    """Dataset for training images with labels."""

    def __init__(self, images_dir, csv_path, transform=None):
        self.images_dir = images_dir
        self.transform = transform
        df = pd.read_csv(csv_path)
        self.samples = df.values.tolist()  # list of [image_id, label]

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
        img_name, label = self.samples[idx]
        img_path = os.path.join(self.images_dir, img_name)
        image = io.imread(img_path)
        if self.transform:
            image = self.transform(image)
        return image, torch.tensor(label, dtype=torch.long)




## === cell 4
class TestDataset(Dataset):
    def __init__(self, root_dir, transform=None):
        self.root_dir = root_dir
        self.transform = transform
        self.images = sorted(os.listdir(root_dir))

    def __len__(self):
        return len(self.images)

    def __getitem__(self, idx):
        if torch.is_tensor(idx):
            idx = idx.tolist()
        img_name = self.images[idx]
        img_path = os.path.join(self.root_dir, img_name)
        image = io.imread(img_path)
        if self.transform:
            image = self.transform(image)
        return img_name, image




## === cell 5
EfficientNet = models.efficientnet_b3
model = EfficientNet(pretrained=True).to(device)

in_features = model.classifier[1].in_features
model.classifier[1] = nn.Linear(in_features, 5).to(device)



## === cell 6
train_tfm = transforms.Compose([Rescale(256), RandomCrop(224), ToTensor()])

train_dataset = TrainDataset(
    images_dir="../input/cassava-leaf-disease-classification/train_images/",
    csv_path="../input/cassava-leaf-disease-classification/train.csv",
    transform=train_tfm,
)

train_loader = DataLoader(
    train_dataset, batch_size=64, shuffle=True, num_workers=2, pin_memory=True
)

criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)

model.train()
epochs = 2  # modest number to keep runtime reasonable
for epoch in range(epochs):
    epoch_loss = 0.0
    for imgs, labels in train_loader:
        imgs = imgs.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        epoch_loss += loss.item() * imgs.size(0)

    avg_loss = epoch_loss / len(train_loader.dataset)
    print(f"Epoch [{epoch+1}/{epochs}]  Loss: {avg_loss:.4f}")



## === cell 7
tsfm = transforms.Compose([Rescale(224), ToTensor()])

test_dataset = TestDataset(
    root_dir="../input/cassava-leaf-disease-classification/test_images/", transform=tsfm
)
test_loader = DataLoader(
    test_dataset, batch_size=64, shuffle=False, num_workers=2, pin_memory=True
)

model.eval()
names = []
predicted = []

with torch.no_grad():
    for names_batch, images_batch in test_loader:
        images_batch = images_batch.to(device)
        outputs = model(images_batch)
        preds = torch.argmax(outputs, dim=1).cpu().numpy()
        names.extend(list(names_batch))
        predicted.extend(preds.tolist())



## === cell 8
submission = pd.DataFrame({"image_id": names, "label": predicted})
submission.to_csv("submission.csv", index=False)
