import os

from os import listdir

from PIL import Image

IMG_HEIGHT = 50
IMG_WIDTH = 50
NUM_CHANNELS = 3
batch_size = 500
num_train_images = 25000
num_test_images = 12500
num_train_threads = int(num_train_images / batch_size)  # 50
num_test_threads = int(num_test_images / batch_size)  # 25



## === cell 1
train_dir_path = os.path.abspath("../input/dogs-vs-cats-redux-kernels-edition/train")
test_dir_path = os.path.abspath("../input/dogs-vs-cats-redux-kernels-edition/test")

test_imgs = glob.glob(join(test_dir_path, "**", "*.jpg"), recursive=True)

print("Number of training images (unused):", num_train_images)
print("Number of test images found:", len(test_imgs))




## === cell 2
def get_img_label(fpath):
    category = fpath.split(".")[-3]
    if category == "dog":
        return [1, 0]
    elif category == "cat":
        return [0, 1]
    else:
        return [0, 0]


def load_image(f):
    img = Image.open(f).convert("RGB")
    img = img.resize((IMG_HEIGHT, IMG_WIDTH), Image.ANTIALIAS)
    return np.asarray(img, dtype=np.uint8)


def get_img_array_labels(fpaths):
    imgs = [load_image(f) for f in fpaths]
    labels = [get_img_label(basename(f)) for f in fpaths]
    return np.stack(imgs), np.array(labels)




## === cell 3
print("Ready to generate constant predictions for test set.")
num_test = len(test_imgs)
predictions = np.full((num_test, 2), 0.5)  # column 0 = dog, column 1 = cat



## === cell 4
submission_path = "submission.csv"
with open(submission_path, "w") as f:
    f.write("id,label\n")
    for idx, img_path in enumerate(test_imgs):
        img_id = basename(img_path).split(".")[0]  # removes extension
        prob_dog = predictions[idx, 0]  # probability for dog class
        f.write(f"{img_id},{prob_dog:.6f}\n")
print(f"Submission written to {submission_path}")
