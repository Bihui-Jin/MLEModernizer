## === cell 0
import os
import pandas as pd
import numpy as np
from glob import glob




## === cell 1
train_dir = "/kaggle/input/dogs-vs-cats-redux-kernels-edition/train"
test_dir = "/kaggle/input/dogs-vs-cats-redux-kernels-edition/test"

if not os.path.isdir(train_dir):
    train_dir = "/kaggle/working/train"
if not os.path.isdir(test_dir):
    test_dir = "/kaggle/working/test"

print("train_dir:", train_dir, "exists:", os.path.isdir(train_dir))
print("test_dir :", test_dir, "exists:", os.path.isdir(test_dir))




## === cell 2
train_files = [
    os.path.basename(f)
    for f in glob(os.path.join(train_dir, "**", "*.jpg"), recursive=True)
]
num_train = len(train_files)
dog_count = sum(1 for name in train_files if name.lower().startswith("dog"))
baseline_prob = dog_count / num_train if num_train > 0 else 0.5
print(
    f"Found {num_train} training images, {dog_count} dogs → baseline probability = {baseline_prob:.4f}"
)




## === cell 3
test_files = sorted(
    [
        os.path.basename(f)
        for f in glob(os.path.join(test_dir, "**", "*.jpg"), recursive=True)
    ]
)
ids = []
for name in test_files:
    base = os.path.splitext(name)[0]
    try:
        ids.append(int(base))
    except ValueError:
        ids.append(len(ids) + 1)

submission = pd.DataFrame(
    {"id": ids, "label": np.full(len(ids), baseline_prob, dtype=np.float32)}
)
print("Submission preview:")
print(submission.head())




## === cell 4
output_path = "/kaggle/working/sample_submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission saved to {output_path}")
