## === cell 0
import os
import pandas as pd
from pathlib import Path




## === cell 1
DATA_ROOT = Path("/kaggle/input/dogs-vs-cats-redux-kernels-edition")

TEST_DIR = DATA_ROOT / "test" / "unknown"

if not TEST_DIR.is_dir():
    TEST_DIR = DATA_ROOT / "test"

test_files = sorted([p for p in TEST_DIR.iterdir() if p.suffix.lower() == ".jpg"])

test_ids = [int(p.stem) for p in test_files]




## === cell 2
submission = pd.DataFrame(
    {
        "id": test_ids,
        "label": [1e-5] * len(test_ids),  # probability that the image is a dog
    }
)

print("Number of test samples:", len(submission))
print(submission.head())




## === cell 3
output_path = Path("/kaggle/working/submission.csv")
submission.to_csv(output_path, index=False)
print(f"Submission file written to {output_path}")
