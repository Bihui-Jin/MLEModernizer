## === cell 0
import os, random, glob
import numpy as np
import pandas as pd
from PIL import Image

random.seed(42)
np.random.seed(42)

TRAIN_DIR = "../input/train/"
TEST_DIR = "../input/test/"

ROWS, COLS, CHANNELS = 64, 64, 3

test_images = glob.glob(os.path.join(TEST_DIR, "**", "*.jpg"), recursive=True)
print(f"Found {len(test_images)} test images.")




## === cell 1
def read_image(file_path):
    """Read an image, resize it, and return a uint8 numpy array."""
    img = Image.open(file_path).convert("RGB")
    img = img.resize((ROWS, COLS), Image.Resampling.LANCZOS)
    return np.array(img, dtype=np.uint8)






## === cell 2
prob_dog = np.full(len(test_images), 0.5, dtype=np.float32)

ids = [int(os.path.splitext(os.path.basename(p))[0]) for p in test_images]

submission = pd.DataFrame({"id": ids, "label": prob_dog})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}, shape: {submission.shape}")
