## === cell 0
import os
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import LabelEncoder

print("Input directory contents:", os.listdir("../input"))




## === cell 1
train_path = "../input/iwildcam-2019-fgvc6/train.csv"
test_path = "../input/iwildcam-2019-fgvc6/test.csv"
sample_sub_path = "../input/iwildcam-2019-fgvc6/sample_submission.csv"

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)
sample_submission = pd.read_csv(sample_sub_path)

print("Train shape:", train_df.shape)
print("Test shape:", test_df.shape)
print("Sample submission shape:", sample_submission.shape)




## === cell 2
categorical_cols = ["location", "rights_holder"]
encoders = {}
for col in categorical_cols:
    le = LabelEncoder()
    le.fit(pd.concat([train_df[col], test_df[col]], axis=0).astype(str))
    train_df[col] = le.transform(train_df[col].astype(str))
    test_df[col] = le.transform(test_df[col].astype(str))
    encoders[col] = le

for df in (train_df, test_df):
    df["date_captured"] = pd.to_datetime(df["date_captured"], errors="coerce")
    df["date_ordinal"] = df["date_captured"].apply(
        lambda x: x.toordinal() if pd.notnull(x) else -1
    )
numeric_cols = [
    "width",
    "height",
    "frame_num",
    "seq_num_frames",
    "date_ordinal",
]

X_train = train_df[categorical_cols + numeric_cols].values
y_train = train_df["category_id"].values
X_test = test_df[categorical_cols + numeric_cols].values




## === cell 3
rf_clf = RandomForestClassifier(
    n_estimators=200,
    max_depth=None,
    n_jobs=5,
    random_state=42,
    class_weight="balanced",
)
rf_clf.fit(X_train, y_train)




## === cell 4
test_pred = rf_clf.predict(X_test)

submission = pd.DataFrame({"Id": test_df["id"], "Category": test_pred})

submission = submission.set_index("Id").loc[sample_submission["Id"]].reset_index()

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
print(submission.head())
