## === cell 0
import pandas as pd
import numpy as np
import os



## === cell 1
train = pd.read_csv("../input/plant-pathology-2021-fgvc8/train.csv")
submissions = pd.read_csv("../input/plant-pathology-2021-fgvc8/sample_submission.csv")



## === cell 2
h_target = 512
w_target = 512
batch_size = 32



## === cell 3
from sklearn.preprocessing import MultiLabelBinarizer

label_split = train.labels.apply(lambda x: x.split())
mlb = MultiLabelBinarizer().fit(label_split)
label_columns = mlb.classes_



## === cell 4
num_test = len(submissions)
num_classes = len(label_columns)

label_matrix = mlb.transform(label_split)  # shape: (num_train, num_classes)
class_freq = label_matrix.mean(axis=0)  # proportion of images containing each label
np.random.seed(42)
preds = np.tile(class_freq, (num_test, 1)) + np.random.normal(
    scale=0.01, size=(num_test, num_classes)
)
preds = np.clip(preds, 0.0, 1.0)  # ensure valid probability range



## === cell 5
thresh = 0.5  # lowered to allow more labels based on class priors
healthy_idx = None
if "healthy" in label_columns:
    healthy_idx = np.where(label_columns == "healthy")[0][0]

for i in range(num_test):
    if healthy_idx is not None and preds[i][healthy_idx] == np.max(preds[i]):
        submissions.at[i, "labels"] = "healthy"
    else:
        chosen = label_columns[preds[i] >= thresh]
        if len(chosen) == 0:
            chosen = [label_columns[np.argmax(preds[i])]]
        chosen = [c for c in chosen if c != "healthy"]
        submissions.at[i, "labels"] = " ".join(chosen)



## === cell 6
submissions.to_csv("submission.csv", index=False)
print("Submission file written to submission.csv")
