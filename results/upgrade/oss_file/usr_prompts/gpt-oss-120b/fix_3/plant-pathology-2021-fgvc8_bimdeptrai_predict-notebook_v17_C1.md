**train_images** - The training set images.

## === cell 0
import pandas as pd
import numpy as np
import os



## === cell 1
train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
sub_path = "../input/plant-pathology-2021-fgvc8/sample_submission.csv"

train = pd.read_csv(train_path)
submissions = pd.read_csv(sub_path)



## === cell 2
from sklearn.preprocessing import MultiLabelBinarizer

label_split = train["labels"].apply(lambda x: x.split())

mlb = MultiLabelBinarizer().fit(label_split)
class_counts = mlb.transform(label_split).sum(axis=0)
class_freq = pd.Series(class_counts, index=mlb.classes_)

top_n = 3
baseline_classes = list(class_freq.sort_values(ascending=False).head(top_n).index)




## === cell 3
def make_prediction():
    pred_str = " ".join(baseline_classes)
    return [pred_str] * len(submissions)


submissions["labels"] = make_prediction()



## === cell 4
output_path = "submission.csv"
submissions.to_csv(output_path, index=False)
print(f"Submission written to {output_path}, shape: {submissions.shape}")
