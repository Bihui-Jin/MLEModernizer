## === cell 0
import pandas as pd
import numpy as np
import os




## === cell 1
train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
train = pd.read_csv(train_path)




## === cell 2
from sklearn.preprocessing import MultiLabelBinarizer

label_split = train.labels.apply(lambda x: x.split())
mlb = MultiLabelBinarizer()
labels_bin = pd.DataFrame(mlb.fit_transform(label_split), columns=mlb.classes_)




## === cell 3
label_prevalence = labels_bin.mean()  # proportion of images containing each label




## === cell 4
sub_path = "../input/plant-pathology-2021-fgvc8/sample_submission.csv"
submissions = pd.read_csv(sub_path)




## === cell 5
threshold = 0.05  # previously 0.20
default_label = "healthy"

pred_labels = []
for _ in range(len(submissions)):
    chosen = [
        lbl
        for lbl, freq in label_prevalence.items()
        if lbl != default_label and freq > threshold
    ]
    if not chosen:
        chosen = [default_label]
    pred_labels.append(" ".join(chosen))

submissions["labels"] = pred_labels




## === cell 6
output_path = "submission.csv"
submissions.to_csv(output_path, index=False)
print(f"Submission written to {output_path}")
