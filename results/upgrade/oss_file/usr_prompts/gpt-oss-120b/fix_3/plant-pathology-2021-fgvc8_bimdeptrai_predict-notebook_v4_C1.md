label_lists = train["labels"].apply(lambda x: x.split())
all_labels = [lbl for sublist in label_lists for lbl in sublist]
most_common_label = pd.Series(all_labels).value_counts().idxmax()
unique_labels = sorted(set(all_labels))
print(f"Most common label: {most_common_label}")



## === cell 5
num_test = len(submissions)

np.random.seed(42)

p_common = 0.2
other_labels = [lbl for lbl in unique_labels if lbl != most_common_label]

preds = []
for _ in range(num_test):
    if np.random.rand() < p_common:
        preds.append(most_common_label)
    else:
        preds.append(np.random.choice(other_labels))
preds = np.array(preds)



## === cell 6
submissions["labels"] = preds
submissions.to_csv("submission.csv", index=False)
