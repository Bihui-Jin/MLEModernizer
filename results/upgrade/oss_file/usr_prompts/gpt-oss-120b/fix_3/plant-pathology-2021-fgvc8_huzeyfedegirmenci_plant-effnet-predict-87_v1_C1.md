# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from collections import Counter
from PIL import Image

train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
train_img_dir = "../input/plant-pathology-2021-fgvc8/train_images/"
test_dir = "../input/plant-pathology-2021-fgvc8/test_images/"

train_df = pd.read_csv(train_path)

label_counter = Counter()
for lbls in train_df["labels"]:
    for l in lbls.split():
        label_counter[l] += 1

most_common_label = label_counter.most_common(1)[0][0]
cats = list(label_counter.keys())


def mean_rgb(image_path):
    """Return mean RGB as a length‑3 numpy array."""
    with Image.open(image_path) as img:
        img = img.convert("RGB")
        arr = np.asarray(img).reshape(-1, 3)
        return arr.mean(axis=0)


max_samples = 3000
if len(train_df) > max_samples:
    sample_df = train_df.sample(n=max_samples, random_state=42).reset_index(drop=True)
else:
    sample_df = train_df.copy()

train_features = []  # list of (mean_rgb, label_string)
for _, row in sample_df.iterrows():
    img_file = os.path.join(train_img_dir, row["image"])
    if not os.path.isfile(img_file):
        continue
    try:
        rgb = mean_rgb(img_file)
    except Exception:
        continue
    first_label = row["labels"].split()[0]
    train_features.append((rgb, first_label))

train_features = np.array(train_features, dtype=object)  # shape (N, 2)



## === cell 1
test_images = sorted(os.listdir(test_dir))

pred_images = []
pred_labels = []

for img_name in test_images:
    img_path = os.path.join(test_dir, img_name)
    if not os.path.isfile(img_path):
        continue
    try:
        test_rgb = mean_rgb(img_path)
    except Exception:
        pred_images.append(img_name)
        pred_labels.append(most_common_label)
        continue

    if len(train_features) == 0:
        nearest_label = most_common_label
    else:
        train_rgbs = np.stack(train_features[:, 0])
        dists = np.linalg.norm(train_rgbs - test_rgb, axis=1)
        nearest_idx = np.argmin(dists)
        nearest_label = train_features[nearest_idx, 1]

    pred_images.append(img_name)
    pred_labels.append(nearest_label)



## === cell 2
sub_df = pd.DataFrame({"image": pred_images, "labels": pred_labels})
submission_path = "./submission.csv"
sub_df.to_csv(submission_path, index=False)
