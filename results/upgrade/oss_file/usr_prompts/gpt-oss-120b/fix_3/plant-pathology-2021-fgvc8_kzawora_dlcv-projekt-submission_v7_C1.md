**train_images** - The training set images.

## === cell 0
import pandas as pd
from collections import Counter

train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
train_df = pd.read_csv(train_path)
labels_series = train_df["labels"].str.split()
label_counter = Counter([lbl for sublist in labels_series for lbl in sublist])
most_common_label = label_counter.most_common(1)[0][0]  # e.g., 'healthy'

if len(label_counter) > 1:
    second_common_label = label_counter.most_common(2)[1][0]
else:
    second_common_label = most_common_label



## === cell 1
from pathlib import Path

test_dir = Path("../input/plant-pathology-2021-fgvc8/test_images")
filenames = [p.name for p in test_dir.iterdir() if p.is_file()]

predictions = [
    most_common_label if i % 2 == 0 else second_common_label
    for i in range(len(filenames))
]



## === cell 2
submission = pd.DataFrame({"image": filenames, "labels": predictions})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path} with {len(submission)} rows.")
