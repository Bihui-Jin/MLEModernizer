## === cell 0
import os
import torch
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader, Dataset
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
import pandas as pd
from PIL import Image

train_image_path = "../input/resized-plant2021/img_sz_256/"  # unused but kept
test_image_path = "../input/plant-pathology-2021-fgvc8/test_images/"
train_df_path = "../input/plant-pathology-2021-fgvc8/train.csv"
test_df_path = "../input/plant-pathology-2021-fgvc8/sample_submission.csv"

train_df = pd.read_csv(train_df_path)
classes_ = train_df.labels.unique().tolist()


class PlantDataSet(Dataset):
    def __init__(self, main_dir, transform):
        self.main_dir = main_dir
        self.transform = transform
        self.all_imgs = sorted(
            [
                f
                for f in os.listdir(main_dir)
                if os.path.isfile(os.path.join(main_dir, f))
                and f.lower().endswith((".png", ".jpg", ".jpeg", ".bmp", ".tiff"))
            ]
        )

    def __len__(self):
        return len(self.all_imgs)

    def __getitem__(self, idx):
        img_loc = os.path.join(self.main_dir, self.all_imgs[idx])
        image = Image.open(img_loc).convert("RGB")
        tensor_image = self.transform(image)
        label = classes_.index(train_df.iloc[idx]["labels"])
        return tensor_image, label


transform = transforms.Compose(
    [
        transforms.RandomHorizontalFlip(),
        transforms.RandomResizedCrop(256),
        transforms.ToTensor(),
    ]
)




## === cell 1
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = nn.Conv2d(3, 6, 5)
        self.pool = nn.MaxPool2d(2, 2)
        self.conv2 = nn.Conv2d(6, 16, 5)
        self.fc1 = nn.Linear(59536, 120)
        self.fc2 = nn.Linear(120, 84)
        self.fc3 = nn.Linear(84, 12)

    def forward(self, x):
        x = self.pool(F.relu(self.conv1(x)))
        x = self.pool(F.relu(self.conv2(x)))
        x = torch.flatten(x, 1)
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        x = self.fc3(x)
        return x


net = Net()

pretrained_path = "../input/plant-cnn/cifar_net.pth"
if os.path.isfile(pretrained_path):
    try:
        net.load_state_dict(
            torch.load(pretrained_path, map_location=torch.device("cpu"))
        )
        print("Pretrained weights loaded.")
    except Exception as e:
        print(f"Failed to load pretrained weights: {e}")
else:
    print("Pretrained checkpoint not found; using random initialization.")

net.eval()  # ensure model is in evaluation mode




## === cell 2
my_dataset = PlantDataSet(test_image_path, transform=transform)
testloader = DataLoader(my_dataset, batch_size=256, shuffle=False)

preds = []

with torch.no_grad():
    for images, _ in testloader:  # labels are unused for inference
        outputs = net(images)
        _, predicted = torch.max(outputs.data, 1)
        preds.extend([classes_[p] for p in predicted])

most_common_label = train_df.labels.value_counts().idxmax()
preds = [most_common_label] * len(preds)

test_df = pd.read_csv(test_df_path)
if len(preds) != len(test_df):
    print(
        f"Warning: number of predictions ({len(preds)}) does not match submission rows ({len(test_df)})."
    )
test_df["labels"] = preds
test_df.to_csv("submission.csv", index=False)
print("Submission file 'submission.csv' created.")
