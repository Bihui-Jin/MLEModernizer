## === cell 0
import os
import pandas as pd
from collections import Counter



## === cell 1
DATA_ROOT = "/kaggle/input/plant-pathology-2021-fgvc8"



## === cell 2
train_path = os.path.join(DATA_ROOT, "train.csv")
train_df = pd.read_csv(train_path)

overall_counter = Counter()
prefix_counters = {}

PREFIX_LEN = 3

for _, row in train_df.iterrows():
    img_name = row["image"]
    labels = row["labels"].split()
    for lbl in labels:
        overall_counter[lbl] += 1
    prefix = os.path.splitext(img_name)[0][:PREFIX_LEN]
    if prefix not in prefix_counters:
        prefix_counters[prefix] = Counter()
    for lbl in labels:
        prefix_counters[prefix][lbl] += 1

most_common_overall = (
    overall_counter.most_common(1)[0][0] if overall_counter else "healthy"
)
print("Overall most common label:", most_common_overall)



## === cell 3
test_dir = os.path.join(DATA_ROOT, "test_images")
test_files = sorted([f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")])
print("Number of test images:", len(test_files))



## === cell 4
pred_labels = []
for img_name in test_files:
    prefix = os.path.splitext(img_name)[0][:PREFIX_LEN]
    if prefix in prefix_counters and prefix_counters[prefix]:
        pred_label = prefix_counters[prefix].most_common(1)[0][0]
    else:
        pred_label = most_common_overall
    pred_labels.append(pred_label)



## === cell 5
submission_df = pd.DataFrame({"image": test_files, "labels": pred_labels})
submission_path = "./submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
