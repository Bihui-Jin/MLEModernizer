## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import tensorflow as tf
import pandas as pd
import numpy as np

W = H = 448  # image size
BATCH_SIZE = 16
N_CLASSES = 11

target_cols = [
    "CVC - Abnormal",
    "CVC - Borderline",
    "CVC - Normal",
    "ETT - Abnormal",
    "ETT - Borderline",
    "ETT - Normal",
    "NGT - Abnormal",
    "NGT - Borderline",
    "NGT - Incompletely Imaged",
    "NGT - Normal",
    "Swan Ganz Catheter Present",
]

DATA_ROOT = "../input/ranzcr-clip-catheter-line-classification"
TRAIN_IMG_DIR = os.path.join(DATA_ROOT, "train")
TEST_IMG_DIR = os.path.join(DATA_ROOT, "test")
SAMPLE_SUB_PATH = os.path.join(DATA_ROOT, "sample_submission.csv")




## === cell 1
def load_and_preprocess(path):
    """Read an image file, decode, resize and normalise."""
    image = tf.io.read_file(path)
    image = tf.image.decode_png(image, channels=3)
    image = tf.image.resize(image, (H, W))
    image = tf.cast(image, tf.float32) / 255.0
    mean = tf.constant([0.485, 0.456, 0.406])
    std = tf.constant([0.229, 0.224, 0.225])
    image = (image - mean) / std
    return image


def make_dataset(img_dir, uids, labels=None, shuffle=False):
    """Create a tf.data.Dataset yielding (image, uid) or (image, label) tuples."""
    paths = [os.path.join(img_dir, f"{uid}.jpg") for uid in uids]

    ds = tf.data.Dataset.from_tensor_slices((paths, uids))
    ds = ds.map(
        lambda p, uid: (load_and_preprocess(p), uid),
        num_parallel_calls=tf.data.experimental.AUTOTUNE,
    )
    if shuffle:
        ds = ds.shuffle(buffer_size=len(uids))
    if labels is not None:
        label_array = np.stack([labels[uid] for uid in uids])
        lbl_ds = tf.data.Dataset.from_tensor_slices(label_array.astype(np.float32))
        ds = tf.data.Dataset.zip((ds.map(lambda img, uid: img), lbl_ds))
    else:
        ds = ds.map(lambda img, uid: (img, uid))
    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.experimental.AUTOTUNE)
    return ds




## === cell 2
def get_model():
    """Build EfficientNetB5‑based model for multi‑label classification."""
    base = tf.keras.applications.EfficientNetB5(
        include_top=False,
        input_shape=(H, W, 3),
        pooling="avg",
        weights="imagenet",  # use pretrained ImageNet weights
    )
    x = tf.keras.layers.Dropout(0.3)(base.output)
    outputs = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base.input, outputs=outputs)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )
    return model




## === cell 3
train_df = pd.read_csv(os.path.join(DATA_ROOT, "train.csv"))
label_dict = {}
for _, row in train_df.iterrows():
    uid = row["StudyInstanceUID"]
    label_vec = row[target_cols].values.astype(np.float32)
    label_dict[uid] = label_vec

train_uids = list(label_dict.keys())
train_ds = make_dataset(TRAIN_IMG_DIR, train_uids, labels=label_dict, shuffle=True)

model = get_model()

model.fit(train_ds, epochs=3, verbose=2)



## === cell 4
test_df = pd.read_csv(SAMPLE_SUB_PATH)
test_uids = test_df["StudyInstanceUID"].tolist()
test_ds = make_dataset(TEST_IMG_DIR, test_uids, labels=None, shuffle=False)

predictions = []
for batch_images, batch_uids in test_ds:
    batch_preds = model.predict_on_batch(batch_images)
    predictions.extend(batch_preds.tolist())

predictions = np.array(predictions)  # (num_test, N_CLASSES)



## === cell 5
submission_df = pd.read_csv(SAMPLE_SUB_PATH).copy()
available_cols = [c for c in target_cols if c in submission_df.columns]
submission_df[available_cols] = predictions[:, : len(available_cols)]

submission_path = "submission.csv"
submission_df.to_csv(submission_path, index=False)

print(f"Submission file written to {submission_path}")
