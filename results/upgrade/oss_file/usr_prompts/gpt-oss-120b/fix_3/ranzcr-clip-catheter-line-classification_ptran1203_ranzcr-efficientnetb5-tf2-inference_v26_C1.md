## === cell 1
def load_and_preprocess(path):
    """Read an image file, decode, resize and normalise."""
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, (H, W))
    img = tf.cast(img, tf.float32) / 255.0
    mean = tf.constant([0.485, 0.456, 0.406])
    std = tf.constant([0.229, 0.224, 0.225])
    img = (img - mean) / std
    return img


def make_dataset(img_dir, ids, labels=None, batch_size=16, shuffle=False):
    """Create tf.data.Dataset yielding (image, label) or (image, id)."""
    img_paths = [os.path.join(img_dir, f"{i}.jpg") for i in ids]
    if labels is None:
        ds = tf.data.Dataset.from_tensor_slices((img_paths, ids))
        ds = ds.map(
            lambda p, i: (load_and_preprocess(p), i), num_parallel_calls=AUTOTUNE
        )
    else:
        ds = tf.data.Dataset.from_tensor_slices((img_paths, labels))
        ds = ds.map(
            lambda p, y: (load_and_preprocess(p), tf.cast(y, tf.float32)),
            num_parallel_calls=AUTOTUNE,
        )
    if shuffle:
        ds = ds.shuffle(buffer_size=1024)
    ds = ds.batch(batch_size).prefetch(AUTOTUNE)
    return ds


def make_test_dataset(img_dir):
    """Create test tf.data.Dataset and return ordered ids."""
    file_names = sorted([f for f in os.listdir(img_dir) if f.lower().endswith(".jpg")])
    ids = [os.path.splitext(f)[0] for f in file_names]
    ds = tf.data.Dataset.from_tensor_slices((ids,))
    ds = ds.map(
        lambda i: (load_and_preprocess(os.path.join(img_dir, f"{i}.jpg")), i),
        num_parallel_calls=AUTOTUNE,
    )
    ds = ds.batch(16).prefetch(AUTOTUNE)
    return ds, ids


test_dataset, test_ids = make_test_dataset(TEST_IMG_DIR)



## === cell 2
train_csv_path = os.path.join(BASE_DIR, "train.csv")
train_df = pd.read_csv(train_csv_path)

train_labels = train_df[target_cols].values.astype(np.float32)
train_ids = train_df["StudyInstanceUID"].values.astype(str)

from sklearn.model_selection import train_test_split

train_ids_split, val_ids_split, train_lbl_split, val_lbl_split = train_test_split(
    train_ids, train_labels, test_size=0.1, random_state=42, stratify=train_labels[:, 0]
)

train_ds = make_dataset(
    TRAIN_IMG_DIR, train_ids_split, train_lbl_split, batch_size=32, shuffle=True
)
val_ds = make_dataset(
    TRAIN_IMG_DIR, val_ids_split, val_lbl_split, batch_size=32, shuffle=False
)




## === cell 3
def get_model(base_fn, weight_path=None, lr=0.001):
    """Build EfficientNet base + dropout + sigmoid dense."""
    base = base_fn(
        include_top=False, input_shape=(H, W, 3), weights="imagenet", pooling="avg"
    )
    x = tf.keras.layers.Dropout(0.3)(base.output)
    out = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base.input, outputs=out)

    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )

    if weight_path and os.path.exists(weight_path):
        try:
            model.load_weights(weight_path)
            print(f"Loaded pretrained weights from {weight_path}")
        except Exception as e:
            print(f"Failed to load weights ({e}); continuing with ImageNet weights.")
    else:
        print("No custom weight file found – using ImageNet pretrained weights.")
    return model


base_fn, weight_path = MODEL_MAP["efficientb5"]
model = get_model(base_fn, weight_path)



## === cell 4
print("Starting fine‑tuning on train data...")
model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=2,
    verbose=2,
)
print("Fine‑tuning completed.")



## === cell 5
all_preds = []
for batch_imgs, _ in test_dataset:
    preds = model.predict_on_batch(batch_imgs)
    all_preds.append(preds)

all_preds = np.concatenate(all_preds, axis=0)  # (num_test, N_CLASSES)

submission = pd.DataFrame()
submission["StudyInstanceUID"] = test_ids
for idx, col in enumerate(target_cols):
    submission[col] = all_preds[:, idx]

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
