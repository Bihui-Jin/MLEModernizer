## === cell 1
def get_data_path(*parts):
    """
    Return a valid path under the Kaggle input directory.
    Tries the relative 'input' folder first; if it does not exist,
    falls back to the absolute '/kaggle/input' location.
    """
    rel_path = os.path.join("input", *parts)
    if os.path.isdir(rel_path):
        return rel_path
    abs_path = os.path.join("/", "kaggle", "input", *parts)
    if os.path.isdir(abs_path):
        return abs_path
    raise FileNotFoundError(f"Data folder not found: {rel_path} nor {abs_path}")

class DummyModel:
    def predict(self, X):
        n = X.shape[0]
        return np.full((n, 2), 0.5, dtype=np.float32)


model_T2 = DummyModel()




## === cell 3
def load_test_T2W_images(path_test):
    """
    Load up to six T2‑weighted slices per case, resize to 150×150,
    stack as 3‑channel images and normalize.
    Returns six numpy arrays (one per slice position).
    """
    IMG_PX_SIZE = 150
    arrays = [[] for _ in range(6)]

    for case_entry in sorted(os.scandir(path_test), key=lambda e: e.name):
        case_path = case_entry.path
        t2_folder = None
        for f in os.scandir(case_path):
            if f.is_dir() and "t2" in f.name.lower():
                t2_folder = f.path
                break
        if t2_folder is None:
            continue

        dicom_files = sorted(
            [
                f.path
                for f in os.scandir(t2_folder)
                if f.is_file() and f.name.lower().endswith(".dcm")
            ]
        )

        count = 0
        for dcm_path in dicom_files:
            ds = dicom.dcmread(dcm_path)
            if ds.pixel_array.sum() <= 100000:
                continue

            img = ds.pixel_array.astype(np.float32)
            resized = cv2.resize(
                img, (IMG_PX_SIZE, IMG_PX_SIZE), interpolation=cv2.INTER_LINEAR
            )
            stacked = np.stack([resized] * 3, axis=-1)  # (H,W,3)

            max_val = stacked.max()
            if max_val != 0:
                stacked = stacked / max_val
            if stacked.sum() <= 2500:
                continue

            if count < 6:
                arrays[count].append(stacked)
                count += 1
            else:
                break

    result = []
    for lst in arrays:
        if len(lst) == 0:
            result.append(np.empty((0, IMG_PX_SIZE, IMG_PX_SIZE, 3), dtype=np.float32))
        else:
            arr = np.stack(lst)
            overall_max = arr.max()
            if overall_max != 0:
                arr = arr / overall_max
            result.append(arr)
    return result

def create_sub(path_test, preds):
    """
    Build the submission dataframe.
    Uses the original folder names (e.g., '00002') as BraTS21ID.
    """
    case_ids = [
        os.path.basename(entry.path)
        for entry in sorted(os.scandir(path_test), key=lambda e: e.name)
        if entry.is_dir()
    ]
    preds = np.asarray(preds, dtype=np.float32)
    if len(case_ids) != len(preds):
        min_len = min(len(case_ids), len(preds))
        case_ids = case_ids[:min_len]
        preds = preds[:min_len]
    return pd.DataFrame({"BraTS21ID": case_ids, "MGMT_value": preds})
