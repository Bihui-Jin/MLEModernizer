# 5. Code solution

## === cell 0
import os, glob
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

print("Root dirs:", os.listdir("../input"))



## === cell 1
train_df = pd.read_csv("../input/train.csv")
sample_sub = pd.read_csv("../input/sample_submission.csv")
print("Train shape:", train_df.shape, "Sample shape:", sample_sub.shape)



## === cell 2
train_dirs = glob.glob("../input/**/train", recursive=True)
test_dirs = glob.glob("../input/**/test", recursive=True)
train_dir = train_dirs[0] if train_dirs else "../input/train/train"
test_dir = test_dirs[0] if test_dirs else "../input/test/test"
print("Using train dir:", train_dir)
print("Using test  dir:", test_dir)




## === cell 3
def load_images(id_series, base_path):
    """Load images given a Series of filenames."""
    paths = [os.path.join(base_path, img_id) for img_id in id_series]
    images = [plt.imread(p) for p in paths]
    return np.stack(images)  # shape: (N, H, W, C)


X_all = load_images(train_df["id"], train_dir).astype("float32") / 255.0
y_all = train_df["has_cactus"].values.reshape(-1, 1)

print("All images shape:", X_all.shape, "Labels shape:", y_all.shape)



## === cell 4
from sklearn.model_selection import train_test_split

X_train, X_val, y_train, y_val = train_test_split(
    X_all, y_all, test_size=0.1, random_state=32, stratify=y_all
)

X_train_flat = X_train.reshape(X_train.shape[0], -1)
X_val_flat = X_val.reshape(X_val.shape[0], -1)

print("Train/Val shapes:", X_train_flat.shape, X_val_flat.shape)



## === cell 5
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import roc_auc_score

clf = LogisticRegression(max_iter=1000, n_jobs=-1, solver="lbfgs")
clf.fit(X_train_flat, y_train.ravel())

val_probs = clf.predict_proba(X_val_flat)[:, 1]
val_auc = roc_auc_score(y_val, val_probs)
print("Validation AUC:", val_auc)



## === cell 6
X_test = load_images(sample_sub["id"], test_dir).astype("float32") / 255.0
X_test_flat = X_test.reshape(X_test.shape[0], -1)

test_probs = clf.predict_proba(X_test_flat)[:, 1]



## === cell 7
submission = pd.DataFrame({"id": sample_sub["id"], "has_cactus": test_probs})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
print(submission.head())
