## === cell 0
import os
import numpy as np
import pandas as pd
from pathlib import Path
import torch
from fastai.vision.all import *

np.random.seed(42)
torch.manual_seed(42)



## === cell 1
data_folder = Path("../input")
train_df = pd.read_csv(data_folder / "train.csv")
test_df = pd.read_csv(data_folder / "sample_submission.csv")  # contains test ids



## === cell 2
dls = ImageDataLoaders.from_df(
    train_df,
    path=data_folder,  # root folder containing both train/ and test/
    folder="train",  # subfolder where training images reside
    fn_col="id",
    label_col="has_cactus",
    valid_pct=0.10,
    seed=42,
    item_tfms=Resize(128),
    batch_tfms=aug_transforms(
        do_flip=True,
        flip_vert=True,
        max_rotate=10.0,
        max_zoom=1.1,
        max_lighting=0.2,
        max_warp=0.2,
        p_affine=0.75,
        p_lighting=0.75,
    ),
)

learn = cnn_learner(dls, models.densenet161, metrics=accuracy, model_dir="/tmp/model/")
learn.fine_tune(3)



## === cell 3
test_paths = [data_folder / "test" / fname for fname in test_df["id"]]

test_dl = dls.test_dl(test_paths)

logits, _ = learn.get_preds(dl=test_dl)

probs = torch.nn.functional.softmax(logits, dim=1)[:, 1].cpu().numpy()

submission = pd.DataFrame({"id": test_df["id"], "has_cactus": probs})
submission_path = "submission_fastai.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
