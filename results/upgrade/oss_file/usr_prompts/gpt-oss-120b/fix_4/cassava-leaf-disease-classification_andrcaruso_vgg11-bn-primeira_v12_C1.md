## === cell 0
path = "../input/cassava-leaf-disease-classification/"



## === cell 1
import os
import numpy as np
import pandas as pd
from fastai.vision.all import *
from fastai.callback.tracker import SaveModelCallback
from fastai.callback.mixup import MixUp  # added MixUp for better generalisation



## === cell 2
df = pd.read_csv(path + "train.csv")
caminho = path + "train_images/"

dls = ImageDataLoaders.from_df(
    df,
    path=caminho,
    valid_pct=0.2,
    seed=42,
    item_tfms=Resize(256),  # slightly larger input size
    batch_tfms=aug_transforms(mult=1.0),
    bs=64,
)



## === cell 3
os.makedirs("/kaggle/working/models", exist_ok=True)



## === cell 4
learn = cnn_learner(
    dls,
    resnet34,  # deeper backbone for higher accuracy
    metrics=[error_rate, accuracy],
    pretrained=False,  # keep offline weights
    path="/kaggle/working/models",
)

save_callback = SaveModelCallback(
    fname="best_model", monitor="valid_loss", with_opt=True, reset_on_fit=False
)

learn.fine_tune(10, cbs=[save_callback, MixUp()])



## === cell 5
img = dls.train_ds[0][0]
_ = learn.predict(img)  # sanity check – ensures the model works



## === cell 6
submission_df = pd.read_csv(path + "sample_submission.csv")
caminho_test = path + "test_images/"

tst_dl = learn.dls.test_dl(
    submission_df["image_id"].apply(lambda x: os.path.join(caminho_test, x))
)

preds, _ = learn.tta(dl=tst_dl, n=10)

submission_df["label"] = np.argmax(preds, axis=1)

submission_df.to_csv("submission.csv", index=False)
