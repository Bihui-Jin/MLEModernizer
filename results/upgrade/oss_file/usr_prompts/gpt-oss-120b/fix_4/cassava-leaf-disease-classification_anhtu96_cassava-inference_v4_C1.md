ROOT_DIR = "/kaggle/input"

    ROOT_DIR, "cassava-leaf-disease-classification", "test_images"
)



## === cell 2
train_df = pd.read_csv(TRAIN_CSV_PATH)
most_common_label = train_df["label"].mode().iloc[0]



## === cell 3
IMG_SIZE = (128, 128)  # modest size for speed / memory


def load_and_preprocess(img_path):
    """Load an image file, resize, and scale to the range expected by MobileNetV2."""
    img = keras_image.load_img(img_path, target_size=IMG_SIZE)
    arr = keras_image.img_to_array(img)
    arr = tf.keras.applications.mobilenet_v2.preprocess_input(arr)
    return arr




## === cell 4
base_model = MobileNetV2(
    weights="imagenet", include_top=False, pooling="avg", input_shape=IMG_SIZE + (3,)
)

train_image_paths = [
    os.path.join(TRAIN_IMAGES_DIR, fid) for fid in train_df["image_id"]
]
train_embeddings = []

batch_size = 256
for i in range(0, len(train_image_paths), batch_size):
    batch_paths = train_image_paths[i : i + batch_size]
    batch_imgs = np.stack([load_and_preprocess(p) for p in batch_paths])
    batch_emb = base_model(batch_imgs, training=False).numpy()
    train_embeddings.append(batch_emb)

train_embeddings = np.concatenate(train_embeddings, axis=0)
train_labels = train_df["label"].values

nn = NearestNeighbors(n_neighbors=1, algorithm="auto", metric="euclidean")
nn.fit(train_embeddings)



## === cell 5
test_filenames = [f for f in os.listdir(TEST_IMAGES_DIR) if f.lower().endswith(".jpg")]
test_filenames.sort()
test_image_paths = [os.path.join(TEST_IMAGES_DIR, fname) for fname in test_filenames]



## === cell 6
test_embeddings = []
for i in range(0, len(test_image_paths), batch_size):
    batch_paths = test_image_paths[i : i + batch_size]
    batch_imgs = np.stack([load_and_preprocess(p) for p in batch_paths])
    batch_emb = base_model(batch_imgs, training=False).numpy()
    test_embeddings.append(batch_emb)

test_embeddings = np.concatenate(test_embeddings, axis=0)



## === cell 7
distances, indices = nn.kneighbors(test_embeddings, return_distance=True)
pred_labels = train_labels[indices.ravel()]



## === cell 8
submission_df = pd.DataFrame(
    {"image_id": test_filenames, "label": pred_labels.astype(int)}
)
submission_df.to_csv("submission.csv", index=False)
