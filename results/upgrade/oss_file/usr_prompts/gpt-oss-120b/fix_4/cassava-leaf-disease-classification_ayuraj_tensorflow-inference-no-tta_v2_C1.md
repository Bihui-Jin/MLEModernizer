import os

## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TEST_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")

train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(np.int32)

train_df["img_path"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)

test_images = sorted([f for f in os.listdir(TEST_DIR) if f.lower().endswith(".jpg")])
test_paths = [os.path.join(TEST_DIR, fname) for fname in test_images]




## === cell 2
IMAGE_SIZE = (224, 224)
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE
NUM_CLASSES = 5
SEED = 42
val_fraction = 0.2


def decode_image(path, label=None):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.convert_image_dtype(img, tf.float32)
    img = tf.image.resize(img, IMAGE_SIZE)
    if label is None:
        return img
    else:
        return img, label


def augment(image, label):
    image = tf.image.random_flip_left_right(image)
    image = tf.image.random_flip_up_down(image)
    image = tf.image.random_brightness(image, max_delta=0.1)
    return image, label


train_paths = train_df["img_path"].tolist()
train_labels = train_df["label"].tolist()
train_size = len(train_paths)
val_size = int(train_size * val_fraction)

val_paths = train_paths[:val_size]
val_labels = train_labels[:val_size]
train_paths_split = train_paths[val_size:]
train_labels_split = train_labels[val_size:]

train_ds = tf.data.Dataset.from_tensor_slices((train_paths_split, train_labels_split))
val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))

train_ds = train_ds.map(decode_image, num_parallel_calls=AUTOTUNE)
train_ds = train_ds.map(augment, num_parallel_calls=AUTOTUNE)
val_ds = val_ds.map(decode_image, num_parallel_calls=AUTOTUNE)

train_ds = train_ds.shuffle(10000, seed=SEED).batch(BATCH_SIZE).prefetch(AUTOTUNE)
val_ds = val_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)




## === cell 3
base_model = tf.keras.applications.EfficientNetB0(
    input_shape=IMAGE_SIZE + (3,),
    include_top=False,
    weights="imagenet",
)
base_model.trainable = True

inputs = tf.keras.Input(shape=IMAGE_SIZE + (3,))
x = tf.keras.applications.efficientnet.preprocess_input(inputs)
x = base_model(x, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)




## === cell 4
EPOCHS = 8
model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS, verbose=2)




## === cell 5
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(lambda p: decode_image(p), num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)

preds = model.predict(test_ds)
pred_labels = np.argmax(preds, axis=1)

submission = pd.DataFrame({"image_id": test_images, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}, shape: {submission.shape}")
