## === cell 0
import os
import json
import pandas as pd
import torch
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms
import cv2
import numpy as np


def get_data_root():
    possible_paths = [
        "/kaggle/input/cassava-leaf-disease-classification",
        "./input/cassava-leaf-disease-classification",
        "./data/cassava-leaf-disease-classification",
    ]
    for p in possible_paths:
        if os.path.isdir(p):
            return p
    raise FileNotFoundError(
        "Cassava dataset directory not found in any expected location."
    )


DATA_ROOT = get_data_root()
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
TRAIN_IMG_DIR = os.path.join(DATA_ROOT, "train_images")
TEST_IMG_DIR = os.path.join(DATA_ROOT, "test_images")
SUBMISSION_PATH = "./submission.csv"



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)

max_per_class = 400
centroids = {}
for label in sorted(train_df["label"].unique()):
    subset = train_df[train_df["label"] == label].sample(
        n=min(max_per_class, len(train_df[train_df["label"] == label])),
        random_state=42,
    )
    colours = []
    for _, row in subset.iterrows():
        img_path = os.path.join(TRAIN_IMG_DIR, row["image_id"])
        if not os.path.isfile(img_path):
            continue
        img = cv2.imread(img_path)
        if img is None:
            continue
        colours.append(img.mean(axis=(0, 1)))
    if colours:
        centroids[label] = np.mean(colours, axis=0)
    else:
        centroids[label] = None

majority_label = train_df["label"].mode().iloc[0]
print(f"Computed colour centroids for {len(centroids)} classes.")
print(f"Fallback majority label: {majority_label}")



## === cell 2
test_filenames = [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
test_filenames.sort()
print(f"Number of test images found: {len(test_filenames)}")




## === cell 3
def predict_label(img_path):
    img = cv2.imread(img_path)
    if img is None:
        return majority_label
    img_mean = img.mean(axis=(0, 1))  # BGR average colour
    best_label = majority_label
    best_dist = float("inf")
    for label, centroid in centroids.items():
        if centroid is None:
            continue
        dist = np.linalg.norm(img_mean - centroid)
        if dist < best_dist:
            best_dist = dist
            best_label = label
    return best_label


with open(SUBMISSION_PATH, "w") as f:
    f.write("image_id,label\n")
    for img_name in test_filenames:
        img_path = os.path.join(TEST_IMG_DIR, img_name)
        pred = predict_label(img_path)
        f.write(f"{img_name},{pred}\n")
print(f"Submission written to {SUBMISSION_PATH}")
