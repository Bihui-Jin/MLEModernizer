## === cell 5
file_paths = tf.strings.join([TRAIN_DIR, "/", train_df["image_id"].values])
labels_int = train_df["label"].astype(int).values

full_ds = tf.data.Dataset.from_tensor_slices((file_paths, labels_int))
full_ds = full_ds.shuffle(
    buffer_size=len(file_paths), seed=SEED, reshuffle_each_iteration=False
)

val_size = int(0.1 * len(file_paths))
val_ds = full_ds.take(val_size)
train_ds = full_ds.skip(val_size)


def _parse_image(path, label, augment=False):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_HEIGHT, IMG_WIDTH])
    if augment:
        img = tf.image.random_flip_left_right(img, seed=SEED)
        img = tf.keras.layers.RandomRotation(0.04)(img)  # ~15° rotation
    img = tf.keras.applications.mobilenet_v2.preprocess_input(img)
    label = tf.one_hot(label, NUM_CLASSES)
    return img, label


train_ds = train_ds.map(
    lambda p, l: _parse_image(p, l, augment=True), num_parallel_calls=tf.data.AUTOTUNE
)
val_ds = val_ds.map(
    lambda p, l: _parse_image(p, l, augment=False), num_parallel_calls=tf.data.AUTOTUNE
)

train_ds = train_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
val_ds = val_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
