class DummyModel:
    def __init__(self, num_classes: int = 5):
        self.num_classes = num_classes

    def predict(self, x):
        """
        Return random class scores for each input in the batch.
        Shape: (batch_size, num_classes)
        """
        if isinstance(x, tf.Tensor):
            batch_size = tf.shape(x)[0].numpy()
        else:
            batch_size = x.shape[0]
        batch_size = int(batch_size)
        return np.random.rand(batch_size, self.num_classes)


model1 = DummyModel()
model2 = DummyModel()
model3 = DummyModel()

norm_constant = 0.87 + 0.86 + 0.84
alpha_1 = 0.86 / norm_constant
alpha_2 = 0.87 / norm_constant
alpha_3 = 0.84 / norm_constant



## === cell 2
preds = []

for image_name in image_ids:
    img = keras.preprocessing.image.load_img(
        f"/kaggle/input/cassava-leaf-disease-classification/test_images/{image_name}"
    )
    img_array = img_to_array(img)  # (H, W, C) as numpy
    img_resized = tf.image.resize(tf.convert_to_tensor(img_array), (512, 512))
    img_batch = tf.expand_dims(img_resized, axis=0)  # shape (1, 512, 512, 3)

    pred1 = model1.predict(img_batch) * alpha_1
    pred2 = model2.predict(img_batch) * alpha_2
    pred3 = model3.predict(img_batch) * alpha_3

    ensemble_pred = pred1 + pred2 + pred3
    preds.append(int(np.argmax(ensemble_pred, axis=1)[0]))



## === cell 3
submission_path = "/kaggle/working/submission.csv"

    writer = csv.writer(f)

    for img_id, label in zip(image_ids, preds):
        writer.writerow([img_id, label])
