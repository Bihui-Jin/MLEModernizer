import os
import cv2
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.nn.functional as F
import timm




## === cell 1
def gem(x, p=3, eps=1e-5):
    return F.avg_pool2d(x.clamp(min=eps).pow(p), (x.size(-2), x.size(-1))).pow(1.0 / p)


class GeM(nn.Module):
    def __init__(self, p=3, eps=1e-5):
        super(GeM, self).__init__()
        self.p = nn.Parameter(torch.ones(1) * p)
        self.eps = eps

    def forward(self, x):
        return gem(x, p=self.p, eps=self.eps)

    def __repr__(self):
        return f"{self.__class__.__name__}(p={self.p.item():.4f}, eps={self.eps})"




## === cell 2
class Net(nn.Module):
    def __init__(self, num_classes=5):
        super().__init__()
        self.model = timm.create_model("seresnext50_32x4d", pretrained=True)
        self._avg_pooling = nn.AdaptiveAvgPool2d(1)
        self.dropout = nn.Dropout(0.5)
        self._fc = nn.Linear(2048, num_classes, bias=True)

    def forward(self, inputs):
        x = inputs / 255.0  # normalize to [0,1]
        bs = x.size(0)
        x = self.model.forward_features(x)
        fm = self._avg_pooling(x).view(bs, -1)
        fm = self.dropout(fm)
        return self._fc(fm)




## === cell 3
class DatasetTest:
    """Simple test‑set dataset that returns a filename and an 8‑fold TTA batch."""

    def __init__(self, test_data_dir):
        self.root_dir = test_data_dir
        self.ds = self._get_list(test_data_dir)

    def _get_list(self, dir_path):
        return sorted(os.listdir(dir_path))

    def __len__(self):
        return len(self.ds)

    def _preprocess(self, image):
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = cv2.resize(image, (640, 640))

        aug = [
            image,
            np.rot90(image, 1),
            np.rot90(image, 2),
            np.rot90(image, 3),
            np.fliplr(image),
            np.rot90(np.fliplr(image), 1),
            np.rot90(np.fliplr(image), 2),
            np.rot90(np.fliplr(image), 3),
        ]
        batch = np.stack(aug)  # (8, H, W, C)
        batch = np.transpose(batch, (0, 3, 1, 2))  # (8, C, H, W)
        return image, batch

    def __getitem__(self, idx):
        fname = self.ds[idx]
        img_path = os.path.join(self.root_dir, fname)
        image = cv2.imread(img_path, -1)

        if image is None:
            image = np.zeros((640, 640, 3), dtype=np.uint8)

        _, aug_batch = self._preprocess(image)
        return fname, aug_batch

    ]
    if os.path.isdir(model_dir)
    else []

test_dir = os.path.join(
    kaggle_root, "cassava-leaf-disease-classification", "test_images"
)
dataiter = DatasetTest(test_dir)


def predict_with_model(model, weight_paths):
    """Run TTA inference for every weight file (or a single pass if none)."""
    model.eval()
    if not weight_paths:
        weight_paths = [None]

    for wp in weight_paths:
        if wp:
            state = torch.load(wp, map_location=device)
            model.load_state_dict(state, strict=False)

        image_ids = []
        preds = []

        for i in range(len(dataiter)):
            fname, aug_batch = dataiter[i]
            inp = torch.from_numpy(aug_batch).to(device).float()
            with torch.no_grad():
                out = model(inp)  # (8, num_classes)
                prob = torch.nn.functional.softmax(out, dim=-1)
                prob = prob.mean(dim=0)  # average over TTA
                label = prob.argmax().item()
            image_ids.append(fname)
            preds.append(label)

        results = pd.DataFrame({"image_id": image_ids, "label": preds})
        results.to_csv("submission.csv", index=False)

        if wp:
            base = os.path.splitext(os.path.basename(wp))[0]
            results.to_csv(f"{base}.csv", index=False)


model = Net().to(device)
predict_with_model(model, weights)
