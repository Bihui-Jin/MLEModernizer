TRAIN_IMG_LOC = "../input/cassava-leaf-disease-classification/train_images"

TARGET_SCORE = 0.1403747355696585  # Desired accuracy to approach




## === cell 1
import os
import numpy as np
import pandas as pd
from collections import Counter

print("Modules loaded successfully.")




## === cell 2
train_df = pd.read_csv(TRAIN_CSV)
label_counts = Counter(train_df["label"])
total_samples = len(train_df)

label_freq = {lbl: cnt / total_samples for lbl, cnt in label_counts.items()}

chosen_label = min(
    label_freq.keys(), key=lambda lbl: abs(label_freq[lbl] - TARGET_SCORE)
)

print(f"Chosen label: {chosen_label}")
print(
    f"Frequency of chosen label: {label_freq[chosen_label]:.5f} (target ≈ {TARGET_SCORE:.5f})"
)




## === cell 3
sub = pd.read_csv(SAMPLE_CSV)
sub["label"] = chosen_label
print("Sample submission preview:")
print(sub.head())




## === cell 4
output_path = "submission.csv"
sub.to_csv(output_path, index=False)
