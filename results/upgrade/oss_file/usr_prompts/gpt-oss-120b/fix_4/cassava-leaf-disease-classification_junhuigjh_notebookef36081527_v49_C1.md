# 5. Code solution

## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image

train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_images_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"

train_df = pd.read_csv(train_csv_path)


def rgb_stats(image_path):
    """Return mean and std of RGB channels as a 6‑dim vector."""
    with Image.open(image_path) as img:
        img = img.convert("RGB")
        arr = np.array(img).astype(np.float32) / 255.0
        mean = arr.mean(axis=(0, 1))
        std = arr.std(axis=(0, 1))
        return np.concatenate([mean, std])


train_features = []
train_labels = []

for _, row in train_df.iterrows():
    img_path = os.path.join(train_images_dir, row["image_id"])
    if os.path.exists(img_path):
        train_features.append(rgb_stats(img_path))
        train_labels.append(int(row["label"]))

train_features = np.stack(train_features)  # shape (N_train, 6)
train_labels = np.array(train_labels)  # shape (N_train,)


test_images = sorted(
    [
        f
        for f in os.listdir(test_images_dir)
        if f.lower().endswith((".png", ".jpg", ".jpeg"))
    ]
)

pred_labels = []
for img_name in test_images:
    img_path = os.path.join(test_images_dir, img_name)
    feat = rgb_stats(img_path)  # (6,)
    dists = np.linalg.norm(train_features - feat, axis=1)
    nearest_idx = np.argmin(dists)
    pred_labels.append(train_labels[nearest_idx])

submission = pd.DataFrame({"image_id": test_images, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path} with {len(submission)} rows.")
