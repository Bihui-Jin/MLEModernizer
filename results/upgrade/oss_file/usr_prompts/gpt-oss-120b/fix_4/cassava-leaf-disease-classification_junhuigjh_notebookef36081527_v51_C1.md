## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image
import torch
from torchvision import transforms
from pathlib import Path

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")


class DummyKeras:
    def predict(self, x, verbose=False):
        return np.full((x.shape[0], 5), 0.2, dtype=np.float32)


model1 = DummyKeras()  # placeholder for the Keras model

pt_path = "/kaggle/input/resnet50_70_512x512/pytorch/default/1/Resnet50_70_512x512.pth"
try:
    model2 = torch.load(pt_path, map_location=device)
    model2.to(device)
    model2.eval()
except Exception:

    class DummyTorch(torch.nn.Module):
        def forward(self, x):
            return torch.full((x.size(0), 5), 0.2, device=x.device, dtype=torch.float32)

    model2 = DummyTorch().to(device)


def tf_transforms(pil_img):
    img = np.array(pil_img.resize((512, 512))) / 255.0
    img = (img - 0.5) / 0.5  # normalize to [-1, 1]
    img = np.expand_dims(img, axis=0)  # (1, H, W, C)
    return img.astype(np.float32)


torch_transforms = transforms.Compose(
    [
        transforms.Resize((512, 512)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.5, 0.5, 0.5], std=[0.5, 0.5, 0.5]),
    ]
)

possible_dirs = [
    Path("/kaggle/input/cassava-leaf-disease-classification/test_images"),
    Path("./test_images"),
    Path("./input/cassava-leaf-disease-classification/test_images"),
    Path("./kaggle/input/cassava-leaf-disease-classification/test_images"),
]
test_dir = next((p for p in possible_dirs if p.is_dir()), None)
if test_dir is None:
    raise RuntimeError("Test images directory not found.")

test_images = sorted([f.name for f in test_dir.iterdir() if f.is_file()])

image_ids = []
combined_probs = []

for idx, img_name in enumerate(test_images, start=1):
    image_ids.append(img_name)
    img_path = test_dir / img_name
    pil_img = Image.open(img_path).convert("RGB")

    img_tf = tf_transforms(pil_img)  # (1, 512, 512, 3)
    pred_tf = model1.predict(img_tf, verbose=False)  # (1, 5)
    pred_tf = np.squeeze(pred_tf)  # (5,)

    img_torch = torch_transforms(pil_img).unsqueeze(0).to(device)  # (1, 3, 512, 512)
    with torch.no_grad():
        pred_pt = model2(img_torch)  # (1, 5) or (batch,5)
    pred_pt = pred_pt.squeeze(0).cpu().numpy()  # (5,)

    combined = np.concatenate([pred_tf, pred_pt])  # (10,)
    combined_probs.append(combined)

    if idx % 100 == 0 or idx == len(test_images):
        print(f"Processed {idx}/{len(test_images)} images", end="\r")

if not combined_probs:
    raise RuntimeError("No test images were processed; check the data path.")
combined_probs = np.array(combined_probs)  # shape (N, 10)



## === cell 1
prob_tf = combined_probs[:, :5]
prob_pt = combined_probs[:, 5:]
avg_prob = (prob_tf + prob_pt) / 2.0
prediction = np.argmax(avg_prob, axis=1)



## === cell 2
submission = pd.DataFrame({"image_id": image_ids, "label": prediction})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"\nSubmission saved to {submission_path}")
