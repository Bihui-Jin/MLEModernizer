## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image




## === cell 1
import torch

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")




## === cell 2
train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"

train_df = pd.read_csv(train_csv_path)




## === cell 3
def average_rgb(image_path):
    """Return the mean RGB vector of an image."""
    try:
        img = Image.open(image_path).convert("RGB")
        arr = np.asarray(img, dtype=np.float32) / 255.0
        return arr.mean(axis=(0, 1))  # shape (3,)
    except Exception:
        return np.array([0.5, 0.5, 0.5], dtype=np.float32)


centroids = {}
train_vectors = []  # average RGB of each training image
train_labels = []  # corresponding label

for label in train_df["label"].unique():
    mask = train_df["label"] == label
    img_paths = [
        os.path.join(train_images_dir, fname)
        for fname in train_df.loc[mask, "image_id"]
    ]
    colours = [average_rgb(p) for p in img_paths if os.path.exists(p)]
    if colours:
        centroids[int(label)] = np.mean(colours, axis=0)
        train_vectors.extend(colours)
        train_labels.extend([int(label)] * len(colours))
    else:
        centroids[int(label)] = np.array([0.5, 0.5, 0.5], dtype=np.float32)

train_vectors = np.stack(train_vectors) if train_vectors else np.empty((0, 3))
train_labels = np.array(train_labels, dtype=int)




## === cell 4
image_ids = sorted(os.listdir(test_image_dir))




## === cell 5
predictions = []

centroid_labels = np.array(list(centroids.keys()))
centroid_vectors = np.stack([centroids[l] for l in centroid_labels])

for img_name in image_ids:
    img_path = os.path.join(test_image_dir, img_name)
    rgb = average_rgb(img_path)

    if train_vectors.shape[0] > 0:
        dists = np.linalg.norm(train_vectors - rgb, axis=1)
        pred_label = int(train_labels[np.argmin(dists)])
    else:
        dists = np.linalg.norm(centroid_vectors - rgb, axis=1)
        pred_label = int(centroid_labels[np.argmin(dists)])

    predictions.append(pred_label)




## === cell 6
submission = pd.DataFrame({"image_id": image_ids, "label": predictions})
submission_path = (
    "submission.csv"  # saved in the working directory as required by Kaggle
)
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path} with {len(submission)} rows.")
