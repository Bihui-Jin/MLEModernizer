## === cell 0
import os, json, numpy as np, pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models
from tensorflow.keras.utils import to_categorical




## === cell 1
ROOT_DIR = "../input/cassava-leaf-disease-classification/"
TRAIN_IMG_DIR = os.path.join(ROOT_DIR, "train_images")
TEST_IMG_DIR = os.path.join(ROOT_DIR, "test_images")
TRAIN_CSV = os.path.join(ROOT_DIR, "train.csv")
SAMPLE_SUB = os.path.join(ROOT_DIR, "sample_submission.csv")




## === cell 2
train_df = pd.read_csv(TRAIN_CSV)
train_df["filepath"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)
train_paths = train_df["filepath"].values
train_labels = train_df["label"].values.astype(np.int32)




## === cell 3
IMG_SIZE = 224
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE


def _parse_image(filename, label=None):
    image = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [IMG_SIZE, IMG_SIZE])
    image = image / 255.0
    if label is None:
        return image
    else:
        return image, label


train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(_parse_image, num_parallel_calls=AUTOTUNE)
train_ds = (
    train_ds.shuffle(buffer_size=10000, seed=42).batch(BATCH_SIZE).prefetch(AUTOTUNE)
)




## === cell 4
model = models.Sequential(
    [
        layers.Conv2D(
            32, (3, 3), activation="relu", input_shape=(IMG_SIZE, IMG_SIZE, 3)
        ),
        layers.MaxPooling2D(),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D(),
        layers.Conv2D(128, (3, 3), activation="relu"),
        layers.MaxPooling2D(),
        layers.Flatten(),
        layers.Dense(128, activation="relu"),
        layers.Dropout(0.3),
        layers.Dense(5, activation="softmax"),
    ]
)
model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)




## === cell 5
EPOCHS = 5
model.fit(train_ds, epochs=EPOCHS, verbose=1)




## === cell 6
test_files = sorted(os.listdir(TEST_IMG_DIR))
test_paths = [os.path.join(TEST_IMG_DIR, f) for f in test_files]
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(lambda x: _parse_image(x), num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)




## === cell 7
pred_probs = model.predict(test_ds, verbose=0)
preds = np.argmax(pred_probs, axis=1)




## === cell 8
assert len(test_files) == len(
    preds
), "Length mismatch between test files and predictions"
submission = pd.DataFrame({"image_id": test_files, "label": preds})
submission.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv (rows:", len(submission), ")")
