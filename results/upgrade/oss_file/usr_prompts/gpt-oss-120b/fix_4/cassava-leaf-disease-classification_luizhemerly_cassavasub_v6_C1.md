import numpy as np
import pandas as pd
import tensorflow as tf
import os
import re
from functools import partial

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

DATA_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
AUTOTUNE = tf.data.experimental.AUTOTUNE
FILENAMES = tf.io.gfile.glob(DATA_DIR + "train_tfrecords/*.tfrec")
split_ind = int(0.9 * len(FILENAMES))
TRAINING_FILENAMES, VALID_FILENAMES = FILENAMES[:split_ind], FILENAMES[split_ind:]

BATCH_SIZE = 64
IMG_HEIGHT = 200
IMG_WIDTH = 150


def build_fallback_model():
    base = tf.keras.applications.MobileNetV2(
        input_shape=(IMG_HEIGHT, IMG_WIDTH, 3), include_top=False, weights="imagenet"
    )
    base.trainable = False
    x = tf.keras.layers.GlobalAveragePooling2D()(base.output)
    outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
    model = tf.keras.Model(inputs=base.input, outputs=outputs)
    model.compile(
        optimizer="adam", loss="categorical_crossentropy", metrics=["accuracy"]
    )
    return model


model = build_fallback_model()
model.summary()

def count_data_items(filenames):
    n = [
        int(re.compile(r"-([0-9]*)\.").search(filename).group(1))
        for filename in filenames
    ]
    return np.sum(n)


TEST_FILENAMES = tf.io.gfile.glob(DATA_DIR + "test_tfrecords/*.tfrec")
NUM_TEST_IMAGES = count_data_items(TEST_FILENAMES)




## === cell 2
def decode_image(image):
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [IMG_HEIGHT, IMG_WIDTH])
    return image


def read_tfrecord(example, labeled):
    feature_description = (
        {
            "target": tf.io.FixedLenFeature([], tf.int64),
            "image_name": tf.io.FixedLenFeature([], tf.string),
            "image": tf.io.FixedLenFeature([], tf.string),
        }
        if labeled
        else {
            "image_name": tf.io.FixedLenFeature([], tf.string),
            "image": tf.io.FixedLenFeature([], tf.string),
        }
    )
    example = tf.io.parse_single_example(example, feature_description)
    image = decode_image(example["image"])
    if labeled:
        label = tf.cast(example["target"], tf.int32)
        label = tf.one_hot(label, depth=5)
        return image, label
    idnum = example["image_name"]
    return image, idnum


def load_dataset(filenames, labeled=True, ordered=False):
    ignore_order = tf.data.Options()
    if not ordered:
        ignore_order.experimental_deterministic = False
    dataset = tf.data.TFRecordDataset(filenames, num_parallel_reads=AUTOTUNE)
    dataset = dataset.with_options(ignore_order)
    dataset = dataset.map(
        partial(read_tfrecord, labeled=labeled), num_parallel_calls=AUTOTUNE
    )
    return dataset


def get_dataset(filenames, labeled=True, ordered=False):
    dataset = load_dataset(filenames, labeled=labeled, ordered=ordered)
    dataset = dataset.batch(BATCH_SIZE)
    dataset = dataset.prefetch(buffer_size=AUTOTUNE)
    return dataset


train_dataset = get_dataset(TRAINING_FILENAMES, labeled=True, ordered=False)
train_dataset = train_dataset.shuffle(1000)

valid_dataset = get_dataset(VALID_FILENAMES, labeled=True, ordered=False)

test_dataset_full = get_dataset(TEST_FILENAMES, labeled=False, ordered=True)



## === cell 3
model.fit(
    train_dataset,
    epochs=2,
    validation_data=valid_dataset,
)

test_images_dataset = test_dataset_full.map(lambda img, _: img)

probabilities = model.predict(test_images_dataset)
predictions = tf.argmax(probabilities, axis=1).numpy()



## === cell 4
test_ids_dataset = test_dataset_full.map(lambda img, idnum: idnum).unbatch()
test_ids = next(iter(test_ids_dataset.batch(NUM_TEST_IMAGES))).numpy().astype(str)

submission = pd.DataFrame({"image_id": test_ids, "label": predictions})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
