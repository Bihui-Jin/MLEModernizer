test_dir = "../input/cassava-leaf-disease-classification/test_images"
sample_sub_path = "../input/cassava-leaf-disease-classification/sample_submission.csv"

sample_sub = pd.read_csv(sample_sub_path)
ordered_image_ids = sample_sub["image_id"].tolist()

imagenet_mean = np.array([0.485, 0.456, 0.406], dtype=np.float32)
imagenet_std = np.array([0.229, 0.224, 0.225], dtype=np.float32)


def preprocess(img_bgr, flip=False):
    """Resize to 224, apply CLAHE, normalize, and optionally horizontally flip."""
    img = cv2.resize(img_bgr, (224, 224))
    lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)
    l = clahe.apply(l)
    lab = cv2.merge((l, a, b))
    img = cv2.cvtColor(lab, cv2.COLOR_LAB2BGR)

    if flip:
        img = cv2.flip(img, 1)  # horizontal flip

    img = img.astype(np.float32) / 255.0
    img = (img - imagenet_mean) / imagenet_std
    tensor = (
        torch.tensor(img.transpose(2, 0, 1), dtype=torch.float).unsqueeze(0).to(device)
    )
    return tensor


names, preds = [], []

for img_name in tqdm.tqdm(ordered_image_ids, desc="Predicting"):
    file_path = os.path.join(test_dir, img_name)
    img = cv2.imread(file_path)
    if img is None:
        pred_label = 0
    else:
        input_tensor = preprocess(img, flip=False)
        input_tensor_flip = preprocess(img, flip=True)

        with torch.no_grad():
            hr_out = hrnet(input_tensor)
            hr_out_flip = hrnet(input_tensor_flip)
            hr_out_avg = (hr_out + hr_out_flip) / 2.0

            if use_efficient:
                eff_out = efficient(input_tensor)
                eff_out_flip = efficient(input_tensor_flip)
                eff_out_avg = (eff_out + eff_out_flip) / 2.0
                total = (hr_out_avg + eff_out_avg) / 2.0
            else:
                total = hr_out_avg

            prob = torch.nn.functional.softmax(total, dim=1)
            pred_label = int(torch.argmax(prob, dim=1).cpu().item())

    names.append(img_name)
    preds.append(pred_label)
