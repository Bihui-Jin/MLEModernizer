# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, optimizers, callbacks
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.applications.efficientnet import preprocess_input
from PIL import Image
import warnings

warnings.filterwarnings("ignore")
tf.get_logger().setLevel("ERROR")
seed = 42
tf.random.set_seed(seed)
np.random.seed(seed)



## === cell 1
BASE = "../input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE, "train_images")
TEST_IMG_DIR = os.path.join(BASE, "test_images")

train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(str)

from sklearn.model_selection import train_test_split

train_df, val_df = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=seed
)

val_df["label"] = val_df["label"].astype(str)

IMG_SIZE = 128  # smaller size for faster training
BATCH_SIZE = 32
NUM_CLASSES = 5


def load_images(df, img_dir):
    """Load, resize, and preprocess images from a dataframe."""
    img_list = []
    for fname in df["image_id"]:
        img_path = os.path.join(img_dir, fname)
        img = Image.open(img_path).convert("RGB")
        img = img.resize((IMG_SIZE, IMG_SIZE))
        arr = np.array(img, dtype=np.float32)
        arr = preprocess_input(arr)
        img_list.append(arr)
    images = np.stack(img_list, axis=0)
    labels = tf.keras.utils.to_categorical(df["label"].astype(int), NUM_CLASSES)
    return images, labels


train_images, train_labels = load_images(train_df, TRAIN_IMG_DIR)
val_images, val_labels = load_images(val_df, TRAIN_IMG_DIR)

base_model = EfficientNetB0(
    include_top=False,
    weights="imagenet",
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
    pooling="avg",
)

base_model.trainable = True  # fine‑tune entire backbone

inputs = layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = base_model(inputs, training=False)
x = layers.Dropout(0.2)(x)
outputs = layers.Dense(NUM_CLASSES, activation="softmax", dtype="float32")(x)
model = models.Model(inputs, outputs)

model.compile(
    optimizer=optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

es = callbacks.EarlyStopping(
    monitor="val_accuracy", patience=4, restore_best_weights=True, verbose=0
)
rlr = callbacks.ReduceLROnPlateau(
    monitor="val_accuracy", factor=0.5, patience=2, verbose=0
)

model.fit(
    train_images,
    train_labels,
    batch_size=BATCH_SIZE,
    epochs=12,
    validation_data=(val_images, val_labels),
    callbacks=[es, rlr],
    verbose=1,
)



## === cell 2
test_files = sorted(os.listdir(TEST_IMG_DIR))
test_imgs = []
batch_ids = []  # keep ordering

for fname in test_files:
    img_path = os.path.join(TEST_IMG_DIR, fname)
    if os.path.isdir(img_path):
        continue  # skip nested folders like "test_images"
    img = Image.open(img_path).convert("RGB")
    img = img.resize((IMG_SIZE, IMG_SIZE))
    img_arr = np.array(img, dtype=np.float32)
    img_arr = preprocess_input(img_arr)
    test_imgs.append(img_arr)
    batch_ids.append(fname)

test_array = np.stack(test_imgs, axis=0)

preds = model.predict(test_array, batch_size=BATCH_SIZE, verbose=0)
pred_labels = np.argmax(preds, axis=1)



## === cell 3
sub = pd.DataFrame({"image_id": batch_ids, "label": pred_labels})
sub.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv (rows:", len(sub), ")")
