**[train/test]_tfrecords** the image files in tfrecord format.

**label_num_to_disease_map.json** The mapping between each disease code and the real disease name.

## === cell 1
import pandas as pd
import json
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split

from tensorflow.keras.preprocessing.image import (
    ImageDataGenerator,
    load_img,
    img_to_array,
)
from tensorflow.keras.utils import to_categorical

from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import ReduceLROnPlateau, EarlyStopping, ModelCheckpoint



## === cell 2
train = pd.read_csv(os.path.join(INPUT_DIR, "train.csv"))
train.head()



## === cell 3
with open(os.path.join(INPUT_DIR, "label_num_to_disease_map.json")) as f:
    classes = json.load(f)



## === cell 4
train["class"] = train["label"].apply(lambda x: classes[str(x)])



## === cell 6
train["path"] = train["image_id"].apply(lambda x: os.path.join("train_images", str(x)))
train = train.astype(str)

train_df, val_df = train_test_split(
    train, test_size=0.05, random_state=100, stratify=train["label"].values
)



## === cell 7
batch_size = 32  # larger batch size reduces number of steps per epoch


def identity_transform(image):
    return image


train_datagen = ImageDataGenerator(
    preprocessing_function=identity_transform, rescale=1.0 / 255
)

val_datagen = ImageDataGenerator(rescale=1.0 / 255)

train_generator = train_datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=TRAIN_PATH,
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    class_mode="categorical",
    batch_size=batch_size,
    shuffle=True,
    workers=4,
    use_multiprocessing=True,
)

val_generator = val_datagen.flow_from_dataframe(
    dataframe=val_df,
    directory=TRAIN_PATH,
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    class_mode="categorical",
    batch_size=batch_size,
    shuffle=False,
    workers=4,
    use_multiprocessing=True,
)



## === cell 8
base_model = EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3)
)
x = base_model.output
x = GlobalAveragePooling2D()(x)
outputs = Dense(5, activation="softmax")(x)  # 5 classes
model = Model(inputs=base_model.input, outputs=outputs)

model.compile(
    optimizer=Adam(learning_rate=1e-4),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 9
callbacks = [
    ReduceLROnPlateau(monitor="val_loss", factor=0.5, patience=2, verbose=1),
    EarlyStopping(monitor="val_loss", patience=4, verbose=1, restore_best_weights=True),
]

model.fit(
    train_generator,
    validation_data=val_generator,
    epochs=10,
    callbacks=callbacks,
    verbose=2,
    workers=4,
    use_multiprocessing=True,
)
