def get_valid_path(*paths):
    """
    Return the first existing path from the supplied alternatives.
    """
    for p in paths:
        if os.path.exists(p):
            return p
    raise FileNotFoundError(f"None of the candidate paths exist: {paths}")


BASE_ROOT = get_valid_path(
    "./input/cassava-leaf-disease-classification",
    "/kaggle/input/cassava-leaf-disease-classification",
)

SAMPLE_SUBMISSION_PATH = os.path.join(BASE_ROOT, "sample_submission.csv")
TEST_IMAGES_ROOT = os.path.join(BASE_ROOT, "test_images")
TRAIN_IMAGES_ROOT = os.path.join(BASE_ROOT, "train_images")
TRAIN_CSV_PATH = os.path.join(BASE_ROOT, "train.csv")


def get_tta(image_size, p=0.5):
    """
    Test‑time augmentation pipeline.
    Normalization is applied via A.Normalize; ToTensorV2 receives no arguments.
    """
    imagenet_mean = [0.485, 0.456, 0.406]
    imagenet_std = [0.229, 0.224, 0.225]

    augs = A.Compose(
        [
            A.Resize(600, 600),
            A.CenterCrop(image_size, image_size),
            A.HorizontalFlip(p=p),
            A.RandomBrightnessContrast(
                brightness_limit=(-0.1, 0.1), contrast_limit=(-0.1, 0.1), p=p
            ),
            A.OneOf(
                [
                    A.MotionBlur(p=0.2),
                    A.MedianBlur(blur_limit=3, p=0.2),
                    A.Blur(blur_limit=3, p=0.2),
                ],
                p=p,
            ),
            A.Normalize(mean=imagenet_mean, std=imagenet_std),
            ToTensorV2(),
        ]
    )
    return augs


class CassavaLeafTestDataset(Dataset):
    """
    Dataset for test images. Reads the sample_submission file to obtain image ids.
    Returns a dummy label (0) because labels are not needed for inference.
    """

    def __init__(self, root_dir, transforms):
        self.root_dir = root_dir
        self.transform = transforms
        self.df = pd.read_csv(SAMPLE_SUBMISSION_PATH)
        self.image_ids = self.df["image_id"].values

    def __len__(self):
        return len(self.image_ids)

    def __getitem__(self, idx):
        img_path = os.path.join(self.root_dir, self.image_ids[idx])
        image = cv2.imread(img_path)
        if image is None:
            raise FileNotFoundError(f"Image not found: {img_path}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = self.transform(image=image)["image"]
        return image, 0


test_transforms = get_tta(528)
test_data = CassavaLeafTestDataset(TEST_IMAGES_ROOT, test_transforms)
test_loader = DataLoader(test_data, batch_size=8, pin_memory=True, num_workers=2)



## === cell 3
model = models.resnext50_32x4d(pretrained=True)
model.fc = nn.Linear(in_features=2048, out_features=5, bias=True)
model.to(device)
model.eval()
print(model)



## === cell 4
preds = []
with torch.no_grad():
    for images, _ in test_loader:
        images = images.to(device)
        outputs = model(images)
        _, predicted = torch.max(outputs, 1)
        preds.extend(predicted.cpu().numpy().tolist())
print(f"Number of predictions: {len(preds)}")



## === cell 5

if len(preds) != len(submission):
    raise ValueError(
        f"Prediction count ({len(preds)}) does not match submission rows ({len(submission)})."
    )
submission["label"] = preds
submission.to_csv("./submission.csv", index=False)
