## === cell 1
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "../input/cassava-leaf-disease-classification/train_images/"

train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(str)
train_df["path"] = train_df["image_id"].apply(
    lambda x: os.path.join(train_images_dir, x)
)

train_split, val_split = train_test_split(
    train_df, test_size=0.2, stratify=train_df["label"], random_state=SEED
)

batch_size = 64
img_size = (256, 256)

train_idg = ImageDataGenerator(
    preprocessing_function=preprocess_input,
    horizontal_flip=True,
    rotation_range=20,
    width_shift_range=0.1,
    height_shift_range=0.1,
    zoom_range=0.1,
)

val_idg = ImageDataGenerator(preprocessing_function=preprocess_input)

train_gen = train_idg.flow_from_dataframe(
    dataframe=train_split,
    x_col="path",
    y_col="label",
    target_size=img_size,
    batch_size=batch_size,
    class_mode="categorical",
    seed=SEED,
    shuffle=True,
)

val_gen = val_idg.flow_from_dataframe(
    dataframe=val_split,
    x_col="path",
    y_col="label",
    target_size=img_size,
    batch_size=batch_size,
    class_mode="categorical",
    seed=SEED,
    shuffle=False,
)

base_model = ResNet50(weights="imagenet", include_top=False, input_shape=(*img_size, 3))
x = GlobalAveragePooling2D()(base_model.output)
output = Dense(5, activation="softmax")(x)
my_model = Model(inputs=base_model.input, outputs=output)

for layer in base_model.layers:
    layer.trainable = False

my_model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

my_model.fit(
    train_gen,
    epochs=3,
    validation_data=val_gen,
    verbose=2,
    workers=4,
    use_multiprocessing=True,
)
