## === cell 0
import os
import glob
import pandas as pd
import numpy as np
from PIL import Image

SEED = 42
np.random.seed(SEED)




## === cell 1
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "../input/cassava-leaf-disease-classification/train_images/"
test_images_dir = "../input/cassava-leaf-disease-classification/test_images/"

train_df = pd.read_csv(train_csv_path)

SUBSET_SIZE = 5000  # you can adjust; larger may improve a bit more
train_subset = train_df.sample(n=min(SUBSET_SIZE, len(train_df)), random_state=SEED)

label_sums = {}
label_counts = {}

for _, row in train_subset.iterrows():
    img_path = os.path.join(train_images_dir, row["image_id"])
    try:
        img = Image.open(img_path).convert("RGB")
        img_arr = np.asarray(img, dtype=np.float32)
        mean_colour = img_arr.mean(axis=(0, 1))  # (R, G, B)
    except Exception:
        continue

    label = row["label"]
    if label not in label_sums:
        label_sums[label] = np.zeros(3, dtype=np.float64)
        label_counts[label] = 0
    label_sums[label] += mean_colour
    label_counts[label] += 1

centroids = {lbl: (label_sums[lbl] / label_counts[lbl]) for lbl in label_sums}

test_image_paths = glob.glob(os.path.join(test_images_dir, "*.jpg"))
pred_labels = []

for img_path in test_image_paths:
    try:
        img = Image.open(img_path).convert("RGB")
        img_arr = np.asarray(img, dtype=np.float32)
        mean_colour = img_arr.mean(axis=(0, 1))
    except Exception:
        pred_labels.append(train_df["label"].mode()[0])
        continue

    best_label = None
    best_dist = float("inf")
    for lbl, ctr in centroids.items():
        dist = np.linalg.norm(mean_colour - ctr)
        if dist < best_dist:
            best_dist = dist
            best_label = lbl
    if best_label is None:
        best_label = train_df["label"].mode()[0]
    pred_labels.append(best_label)

submission = pd.DataFrame(
    {"image_id": [os.path.basename(p) for p in test_image_paths], "label": pred_labels}
)




## === cell 2
output_path = "submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission file written to {output_path} with {len(submission)} rows.")
