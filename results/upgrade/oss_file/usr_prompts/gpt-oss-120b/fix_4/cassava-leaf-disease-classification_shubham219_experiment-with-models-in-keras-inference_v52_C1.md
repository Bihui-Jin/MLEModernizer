train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
train_img_dir = "../input/cassava-leaf-disease-classification/train_images"

train_df = pd.read_csv(train_csv_path)
train_df["filepath"] = train_df["image_id"].apply(
    lambda x: os.path.join(train_img_dir, x)
)

train_df, val_df = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=SEED
)


def make_dataset(df, training=False):
    paths = df["filepath"].values
    labels = df["label"].values if "label" in df.columns else None
    ds = tf.data.Dataset.from_tensor_slices(
        (paths, labels) if labels is not None else paths
    )

    def _load_image(path, label=None):
        img = tf.io.read_file(path)
        img = tf.image.decode_jpeg(img, channels=3)
        img = tf.image.resize(img, [224, 224])
        img = img / 255.0  # rescale

        if training:
            img = tf.image.random_flip_left_right(img, seed=SEED)
            img = tf.image.random_flip_up_down(img, seed=SEED)
            img = tf.image.random_brightness(img, max_delta=0.2, seed=SEED)

        if label is None:
            return img
        else:
            return img, tf.cast(label, tf.int32)

    if training:
        ds = ds.map(lambda p, l: _load_image(p, l), num_parallel_calls=tf.data.AUTOTUNE)
        ds = ds.shuffle(buffer_size=len(df), seed=SEED)
    else:
        ds = ds.map(lambda p: _load_image(p), num_parallel_calls=tf.data.AUTOTUNE)

    ds = ds.batch(64)
    ds = ds.prefetch(tf.data.AUTOTUNE)
    return ds


train_ds = make_dataset(train_df, training=True)
val_ds = make_dataset(val_df, training=False)

base = EfficientNetB0(weights="imagenet", include_top=False, input_shape=(224, 224, 3))
x = GlobalAveragePooling2D()(base.output)
x = Dropout(0.2, seed=SEED)(x)
outputs = Dense(5, activation="softmax")(x)
model = Model(base.input, outputs)

model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)

model.fit(train_ds, epochs=3, validation_data=val_ds, verbose=1)
