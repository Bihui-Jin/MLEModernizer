## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["label"] = df_train["label"].astype(str)
df_train["filepath"] = df_train["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)

train_df, val_df = train_test_split(
    df_train, test_size=0.2, random_state=SEED, stratify=df_train["label"]
)



## === cell 2
preprocess = tf.keras.applications.resnet50.preprocess_input
train_idg = ImageDataGenerator(
    preprocessing_function=preprocess,
    horizontal_flip=True,
    rotation_range=20,
    width_shift_range=0.1,
    height_shift_range=0.1,
)

val_idg = ImageDataGenerator(preprocessing_function=preprocess)

batch_size = 64
target_size = (256, 256)

train_gen = train_idg.flow_from_dataframe(
    dataframe=train_df,
    x_col="filepath",
    y_col="label",
    target_size=target_size,
    batch_size=batch_size,
    class_mode="categorical",
    seed=SEED,
    shuffle=True,
)

val_gen = val_idg.flow_from_dataframe(
    dataframe=val_df,
    x_col="filepath",
    y_col="label",
    target_size=target_size,
    batch_size=batch_size,
    class_mode="categorical",
    seed=SEED,
    shuffle=False,
)



## === cell 3
base_model = tf.keras.applications.ResNet50(
    weights="imagenet", include_top=False, input_shape=(256, 256, 3)
)
base_model.trainable = False  # freeze base

inputs = tf.keras.Input(shape=(256, 256, 3))
x = base_model(inputs, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
my_model = tf.keras.Model(inputs, outputs)

my_model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

my_model.fit(
    train_gen,
    epochs=5,
    validation_data=val_gen,
    verbose=2,
    workers=4,
    use_multiprocessing=True,
)
