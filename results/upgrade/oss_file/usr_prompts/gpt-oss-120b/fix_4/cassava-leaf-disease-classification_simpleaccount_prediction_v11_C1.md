## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np, pandas as pd, os, cv2
import tensorflow as tf
from sklearn.model_selection import train_test_split
from tensorflow.keras.layers import Input, Lambda, Dense
from tensorflow.keras.models import Model
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.optimizers import Adam



## === cell 1
gm_exp = tf.Variable(3.0, dtype=tf.float32)


def generalized_mean_pool_2d(X):
    pool = (
        tf.reduce_mean(tf.abs(X ** (gm_exp)), axis=[1, 2], keepdims=False) + 1e-7
    ) ** (1.0 / gm_exp)
    return pool


def create_model(input_shape):
    inp = Input(shape=input_shape)
    base = ResNet50(
        weights="imagenet", include_top=False, input_tensor=inp, pooling=None
    )
    for l in base.layers:
        l.trainable = True
    x = Lambda(generalized_mean_pool_2d)(base.output)
    out = Dense(5, activation="softmax", name="label")(x)
    model = Model(inputs=inp, outputs=out)
    model.compile(
        optimizer=Adam(1e-4),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )
    return model




## === cell 2
class ImageDataGenerator(tf.keras.utils.Sequence):
    def __init__(
        self,
        df,
        img_dir,
        batch_size=64,  # ↑ increased batch size to reduce iteration overhead
        img_size=(224, 224, 3),
        shuffle=True,
        augment=False,
    ):
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.batch_size = batch_size
        self.img_h, self.img_w, self.img_c = img_size
        self.shuffle = shuffle
        self.augment = augment
        self.indices = np.arange(len(self.df))
        self.on_epoch_end()

    def __len__(self):
        return int(np.ceil(len(self.df) / self.batch_size))

    def on_epoch_end(self):
        if self.shuffle:
            np.random.shuffle(self.indices)

    def __getitem__(self, idx):
        batch_idx = self.indices[idx * self.batch_size : (idx + 1) * self.batch_size]
        batch_df = self.df.iloc[batch_idx]
        X = np.empty(
            (len(batch_df), self.img_h, self.img_w, self.img_c), dtype=np.float32
        )
        y = np.empty(len(batch_df), dtype=np.int32)
        for i, row in enumerate(batch_df.itertuples()):
            img_path = os.path.join(self.img_dir, row.image_id)
            img = cv2.imread(img_path)
            if img is None:
                img = np.zeros((self.img_h, self.img_w, self.img_c), dtype=np.uint8)
            img = cv2.resize(img, (self.img_w, self.img_h))
            if self.augment and i % 2 == 0:  # simple horizontal flip augmentation
                img = cv2.flip(img, 1)
            X[i] = img / 255.0
            if hasattr(row, "label"):
                y[i] = row.label
        if hasattr(self.df.columns, "__contains__") and "label" in self.df.columns:
            return X, y
        else:
            return X




## === cell 3
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")

train_df = pd.read_csv(os.path.join(BASE_PATH, "train.csv"))
sample_submission = pd.read_csv(os.path.join(BASE_PATH, "sample_submission.csv"))

train_split, val_split = train_test_split(
    train_df, test_size=0.2, stratify=train_df["label"], random_state=42
)

train_gen = ImageDataGenerator(train_split, TRAIN_IMG_DIR, augment=True)
val_gen = ImageDataGenerator(val_split, TRAIN_IMG_DIR, augment=False)

model = create_model((224, 224, 3))
model.fit(
    train_gen,
    validation_data=val_gen,
    epochs=3,
    verbose=2,
)



## === cell 4
test_gen = ImageDataGenerator(
    sample_submission.drop(columns=["label"]),
    TEST_IMG_DIR,
    batch_size=64,
    augment=False,
    shuffle=False,
)

preds = model.predict(test_gen, verbose=0)  # workers args removed
pred_labels = np.argmax(preds, axis=1)
sample_submission["label"] = pred_labels
sample_submission.to_csv("submission.csv", index=False)
