## === cell 0
import os
import numpy as np
import pandas as pd
import torch
import torchvision.transforms as T
import torchvision.models as models
from tqdm import tqdm
from PIL import Image  # reliable image handling




## === cell 1
model_path = "../input/cassavamodel2/model.pt"
if os.path.exists(model_path):
    model = torch.jit.load(model_path, map_location="cpu")
else:
    backbone = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)
    num_ftrs = backbone.fc.in_features
    backbone.fc = torch.nn.Linear(num_ftrs, 5)
    model = backbone
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = model.to(device).eval()




## === cell 2
preprocess = T.Compose(
    [
        T.Resize((224, 224)),
        T.ToTensor(),
        T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ]
)


def predict(model, img_paths, batch_size=32):
    """Run inference on a list of image file paths using PIL."""
    model.eval()
    all_preds = []
    with torch.no_grad():
        for i in tqdm(range(0, len(img_paths), batch_size), desc="Predict"):
            batch_paths = img_paths[i : i + batch_size]
            batch_tensors = []
            for p in batch_paths:
                pil_img = Image.open(p).convert("RGB")
                tensor = preprocess(pil_img)
                batch_tensors.append(tensor)
            batch = torch.stack(batch_tensors).to(device)
            outputs = model(batch)
            preds = torch.argmax(outputs, dim=1).cpu().numpy()
            all_preds.extend(preds)
    return np.array(all_preds)




## === cell 3
test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
filenames = sorted(
    [f for f in os.listdir(test_dir) if f.lower().endswith((".jpg", ".jpeg", ".png"))]
)
test_paths = [os.path.join(test_dir, f) for f in filenames]




## === cell 4
labels = predict(model, test_paths)




## === cell 5
submission = pd.DataFrame({"image_id": filenames, "label": labels.astype(int)})
submission.head()




## === cell 6
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
