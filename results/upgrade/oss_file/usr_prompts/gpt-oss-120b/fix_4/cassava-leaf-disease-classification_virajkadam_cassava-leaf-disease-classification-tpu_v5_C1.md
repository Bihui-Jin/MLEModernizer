# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, optimizers

BASE_DIR = os.getenv("KAGGLE_INPUT_DIR", "/kaggle/input")
DATA_DIR = os.path.join(BASE_DIR, "cassava-leaf-disease-classification")
TRAIN_IMG_DIR = os.path.join(DATA_DIR, "train_images")
TEST_IMG_DIR = os.path.join(DATA_DIR, "test_images")
TRAIN_CSV = os.path.join(DATA_DIR, "train.csv")
SAMPLE_SUB = os.path.join(DATA_DIR, "sample_submission.csv")



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
sample_sub = pd.read_csv(SAMPLE_SUB)



## === cell 2
IMG_SIZE = 224  # EfficientNetB0 default input size

train_paths = (
    train_df["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x)).values
)
train_labels = train_df["label"].values.astype(np.int32)

val_split = 0.2
val_size = int(len(train_paths) * val_split)


def preprocess_image(path, label):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [IMG_SIZE, IMG_SIZE])
    image = image / 255.0
    return image, label


train_ds = tf.data.Dataset.from_tensor_slices(
    (train_paths[:-val_size], train_labels[:-val_size])
)
train_ds = train_ds.map(preprocess_image, num_parallel_calls=tf.data.AUTOTUNE)
train_ds = train_ds.cache().shuffle(1000).batch(32).prefetch(tf.data.AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices(
    (train_paths[-val_size:], train_labels[-val_size:])
)
val_ds = val_ds.map(preprocess_image, num_parallel_calls=tf.data.AUTOTUNE)
val_ds = val_ds.cache().batch(32).prefetch(tf.data.AUTOTUNE)



## === cell 3
base_model = tf.keras.applications.EfficientNetB0(
    include_top=False,
    weights="imagenet",
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
    pooling="avg",
)
base_model.trainable = False  # freeze for initial training

model = models.Sequential([base_model, layers.Dense(5, activation="softmax")])

model.compile(
    optimizer=optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

EPOCHS_STAGE1 = 4
model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS_STAGE1, verbose=2)

base_model.trainable = True
model.compile(
    optimizer=optimizers.Adam(learning_rate=1e-4),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

EPOCHS_STAGE2 = 4
model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS_STAGE2, verbose=2)




## === cell 4
def preprocess_test_image(path):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [IMG_SIZE, IMG_SIZE])
    image = image / 255.0
    return image


test_paths = (
    sample_sub["image_id"].apply(lambda x: os.path.join(TEST_IMG_DIR, x)).values
)
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(preprocess_test_image, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(32).prefetch(tf.data.AUTOTUNE)

pred_probs = model.predict(test_ds, verbose=0)
predictions = np.argmax(pred_probs, axis=1).astype(int).tolist()

submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": predictions})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
