## === cell 0
import os
import numpy as np
import pandas as pd
from pathlib import Path




## === cell 1
BASE_DIR = Path("/") / "kaggle" / "input"
DATA_DIRS = [
    BASE_DIR / "dog-breed-identification",
    BASE_DIR / "data",
    Path("."),
]


def find_file(filename: str) -> Path:
    for d in DATA_DIRS:
        candidate = d / filename
        if candidate.is_file():
            return candidate
    raise FileNotFoundError(f"{filename} not found in any of {DATA_DIRS}")


labels_csv_path = find_file("labels.csv")
sample_submission_csv_path = find_file("sample_submission.csv")




## === cell 2
sample_sub = pd.read_csv(sample_submission_csv_path)

breed_columns = sample_sub.columns.tolist()[1:]

num_breeds = len(breed_columns)

rng = np.random.default_rng(42)

alpha = 0.5
dirichlet_samples = rng.dirichlet([alpha] * num_breeds, size=sample_sub.shape[0])

submission = pd.DataFrame(dirichlet_samples, columns=breed_columns)
submission.insert(0, "id", sample_sub["id"])




## === cell 3
output_path = Path("output_submission.csv")
submission.to_csv(output_path, index=False)
print(f"Submission written to {output_path.resolve()}")
