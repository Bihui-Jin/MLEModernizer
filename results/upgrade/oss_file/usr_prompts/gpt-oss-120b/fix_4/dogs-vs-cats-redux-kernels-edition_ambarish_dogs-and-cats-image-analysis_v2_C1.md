## === cell 0
import os
import glob
import pandas as pd
import numpy as np

print("Input root contents:", os.listdir("../input"))




## === cell 1
test_image_paths = glob.glob("../input/**/test/**/*.jpg", recursive=True)
print(f"Found {len(test_image_paths)} test images")

test_ids = [os.path.splitext(os.path.basename(p))[0] for p in test_image_paths]




## === cell 2
train_image_paths = glob.glob("../input/**/train/**/*.jpg", recursive=True)
dog_count = sum("dog" in os.path.basename(p).lower() for p in train_image_paths)
cat_count = sum("cat" in os.path.basename(p).lower() for p in train_image_paths)
total = dog_count + cat_count

if total == 0:
    prob_dog = 0.5
else:
    prob_dog = dog_count / total

print(
    f"Training set: {dog_count} dogs, {cat_count} cats, probability of dog = {prob_dog:.6f}"
)

submission = pd.DataFrame({"id": test_ids, "label": prob_dog})
submission = submission.sort_values("id").reset_index(drop=True)

print("Submission preview (first 5 rows):")
print(submission.head())




## === cell 3
submission_path = "predictions.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")




## === cell 4
assert os.path.exists(submission_path), "CSV file was not created."
df_check = pd.read_csv(submission_path)
assert ("id" in df_check.columns) and (
    "label" in df_check.columns
), "Missing required columns."
print("CSV format verified successfully.")
