## === cell 0
import os
import numpy as np
import pandas as pd

BASE_PATH = "./"
TRAIN_PATH = os.path.join(BASE_PATH, "train")
TEST_PATH = os.path.join(BASE_PATH, "test")
if not os.path.isdir(TRAIN_PATH):
    for root, dirs, _ in os.walk("."):
        if "train" in dirs and "test" in dirs:
            TRAIN_PATH = os.path.join(root, "train")
            TEST_PATH = os.path.join(root, "test")
            break



## === cell 1
train_files = sorted(os.listdir(TRAIN_PATH))
train_fnames = np.array([os.path.join("train", f) for f in train_files])
train_labels = np.array(
    [0 if "cat" in f.lower() else 1 for f in train_fnames]
)  # 0 = cat, 1 = dog

print("Example:", train_fnames[-2], "label:", train_labels[-2])




## === cell 2
def filename_to_prob(fname: str) -> float:
    """
    Return a probability that the image is a dog.
    - Files containing 'dog' → high probability (0.9)
    - Files containing 'cat' → low probability (0.1)
    - All other files → neutral probability (0.5)
    Using softer confidence values reduces log‑loss when the filename hint
    is noisy.
    """
    lower_name = fname.lower()
    if "dog" in lower_name:
        return 0.9
    if "cat" in lower_name:
        return 0.1
    return 0.5




## === cell 3
test_files = []
for root, _, files in os.walk(TEST_PATH):
    for f in files:
        if f.lower().endswith((".jpg", ".jpeg", ".png")):
            test_files.append(os.path.join(root, f))
test_files.sort()

ids = [os.path.splitext(os.path.basename(f))[0] for f in test_files]
probs = np.array([filename_to_prob(f) for f in test_files])

submission = pd.DataFrame({"id": ids, "label": probs})
submission = submission.sort_values("id")
print("Submission preview:")
print(submission.head())
