submissions.head()

## === cell 3
label_split = train.labels.apply(lambda x: x.split())
mlb = MultiLabelBinarizer().fit(label_split)
labels_matrix = mlb.transform(label_split)
label_names = mlb.classes_

class_counts = labels_matrix.sum(axis=0)

top_k = 5
most_common_idxs = np.argsort(-class_counts)[:top_k]
most_common_labels = label_names[most_common_idxs]

num_tests = len(submissions)



## === cell 4
for i in range(num_tests):
    submissions.at[i, "labels"] = " ".join(most_common_labels)

submissions.to_csv("submission.csv", index=False)
