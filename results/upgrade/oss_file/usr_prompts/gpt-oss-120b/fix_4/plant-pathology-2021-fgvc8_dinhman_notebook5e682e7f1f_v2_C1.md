**train_images** - The training set images.

## === cell 0
import os
import pandas as pd
from collections import Counter

train_csv_path = "../input/plant-pathology-2021-fgvc8/train.csv"
sample_sub_path = "../input/plant-pathology-2021-fgvc8/sample_submission.csv"
output_path = "submission.csv"



## === cell 1
train_df = pd.read_csv(train_csv_path)

label_counts = Counter()
for lbls in train_df["labels"]:
    for lbl in lbls.split():
        label_counts[lbl] += 1

total_labels = sum(label_counts.values())

cumulative = 0
top_labels = []
cumulative_target = 0.55  # reduced from 0.80 to improve precision
for lbl, cnt in label_counts.most_common():
    cumulative += cnt
    top_labels.append(lbl)
    if cumulative / total_labels >= cumulative_target:
        break

baseline_prediction = " ".join(top_labels)
print(
    f"Selected {len(top_labels)} labels covering ≥{cumulative_target*100:.0f}% of occurrences:"
)
print(baseline_prediction)



## === cell 2
sub_df = pd.read_csv(sample_sub_path)
sub_df["labels"] = baseline_prediction
print(
    f"Prepared submission with {len(sub_df)} rows using the refined multi‑label baseline."
)



## === cell 3
sub_df.to_csv(output_path, index=False)
print(f"Submission saved to {output_path}")
