## Metric

labels should be a space-delimited list.

The file should contain a header and have the following format:

```

```

**sample_submission.csv** - A sample submission file in the correct format.

- `image`
- `labels`

**train_images** - The training set images.

**test_images** - The test set images. This competition has a hidden test set: only three images are provided here as samples while the remaining 5,000 images will be available to your notebook once it is submitted.

import os
import pandas as pd
import numpy as np
import tensorflow as tf

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

output_dir = "./"
test_dir = "../input/plant-pathology-2021-fgvc8/test_images/"
train_dir = "../input/plant-pathology-2021-fgvc8/train_images/"

train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
data_set = pd.read_csv(train_path)

df_labels = data_set["labels"]
one_hot = df_labels.str.get_dummies(sep=" ")
dataset_labels = one_hot.columns.tolist()  # ordered list of class names

class_counts = one_hot.sum()
total_images = len(one_hot)
class_freq = class_counts / total_images

freq_threshold = 0.10
default_labels = [cls for cls, freq in class_freq.items() if freq >= freq_threshold]
default_label_str = " ".join(default_labels).strip()
if not default_label_str:
    default_label_str = class_counts.idxmax()

IMG_SIZE = (224, 224)


def _load_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = tf.keras.applications.mobilenet_v2.preprocess_input(img)
    return img


encoder = tf.keras.applications.MobileNetV2(
    weights="imagenet", include_top=False, pooling="avg"
)

train_image_names = data_set["image"].tolist()
train_image_paths = [os.path.join(train_dir, name) for name in train_image_names]

train_ds = tf.data.Dataset.from_tensor_slices(train_image_paths)
train_ds = train_ds.map(_load_and_preprocess, num_parallel_calls=tf.data.AUTOTUNE)
train_ds = train_ds.batch(64).prefetch(tf.data.AUTOTUNE)

train_embeddings = encoder.predict(train_ds, verbose=0)  # shape: (N, 1280)

train_labels = data_set["labels"].apply(lambda s: s.split()[0]).tolist()
train_labels = np.array(train_labels)


def _nearest_label(test_emb_batch):
    dists = np.linalg.norm(
        train_embeddings[None, :, :] - test_emb_batch[:, None, :], axis=2
    )
    nearest_idx = np.argmin(dists, axis=1)
    return train_labels[nearest_idx]
