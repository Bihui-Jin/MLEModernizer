dataset_labels = one_hot.columns.to_list()
num_classes = len(dataset_labels)




## === cell 2
from tensorflow.keras.applications import EfficientNetB7
from tensorflow.keras import Model
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D, Input


class MultiLabel(Model):
    def __init__(self, num_classes, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.backbone = EfficientNetB7(
            include_top=False,
            weights="imagenet",
            input_shape=image_dims,
        )
        self.pool = GlobalAveragePooling2D()
        self.classifier = Dense(num_classes, activation="sigmoid")

    def call(self, inputs, training=False):
        x = self.backbone(inputs, training=training)
        x = self.pool(x)
        return self.classifier(x)




## === cell 3
if __name__ == "__main__":
    model = MultiLabel(num_classes=num_classes)
    model.build(input_shape=[None, *image_dims])

    image_filenames = sorted(os.listdir(test_dir))
    image_paths = [os.path.join(test_dir, fname) for fname in image_filenames]

    def _preprocess(path):
        raw = tf.io.read_file(path)
        img = tf.io.decode_image(contents=raw, channels=3, dtype=tf.dtypes.float32)
        img = tf.image.resize(img, [image_dims[0], image_dims[1]])
        img = img * 255.0  # preserve original scaling
        return img

    ds = tf.data.Dataset.from_tensor_slices(image_paths)
    ds = ds.map(lambda p: (p, _preprocess(p)), num_parallel_calls=tf.data.AUTOTUNE)
    batch_size = 32
    ds = ds.batch(batch_size)

    values = []
    for batch_paths, batch_imgs in ds:
        preds = model(batch_imgs)  # (B, num_classes)
        preds_np = tf.squeeze(preds).numpy()  # (B, num_classes) -> numpy
        if preds_np.ndim == 1:
            preds_np = preds_np[None, :]

        for path_tensor, pred_vec in zip(batch_paths, preds_np):
            idxs = [idx for idx, v in enumerate(pred_vec) if v > 0.7]
            classes_img = " ".join([dataset_labels[idx] for idx in idxs])
            filename = os.path.basename(path_tensor.numpy().decode())
            values.append([filename, classes_img.strip()])

    submission_df = pd.DataFrame(values, columns=["image", "labels"])
    submission_df.to_csv(os.path.join(output_dir, "submission.csv"), index=False)
