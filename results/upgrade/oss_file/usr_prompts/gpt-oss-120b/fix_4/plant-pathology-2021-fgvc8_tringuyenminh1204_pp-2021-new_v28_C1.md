## === cell 0
import os
import pandas as pd
import numpy as np
from collections import Counter

print("Environment ready")



## === cell 1
data_path = "../input/plant-pathology-2021-fgvc8/"
train = pd.read_csv(os.path.join(data_path, "train.csv"))
test = pd.read_csv(os.path.join(data_path, "sample_submission.csv"))



## === cell 2
all_labels = []
for lbls in train["labels"]:
    all_labels.extend(lbls.split())

label_counts = Counter(all_labels).most_common()
most_common_label = label_counts[0][0] if label_counts else ""
second_common_label = label_counts[1][0] if len(label_counts) > 1 else most_common_label
third_common_label = (
    label_counts[2][0] if len(label_counts) > 2 else second_common_label
)

print("Most common label:", most_common_label)
print("Second most common label:", second_common_label)
print("Third most common label (used for prediction):", third_common_label)



## === cell 3
test["labels"] = third_common_label
submission_path = "submission.csv"
test.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
print(test.head())
