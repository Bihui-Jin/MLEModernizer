def get_label(image_path, thresh=120):
    """
    Heuristic labeler using average intensity and channel information.
    - Bright images (mean > thresh) → healthy
    - Dark images (mean <= thresh) → complex
    - If the image is not extremely bright/dark, use dominant color:
        * Red‑dominant → rust
        * Blue‑dominant → powdery_mildew
        * Otherwise → complex
    This adds inexpensive cues for additional disease classes while keeping the original
    simple logic, aiming to improve the mean F1‑score toward the target.
    """
    try:
        img = Image.open(image_path).convert("RGB")
        arr = np.array(img).astype(np.float32)
        mean_intensity = arr.mean()
        if mean_intensity > thresh:
            return "healthy"
        r_mean = arr[:, :, 0].mean()
        g_mean = arr[:, :, 1].mean()
        b_mean = arr[:, :, 2].mean()
        if r_mean > g_mean and r_mean > b_mean and r_mean > 100:
            return "rust"
        if b_mean > r_mean and b_mean > g_mean and b_mean > 100:
            return "powdery_mildew"
        return "complex"
    except Exception:
        return "healthy"




## === cell 5
def predict(test_path, default_label="healthy"):
    """
    Predict labels for all images in the test directory using the
    get_label heuristic. Returns two lists: image filenames and their
    predicted labels.
    """
    images = []
    labels = []
    for image_name in sorted(os.listdir(test_path)):
        if image_name.startswith("."):
            continue
        full_path = os.path.join(test_path, image_name)
        images.append(image_name)
        label = get_label(full_path) if os.path.isfile(full_path) else default_label
        labels.append(label)
    return images, labels
