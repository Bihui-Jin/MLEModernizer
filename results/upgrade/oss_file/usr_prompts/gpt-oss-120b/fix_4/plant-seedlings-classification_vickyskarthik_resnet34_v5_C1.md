## === cell 0
import os
import numpy as np
import pandas as pd
from fastai.vision.all import *

for dirname, _, filenames in os.walk("/kaggle/input"):
    for filename in filenames:
        pass  # placeholder for potential debugging



## === cell 1
train_dir = "/kaggle/input/plant-seedlings-classification/train"
test_dir = "/kaggle/input/plant-seedlings-classification/test"



## === cell 2
dls = ImageDataLoaders.from_folder(
    path=train_dir,
    valid_pct=0.2,
    seed=42,
    item_tfms=Resize(336),
    batch_tfms=aug_transforms() + [Normalize.from_stats(*imagenet_stats)],
    bs=16,
    num_workers=2,  # reasonable for Kaggle environment
)

print("Classes:", dls.vocab)
dls.show_batch(max_n=4, figsize=(6, 6))



## === cell 3
learn = cnn_learner(dls, resnet34, metrics=accuracy, model_dir="/tmp/models")
learn.fine_tune(1)  # changed from 3 epochs to 1 epoch



## === cell 4
test_files = get_image_files(test_dir)
test_dl = learn.dls.test_dl(test_files)
preds, _ = learn.get_preds(dl=test_dl)

pred_classes = [learn.dls.vocab[i] for i in preds.argmax(dim=1)]

submission = pd.DataFrame(
    {"file": [f.name for f in test_files], "species": pred_classes}
)

submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
print(submission.head())
