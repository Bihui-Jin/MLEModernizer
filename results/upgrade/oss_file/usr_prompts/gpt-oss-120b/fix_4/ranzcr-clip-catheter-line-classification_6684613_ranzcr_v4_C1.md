## Dataset
`train.csv` contains image IDs, binary labels, and patient IDs.

TFRecords are available for both train and test.

We've also included `train_annotations.csv`. These are segmentation annotations for training samples that have them. They are included solely as additional information for competitors.

- train.csv - contains image IDs, binary labels, and patient IDs.
- sample_submission.csv - a sample submission file in the correct format
- test - test images
- train - training images

### Columns
- `StudyInstanceUID` - unique ID for each image
- `ETT - Abnormal` - endotracheal tube placement abnormal
- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

## === cell 0
import os

try:
    from google.protobuf.message_factory import MessageFactory

    if not hasattr(MessageFactory, "GetPrototype"):

        def _GetPrototype(self, descriptor):
            return self.GetMessageClass(descriptor)

        MessageFactory.GetPrototype = _GetPrototype
except Exception:
    pass

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models




## === cell 1
DATA_ROOT = "../input/ranzcr-clip-catheter-line-classification"
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
TEST_CSV = os.path.join(DATA_ROOT, "sample_submission.csv")
TRAIN_IMG_DIR = os.path.join(DATA_ROOT, "train")
TEST_IMG_DIR = os.path.join(DATA_ROOT, "test")

IMG_SIZE = 224
BATCH_SIZE = 32
EPOCHS_TOP = 5  # epochs with frozen backbone
EPOCHS_FINE = 5  # additional epochs after unfreezing
SAMPLE_TRAIN_SIZE = 27074

classes = [
    "ETT - Abnormal",
    "ETT - Borderline",
    "ETT - Normal",
    "NGT - Abnormal",
    "NGT - Borderline",
    "NGT - Incompletely Imaged",
    "NGT - Normal",
    "CVC - Abnormal",
    "CVC - Borderline",
    "CVC - Normal",
    "Swan Ganz Catheter Present",
]
num_classes = len(classes)




## === cell 2
train_df = pd.read_csv(TRAIN_CSV)
train_df = train_df.sample(
    n=min(SAMPLE_TRAIN_SIZE, len(train_df)), random_state=42
).reset_index(drop=True)

y_train = train_df[classes].values.astype(np.float32)

train_img_paths = (
    train_df["StudyInstanceUID"]
    .apply(lambda uid: os.path.join(TRAIN_IMG_DIR, f"{uid}.jpg"))
    .values
)




## === cell 3
def preprocess_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = img / 255.0
    return img


train_ds = (
    tf.data.Dataset.from_tensor_slices((train_img_paths, y_train))
    .map(lambda p, l: (preprocess_image(p), l), num_parallel_calls=tf.data.AUTOTUNE)
    .shuffle(1024, seed=42)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)




## === cell 4
augmentation = tf.keras.Sequential(
    [
        layers.RandomFlip("horizontal"),
        layers.RandomRotation(0.1),
    ]
)

base_model = tf.keras.applications.EfficientNetB0(
    include_top=False,
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
    weights="imagenet",
    pooling="avg",
)
base_model.trainable = False

inputs = layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = augmentation(inputs)  # <-- added augmentation
x = base_model(x, training=False)
outputs = layers.Dense(num_classes, activation="sigmoid")(x)
model = models.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
    loss="binary_crossentropy",
    metrics=[tf.keras.metrics.AUC(name="auc")],
)

model.fit(train_ds, epochs=EPOCHS_TOP, verbose=1)




## === cell 5
base_model.trainable = True
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
    loss="binary_crossentropy",
    metrics=[tf.keras.metrics.AUC(name="auc")],
)
model.fit(train_ds, epochs=EPOCHS_FINE, verbose=1)




## === cell 6
test_df = pd.read_csv(TEST_CSV)

test_img_paths = (
    test_df["StudyInstanceUID"]
    .apply(lambda uid: os.path.join(TEST_IMG_DIR, f"{uid}.jpg"))
    .values
)

test_ds = (
    tf.data.Dataset.from_tensor_slices(test_img_paths)
    .map(preprocess_image, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

predictions = model.predict(test_ds, verbose=1)
assert predictions.shape[0] == len(test_df)




## === cell 7
submission = pd.DataFrame(predictions, columns=classes)
submission.insert(0, "StudyInstanceUID", test_df["StudyInstanceUID"])
submission.to_csv("submission.csv", index=False)
