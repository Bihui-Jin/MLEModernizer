BASE_INPUT = os.getenv("KAGGLE_INPUT_PATH", "/kaggle/input")
DATA_ROOT = os.path.join(BASE_INPUT, "ranzcr-clip-catheter-line-classification")
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
TEST_DIR = os.path.join(DATA_ROOT, "test")
SAMPLE_SUBMIT = os.path.join(DATA_ROOT, "sample_submission.csv")

train_df = pd.read_csv(TRAIN_CSV)
sample_sub = pd.read_csv(SAMPLE_SUBMIT)

label_cols = [c for c in train_df.columns if c not in ["StudyInstanceUID", "PatientID"]]
submit_label_cols = [c for c in sample_sub.columns if c != "StudyInstanceUID"]
label_cols = [c for c in label_cols if c in submit_label_cols]

IMG_SIZE = 260
BATCH_SIZE = 16


def load_image(path):
    """Read an image, resize to IMG_SIZE and normalize to [0,1]."""
    img = cv2.imread(path)
    if img is None:
        img = np.zeros((IMG_SIZE, IMG_SIZE, 3), dtype=np.uint8)
    else:
        img = cv2.resize(img, (IMG_SIZE, IMG_SIZE))
    img = img.astype("float32") / 255.0
    return img


def df_to_dataset(df, shuffle=True):
    """Convert a dataframe with UIDs to a tf.data.Dataset of (image, label).

    Added .cache() to avoid re‑reading images each epoch while keeping the
    same preprocessing and deterministic shuffling.
    """
    image_paths = (
        df["StudyInstanceUID"]
        .apply(lambda uid: os.path.join(TEST_DIR, "..", "train", uid + ".jpg"))
        .values
    )
    labels = df[label_cols].values.astype("float32")
    ds = tf.data.Dataset.from_tensor_slices((image_paths, labels))

    def _process(path, label):
        img = tf.numpy_function(load_image, [path], tf.float32)
        img.set_shape((IMG_SIZE, IMG_SIZE, 3))
        return img, label

    ds = ds.map(_process, num_parallel_calls=tf.data.AUTOTUNE)
    ds = ds.cache()  # <-- cache processed images
    if shuffle:
        ds = ds.shuffle(buffer_size=len(df), seed=seed)
    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
    return ds


from sklearn.model_selection import train_test_split

train_split_df, val_split_df = train_test_split(
    train_df, test_size=0.10, random_state=seed
)

train_ds = df_to_dataset(train_split_df, shuffle=True)
val_ds = df_to_dataset(val_split_df, shuffle=False)




## === cell 2
def build_model(input_shape=(IMG_SIZE, IMG_SIZE, 3), num_labels=len(label_cols)):
    inputs = layers.Input(shape=input_shape)
    x = layers.Conv2D(32, (3, 3), activation="relu")(inputs)
    x = layers.MaxPooling2D()(x)
    x = layers.Conv2D(64, (3, 3), activation="relu")(x)
    x = layers.MaxPooling2D()(x)
    x = layers.Conv2D(128, (3, 3), activation="relu")(x)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.Dense(256, activation="relu")(x)
    outputs = layers.Dense(num_labels, activation="sigmoid")(x)
    model = models.Model(inputs, outputs)
    return model


model = build_model()
model.compile(
    optimizer=optimizers.Adam(learning_rate=1e-3),
    loss="binary_crossentropy",
    metrics=[tf.keras.metrics.AUC(name="auc")],
)




## === cell 3
es = callbacks.EarlyStopping(
    monitor="val_auc", patience=3, mode="max", restore_best_weights=True
)

model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=10,
    callbacks=[es],
    verbose=2,
)




## === cell 4
test_ids = sample_sub["StudyInstanceUID"].values
test_paths = [os.path.join(TEST_DIR, uid + ".jpg") for uid in test_ids]


def test_generator(paths):
    for p in paths:
        img = load_image(p)
        yield img


test_ds = tf.data.Dataset.from_generator(
    lambda: test_generator(test_paths),
    output_signature=tf.TensorSpec(shape=(IMG_SIZE, IMG_SIZE, 3), dtype=tf.float32),
).batch(BATCH_SIZE)

preds = model.predict(test_ds, verbose=0)  # shape (num_test, num_labels)

pred_df = pd.DataFrame(preds, columns=label_cols)
pred_df.insert(0, "StudyInstanceUID", test_ids)
pred_df = pred_df[sample_sub.columns]  # reorder to match the sample submission
pred_df.to_csv("submission.csv", index=False)
