## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image

possible_paths = [
    os.path.join(os.getcwd(), "data", "ranzcr-clip-catheter-line-classification"),
    "/kaggle/input/ranzcr-clip-catheter-line-classification",
    "/kaggle/input",
]
WORK_DIR = None
for p in possible_paths:
    if os.path.isdir(p) and os.path.isfile(os.path.join(p, "train.csv")):
        WORK_DIR = p
        break
if WORK_DIR is None:
    raise FileNotFoundError(
        "Could not locate the dataset directory containing train.csv"
    )

train_path = os.path.join(WORK_DIR, "train.csv")
sample_path = os.path.join(WORK_DIR, "sample_submission.csv")
train_df = pd.read_csv(train_path)
sample_sub = pd.read_csv(sample_path)

label_cols = sample_sub.columns[1:]
print(f"Label columns ({len(label_cols)}): {list(label_cols)}")



## === cell 1
train_img_dir = os.path.join(WORK_DIR, "train")
test_img_dir = os.path.join(WORK_DIR, "test")


def image_mean_intensity(img_path):
    """Return the mean grayscale intensity of an image; fallback to np.nan on error."""
    try:
        with Image.open(img_path) as img:
            img_gray = img.convert("L")  # convert to 8‑bit grayscale
            return np.array(img_gray).mean()
    except Exception:
        return np.nan


def compute_intensity_series(df, img_dir, id_col="StudyInstanceUID"):
    """Map each StudyInstanceUID to its mean intensity."""
    intensities = {}
    for uid in df[id_col].values:
        img_path = os.path.join(img_dir, f"{uid}.jpg")
        intensities[uid] = image_mean_intensity(img_path)
    return pd.Series(intensities, name="intensity")


train_intensity = compute_intensity_series(train_df, train_img_dir)
test_intensity = compute_intensity_series(sample_sub, test_img_dir)

overall_mean_intensity = train_intensity.mean()
train_intensity.fillna(overall_mean_intensity, inplace=True)
test_intensity.fillna(overall_mean_intensity, inplace=True)

submission = sample_sub.copy()
for col in label_cols:
    pos_mean = train_intensity[train_df[col] == 1].mean()
    neg_mean = train_intensity[train_df[col] == 0].mean()
    if np.isnan(pos_mean) or np.isnan(neg_mean) or pos_mean == neg_mean:
        fallback_prob = train_df[col].mean()
        submission[col] = fallback_prob
        continue
    if pos_mean > neg_mean:
        prob = (test_intensity - neg_mean) / (pos_mean - neg_mean)
    else:
        prob = (neg_mean - test_intensity) / (neg_mean - pos_mean)
    prob = prob.clip(0.0, 1.0)  # ensure valid probability range
    submission[col] = prob.values



## === cell 2
output_path = "submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission file written to {output_path}")
print("First few rows of the submission:")
print(submission.head())
