import os
import numpy as np
import pandas as pd
import tensorflow as tf
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score

DATA_ROOT = "../input/ranzcr-clip-catheter-line-classification"
TRAIN_CSV = os.path.join(DATA_ROOT, "train.csv")
SAMPLE_SUBMISSION = os.path.join(DATA_ROOT, "sample_submission.csv")
TRAIN_IMG_DIR = os.path.join(DATA_ROOT, "train")
TEST_IMG_DIR = os.path.join(DATA_ROOT, "test")
OUTPUT_SUBMISSION = "submission.csv"

sample_sub_df = pd.read_csv(SAMPLE_SUBMISSION)
TARGET_COLS = [c for c in sample_sub_df.columns if c != "StudyInstanceUID"]

train_df = pd.read_csv(TRAIN_CSV)
assert set(TARGET_COLS).issubset(
    set(train_df.columns)
), "Some target columns missing in train.csv"

labels = train_df[TARGET_COLS].values.astype(np.float32)

## === cell 1
IMG_SIZE = (224, 224)


def preprocess_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = tf.keras.applications.mobilenet_v2.preprocess_input(img)
    return img


def make_dataset(img_dir, ids):
    paths = [os.path.join(img_dir, f"{uid}.jpg") for uid in ids]
    ds = tf.data.Dataset.from_tensor_slices(paths)
    ds = ds.map(lambda p: preprocess_image(p), num_parallel_calls=tf.data.AUTOTUNE)
    ds = ds.batch(64).prefetch(tf.data.AUTOTUNE)
    return ds


train_ids = train_df["StudyInstanceUID"].values
train_ds = make_dataset(TRAIN_IMG_DIR, train_ids)



## === cell 2
base_model = tf.keras.applications.MobileNetV2(
    input_shape=IMG_SIZE + (3,), include_top=False, weights="imagenet", pooling="avg"
)
base_model.trainable = False

train_features = base_model.predict(train_ds, verbose=0)
assert train_features.shape[0] == len(train_ids), "Feature count mismatch"



## === cell 3
X_tr, X_val, y_tr, y_val = train_test_split(
    train_features,
    labels,
    test_size=0.2,
    random_state=42,
    stratify=labels.argmax(axis=1),
)

input_dim = X_tr.shape[1]
num_labels = len(TARGET_COLS)

model = tf.keras.Sequential(
    [
        tf.keras.layers.Input(shape=(input_dim,)),
        tf.keras.layers.Dense(num_labels, activation="sigmoid"),
    ]
)
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3), loss="binary_crossentropy"
)

model.fit(
    X_tr, y_tr, epochs=5, batch_size=128, validation_data=(X_val, y_val), verbose=0
)

val_preds = model.predict(X_val, verbose=0)
auc_per_label = [roc_auc_score(y_val[:, i], val_preds[:, i]) for i in range(num_labels)]
mean_auc = np.mean(auc_per_label)
print(f"Validation mean AUC: {mean_auc:.5f}")



## === cell 4
test_ids = pd.read_csv(SAMPLE_SUBMISSION)["StudyInstanceUID"].values
test_ds = make_dataset(TEST_IMG_DIR, test_ids)
test_features = base_model.predict(test_ds, verbose=0)

test_preds = model.predict(test_features, verbose=0)



## === cell 5
submission_df = pd.DataFrame(test_preds, columns=TARGET_COLS)
submission_df.insert(0, "StudyInstanceUID", test_ids)

expected_cols = ["StudyInstanceUID"] + TARGET_COLS
assert list(submission_df.columns) == expected_cols, "Column order mismatch"

submission_df.to_csv(OUTPUT_SUBMISSION, index=False)
print(
    f"Submission file written to '{OUTPUT_SUBMISSION}' with shape {submission_df.shape}"
)
