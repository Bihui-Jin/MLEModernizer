def extract_features(img_path):
    img = cv2.imread(img_path)
    if img is None:
        return np.zeros(3 * 2 + 3 * 32, dtype=np.float32)
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img_resized = cv2.resize(img, (224, 224))
    mean = img_resized.mean(axis=(0, 1))
    std = img_resized.std(axis=(0, 1))
    hist_features = []
    for ch in range(3):
        hist = np.histogram(img_resized[:, :, ch], bins=32, range=(0, 255))[0]
        hist = hist.astype(np.float32) / (224 * 224)  # normalize
        hist_features.append(hist)
    hist_features = np.concatenate(hist_features)
    return np.concatenate([mean, std, hist_features])




## === cell 2
train_features = []
valid_ids = []  # keep track of IDs for which we could load an image
for uid in train_df["StudyInstanceUID"]:

    n_jobs=-1,
)
