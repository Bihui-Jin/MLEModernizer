- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import tensorflow as tf
import pandas as pd
import numpy as np

W = H = 338
autotune = tf.data.experimental.AUTOTUNE

base_path = "../input/ranzcr-clip-catheter-line-classification"
train_dir = os.path.join(base_path, "train")
test_dir = os.path.join(base_path, "test")
train_csv_path = os.path.join(base_path, "train.csv")
sample_submission_path = os.path.join(base_path, "sample_submission.csv")
submission_path = "submission.csv"




## === cell 1
def triple_image(image):
    """Convert single‑channel image to 3‑channel by concatenation."""
    return tf.concat([image] * 3, axis=-1)


def decode_and_resize(path):
    """Read a JPEG file, decode, resize and convert to 3‑channel."""
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=1)
    img = tf.image.resize(img, (H, W))
    img = triple_image(img)
    return img


def preprocess(images, labels=None):
    """Standardise images for EfficientNet."""
    images = tf.cast(images, tf.float32) / 255.0
    mean = tf.constant([0.485, 0.456, 0.406], dtype=tf.float32)
    std = tf.constant([0.229, 0.224, 0.225], dtype=tf.float32)
    images = (images - mean) / std
    if labels is None:
        return images
    else:
        return images, labels


def get_model(
    num_classes,
    baseline_weight="imagenet",
    init_weight=None,
    lr=1e-4,
    optimizer=tf.optimizers.Adam,
):
    """EfficientNetB3 backbone with a sigmoid output layer."""
    base_model = tf.keras.applications.EfficientNetB3(
        include_top=False,
        input_shape=(W, H, 3),
        pooling="avg",
        weights=baseline_weight,
    )
    x = tf.keras.layers.Dropout(0.3)(base_model.output)
    outputs = tf.keras.layers.Dense(num_classes, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base_model.input, outputs=outputs)

    model.compile(
        optimizer=optimizer(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )

    if init_weight and os.path.exists(init_weight):
        try:
            model.load_weights(init_weight)
            print(f"Weight loaded from {init_weight}")
        except Exception as e:
            print(f"Failed to load weights from {init_weight}: {e}")
    return model




## === cell 2
train_df = pd.read_csv(train_csv_path)
target_cols = [
    c for c in train_df.columns if c not in ("StudyInstanceUID", "PatientID")
]
num_classes = len(target_cols)

train_df["filepath"] = train_df["StudyInstanceUID"].apply(
    lambda uid: os.path.join(train_dir, f"{uid}.jpg")
)

train_df = train_df[train_df["filepath"].apply(os.path.exists)].reset_index(drop=True)

train_images = train_df["filepath"].values
train_labels = train_df[target_cols].values.astype(np.float32)

train_ds = tf.data.Dataset.from_tensor_slices((train_images, train_labels))
train_ds = train_ds.map(
    lambda p, l: (decode_and_resize(p), l), num_parallel_calls=autotune
)
train_ds = train_ds.map(preprocess, num_parallel_calls=autotune)
train_ds = train_ds.shuffle(1024).batch(32).prefetch(autotune)

model = get_model(num_classes=num_classes, baseline_weight="imagenet")
model.fit(train_ds, epochs=5, verbose=1)



## === cell 3
test_files = sorted([f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")])
test_paths = [os.path.join(test_dir, f) for f in test_files]
test_ids = [os.path.splitext(f)[0] for f in test_files]

test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(decode_and_resize, num_parallel_calls=autotune)
test_ds = test_ds.map(preprocess, num_parallel_calls=autotune)
test_ds = test_ds.batch(16).prefetch(autotune)

preds = model.predict(test_ds, verbose=1)
if preds.shape[1] != num_classes:
    if preds.shape[1] > num_classes:
        preds = preds[:, :num_classes]
    else:
        pad = np.zeros((preds.shape[0], num_classes - preds.shape[1]))
        preds = np.hstack([preds, pad])



## === cell 4
submission_df = pd.read_csv(sample_submission_path).iloc[0:0]  # get header only
submission_df = pd.DataFrame(preds, columns=target_cols)
submission_df["StudyInstanceUID"] = test_ids
submission_df = submission_df[["StudyInstanceUID"] + target_cols]

submission_df.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
