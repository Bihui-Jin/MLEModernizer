## === cell 0
import os, glob
import numpy as np
import pandas as pd

database_base_path = "/kaggle/input/ranzcr-clip-catheter-line-classification/"
train_csv_path = os.path.join(database_base_path, "train.csv")
sample_submission_path = os.path.join(database_base_path, "sample_submission.csv")
test_images_dir = os.path.join(database_base_path, "test")
label_cols = [
    "ETT - Abnormal",
    "ETT - Borderline",
    "ETT - Normal",
    "NGT - Abnormal",
    "NGT - Borderline",
    "NGT - Incompletely Imaged",
    "NGT - Normal",
    "CVC - Abnormal",
    "CVC - Borderline",
    "CVC - Normal",
    "Swan Ganz Catheter Present",
]



## === cell 1
train_df = pd.read_csv(train_csv_path)
baseline_means = train_df[label_cols].mean().values.astype(np.float32)  # shape (11,)



## === cell 2
test_file_paths = glob.glob(os.path.join(test_images_dir, "*.jpg"))
test_ids = [os.path.splitext(os.path.basename(p))[0] for p in sorted(test_file_paths)]
NUM_TEST_IMAGES = len(test_ids)
print(f"Number of test images found: {NUM_TEST_IMAGES}")



## === cell 3
probabilities = np.tile(baseline_means, (NUM_TEST_IMAGES, 1))  # shape (num_images, 11)
print(f"Probabilities shape: {probabilities.shape}")



## === cell 4
submission_df = pd.DataFrame(probabilities, columns=label_cols)
submission_df.insert(0, "StudyInstanceUID", test_ids)
output_path = "submission.csv"
submission_df.to_csv(output_path, index=False)
print(f"Submission file written to {output_path}")
