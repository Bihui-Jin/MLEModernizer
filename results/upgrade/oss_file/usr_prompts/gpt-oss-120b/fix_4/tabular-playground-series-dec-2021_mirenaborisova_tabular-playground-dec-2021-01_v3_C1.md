import os
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score


def resolve_path(filename: str) -> str:
    """Return the first existing path for `filename` among common Kaggle locations."""
    candidates = [
        os.path.join("data", filename),
        os.path.join("/kaggle/input/tabular-playground-series-dec-2021", filename),
        os.path.join("/kaggle/input", filename),
    ]
    for p in candidates:
        if os.path.exists(p):
            return p
    raise FileNotFoundError(f"Unable to locate {filename}")


train_path = resolve_path("train.csv")
test_path = resolve_path("test.csv")

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

X = train_df.drop(columns=["Cover_Type"])
y = train_df["Cover_Type"]

X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, random_state=42, stratify=y
)

subset_size = 400_000
if len(X_train) > subset_size:
    X_train_sub = X_train.sample(n=subset_size, random_state=42)
    y_train_sub = y_train.loc[X_train_sub.index]
else:
    X_train_sub = X_train
    y_train_sub = y_train

rf = RandomForestClassifier(
    n_estimators=600,  # more trees for better performance
    max_depth=None,
    n_jobs=-1,
    random_state=42,
    min_samples_split=2,
    min_samples_leaf=1,
    class_weight="balanced",  # handle possible class imbalance
)

rf.fit(X_train_sub, y_train_sub)

val_pred = rf.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy (approx.): {val_acc:.5f}")
