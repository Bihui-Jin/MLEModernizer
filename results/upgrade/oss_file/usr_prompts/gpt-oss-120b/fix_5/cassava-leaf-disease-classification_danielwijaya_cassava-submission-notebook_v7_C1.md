import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION_VERSION"] = "2"

import re
import numpy as np
import pandas as pd
import tensorflow as tf
from functools import partial

for dirname, _, filenames in os.walk("/kaggle/input"):
    for filename in filenames:
        print(os.path.join(dirname, filename))




## === cell 1
GCS_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_TFREC_PATTERN = GCS_PATH + "/train_tfrecords/ld_train*.tfrec"
TEST_TFREC_PATTERN = GCS_PATH + "/test_tfrecords/ld_test*.tfrec"

BATCH_SIZE = 64
AUTOTUNE = tf.data.experimental.AUTOTUNE
IMAGE_SIZE = (224, 224)
NUM_CLASSES = 5




## === cell 2
def dataset_sizes(filenames):
    """Utility to compute total number of images from TFRecord filenames."""
    nums = [int(re.search(r"-([0-9]*)\.", fn).group(1)) for fn in filenames]
    return int(np.sum(nums))


def decode_img(img_bytes):
    img = tf.io.decode_jpeg(img_bytes, channels=3)
    img = tf.image.resize(img, IMAGE_SIZE)
    img = tf.cast(img, tf.float32) / 255.0
    return img


def read_tfrecord(example, labeled=True):
    if labeled:
        fmt = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "target": tf.io.FixedLenFeature([], tf.int64),
        }
    else:
        fmt = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
        }
    parsed = tf.io.parse_single_example(example, fmt)
    img = decode_img(parsed["image"])
    if labeled:
        label = tf.cast(parsed["target"], tf.int32)
        return img, label
    else:
        return img, parsed["image_name"]


def load_dataset(filenames, labeled=True, ordered=False):
    options = tf.data.Options()
    if not ordered:
        options.experimental_deterministic = False
    ds = tf.data.TFRecordDataset(filenames, num_parallel_reads=AUTOTUNE)
    ds = ds.with_options(options)
    ds = ds.map(partial(read_tfrecord, labeled=labeled), num_parallel_calls=AUTOTUNE)
    return ds




## === cell 3
train_fnames = tf.io.gfile.glob(TRAIN_TFREC_PATTERN)
split_idx = int(0.9 * len(train_fnames))
train_files, val_files = train_fnames[:split_idx], train_fnames[split_idx:]

train_ds = (
    load_dataset(train_files, labeled=True)
    .shuffle(2048)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)
val_ds = load_dataset(val_files, labeled=True).batch(BATCH_SIZE).prefetch(AUTOTUNE)

base_model = tf.keras.applications.EfficientNetB0(
    include_top=False,
    weights="imagenet",
    input_shape=(*IMAGE_SIZE, 3),
    pooling="avg",
)
base_model.trainable = True  # enable training of all backbone layers

inputs = tf.keras.Input(shape=(*IMAGE_SIZE, 3))
x = base_model(inputs, training=True)
outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)

model.fit(train_ds, validation_data=val_ds, epochs=30, verbose=2)




## === cell 4
test_fnames = tf.io.gfile.glob(TEST_TFREC_PATTERN)
NUM_TEST_IMAGES = dataset_sizes(test_fnames)

test_ds = load_dataset(test_fnames, labeled=False, ordered=True)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)

test_images = test_ds.map(lambda img, _: img)
test_ids_ds = test_ds.map(lambda _, idnum: idnum).unbatch()

test_ids = [id_bytes.decode("utf-8") for id_bytes in test_ids_ds.as_numpy_iterator()]

probabilities = model.predict(test_images)
predictions = np.argmax(probabilities, axis=-1)




## === cell 5
submission = pd.DataFrame({"image_id": test_ids, "label": predictions})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)

print(f"Submission written to {submission_path}")
print(submission.head())
