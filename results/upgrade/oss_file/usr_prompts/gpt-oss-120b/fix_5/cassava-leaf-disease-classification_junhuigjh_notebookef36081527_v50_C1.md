# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from pathlib import Path
from PIL import Image
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score




## === cell 1
def extract_features(pil_img):
    """
    Compute simple color statistics:
    - mean of each RGB channel
    - standard deviation of each RGB channel
    Returns a 6‑element numpy array.
    """
    img_arr = np.asarray(pil_img).astype(np.float32) / 255.0  # (H, W, 3)
    flat = img_arr.reshape(-1, 3)
    means = flat.mean(axis=0)  # (3,)
    stds = flat.std(axis=0)  # (3,)
    return np.concatenate([means, stds])  # (6,)




## === cell 2
base_path = "/kaggle/input/cassava-leaf-disease-classification"
train_csv_path = os.path.join(base_path, "train.csv")
train_img_dir = os.path.join(base_path, "train_images")

train_df = pd.read_csv(train_csv_path)
train_image_ids = train_df["image_id"].tolist()
train_labels = train_df["label"].values

train_features = []
for img_id in train_image_ids:
    img_path = os.path.join(train_img_dir, img_id)
    pil_img = Image.open(img_path).convert("RGB")
    feats = extract_features(pil_img)
    train_features.append(feats)
train_features = np.stack(train_features, axis=0)  # (N_train, 6)

X_tr, X_val, y_tr, y_val = train_test_split(
    train_features, train_labels, test_size=0.2, random_state=42, stratify=train_labels
)

best_depth = None
best_acc = 0.0
for depth in [8, 12, 16, None]:
    clf = DecisionTreeClassifier(
        criterion="gini",
        max_depth=depth,
        min_samples_split=2,
        random_state=42,
    )
    clf.fit(X_tr, y_tr)
    preds = clf.predict(X_val)
    acc = accuracy_score(y_val, preds)
    if acc > best_acc:
        best_acc = acc
        best_depth = depth

decision_tree = DecisionTreeClassifier(
    criterion="gini",
    max_depth=best_depth,
    min_samples_split=2,
    random_state=42,
)
decision_tree.fit(train_features, train_labels)



## === cell 3
test_root = Path(base_path) / "test_images"
if not test_root.is_dir():
    candidates = list(Path(base_path).rglob("test_images"))
    test_root = next((p for p in candidates if p.is_dir()), None)
    if test_root is None:
        raise FileNotFoundError("Test image directory not found.")

valid_ext = {".jpg", ".jpeg", ".png"}
test_image_paths = sorted(
    [p for p in test_root.rglob("*") if p.suffix.lower() in valid_ext]
)

test_image_ids = [p.name for p in test_image_paths]

test_features = []
for p in test_image_paths:
    pil_img = Image.open(p).convert("RGB")
    feats = extract_features(pil_img)
    test_features.append(feats)
test_features = np.stack(test_features, axis=0)  # (N_test, 6)

test_predictions = decision_tree.predict(test_features)



## === cell 4
submission = pd.DataFrame(
    {"image_id": test_image_ids, "label": test_predictions.astype(int)}
)
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
