train_df = pd.read_csv(TRAIN_CSV)

train_tfrecord_files = tf.io.gfile.glob(os.path.join(TRAIN_TFREC_DIR, "*.tfrec"))


def _parse_tfrecord(example_proto):
    feature_description = {
        "image": tf.io.FixedLenFeature([], tf.string),
        "label": tf.io.FixedLenFeature([], tf.int64),
    }
    parsed = tf.io.parse_single_example(example_proto, feature_description)
    img = tf.image.decode_jpeg(parsed["image"], channels=3)
    img = tf.image.resize(img, [128, 128])
    img = img / 255.0
    label = tf.cast(parsed["label"], tf.int32)
    return img, label


train_ds = tf.data.TFRecordDataset(
    train_tfrecord_files, num_parallel_reads=tf.data.AUTOTUNE
)
train_ds = train_ds.map(_parse_tfrecord, num_parallel_calls=tf.data.AUTOTUNE)
train_ds = train_ds.cache()  # cache decoded images in memory for faster epochs
train_ds = train_ds.shuffle(buffer_size=10000, reshuffle_each_iteration=True)
train_ds = train_ds.batch(32).prefetch(tf.data.AUTOTUNE)



## === cell 2
model = tf.keras.Sequential(
    [
        tf.keras.layers.Conv2D(
            32, (3, 3), activation="relu", input_shape=(128, 128, 3)
        ),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, (3, 3), activation="relu"),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(128, activation="relu"),
        tf.keras.layers.Dropout(0.2),
        tf.keras.layers.Dense(5, activation="softmax"),  # 5 disease classes
    ]
)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(train_ds, epochs=5, verbose=1)
