# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier




## === cell 1
ROOT_DIR = "../input/cassava-leaf-disease-classification/"
TRAIN_CSV = os.path.join(ROOT_DIR, "train.csv")
TEST_DIR = os.path.join(ROOT_DIR, "test_images/")

train_df = pd.read_csv(TRAIN_CSV)




## === cell 2
IMG_SIZE = 64  # small size for quick training


def load_and_preprocess(image_path):
    """Load an image, resize, normalize, and return as float array."""
    img = Image.open(image_path).convert("RGB")
    img = img.resize((IMG_SIZE, IMG_SIZE))
    return np.array(img, dtype=np.float32) / 255.0  # [0,1] range


train_images = []
train_labels = []

for _, row in train_df.iterrows():
    img_path = os.path.join(ROOT_DIR, "train_images", row["image_id"])
    if os.path.exists(img_path):
        train_images.append(load_and_preprocess(img_path))
        train_labels.append(row["label"])

X = np.stack(train_images, axis=0)  # (N, H, W, 3)
y = np.array(train_labels, dtype=np.int64)

mean_rgb = X.mean(axis=(1, 2))  # (N, 3)
std_rgb = X.std(axis=(1, 2))  # (N, 3)
color_stats = np.concatenate([mean_rgb, std_rgb], axis=1)  # (N, 6)

X_flat = X.reshape(len(X), -1)  # raw pixel features
X_feat = np.concatenate([X_flat, color_stats], axis=1)  # combine

X_train_feat, X_val_feat, y_train, y_val = train_test_split(
    X_feat, y, test_size=0.1, random_state=42, stratify=y
)




## === cell 3
rf = RandomForestClassifier(
    n_estimators=500,
    max_depth=30,
    class_weight="balanced",
    n_jobs=-1,
    random_state=42,
    verbose=0,
)
rf.fit(X_train_feat, y_train)

val_acc = rf.score(X_val_feat, y_val)
print(f"Validation accuracy (scikit‑learn RF with extra stats): {val_acc:.4f}")




## === cell 4
test_images = [
    f
    for f in os.listdir(TEST_DIR)
    if f.lower().endswith(".jpg") and os.path.isfile(os.path.join(TEST_DIR, f))
]
test_images.sort()  # deterministic order

test_preds = []

for img_name in test_images:
    img_path = os.path.join(TEST_DIR, img_name)
    img_arr = load_and_preprocess(img_path)  # (64, 64, 3)

    img_flat = img_arr.reshape(1, -1)  # (1, 12288)

    mean_rgb = img_arr.mean(axis=(0, 1))  # (3,)
    std_rgb = img_arr.std(axis=(0, 1))  # (3,)

    color_stats = np.hstack([mean_rgb, std_rgb])[np.newaxis, :]  # (1, 6)

    img_feat = np.concatenate([img_flat, color_stats], axis=1)  # (1, 12294)

    pred_label = int(rf.predict(img_feat)[0])
    test_preds.append(pred_label)




## === cell 5
assert len(test_images) == len(
    test_preds
), "Length mismatch between images and predictions"

submission = pd.DataFrame({"image_id": test_images, "label": test_preds})
display(submission.head())

submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
