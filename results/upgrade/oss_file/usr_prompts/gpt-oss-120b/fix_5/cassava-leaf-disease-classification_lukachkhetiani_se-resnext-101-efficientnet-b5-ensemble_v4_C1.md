## === cell 0
import os
import glob
import torch
import torchvision.models as models
import cv2
import numpy as np
import pandas as pd
from tqdm import tqdm
from PIL import Image

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")



## === cell 1
midas = torch.hub.load("intel-isl/MiDaS", "MiDaS", pretrained=True)
midas.to(device)
midas.eval()
midas_transforms = torch.hub.load("intel-isl/MiDaS", "transforms")
transform = midas_transforms.default_transform

model_path = "model.pth"
if os.path.isfile(model_path):
    model = models.efficientnet_b1(pretrained=False)
    in_features = model.classifier[1].in_features
    model.classifier[1] = torch.nn.Linear(in_features, 5)
    model.load_state_dict(torch.load(model_path, map_location=device))
else:
    print("Checkpoint not found, initializing EfficientNet-B1 with ImageNet weights.")
    model = models.efficientnet_b1(pretrained=True)
    in_features = model.classifier[1].in_features
    model.classifier[1] = torch.nn.Linear(in_features, 5)

model.to(device)
model.eval()




## === cell 2
def prepare_image(image: np.ndarray) -> torch.Tensor:
    """
    Resize to 299x299 (EfficientNet‑B1 input size), convert to torch tensor,
    and move to the selected device.
    """
    img_resized = cv2.resize(image, (299, 299))
    tensor = torch.tensor(img_resized.transpose(2, 0, 1), dtype=torch.float32)
    tensor = tensor.unsqueeze(0).to(device)
    return tensor


base_input = "/kaggle/input/cassava-leaf-disease-classification"
test_images_dir = os.path.join(base_input, "test_images")
sample_submission_path = os.path.join(base_input, "sample_submission.csv")

sample_df = pd.read_csv(sample_submission_path)
image_ids = sample_df["image_id"].tolist()

names, preds = [], []

for img_name in tqdm(image_ids, desc="Processing test images"):
    img_path = os.path.join(test_images_dir, img_name)
    img = cv2.imread(img_path)
    if img is None:
        names.append(img_name)
        preds.append(0)
        continue

    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

    transformed = transform(img_rgb)
    if isinstance(transformed, dict):
        input_batch = transformed["image"].unsqueeze(0).to(device)
    else:
        input_batch = transformed.unsqueeze(0).to(device)

    with torch.no_grad():
        depth = midas(input_batch)

    depth_upsampled = torch.nn.functional.interpolate(
        depth.unsqueeze(1),
        size=img_rgb.shape[:2],
        mode="bicubic",
        align_corners=False,
    ).squeeze()

    depth_np = depth_upsampled.cpu().numpy()
    mask = (depth_np > 4000).astype(np.int32)
    mask_3d = np.stack((mask, mask, mask), axis=2)

    masked_img = np.where(mask_3d == 1, img_rgb, mask_3d).astype(np.uint8)

    input_tensor = prepare_image(masked_img)
    with torch.no_grad():
        logits = model(input_tensor)

    probs = torch.nn.functional.softmax(logits, dim=1)
    pred_label = int(torch.argmax(probs, dim=1).cpu().item())

    names.append(img_name)
    preds.append(pred_label)
