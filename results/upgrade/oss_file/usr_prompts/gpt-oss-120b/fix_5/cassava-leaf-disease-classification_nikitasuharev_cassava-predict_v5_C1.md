## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_PATH = os.path.join(BASE_PATH, "train_images")
TEST_IMG_PATH = os.path.join(BASE_PATH, "test_images")

train_df = pd.read_csv(os.path.join(BASE_PATH, "train.csv"))
train_df["label"] = train_df["label"].astype(str)
train_df["filename"] = train_df[
    "image_id"
]  # column name required by flow_from_dataframe
train_df = train_df[["filename", "label"]]

train_split, val_split = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=42
)

IMG_SIZE = (224, 224)  # matches MobileNetV2 default
BATCH_SIZE = 32
WORKERS = 8  # parallel image loading

train_datagen = tf.keras.preprocessing.image.ImageDataGenerator(
    rescale=1.0 / 255,
    horizontal_flip=True,
    rotation_range=15,
    width_shift_range=0.1,
    height_shift_range=0.1,
    zoom_range=0.1,
)
val_datagen = tf.keras.preprocessing.image.ImageDataGenerator(rescale=1.0 / 255)

train_gen = train_datagen.flow_from_dataframe(
    dataframe=train_split,
    directory=TRAIN_IMG_PATH,
    x_col="filename",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="sparse",
    shuffle=True,
    workers=WORKERS,
    use_multiprocessing=True,
)

val_gen = val_datagen.flow_from_dataframe(
    dataframe=val_split,
    directory=TRAIN_IMG_PATH,
    x_col="filename",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="sparse",
    shuffle=False,
    workers=WORKERS,
    use_multiprocessing=True,
)

base_model = tf.keras.applications.MobileNetV2(
    input_shape=IMG_SIZE + (3,), include_top=False, weights="imagenet"
)
base_model.trainable = False  # freeze for quick initial training

inputs = tf.keras.Input(shape=IMG_SIZE + (3,))
x = tf.keras.applications.mobilenet_v2.preprocess_input(inputs)
x = base_model(x, training=False)
x = tf.keras.layers.GlobalAveragePooling2D()(x)
outputs = tf.keras.layers.Dense(5, activation="softmax")(x)

model = tf.keras.Model(inputs, outputs)
model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)

model.fit(
    train_gen,
    epochs=3,
    validation_data=val_gen,
    verbose=2,
    workers=WORKERS,
    use_multiprocessing=True,
)

base_model.trainable = True
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)
model.fit(
    train_gen,
    epochs=5,
    validation_data=val_gen,
    verbose=2,
    workers=WORKERS,
    use_multiprocessing=True,
)

import tensorflow as tf
from tqdm import tqdm

submission = pd.read_csv(os.path.join(BASE_PATH, "sample_submission.csv"))
test_filenames = [
    os.path.join(TEST_IMG_PATH, img_id) for img_id in submission["image_id"].values
]


def load_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    return img


test_ds = tf.data.Dataset.from_tensor_slices(test_filenames)
test_ds = test_ds.map(load_and_preprocess, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)



## === cell 3
preds = model.predict(test_ds, verbose=0)
all_preds = np.argmax(preds, axis=-1)



## === cell 4
submission["label"] = all_preds
submission.to_csv("submission.csv", index=False)
