## === cell 0
import os, glob, sys, random
import numpy as np
import pandas as pd

try:
    from google.protobuf import message_factory

    if not hasattr(message_factory.MessageFactory, "GetPrototype"):

        def _GetPrototype(self, descriptor):
            return self.GetMessageClass(descriptor)

        message_factory.MessageFactory.GetPrototype = _GetPrototype
except Exception:
    pass

import tensorflow as tf

tf.random.set_seed(42)
np.random.seed(42)
random.seed(42)

work_dir = "../input/cassava-leaf-disease-classification/"

train_csv_path = os.path.join(work_dir, "train.csv")
train_img_dir = os.path.join(work_dir, "train_images")
test_img_dir = os.path.join(work_dir, "test_images")

train_df = pd.read_csv(train_csv_path)

train_image_paths = glob.glob(
    os.path.join(train_img_dir, "**", "*.jpg"), recursive=True
)
train_path_map = {os.path.basename(p): p for p in train_image_paths}
train_paths = [train_path_map[fname] for fname in train_df["image_id"]]
train_labels = train_df["label"].values

test_image_paths = glob.glob(os.path.join(test_img_dir, "**", "*.jpg"), recursive=True)
test_files = sorted([os.path.basename(p) for p in test_image_paths])
test_paths = [os.path.join(test_img_dir, fname) for fname in test_files]



## === cell 1
IMG_SIZE = 224  # larger resolution for better feature extraction
BATCH_SIZE = 32

from tensorflow.keras.applications.efficientnet import preprocess_input


def _parse_image_train(filename, label):
    img = tf.io.read_file(filename)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = tf.cast(img, tf.float32)
    img = tf.image.random_flip_left_right(img)
    img = tf.image.random_flip_up_down(img)
    img = preprocess_input(img)  # scale to EfficientNet range
    return img, label


train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(_parse_image_train, num_parallel_calls=tf.data.AUTOTUNE)
train_ds = (
    train_ds.shuffle(buffer_size=1000).batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
)



## === cell 2
base_model = tf.keras.applications.EfficientNetB0(
    include_top=False,
    weights="imagenet",
    input_shape=(IMG_SIZE, IMG_SIZE, 3),
    pooling="avg",
)

inputs = tf.keras.Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = base_model(inputs, training=False)
outputs = tf.keras.layers.Dense(5, activation="softmax")(x)  # 5 classes
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
    loss=tf.keras.losses.SparseCategoricalCrossentropy(),
    metrics=["accuracy"],
)



## === cell 3
EPOCHS = 10  # more epochs for better convergence
model.fit(train_ds, epochs=EPOCHS, verbose=2)




## === cell 4
def _parse_test_image(filename):
    img = tf.io.read_file(filename)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = tf.cast(img, tf.float32)
    img = preprocess_input(img)
    return img


test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(_parse_test_image, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)



## === cell 5
submission = pd.DataFrame({"image_id": test_files, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}, shape: {submission.shape}")
