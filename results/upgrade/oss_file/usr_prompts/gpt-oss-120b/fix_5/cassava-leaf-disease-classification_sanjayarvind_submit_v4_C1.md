## === cell 0
import os
import cv2
import pandas as pd
import torch
from torch import nn
from torch.utils.data import Dataset, DataLoader
import albumentations as A
from albumentations.pytorch import ToTensorV2
from torchvision import models

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Using device:", device)




## === cell 1
def get_tta(image_size, p=0.5):
    imagenet_stats = {"mean": [0.485, 0.456, 0.406], "std": [0.229, 0.224, 0.225]}
    augs = A.Compose(
        [
            A.Resize(600, 600),
            A.RandomCrop(image_size, image_size),
            A.HorizontalFlip(p=p),
            A.RandomBrightnessContrast(brightness_limit=0.2, contrast_limit=0.2, p=p),
            A.OneOf(
                [
                    A.MotionBlur(p=0.3),
                    A.MedianBlur(blur_limit=3, p=0.3),
                    A.Blur(blur_limit=3, p=0.3),
                ],
                p=p,
            ),
            A.OneOf(
                [
                    A.OpticalDistortion(p=0.3),
                    A.GridDistortion(p=0.3),
                    A.PiecewiseAffine(p=0.3),
                ],
                p=p,
            ),
            A.OneOf(
                [
                    A.GaussNoise(p=0.5),
                ],
                p=p,
            ),
            A.Normalize(mean=imagenet_stats["mean"], std=imagenet_stats["std"]),
            ToTensorV2(),
        ]
    )
    return augs




## === cell 2
class CassavaTrainDataset(Dataset):
    """Dataset for training images with labels."""

    def __init__(self, img_dir, csv_path, transforms):
        self.img_dir = img_dir
        self.df = pd.read_csv(csv_path)
        self.transform = transforms

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = os.path.join(self.img_dir, row["image_id"])
        image = cv2.imread(img_path)
        if image is None:
            raise FileNotFoundError(f"Image not found: {img_path}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = self.transform(image=image)["image"]
        label = int(row["label"])
        return image, label


class CassavaTestDataset(Dataset):
    """Dataset for test images (labels are dummy)."""

    def __init__(self, img_dir, csv_path, transforms):
        self.img_dir = img_dir
        self.df = pd.read_csv(csv_path)
        self.transform = transforms

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = os.path.join(self.img_dir, row["image_id"])
        image = cv2.imread(img_path)
        if image is None:
            raise FileNotFoundError(f"Image not found: {img_path}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = self.transform(image=image)["image"]
        return image, 0




## === cell 3
train_img_dir = "../input/cassava-leaf-disease-classification/train_images"
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"

test_img_dir = "../input/cassava-leaf-disease-classification/test_images"
test_csv_path = "../input/cassava-leaf-disease-classification/sample_submission.csv"  # contains image_id list

train_transforms = get_tta(image_size=224)  # smaller size for faster training
test_transforms = get_tta(image_size=546)  # keep larger crop for inference

train_dataset = CassavaTrainDataset(train_img_dir, train_csv_path, train_transforms)
train_loader = DataLoader(
    train_dataset, batch_size=64, shuffle=True, num_workers=0, pin_memory=True
)

test_dataset = CassavaTestDataset(test_img_dir, test_csv_path, test_transforms)
test_loader = DataLoader(
    test_dataset, batch_size=32, shuffle=False, num_workers=0, pin_memory=True
)



## === cell 4
model = models.mobilenet_v2(pretrained=True)
model.classifier[1] = nn.Linear(in_features=1280, out_features=5, bias=True)
model = model.to(device)

criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)

model.train()
for epoch in range(5):
    epoch_loss = 0.0
    for images, labels in train_loader:
        images = images.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        epoch_loss += loss.item() * images.size(0)
    avg_loss = epoch_loss / len(train_loader.dataset)
    print(f"Epoch [{epoch+1}/5] - Loss: {avg_loss:.4f}")



## === cell 5
model.eval()
num_augmentations = 5
with torch.no_grad():
    agg_scores = torch.zeros((len(test_dataset), 5), device=device)
    for aug in range(num_augmentations):
        all_scores = []
        for images, _ in test_loader:
            images = images.to(device)
            logits = model(images)
            probs = torch.nn.functional.softmax(logits, dim=1)
            all_scores.append(probs)
        batch_scores = torch.cat(all_scores, dim=0)
        agg_scores += batch_scores
    agg_scores /= num_augmentations
    pred = torch.argmax(agg_scores, dim=1).cpu().numpy()



## === cell 6
submission_df = pd.read_csv(test_csv_path)
submission_df["label"] = pred
submission_path = "./submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
print(submission_df.head())
