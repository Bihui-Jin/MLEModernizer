import os
import glob
import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms
from transformers import ViTImageProcessor, ViTModel
import pandas as pd
from PIL import Image


class CassavaLeafTestDataset(Dataset):
    def __init__(self, image_dir, image_processor):
        self.image_paths = sorted(glob.glob(os.path.join(image_dir, "*.jpg")))
        self.image_processor = image_processor

    def __len__(self):
        return len(self.image_paths)

    def __getitem__(self, idx):
        path = self.image_paths[idx]
        image = Image.open(path).convert("RGB")
        inputs = self.image_processor(images=image, return_tensors="pt")
        pixel_values = inputs["pixel_values"].squeeze(0)  # [C, H, W]
        image_id = os.path.basename(path)
        return pixel_values, image_id


class ViTForImageClassification(nn.Module):
    def __init__(self, num_labels=5):
        super(ViTForImageClassification, self).__init__()
        self.num_labels = num_labels
        self.vit = ViTModel.from_pretrained(
            "/kaggle/input/google-vit/google_vit", local_files_only=True
        )
        self.dropout = nn.Dropout(0.1)
        self.classifier = nn.Linear(self.vit.config.hidden_size, num_labels)

    def forward(self, pixel_values, labels=None):
        outputs = self.vit(pixel_values=pixel_values)
        cls_output = outputs.last_hidden_state[:, 0]  # [batch, hidden]
        hidden = self.dropout(cls_output)
        logits = self.classifier(hidden)

        loss = None
        if labels is not None:
            loss_fct = nn.CrossEntropyLoss()
            loss = loss_fct(logits.view(-1, self.num_labels), labels.view(-1))
        return logits, loss


device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = ViTForImageClassification().to(device)

model_path = "/kaggle/input/new-pth/new_v2.pth"
if os.path.exists(model_path):
    try:
        model.load_state_dict(torch.load(model_path, map_location=device))
    except Exception as e:
        print(f"Failed to load pretrained weights: {e}. Using base ViT weights.")
else:
    print("Pretrained .pth file not found; proceeding with base ViT weights.")
model.eval()

try:
    image_processor = ViTImageProcessor.from_pretrained(
        "/kaggle/input/google-vit/google_vit", local_files_only=True
    )
except Exception as e:
    print(f"Local ViT processor not found ({e}); loading default processor from hub.")
    image_processor = ViTImageProcessor.from_pretrained(
        "google/vit-base-patch16-224", local_files_only=False
    )

TEST_IMAGES_DIR = "/kaggle/input/cassava-leaf-disease-classification/test_images"
test_dataset = CassavaLeafTestDataset(TEST_IMAGES_DIR, image_processor=image_processor)
test_dataloader = DataLoader(test_dataset, batch_size=12, shuffle=False, num_workers=0)
