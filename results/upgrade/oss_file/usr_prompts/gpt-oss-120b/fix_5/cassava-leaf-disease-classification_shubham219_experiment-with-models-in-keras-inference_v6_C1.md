import os
import glob
import pandas as pd
from collections import Counter


def locate_path(package_name: str, sub_dir: str) -> str:
    """
    Find the directory for a given Kaggle package and sub‑folder.
    Checks common Kaggle roots; raises FileNotFoundError if not found.
    """
    possible_roots = [
        ".",  # current working directory
        "/kaggle/input",  # standard input mount
        "/kaggle/working",  # notebook working dir
        "/kaggle/data",  # alternative data mount
    ]
    for root in possible_roots:
        candidate = os.path.join(root, package_name, sub_dir)
        if os.path.isdir(candidate):
            return candidate
    raise FileNotFoundError(f"Could not locate {sub_dir} inside {package_name}")
