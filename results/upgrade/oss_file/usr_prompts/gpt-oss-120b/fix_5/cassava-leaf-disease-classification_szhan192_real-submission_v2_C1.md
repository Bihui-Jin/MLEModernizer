## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION_VERSION"] = "2"

import numpy as np
import pandas as pd
import tensorflow as tf
from sklearn.model_selection import train_test_split

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

df = pd.read_csv(TRAIN_CSV)
df["filepath"] = df["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))
df["label"] = df["label"].astype(int)  # integer labels for sparse loss

train_df, val_df = train_test_split(
    df, test_size=0.1, stratify=df["label"], random_state=42
)

AUTOTUNE = tf.data.AUTOTUNE
IMG_SIZE = (224, 224)


def _load_and_preprocess_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    return img


def _process_path_label(path, label):
    img = _load_and_preprocess_image(path)
    return img, tf.cast(label, tf.int32)


train_ds = tf.data.Dataset.from_tensor_slices(
    (train_df["filepath"].values, train_df["label"].values)
)
train_ds = train_ds.map(_process_path_label, num_parallel_calls=AUTOTUNE)
train_ds = train_ds.shuffle(1000).batch(32).prefetch(AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices(
    (val_df["filepath"].values, val_df["label"].values)
)
val_ds = val_ds.map(_process_path_label, num_parallel_calls=AUTOTUNE)
val_ds = val_ds.batch(32).prefetch(AUTOTUNE)

base_model = tf.keras.applications.MobileNetV2(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3), pooling="avg"
)
base_model.trainable = False  # freeze base

inputs = tf.keras.Input(shape=(224, 224, 3))
x = base_model(inputs, training=False)
outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
model = tf.keras.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",  # accepts integer labels
    metrics=["accuracy"],
)




## === cell 2
model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=2)




## === cell 3
sample_sub = pd.read_csv(SAMPLE_SUB)
test_paths = (
    sample_sub["image_id"].apply(lambda x: os.path.join(TEST_IMG_DIR, x)).values
)

test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(_load_and_preprocess_image, num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(32).prefetch(AUTOTUNE)




## === cell 4
pred_probs = model.predict(test_ds, verbose=1)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": pred_labels})
submission.to_csv("/kaggle/working/submission.csv", index=False)
