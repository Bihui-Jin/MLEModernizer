# 5. Code solution

## === cell 0
import os
import json
import numpy as np
import pandas as pd
import cv2
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score
from concurrent.futures import ThreadPoolExecutor



## === cell 1
BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
train_dir = os.path.join(BASE_PATH, "train_images")
train_csv = os.path.join(BASE_PATH, "train.csv")
label_json = os.path.join(BASE_PATH, "label_num_to_disease_map.json")
test_dir = os.path.join(BASE_PATH, "test_images")

with open(label_json) as f:
    label_name = json.load(f)

data_label = pd.read_csv(train_csv)




## === cell 2
def load_image(path):
    img = cv2.imread(path)
    if img is None:
        return None
    img = cv2.resize(img, (100, 100))
    img = img.astype(np.float32) / 255.0
    return img.flatten()  # 100*100*3 = 30000 features


train_paths = [os.path.join(train_dir, img_id) for img_id in data_label["image_id"]]
train_labels = data_label["label"].astype(int).tolist()

with ThreadPoolExecutor(max_workers=os.cpu_count()) as executor:
    loaded_imgs = list(executor.map(load_image, train_paths))

train_images = []
train_labels_filtered = []
for img_vec, lbl in zip(loaded_imgs, train_labels):
    if img_vec is not None:
        train_images.append(img_vec)
        train_labels_filtered.append(lbl)

X = np.stack(train_images, axis=0)
y = np.array(train_labels_filtered)



## === cell 3
X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, random_state=42, stratify=y
)

clf = LogisticRegression(
    multi_class="multinomial",
    solver="saga",
    max_iter=200,
    n_jobs=-1,
    verbose=0,
    penalty="l2",
    C=1.0,
    random_state=42,
)
clf.fit(X_train, y_train)

val_pred = clf.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy: {val_acc:.5f}")



## === cell 4
test_images = os.listdir(test_dir)

test_paths = [os.path.join(test_dir, img_name) for img_name in test_images]

with ThreadPoolExecutor(max_workers=os.cpu_count()) as executor:
    test_vecs = list(executor.map(load_image, test_paths))

pred_ids = []
pred_labels = []
for img_name, img_vec in zip(test_images, test_vecs):
    if img_vec is None:
        continue
    label = int(clf.predict(img_vec.reshape(1, -1))[0])
    pred_ids.append(img_name)
    pred_labels.append(label)

submission_df = pd.DataFrame({"image_id": pred_ids, "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
