## === cell 0
import os
import numpy as np
import pandas as pd

if os.path.isdir("/kaggle/input"):
    PATH = "/kaggle/input"
else:
    PATH = "../input"

TRAIN_DIR = os.path.join(PATH, "train")
TEST_DIR = os.path.join(PATH, "test")

if not os.path.isdir(TRAIN_DIR):
    alt_train = os.path.join(PATH, "dogs-vs-cats-redux-kernels-edition", "train")
    if os.path.isdir(alt_train):
        TRAIN_DIR = alt_train

if not os.path.isdir(TEST_DIR):
    alt_test = os.path.join(PATH, "dogs-vs-cats-redux-kernels-edition", "test")
    if os.path.isdir(alt_test):
        TEST_DIR = alt_test




## === cell 1
train_fnames = []
train_labels = []

for cls_name, cls_label in [("cat", 0), ("dog", 1)]:
    cls_path = os.path.join(TRAIN_DIR, cls_name)
    if os.path.isdir(cls_path):
        for f in sorted(os.listdir(cls_path)):
            if f.lower().endswith((".jpg", ".jpeg", ".png")):
                train_fnames.append(os.path.join("train", cls_name, f))
                train_labels.append(cls_label)

train_fnames = np.array(train_fnames)
train_labels = np.array(train_labels)

print(f"Found {len(train_fnames)} training images.")
print(f"Sample: {train_fnames[:2]}, labels: {train_labels[:2]}")




## === cell 2
test_files = [
    f
    for f in sorted(os.listdir(TEST_DIR))
    if f.lower().endswith((".jpg", ".jpeg", ".png"))
]
ids = [os.path.splitext(fname)[0] for fname in test_files]

print(f"Found {len(test_files)} test images. First id: {ids[0] if ids else 'N/A'}")




## === cell 3
if len(train_labels) > 0:
    dog_prob = float(train_labels.mean())  # proportion of label==1
else:
    dog_prob = 0.5  # fallback if training data could not be loaded

preds = np.full(len(ids), dog_prob, dtype=float)




## === cell 4
submission = pd.DataFrame({"id": ids, "label": preds})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)

print(f"Submission file written to {submission_path}")
print(submission.head())
