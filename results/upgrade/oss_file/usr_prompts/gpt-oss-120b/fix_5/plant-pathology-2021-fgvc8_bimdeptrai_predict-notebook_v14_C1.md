## === cell 0
import pandas as pd
import numpy as np
from sklearn.preprocessing import MultiLabelBinarizer



## === cell 1
train = pd.read_csv("../input/plant-pathology-2021-fgvc8/train.csv")



## === cell 2
submissions = pd.read_csv("../input/plant-pathology-2021-fgvc8/sample_submission.csv")



## === cell 3
label_split = train["labels"].apply(lambda x: x.split())
mlb = MultiLabelBinarizer().fit(label_split)
labels_onehot = pd.DataFrame(mlb.transform(label_split), columns=mlb.classes_)

label_counts = labels_onehot.sum().sort_values(ascending=False)

avg_labels_per_image = label_split.apply(len).mean()
k = max(1, int(round(avg_labels_per_image)))  # ensure at least one label per prediction

probs = label_counts.values / label_counts.values.sum()
label_names = label_counts.index.to_numpy()

rng = np.random.default_rng(42)


def sample_labels():
    """Sample k distinct labels according to their training prevalence."""
    return " ".join(rng.choice(label_names, size=k, replace=False, p=probs))


submissions["labels"] = submissions.apply(lambda _: sample_labels(), axis=1)



## === cell 4
submissions.to_csv("submission.csv", index=False)
