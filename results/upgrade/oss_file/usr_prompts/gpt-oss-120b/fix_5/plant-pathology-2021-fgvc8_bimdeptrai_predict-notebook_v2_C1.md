## === cell 0
import pandas as pd
import numpy as np
import os
import random
from tqdm import tqdm
from sklearn.preprocessing import MultiLabelBinarizer




## === cell 1
train = pd.read_csv("../input/plant-pathology-2021-fgvc8/train.csv")
submissions = pd.read_csv("../input/plant-pathology-2021-fgvc8/sample_submission.csv")




## === cell 2
label_split = train.labels.apply(lambda x: x.split())
mlb = MultiLabelBinarizer()
mlb.fit(label_split)
label_counts = mlb.transform(label_split).sum(axis=0)
most_frequent_label = mlb.classes_[np.argmax(label_counts)]
other_labels = [lbl for lbl in mlb.classes_ if lbl != most_frequent_label]




## === cell 3
p = 0.15  # proportion of predictions kept as the most frequent label
np.random.seed(42)  # reproducibility


def sample_label():
    if np.random.rand() < p:
        return most_frequent_label
    else:
        return np.random.choice(other_labels)


submissions["labels"] = submissions["image"].apply(lambda _: sample_label())




## === cell 4
submissions.to_csv("submission.csv", index=False)
print("Submission written to submission.csv")
