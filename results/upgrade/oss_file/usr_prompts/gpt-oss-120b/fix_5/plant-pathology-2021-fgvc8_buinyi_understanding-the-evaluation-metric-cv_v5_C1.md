**sample_submission.csv** - A sample submission file in the correct format.

- `image`
- `labels`

**train_images** - The training set images.

**test_images** - The test set images. This competition has a hidden test set: only three images are provided here as samples while the remaining 5,000 images will be available to your notebook once it is submitted.

## === cell 0
import numpy as np  # linear algebra
import pandas as pd  # data processing, CSV file I/O (e.g. pd.read_csv)
import os

BASE_PATH = "/kaggle/input/plant-pathology-2021-fgvc8"



## === cell 1
train_path = os.path.join(BASE_PATH, "train.csv")
train = pd.read_csv(train_path)

if train["labels"].isnull().any():
    raise ValueError("Missing label entries in training data.")
all_labels = train["labels"].str.split().explode()
most_common_label = all_labels.value_counts().idxmax()
if pd.isna(most_common_label):
    raise ValueError("Could not determine a most common label.")



## === cell 2
TEST_FOLDER = os.path.join(BASE_PATH, "test_images")
images = os.listdir(TEST_FOLDER)
sub = pd.DataFrame(images, columns=["image"])



## === cell 3
sub["labels"] = most_common_label

sub = sub.sort_values("image").reset_index(drop=True)

submission_path = "submission.csv"
sub.to_csv(submission_path, index=False)
