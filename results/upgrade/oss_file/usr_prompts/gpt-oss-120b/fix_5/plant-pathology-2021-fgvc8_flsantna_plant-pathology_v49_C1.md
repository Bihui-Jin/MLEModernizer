## === cell 0
import os
import pandas as pd

output_dir = "./"
test_dir = "../input/plant-pathology-2021-fgvc8/test_images/"

train_path = "../input/plant-pathology-2021-fgvc8/train.csv"
train_df = pd.read_csv(train_path)

label_counts = {}
for lbls in train_df["labels"].astype(str):
    for lab in lbls.split():
        label_counts[lab] = label_counts.get(lab, 0) + 1

total_label_occurrences = sum(label_counts.values())
PROPORTION_THRESHOLD = 0.05  # 5%

default_labels = [
    lbl
    for lbl, cnt in label_counts.items()
    if cnt / total_label_occurrences >= PROPORTION_THRESHOLD
]

if not default_labels:
    most_common_label = max(label_counts.items(), key=lambda x: x[1])[0]
    default_labels = [most_common_label]

default_label_str = " ".join(default_labels)



## === cell 1
test_images = sorted(
    [f for f in os.listdir(test_dir) if f.lower().endswith((".jpg", ".png", ".jpeg"))]
)

submission_rows = [[img_name, default_label_str] for img_name in test_images]
submission_df = pd.DataFrame(submission_rows, columns=["image", "labels"])

submission_path = os.path.join(output_dir, "submission.csv")
submission_df.to_csv(submission_path, index=False)

print(f"Submission file written to {submission_path} with {len(submission_df)} rows.")
