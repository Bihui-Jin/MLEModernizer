`train.csv` contains image IDs, binary labels, and patient IDs.

TFRecords are available for both train and test.

### Columns
- `StudyInstanceUID` - unique ID for each image
- `ETT - Abnormal` - endotracheal tube placement abnormal
- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

W = H = 456
N_CLASSES = 11
BATCH_SIZE = 16
AUTOTUNE = tf.data.experimental.AUTOTUNE

target_cols = [
    "CVC - Abnormal",
    "CVC - Borderline",
    "CVC - Normal",
    "ETT - Abnormal",
    "ETT - Borderline",
    "ETT - Normal",
    "NGT - Abnormal",

    "Swan Ganz Catheter Present",
]

data_dir = "../input/ranzcr-clip-catheter-line-classification"
train_dir = os.path.join(data_dir, "train")
test_dir = os.path.join(data_dir, "test")
sample_submission_path = os.path.join(data_dir, "sample_submission.csv")
train_csv_path = os.path.join(data_dir, "train.csv")




## === cell 1
def load_and_preprocess(path):
    image = tf.io.read_file(path)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, (H, W))
    image = tf.cast(image, tf.float32) / 255.0  # scale to [0,1]
    mean = tf.constant([0.485, 0.456, 0.406])
    std = tf.constant([0.229, 0.224, 0.225])
    image = (image - mean) / std
    return image


def build_image_dataset(image_dir, ids):
    file_paths = [os.path.join(image_dir, f"{uid}.jpg") for uid in ids]
    ds = tf.data.Dataset.from_tensor_slices(file_paths)
    ds = ds.map(load_and_preprocess, num_parallel_calls=AUTOTUNE)
    return ds


def build_train_dataset(df, image_dir):
    ids = df["StudyInstanceUID"].values
    labels = df[target_cols].values.astype(np.float32)

    img_ds = build_image_dataset(image_dir, ids)
    lbl_ds = tf.data.Dataset.from_tensor_slices(labels)

    ds = tf.data.Dataset.zip((img_ds, lbl_ds))

    ds = ds.cache()

    ds = ds.shuffle(buffer_size=1024, reshuffle_each_iteration=True)
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds, ids


def build_test_dataset(image_dir):
    file_names = sorted(
        [f for f in os.listdir(image_dir) if f.lower().endswith(".jpg")]
    )
    ids = [os.path.splitext(f)[0] for f in file_names]
    ds = build_image_dataset(image_dir, ids)
    ds = ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
    return ds, ids




## === cell 2
train_df = pd.read_csv(train_csv_path)
train_dataset, _ = build_train_dataset(train_df, train_dir)

test_dataset, test_ids = build_test_dataset(test_dir)




## === cell 3
def get_model(base_model_cls, lr=0.001):
    base = base_model_cls(
        include_top=False, input_shape=(H, W, 3), pooling="avg", weights="imagenet"
    )
    x = tf.keras.layers.Dropout(0.3)(base.output)
    out = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.models.Model(inputs=base.input, outputs=out)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )
    return model


model = get_model(tf.keras.applications.EfficientNetB5)



## === cell 4
model.fit(train_dataset, epochs=3, verbose=1)



## === cell 5
all_preds = model.predict(test_dataset, verbose=0)



## === cell 6
submission = pd.read_csv(sample_submission_path, nrows=0)  # get header only
submission = pd.DataFrame(columns=submission.columns)

submission["StudyInstanceUID"] = test_ids
for i, col in enumerate(target_cols):
    submission[col] = all_preds[:, i]

submission = submission[["StudyInstanceUID"] + target_cols]
