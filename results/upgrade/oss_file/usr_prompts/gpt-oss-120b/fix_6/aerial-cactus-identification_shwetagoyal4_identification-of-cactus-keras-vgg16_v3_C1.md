## === cell 0
import os
import warnings

warnings.filterwarnings("ignore")

import numpy as np
import pandas as pd
import cv2
import concurrent.futures  # added for parallel image loading

from sklearn.model_selection import train_test_split
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import roc_auc_score


def _load_one(image_dir, fname, img_size):
    """Load, optionally resize, and normalize a single image."""
    path = os.path.join(image_dir, fname)
    img = cv2.imread(path)
    if img is None:
        img = np.zeros((*img_size, 3), dtype=np.uint8)
    else:
        if img.shape[0] != img_size[1] or img.shape[1] != img_size[0]:
            img = cv2.resize(img, img_size)
    return img.astype(np.float32) / 255.0




## === cell 1
train_df = pd.read_csv("../input/train.csv")
train_df["has_cactus"] = train_df["has_cactus"].astype(int)




## === cell 2
def load_images(image_dir, filenames, img_size=(32, 32)):
    """Read images in parallel, resize when needed, and normalize to [0,1].
    Returns a float32 NumPy array with shape (N, H, W, 3)."""
    n = len(filenames)
    imgs = np.empty((n, img_size[1], img_size[0], 3), dtype=np.float32)
    with concurrent.futures.ThreadPoolExecutor() as executor:
        future_to_idx = {
            executor.submit(_load_one, image_dir, fname, img_size): i
            for i, fname in enumerate(filenames)
        }
        for future in concurrent.futures.as_completed(future_to_idx):
            idx = future_to_idx[future]
            imgs[idx] = future.result()
    return imgs




## === cell 3
train_image_dir = "../input/train/train"
X = load_images(train_image_dir, train_df["id"].values)
y = train_df["has_cactus"].values




## === cell 4
X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, stratify=y, random_state=42
)

X_train_flat = X_train.reshape(len(X_train), -1)
X_val_flat = X_val.reshape(len(X_val), -1)




## === cell 5
model = GradientBoostingClassifier(
    n_estimators=200,
    learning_rate=0.1,
    max_depth=3,
    random_state=42,
)
model.fit(X_train_flat, y_train)

val_pred = model.predict_proba(X_val_flat)[:, 1]
val_auc = roc_auc_score(y_val, val_pred)
print(f"Validation AUC: {val_auc:.5f}")




## === cell 6
test_dir = "../input/test/test"
test_filenames = sorted(os.listdir(test_dir))
X_test = load_images(test_dir, test_filenames)
X_test_flat = X_test.reshape(len(X_test), -1)




## === cell 7
test_pred = model.predict_proba(X_test_flat)[:, 1]




## === cell 8
submission = pd.DataFrame({"id": test_filenames, "has_cactus": test_pred})
submission.to_csv("submission.csv", index=False)
