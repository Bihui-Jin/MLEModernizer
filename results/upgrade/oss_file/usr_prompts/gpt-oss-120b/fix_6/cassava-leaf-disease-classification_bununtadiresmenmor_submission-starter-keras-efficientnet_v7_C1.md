## === cell 0
import os
import json
import numpy as np
import pandas as pd
import cv2
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUBMIT = os.path.join(BASE_PATH, "sample_submission.csv")


def extract_features(img_path):
    img = cv2.imread(img_path)
    if img is None:
        return np.zeros(32 * 32 * 3 + 3, dtype=np.float32)
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    resized = cv2.resize(img, (32, 32))
    flat = resized.flatten().astype(np.float32) / 255.0
    mean_rgb = img.mean(axis=(0, 1)).astype(np.float32) / 255.0
    return np.concatenate([flat, mean_rgb])


train_df = pd.read_csv(TRAIN_CSV)

features = []
labels = []
for _, row in train_df.iterrows():
    img_path = os.path.join(TRAIN_IMG_DIR, row["image_id"])
    feats = extract_features(img_path)
    features.append(feats)
    labels.append(row["label"])
X = np.stack(features)
y = np.array(labels)

X_tr, X_val, y_tr, y_val = train_test_split(
    X, y, test_size=0.1, random_state=42, stratify=y
)

rf = RandomForestClassifier(
    n_estimators=1,  # single tree
    max_depth=1,  # stump
    min_samples_split=20,
    min_samples_leaf=10,
    max_features=0.2,  # use only 20% of features per split
    bootstrap=False,
    n_jobs=-1,
    random_state=42,
)
rf.fit(X_tr, y_tr)

val_pred = rf.predict(X_val)
print("Validation accuracy:", accuracy_score(y_val, val_pred))




## === cell 1
sample_sub = pd.read_csv(SAMPLE_SUBMIT)

submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
