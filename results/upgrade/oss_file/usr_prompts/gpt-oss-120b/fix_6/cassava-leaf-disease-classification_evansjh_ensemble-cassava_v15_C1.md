# 5. Code solution

## === cell 0
import os
import pandas as pd
import numpy as np
from collections import Counter
from PIL import Image
from sklearn.linear_model import LogisticRegression
import concurrent.futures  # thread pool for faster I/O‑bound feature extraction



## === cell 1
train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_image_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"
submission_path = "/kaggle/working/submission.csv"



## === cell 2
train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(int)




## === cell 3
def _extract_features(img_path: str) -> np.ndarray:
    """Read an image and return a 12‑dimensional feature vector:
    per‑channel mean, std, min, max (3 × 4 = 12)."""
    if os.path.isfile(img_path):
        with Image.open(img_path).convert("RGB") as im:
            arr = np.array(im).astype(np.float32)  # shape (H, W, 3)
        mean = arr.mean(axis=(0, 1))
        std = arr.std(axis=(0, 1))
        mins = arr.min(axis=(0, 1))
        maxs = arr.max(axis=(0, 1))
        return np.concatenate([mean, std, mins, maxs])
    else:
        return np.zeros(12, dtype=float)


train_image_paths = [
    os.path.join(train_image_dir, row["image_id"]) for _, row in train_df.iterrows()
]

max_workers = min(8, os.cpu_count() or 1)
with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
    train_features = list(executor.map(_extract_features, train_image_paths))

X_train = np.stack(train_features)  # shape (n_samples, 12)
y_train = train_df["label"].values.astype(int)



## === cell 4
model = LogisticRegression(
    multi_class="multinomial",
    solver="lbfgs",
    max_iter=200,
    n_jobs=1,  # single thread – model is tiny, avoids CPU oversubscription
    class_weight="balanced",
    random_state=42,  # deterministic training
)
model.fit(X_train, y_train)




## === cell 5
def _extract_test_features(img_name: str) -> np.ndarray:
    img_path = os.path.join(test_image_dir, img_name)
    return _extract_features(img_path)


test_files = [
    entry.name
    for entry in os.scandir(test_image_dir)
    if entry.is_file() and entry.name.lower().endswith((".jpg", ".jpeg", ".png"))
]

with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
    test_features = list(executor.map(_extract_test_features, test_files))

X_test = np.stack(test_features)  # shape (n_test, 12)
test_preds = model.predict(X_test)



## === cell 6
submission_df = pd.DataFrame({"image_id": test_files, "label": test_preds.astype(int)})
submission_df.to_csv(submission_path, index=False)

print("Submission file created at:", submission_path)
print(submission_df.head())
