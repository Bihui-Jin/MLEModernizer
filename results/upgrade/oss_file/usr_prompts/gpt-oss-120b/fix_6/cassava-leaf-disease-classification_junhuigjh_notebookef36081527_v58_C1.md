## === cell 0
import os
import pandas as pd
import numpy as np
from PIL import Image



## === cell 1
import torch

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")



## === cell 2
train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images"
test_image_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images"

train_df = pd.read_csv(train_csv_path)




## === cell 3
def rgb_stats(image_path):
    """Return mean and std RGB vectors of an image."""
    try:
        img = Image.open(image_path).convert("RGB")
        arr = np.asarray(img, dtype=np.float32) / 255.0  # shape HxWx3
        mean = arr.mean(axis=(0, 1))
        std = arr.std(axis=(0, 1))
        return np.concatenate([mean, std])  # 6‑dim feature
    except Exception:
        return np.full(6, 0.5, dtype=np.float32)


train_features = []
train_labels = []

for idx, row in train_df.iterrows():
    img_path = os.path.join(train_images_dir, row["image_id"])
    if os.path.exists(img_path):
        feats = rgb_stats(img_path)
        train_features.append(feats)
        train_labels.append(int(row["label"]))
    else:
        continue

train_features = np.stack(train_features) if train_features else np.empty((0, 6))
train_labels = np.array(train_labels, dtype=int)



## === cell 4
from sklearn.linear_model import LogisticRegression

logreg = LogisticRegression(
    multi_class="multinomial",
    solver="lbfgs",
    max_iter=500,
    random_state=42,
    n_jobs=-1,
)
logreg.fit(train_features, train_labels)



## === cell 5
image_ids = sorted(os.listdir(test_image_dir))

test_features = []
valid_ids = []  # keep only ids for which we can read the image

for img_name in image_ids:
    img_path = os.path.join(test_image_dir, img_name)
    if os.path.exists(img_path):
        test_features.append(rgb_stats(img_path))
        valid_ids.append(img_name)
    else:
        continue

test_features = np.stack(test_features) if test_features else np.empty((0, 6))



## === cell 6
predictions = logreg.predict(test_features).astype(int).tolist()



## === cell 7
submission = pd.DataFrame({"image_id": valid_ids, "label": predictions})
submission_path = "submission.csv"  # Kaggle expects this in the working directory
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path} with {len(submission)} rows.")
