# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd

train_path = "../input/cassava-leaf-disease-classification/train.csv"
train_dir = "../input/cassava-leaf-disease-classification/train_images/"
test_dir = "../input/cassava-leaf-disease-classification/test_images/"

train_df = pd.read_csv(train_path)

train_df["label"] = train_df["label"].astype(str)

most_common_label = train_df["label"].mode()[0]
fallback_label_str = str(most_common_label)



## === cell 1
try:
    import tensorflow as tf

    tf_version = tf.__version__
    tf.compat.v1.logging.set_verbosity(tf.compat.v1.logging.ERROR)
    tf_available = True
except Exception as e:
    print("TensorFlow import failed:", e)
    tf_available = False



## === cell 2
if tf_available:
    img_size = (128, 128)
    batch_size = 32
    epochs = 5  # modest increase to improve accuracy

    train_datagen = tf.keras.preprocessing.image.ImageDataGenerator(
        rescale=1.0 / 255,
        validation_split=0.1,
        horizontal_flip=True,
        rotation_range=20,
        width_shift_range=0.1,
        height_shift_range=0.1,
    )

    train_generator = train_datagen.flow_from_dataframe(
        dataframe=train_df,
        directory=train_dir,
        x_col="image_id",
        y_col="label",
        subset="training",
        batch_size=batch_size,
        seed=42,
        shuffle=True,
        class_mode="categorical",
        target_size=img_size,
    )

    val_generator = train_datagen.flow_from_dataframe(
        dataframe=train_df,
        directory=train_dir,
        x_col="image_id",
        y_col="label",
        subset="validation",
        batch_size=batch_size,
        seed=42,
        shuffle=False,
        class_mode="categorical",
        target_size=img_size,
    )

    base_model = tf.keras.applications.MobileNetV2(
        input_shape=img_size + (3,),
        include_top=False,
        weights="imagenet",
        pooling="avg",
    )
    base_model.trainable = False  # freeze base

    inputs = tf.keras.Input(shape=img_size + (3,))
    x = tf.keras.applications.mobilenet_v2.preprocess_input(inputs)
    x = base_model(x, training=False)
    outputs = tf.keras.layers.Dense(5, activation="softmax")(x)
    model = tf.keras.Model(inputs, outputs)

    model.compile(
        optimizer=tf.keras.optimizers.Adam(),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
    )

    model.fit(
        train_generator,
        epochs=epochs,
        validation_data=val_generator,
        verbose=0,
    )

    test_filenames = sorted(
        [f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")]
    )
    test_df = pd.DataFrame({"image_id": test_filenames})

    test_datagen = tf.keras.preprocessing.image.ImageDataGenerator(rescale=1.0 / 255)

    test_generator = test_datagen.flow_from_dataframe(
        dataframe=test_df,
        directory=test_dir,
        x_col="image_id",
        y_col=None,
        batch_size=batch_size,
        shuffle=False,
        class_mode=None,
        target_size=img_size,
    )

    preds = model.predict(test_generator, verbose=0)
    pred_labels = np.argmax(preds, axis=1).astype(str)

else:
    test_filenames = sorted(
        [f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")]
    )
    pred_labels = np.array([fallback_label_str] * len(test_filenames))



## === cell 3
submission = pd.DataFrame({"image_id": test_filenames, "label": pred_labels})

submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
