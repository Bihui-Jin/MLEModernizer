# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image  # added for lightweight image feature extraction

np.random.seed(42)




## === cell 1
train_csv_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
train_df = pd.read_csv(train_csv_path)




## === cell 2
train_images_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images/"
train_features = []  # list of (mean_std_rgb, label) tuples
for idx, row in train_df.iterrows():
    img_path = os.path.join(train_images_dir, row["image_id"])
    try:
        with Image.open(img_path) as img:
            img = img.convert("RGB")
            arr = np.array(img).astype(np.float32)
            mean_rgb = arr.mean(axis=(0, 1))  # (3,)
            std_rgb = arr.std(axis=(0, 1))  # (3,)
            feature = np.concatenate([mean_rgb, std_rgb])  # (6,)
        train_features.append((feature, int(row["label"])))
    except Exception:
        continue

train_features_arr = np.stack([f[0] for f in train_features])  # (N_train, 6)
train_labels_arr = np.array([f[1] for f in train_features])  # (N_train,)




## === cell 3
test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"
test_filenames = sorted([f for f in os.listdir(test_dir) if f.lower().endswith(".jpg")])
num_test = len(test_filenames)

fallback_label = int(train_df["label"].value_counts().idxmax())

k = 3  # number of neighbours for majority vote
predictions = []

for filename in test_filenames:
    img_path = os.path.join(test_dir, filename)
    try:
        with Image.open(img_path) as img:
            img = img.convert("RGB")
            arr = np.array(img).astype(np.float32)
            mean_rgb = arr.mean(axis=(0, 1))
            std_rgb = arr.std(axis=(0, 1))
            test_feat = np.concatenate([mean_rgb, std_rgb])  # (6,)
        dists = np.linalg.norm(train_features_arr - test_feat, axis=1)
        kn = min(k, len(dists))
        nearest_idxs = np.argpartition(dists, kn - 1)[:kn]
        nearest_labels = train_labels_arr[nearest_idxs]
        counts = np.bincount(nearest_labels, minlength=5)
        pred_label = int(np.argmax(counts))
    except Exception:
        pred_label = fallback_label
    predictions.append(pred_label)




## === cell 4
submission_path = "/kaggle/working/submission.csv"
results = pd.DataFrame({"image_id": test_filenames, "label": predictions})
results.to_csv(submission_path, index=False)
