## === cell 1
base_path = "/kaggle/input/cassava-leaf-disease-classification"
if not os.path.isdir(base_path):
    base_path = "../input/cassava-leaf-disease-classification"

train_dir = os.path.join(base_path, "train_images")
test_dir = os.path.join(base_path, "test_images")
train_csv_path = os.path.join(base_path, "train.csv")
sample_submission_path = os.path.join(base_path, "sample_submission.csv")

train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(str)

num_workers = multiprocessing.cpu_count()



## === cell 2
datagen = ImageDataGenerator(rescale=1.0 / 255, validation_split=0.2)

train_generator = datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=train_dir,
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    batch_size=32,
    class_mode="categorical",
    subset="training",
    shuffle=True,
    seed=42,
    workers=num_workers,
    use_multiprocessing=True,
)

val_generator = datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=train_dir,
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    batch_size=32,
    class_mode="categorical",
    subset="validation",
    shuffle=False,
    seed=42,
    workers=num_workers,
    use_multiprocessing=True,
)



## === cell 3
base_model = applications.EfficientNetB0(
    weights="imagenet", include_top=False, input_shape=(224, 224, 3), pooling="avg"
)

base_model.trainable = True

model = models.Sequential([base_model, layers.Dense(5, activation="softmax")])

model.compile(
    optimizer=optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(
    train_generator,
    epochs=30,
    validation_data=val_generator,
    verbose=2,
    workers=num_workers,
    use_multiprocessing=True,
)
