## Task
Classify each cassava image into four disease categories or a fifth category indicating a healthy leaf.

## Metric
Categorization accuracy.

## Submission Format
```

1000840542.jpg,4
etc.
```

## Dataset
**[train/test]_images** the image files.

**train.csv**

- `image_id` the image file name.

- `label` the ID code for the disease.

- `image_id` the image file name.

- `label` the predicted ID code for the disease.

**[train/test]_tfrecords** the image files in tfrecord format.

**label_num_to_disease_map.json** The mapping between each disease code and the real disease name.

import os
import albumentations as A
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from tensorflow.keras.applications import EfficientNetB0
from tensorflow.keras.preprocessing.image import ImageDataGenerator

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

INPUT_DIR = "/kaggle/input/cassava-leaf-disease-classification"

train_df = pd.read_csv(os.path.join(INPUT_DIR, "train.csv"))
train_df["label"] = train_df["label"].astype(str)

test_images_path = os.path.join(INPUT_DIR, "test_images")
image_files = [
    f
    for f in os.listdir(test_images_path)
    if f.lower().endswith((".jpg", ".jpeg", ".png"))
]
test_df = pd.DataFrame({"image_id": sorted(image_files)})


def transform(image):
    aug = A.Compose(
        [
            A.HorizontalFlip(p=0.5),
            A.RandomRotate90(p=0.5),
            A.VerticalFlip(p=0.5),
        ]
    )
    return aug(image=image)["image"]


train_split, val_split = train_test_split(
    train_df,
    test_size=0.20,  # was 0.10
    stratify=train_df["label"],
    random_state=42,
)

train_datagen = ImageDataGenerator(
    preprocessing_function=None  # was transform
).flow_from_dataframe(
    dataframe=train_split,
    directory=os.path.join(INPUT_DIR, "train_images"),
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    class_mode="categorical",
    batch_size=32,
    shuffle=True,
    seed=42,
)

val_datagen = ImageDataGenerator(preprocessing_function=None).flow_from_dataframe(
    dataframe=val_split,
    directory=os.path.join(INPUT_DIR, "train_images"),
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    class_mode="categorical",
    batch_size=32,
    shuffle=False,
)

model = EfficientNetB0(classes=5, weights=None, input_shape=(224, 224, 3))
model.compile(optimizer="adam", loss="categorical_crossentropy", metrics=["accuracy"])

model.fit(
    train_datagen,
    validation_data=val_datagen,
    epochs=2,  # was 5
    verbose=2,
)

test_datagen = ImageDataGenerator().flow_from_dataframe(
    dataframe=test_df,
    directory=test_images_path,
    x_col="image_id",
    y_col=None,
    target_size=(224, 224),
    class_mode=None,
    batch_size=32,
    shuffle=False,
)

y_pred = model.predict(test_datagen, verbose=0)
test_df["label"] = np.argmax(y_pred, axis=1).astype(int)

submission_path = "submission.csv"
test_df.to_csv(submission_path, index=False)
print("Done – submission written to", submission_path)
