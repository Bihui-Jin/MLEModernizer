## === cell 1
sample_sub_path = "../input/cassava-leaf-disease-classification/sample_submission.csv"

## === cell 2
device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
model = models.resnext50_32x4d(pretrained=True)
model.fc = nn.Linear(model.fc.in_features, 5)
model = model.to(device)




## === cell 3
train_aug = A.Compose(
    [

        A.HorizontalFlip(p=0.5),
        A.ShiftScaleRotate(shift_limit=0.1, scale_limit=0.1, rotate_limit=15, p=0.5),
        A.RandomBrightnessContrast(p=0.2),
        A.Normalize(
            mean=[0.485, 0.456, 0.406],
            std=[0.229, 0.224, 0.225],
            max_pixel_value=255.0,
            p=1.0,
        ),
    ]
)

test_aug = A.Compose(
    [
        A.Resize(256, 256),
        A.HorizontalFlip(p=0.5),
        A.Normalize(
            mean=[0.485, 0.456, 0.406],
            std=[0.229, 0.224, 0.225],
            max_pixel_value=255.0,
            p=1.0,
        ),
    ]
)




## === cell 4
class CassavaDataset(Dataset):
    def __init__(self, df, img_dir, augment):
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.augment = augment

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = os.path.join(self.img_dir, row["image_id"])
        with Image.open(img_path) as img:
            img = img.convert("RGB")
            img_np = np.array(img)

        aug = self.augment(image=img_np)["image"]
        tensor = torch.from_numpy(aug.transpose(2, 0, 1)).float()
        label = int(row["label"])
        return tensor, label


train_df = pd.read_csv(train_csv_path)

train_split, val_split = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=42
)

train_dataset = CassavaDataset(train_split, train_images_path, train_aug)
val_dataset = CassavaDataset(val_split, train_images_path, test_aug)

train_loader = DataLoader(
    train_dataset,
    batch_size=64,
    shuffle=True,
    num_workers=4,
    pin_memory=True,
    persistent_workers=True,
)
val_loader = DataLoader(
    val_dataset,
    batch_size=64,
    shuffle=False,
    num_workers=4,
    pin_memory=True,
    persistent_workers=True,
)

criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)
scheduler = torch.optim.lr_scheduler.StepLR(optimizer, step_size=3, gamma=0.5)

num_epochs = 12  # increased from 5 to give the model more learning

model.train()
for epoch in range(1, num_epochs + 1):
    running_loss = 0.0
    for imgs, lbls in train_loader:
        imgs = imgs.to(device)
        lbls = lbls.to(device)

        optimizer.zero_grad()
        outputs = model(imgs)
        loss = criterion(outputs, lbls)
        loss.backward()
        optimizer.step()
        running_loss += loss.item() * imgs.size(0)

    epoch_loss = running_loss / len(train_loader.dataset)

    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for imgs, lbls in val_loader:
            imgs = imgs.to(device)
            lbls = lbls.to(device)
            outputs = model(imgs)
            _, preds = torch.max(outputs, 1)
            correct += (preds == lbls).sum().item()
            total += lbls.size(0)
    val_acc = correct / total if total > 0 else 0.0
    print(
        f"Epoch {epoch}/{num_epochs} - Loss: {epoch_loss:.4f} - Val Acc: {val_acc:.4f}"
    )

    model.train()
    scheduler.step()  # update learning rate

class TestDataset(Dataset):
    """Repeat each test image `tta` times; each __getitem__ returns an augmented tensor and its image_id.
    Images are pre‑loaded into memory once to avoid repeated disk I/O during test‑time augmentation.
    """

    def __init__(self, df, img_dir, augment, tta=10):
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.augment = augment
        self.tta = tta
        self.total_len = len(self.df) * self.tta

        self.preloaded_imgs = []
        for image_id in self.df["image_id"]:
            img_path = os.path.join(self.img_dir, image_id)
            with Image.open(img_path) as img:
                img = img.convert("RGB")
                self.preloaded_imgs.append(np.array(img))

    def __len__(self):
        return self.total_len

    def __getitem__(self, idx):
        base_idx = idx // self.tta
        image_id = self.df.iloc[base_idx]["image_id"]
        img_np = self.preloaded_imgs[base_idx]

        aug = self.augment(image=img_np)["image"]
        tensor = torch.from_numpy(aug.transpose(2, 0, 1)).float()
        return tensor, image_id


sample_sub = pd.read_csv(sample_sub_path)

test_dataset = TestDataset(sample_sub, test_images_path, test_aug, tta=10)
test_loader = DataLoader(
    test_dataset,
    batch_size=64,
    shuffle=False,
    num_workers=4,  # keep parallelism for augmentation
    pin_memory=True,
)

model.eval()
accum = {img_id: torch.zeros(5, device=device) for img_id in sample_sub["image_id"]}

with torch.no_grad():
    for imgs, img_ids in test_loader:
        imgs = imgs.to(device)
        outputs = model(imgs)  # shape: (batch, 5)
        for out, img_id in zip(outputs, img_ids):
            accum[img_id] += out

predictions = []
for img_id in sample_sub["image_id"]:
    avg_logits = accum[img_id] / 10.0
    _, pred_label = torch.max(avg_logits, dim=0)
    predictions.append([img_id, int(pred_label.item())])

sub_df = pd.DataFrame(predictions, columns=["image_id", "label"])
sub_df.to_csv("submission.csv", index=False)
print(sub_df.head())
