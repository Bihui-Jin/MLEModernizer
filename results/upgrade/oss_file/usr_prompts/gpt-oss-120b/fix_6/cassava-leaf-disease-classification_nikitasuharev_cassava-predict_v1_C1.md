## === cell 1
class DummyModel:
    def __init__(self, num_classes=5, seed=42):
        self.num_classes = num_classes
        random.seed(seed)
        np.random.seed(seed)

    def predict(self, data_generator, steps=None, verbose=0):
        preds = []
        count = 0
        for batch in data_generator:
            batch_size = batch.shape[0]
            batch_preds = np.random.rand(batch_size, self.num_classes)
            preds.append(batch_preds)
            count += batch_size
            if steps is not None and count >= steps * batch_size:
                break
        return np.vstack(preds)


model = DummyModel()




## === cell 2
def dummy_image_generator(image_ids, batch_size=32):
    total = len(image_ids)
    for i in range(0, total, batch_size):
        cur_batch = min(batch_size, total - i)
        dummy_batch = np.zeros((cur_batch, 512, 512, 3), dtype=np.float32)
        yield dummy_batch
