import pandas as pd

import albumentations as A
from albumentations.pytorch import ToTensorV2
from torchvision import models



## === cell 1
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")




## === cell 2
def get_tta(image_size, p=0.5):
    imagenet_stats = {"mean": [0.485, 0.456, 0.406], "std": [0.229, 0.224, 0.225]}
    augs = A.Compose(
        [
            A.Resize(600, 600),
            A.RandomCrop(image_size, image_size),
            A.HorizontalFlip(p=p),
            A.RandomBrightnessContrast(brightness_limit=0.2, contrast_limit=0.2, p=p),
            A.OneOf(
                [
                    A.MotionBlur(p=0.3),
                    A.MedianBlur(blur_limit=3, p=0.3),
                    A.Blur(blur_limit=3, p=0.3),
                ],
                p=p,
            ),
            A.OneOf(
                [
                    A.OpticalDistortion(p=0.3),
                    A.GridDistortion(p=0.3),
                    A.Affine(p=0.3),
                ],
                p=p,
            ),
            A.OneOf(
                [
                    A.GaussNoise(p=0.5),
                ],
                p=p,
            ),
            A.Normalize(
                mean=imagenet_stats["mean"],
                std=imagenet_stats["std"],
                max_pixel_value=255.0,
            ),
            ToTensorV2(),
        ]
    )
    return augs




## === cell 3
class CassavaLeafDataset(Dataset):
    def __init__(self, root_dir, transforms):
        self.root_dir = root_dir
        self.transform = transforms
        self.dataframe = pd.read_csv(
            "../input/cassava-leaf-disease-classification/sample_submission.csv"
        )

    def __len__(self):
        return len(self.dataframe)

    def __getitem__(self, idx):
        row = self.dataframe.iloc[idx]
        img_path = os.path.join(self.root_dir, row["image_id"])
        image = cv2.imread(img_path)
        if image is None:
            raise FileNotFoundError(f"Image not found: {img_path}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = self.transform(image=image)["image"]
        return image, 0




## === cell 4
test_transforms = get_tta(546)
test_data = CassavaLeafDataset(
    "../input/cassava-leaf-disease-classification/test_images", test_transforms
)
test_loader = DataLoader(test_data, batch_size=32, pin_memory=True, num_workers=0)



## === cell 5
model = models.mobilenet_v2(pretrained=True)
model.classifier[1] = nn.Linear(in_features=1280, out_features=5, bias=True)
model.to(device)
model.eval()
print(model)



## === cell 6
df = pd.read_csv("../input/cassava-leaf-disease-classification/sample_submission.csv")
print(f"Number of test samples: {df.shape[0]}")



## === cell 7
epochs = 10
res = torch.zeros((df.shape[0], 5), device=device)

with torch.no_grad():
    for _ in range(epochs):
        preds_epoch = []
        for images, _ in test_loader:
            images = images.to(device, non_blocking=True)
            outputs = model(images)
            preds_epoch.append(outputs)
        preds_epoch = torch.cat(preds_epoch, dim=0)
        res += preds_epoch

res /= epochs
pred = torch.argmax(res, dim=1)



## === cell 8
df["label"] = pred.cpu().numpy()
df.to_csv("./submission.csv", index=False)
print("Submission saved to ./submission.csv")
