## === cell 0
import os
import glob
import pandas as pd
import numpy as np
from PIL import Image

SEED = 42
np.random.seed(SEED)




## === cell 1
train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "../input/cassava-leaf-disease-classification/train_images/"
test_images_dir = "../input/cassava-leaf-disease-classification/test_images/"

train_df = pd.read_csv(train_csv_path)

train_means = []
train_labels = []

for _, row in train_df.iterrows():
    img_path = os.path.join(train_images_dir, row["image_id"])
    try:
        img = Image.open(img_path).convert("RGB")
        img_arr = np.asarray(img, dtype=np.float32)
        mean_colour = img_arr.mean(axis=(0, 1))  # shape (3,)
    except Exception:
        continue
    train_means.append(mean_colour)
    train_labels.append(int(row["label"]))  # ensure integer label

train_means = np.stack(train_means)  # shape (N_train, 3)
train_labels = np.array(train_labels)  # shape (N_train,)

test_image_paths = sorted(glob.glob(os.path.join(test_images_dir, "*.jpg")))
pred_labels = []

fallback_label = int(train_df["label"].mode()[0])

for img_path in test_image_paths:
    try:
        img = Image.open(img_path).convert("RGB")
        img_arr = np.asarray(img, dtype=np.float32)
        test_mean = img_arr.mean(axis=(0, 1))  # shape (3,)
    except Exception:
        pred_labels.append(fallback_label)
        continue

    diffs = train_means - test_mean  # (N_train, 3)
    dists = np.linalg.norm(diffs, axis=1)  # (N_train,)
    nearest_idx = np.argmin(dists)
    pred_labels.append(int(train_labels[nearest_idx]))

submission = pd.DataFrame(
    {"image_id": [os.path.basename(p) for p in test_image_paths], "label": pred_labels}
)




## === cell 2
output_path = "submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission file written to {output_path} with {len(submission)} rows.")
