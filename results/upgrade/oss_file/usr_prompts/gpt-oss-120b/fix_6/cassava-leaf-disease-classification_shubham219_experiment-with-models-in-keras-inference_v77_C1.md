IMG_SIZE = (224, 224)  # EfficientNetB0 default input size
BATCH_SIZE = 32


def _decode_and_resize(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0  # rescale
    return img


def _augment(img):
    img = tf.image.random_flip_left_right(img, seed=SEED)

    img = tfa.image.rotate(img, rot, fill_mode="reflect")
    scales = tf.random.uniform([], 0.8, 1.2, seed=SEED)
    new_sz = tf.cast(scales * tf.cast(IMG_SIZE, tf.float32), tf.int32)
    img = tf.image.resize(img, new_sz)
    img = tf.image.resize_with_crop_or_pad(img, IMG_SIZE[0], IMG_SIZE[1])
    return img


def process_train(path, label):
    img = _decode_and_resize(path)
    img = _augment(img)
    label = tf.cast(label, tf.int32)
    label = tf.one_hot(label, NUM_CLASSES)
    return img, label


def process_val(path, label):
    img = _decode_and_resize(path)
    label = tf.cast(label, tf.int32)
    label = tf.one_hot(label, NUM_CLASSES)
    return img, label


def process_test(path):
    img = _decode_and_resize(path)
