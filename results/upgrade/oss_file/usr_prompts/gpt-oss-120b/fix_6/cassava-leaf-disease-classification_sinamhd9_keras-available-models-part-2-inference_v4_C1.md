# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models
from tensorflow.keras.applications import MobileNetV2

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

IMG_SIZE = 224
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE
NUM_CLASSES = 5
SEED = 42
np.random.seed(SEED)
tf.random.set_seed(SEED)



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
train_df["filepath"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)
train_df["label"] = train_df["label"].astype(np.int32)

train_paths = train_df["filepath"].values
train_labels = train_df["label"].values

indices = np.arange(len(train_paths))
np.random.shuffle(indices)
val_split = int(0.10 * len(indices))  # 10% for validation
val_idx, train_idx = indices[:val_split], indices[val_split:]

train_paths_split = train_paths[train_idx]
train_labels_split = train_labels[train_idx]
val_paths_split = train_paths[val_idx]
val_labels_split = train_labels[val_idx]




## === cell 2
def _parse_image(filename, label=None):
    image = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [IMG_SIZE, IMG_SIZE])
    image = tf.cast(image, tf.float32) / 255.0
    if label is None:
        return image
    else:
        return image, label


options = tf.data.Options()
options.experimental_deterministic = False

train_ds = tf.data.Dataset.from_tensor_slices((train_paths_split, train_labels_split))
train_ds = train_ds.map(_parse_image, num_parallel_calls=AUTOTUNE)
train_ds = train_ds.cache("/tmp/train_cache.tfrecord")
train_ds = train_ds.shuffle(buffer_size=len(train_paths_split), seed=SEED)
train_ds = train_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
train_ds = train_ds.with_options(options)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths_split, val_labels_split))
val_ds = val_ds.map(_parse_image, num_parallel_calls=AUTOTUNE)
val_ds = val_ds.cache("/tmp/val_cache.tfrecord")
val_ds = val_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)
val_ds = val_ds.with_options(options)



## === cell 3
base_model = MobileNetV2(
    weights="imagenet", include_top=False, input_shape=(IMG_SIZE, IMG_SIZE, 3)
)
base_model.trainable = False  # freeze base

model = models.Sequential(
    [
        base_model,
        layers.GlobalAveragePooling2D(),
        layers.Dense(128, activation="relu"),
        layers.Dropout(0.2),
        layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)



## === cell 4
model.fit(train_ds, validation_data=val_ds, epochs=5, verbose=1)

base_model.trainable = True
for layer in base_model.layers[:-30]:
    layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(1e-4),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(train_ds, validation_data=val_ds, epochs=3, verbose=1)



## === cell 5
test_images = sorted(
    [f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")]
)
test_paths = [os.path.join(TEST_IMG_DIR, img) for img in test_images]

test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(lambda x: _parse_image(x), num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

df_sub = pd.DataFrame({"image_id": test_images, "label": pred_labels})
df_sub.to_csv("submission.csv", index=False)
print("Submission file written to submission.csv")
