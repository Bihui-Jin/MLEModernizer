# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, optimizers, mixed_precision
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.applications.efficientnet import EfficientNetB0, preprocess_input
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau

policy = mixed_precision.Policy("mixed_float16")
mixed_precision.set_global_policy(policy)

np.random.seed(42)
tf.random.set_seed(42)



## === cell 1
WORK_DIR = "../input/cassava-leaf-disease-classification/"

train_df = pd.read_csv(os.path.join(WORK_DIR, "train.csv"))
train_df["label"] = train_df["label"].astype(str)

datagen = ImageDataGenerator(
    preprocessing_function=preprocess_input,
    validation_split=0.1,
    horizontal_flip=True,
    rotation_range=15,
    zoom_range=0.1,
)

cpu_count = os.cpu_count() or 1

train_generator = datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=os.path.join(WORK_DIR, "train_images"),
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    batch_size=64,
    class_mode="categorical",
    subset="training",
    shuffle=True,
    workers=cpu_count,
    use_multiprocessing=True,
)

val_generator = datagen.flow_from_dataframe(
    dataframe=train_df,
    directory=os.path.join(WORK_DIR, "train_images"),
    x_col="image_id",
    y_col="label",
    target_size=(224, 224),
    batch_size=64,
    class_mode="categorical",
    subset="validation",
    shuffle=False,
    workers=cpu_count,
    use_multiprocessing=True,
)

num_classes = len(train_generator.class_indices)



## === cell 2
base_model = EfficientNetB0(
    include_top=False, input_shape=(224, 224, 3), weights="imagenet", pooling="avg"
)

base_model.trainable = False

inputs = layers.Input(shape=(224, 224, 3))
x = base_model(inputs, training=False)
outputs = layers.Dense(num_classes, activation="softmax")(x)
model = models.Model(inputs, outputs)

model.compile(
    optimizer=optimizers.Adam(learning_rate=1e-3),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)



## === cell 3
callbacks = [
    EarlyStopping(
        monitor="val_accuracy", patience=2, restore_best_weights=True, verbose=1
    ),
    ReduceLROnPlateau(monitor="val_accuracy", factor=0.5, patience=1, verbose=1),
]

model.fit(
    train_generator,
    epochs=5,
    validation_data=val_generator,
    callbacks=callbacks,
    verbose=2,
)

base_model.trainable = True
fine_tune_at = int(len(base_model.layers) * 0.3)  # fine‑tune top 30%
for layer in base_model.layers[:fine_tune_at]:
    layer.trainable = False

model.compile(
    optimizer=optimizers.Adam(learning_rate=1e-4),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(
    train_generator,
    epochs=3,
    validation_data=val_generator,
    callbacks=callbacks,
    verbose=2,
)



## === cell 4
test_dir = os.path.join(WORK_DIR, "test_images")
test_paths = [
    os.path.join(test_dir, fname)
    for fname in sorted(os.listdir(test_dir))
    if not fname.startswith(".")
]


def load_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [224, 224])
    img = preprocess_input(img)
    filename = tf.strings.split(path, os.sep)[-1]
    return img, filename


test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(load_preprocess, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(64).prefetch(tf.data.AUTOTUNE)

pred_labels = []
filenames = []

for batch_imgs, batch_names in test_ds:
    preds = model.predict(batch_imgs, verbose=0)
    batch_pred = np.argmax(preds, axis=-1).astype(int)
    pred_labels.extend(batch_pred)
    filenames.extend(batch_names.numpy())

submission = pd.DataFrame(
    {"image_id": [name.decode() for name in filenames], "label": pred_labels}
)
submission.to_csv("submission.csv", index=False)
