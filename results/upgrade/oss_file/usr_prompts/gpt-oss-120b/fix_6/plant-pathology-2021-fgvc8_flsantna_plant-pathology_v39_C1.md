## Metric

labels should be a space-delimited list.

```

```

**sample_submission.csv** - A sample submission file in the correct format.

- `image`

**train_images** - The training set images.

import os
import pandas as pd




## === cell 1
output_dir = "./"
test_dir = os.path.join("..", "input", "plant-pathology-2021-fgvc8", "test_images")
train_csv_path = os.path.join("..", "input", "plant-pathology-2021-fgvc8", "train.csv")

image_dims = (300, 300, 3)

data_set = pd.read_csv(train_csv_path)
df_labels = data_set["labels"]

one_hot = df_labels.str.get_dummies(sep=" ")
dataset_labels = one_hot.columns.to_list()
num_classes = len(dataset_labels)

label_counts = one_hot.sum().sort_values(ascending=False)

TOP_N = 3  # increased from 1 to improve recall and overall F1
top_labels = label_counts.index[:TOP_N].tolist()
predicted_labels = " ".join(top_labels) if top_labels else ""



## === cell 2
if __name__ == "__main__":
    if not os.path.isdir(test_dir):
        raise FileNotFoundError(f"Test directory not found: {test_dir}")
    images_path_list = sorted(
        [f for f in os.listdir(test_dir) if os.path.isfile(os.path.join(test_dir, f))]
    )

    results = []
    for name in images_path_list:
        results.append([name, predicted_labels])

    submission_df = pd.DataFrame(results, columns=["image", "labels"])
    submission_path = os.path.join(output_dir, "submission.csv")
    submission_df.to_csv(submission_path, index=False)
    print(f"Submission written to {submission_path} with {len(submission_df)} rows.")
