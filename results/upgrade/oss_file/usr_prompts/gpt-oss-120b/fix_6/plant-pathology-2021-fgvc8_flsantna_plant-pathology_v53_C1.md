from tensorflow.keras.applications import EfficientNetB7
from tensorflow.keras import Sequential, Model
from tensorflow.keras.layers import (
    Dense,
    GlobalAveragePooling2D,
    BatchNormalization,
    Dropout,
    InputLayer,
)


class MultiLabel(Model):
    """
    Simple multi‑label classifier based on EfficientNetB7 backbone.
    The model outputs one sigmoid unit per class.
    """

    def __init__(self, num_classes, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.backbone = EfficientNetB7(
            include_top=False, weights=None, input_shape=image_dims
        )
        self.model = Sequential(
            [
                InputLayer(input_shape=image_dims),
                self.backbone,
                GlobalAveragePooling2D(),
                BatchNormalization(),
                Dropout(0.2),
                Dense(num_classes, activation="sigmoid"),
            ]
        )

    def call(self, inputs, **kwargs):
        return self.model(inputs)

import numpy as np

if __name__ == "__main__":
    model = MultiLabel(num_classes=num_classes)
    model.build(input_shape=[None, image_dims[0], image_dims[1], image_dims[2]])

    try:
        model.load_weights(model_dir)
        print("Custom weights loaded.")
    except Exception as e:
        print(f"Could not load weights from {model_dir}: {e}")
        print("Proceeding with random initialization (no ImageNet weights).")

    images_path_list = sorted(
        [
            f
            for f in os.listdir(test_dir)
            if f.lower().endswith((".jpg", ".jpeg", ".png"))
        ]
    )

    full_paths = [os.path.join(test_dir, fname) for fname in images_path_list]

    def _process(path):
        img_raw = tf.io.read_file(path)
        img = tf.io.decode_jpeg(img_raw, channels=3)
        img = tf.image.resize(img, [image_dims[0], image_dims[1]])
        img = tf.cast(img, tf.float32)  # keep 0‑255 range
        filename = tf.io.basename(path)
        return img, filename

    dataset = (
        tf.data.Dataset.from_tensor_slices(full_paths)
        .map(_process, num_parallel_calls=tf.data.AUTOTUNE)
        .batch(32)  # batch size matches original setting
        .prefetch(tf.data.AUTOTUNE)
    )

    predictions = []

    for batch_images, batch_filenames in dataset:
        probs_batch = model(batch_images, training=False)  # (batch, num_classes)
        probs_np = probs_batch.numpy()
        present_mask = probs_np > 0.5  # bool mask

        for i in range(probs_np.shape[0]):
            present_indices = np.where(present_mask[i])[0]
            label_str = " ".join([dataset_labels[idx] for idx in present_indices])
            fname = batch_filenames[i].numpy().decode()
            predictions.append([fname, label_str])

    submission_df = pd.DataFrame(predictions, columns=["image", "labels"])
    submission_path = os.path.join(output_dir, "submission.csv")
    submission_df.to_csv(submission_path, index=False)
    print(f"Submission written to {submission_path}")
