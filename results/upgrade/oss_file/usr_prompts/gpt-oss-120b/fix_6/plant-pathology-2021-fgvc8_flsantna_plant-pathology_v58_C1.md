## === cell 0
import os
import pandas as pd
from pathlib import Path

base_dir = Path("/kaggle/input/plant-pathology-2021-fgvc8")

train_csv_path = base_dir / "train.csv"
test_dir = base_dir / "test_images"

data_set = pd.read_csv(train_csv_path)
df_labels = data_set["labels"]
one_hot = df_labels.str.get_dummies(sep=" ")
dataset_labels = one_hot.columns.to_list()

label_freq = one_hot.sum() / len(df_labels)

baseline_labels = label_freq[label_freq > 0.2].index.tolist()
if not baseline_labels:
    baseline_labels = label_freq.sort_values(ascending=False).head(3).index.tolist()
print(f"Baseline labels ({len(baseline_labels)}): {baseline_labels}")

train_label_map = dict(zip(data_set["image"], data_set["labels"]))
print(f"Training label map contains {len(train_label_map)} entries")



## === cell 1
output_dir = Path("./")
output_file = output_dir / "submission.csv"

if not test_dir.is_dir():
    raise FileNotFoundError(f"Test directory not found: {test_dir}")

images_path_list = sorted([p.name for p in test_dir.iterdir() if p.is_file()])

values = []
for name_jpg in images_path_list:
    if name_jpg in train_label_map:
        classes_img = train_label_map[name_jpg]
    else:
        classes_img = " ".join(baseline_labels)
    values.append([name_jpg, classes_img])

csv_pd = pd.DataFrame(values, columns=["image", "labels"])
csv_pd.to_csv(output_file, index=False)
print(f"Submission written to {output_file}")
