def decode_image(filename, label=None, image_size=(224, 224)):
    bits = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(bits, channels=3)
    image = tf.cast(image, tf.float32) / 255.0
    image = tf.image.resize(image, image_size)
    if label is None:
        return image
    else:
        return image, label

class FixedDropout(tf.keras.layers.Dropout):
    def _get_noise_shape(self, inputs):
        if self.noise_shape is None:
            return self.noise_shape
        symbolic_shape = K.shape(inputs)
        noise_shape = [
            symbolic_shape[axis] if shape is None else shape
            for axis, shape in enumerate(self.noise_shape)
        ]
        return tuple(noise_shape)

possible_train_paths = [

train_df = pd.read_csv(train_csv_path)

label_to_index = {lbl: idx for idx, lbl in enumerate(all_labels)}
num_classes = len(all_labels)


def label_to_vector(label_str):
    vec = np.zeros(num_classes, dtype=np.float32)
    for lbl in label_str.split():
        vec[label_to_index[lbl]] = 1.0
    return vec


train_df["label_vec"] = train_df["labels"].apply(label_to_vector)

possible_image_dirs = [
    "../input/plant-pathology-2021-fgvc8/train_images",
    "/kaggle/input/plant-pathology-2021-fgvc8/train_images",
    "input/plant-pathology-2021-fgvc8/train_images",
    "train_images",
]
train_image_dir = None
for p in possible_image_dirs:
    if os.path.isdir(p):
        train_image_dir = p
        break
if train_image_dir is None:
    raise FileNotFoundError("train_images directory not found.")

train_image_paths = np.array(
    [os.path.join(train_image_dir, fname) for fname in train_df["image"]]
)
train_label_vectors = np.stack(train_df["label_vec"].values)

val_size = int(0.1 * len(train_image_paths))

val_label_vectors = train_label_vectors[:val_size]

train_label_vectors = train_label_vectors[val_size:]

base_model = MobileNetV2(
    input_shape=(224, 224, 3), include_top=False, weights="imagenet"
)
base_model.trainable = False  # freeze

feature_extractor = Model(
    inputs=base_model.input, outputs=GlobalAveragePooling2D()(base_model.output)
)


def preprocess_image(path, label):
    img, _ = decode_image(path)
    return img, label


train_ds = (
    tf.data.Dataset.from_tensor_slices((train_image_paths, train_label_vectors))
    .map(decode_image, num_parallel_calls=AUTO)
    .map(
        lambda img, lbl: (feature_extractor(img, training=False), lbl),
        num_parallel_calls=AUTO,
    )
    .cache()
    .shuffle(10000, seed=42)
    .batch(BATCH_SIZE)
    .prefetch(AUTO)
)

val_ds = (
    tf.data.Dataset.from_tensor_slices((val_image_paths, val_label_vectors))
    .map(decode_image, num_parallel_calls=AUTO)

        lambda img, lbl: (feature_extractor(img, training=False), lbl),

    .cache()

    .prefetch(AUTO)

model = Model(inputs=head_input, outputs=output_layer)

    optimizer=optimizers.Adam(),

    metrics=["accuracy"],

model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=3,
