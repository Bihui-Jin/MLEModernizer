## === cell 0
import numpy as np, pandas as pd, os, random, torch
from pathlib import Path
from fastai.vision.all import *

seed = 42
random.seed(seed)
np.random.seed(seed)
torch.manual_seed(seed)
if torch.cuda.is_available():
    torch.cuda.manual_seed_all(seed)

torch.backends.cudnn.benchmark = True

PATH = Path("/kaggle/input/plant-pathology-2021-fgvc8")
if not PATH.exists():
    PATH = Path("data/plant-pathology-2021-fgvc8")
print(f"Using data path: {PATH}")



## === cell 1
df = pd.read_csv(PATH / "train.csv")
print("Training data shape:", df.shape)
df.head()



## === cell 2
max_workers = min(8, os.cpu_count() or 1)

norm_tfms = Normalize.from_stats(*imagenet_stats)

dls = ImageDataLoaders.from_df(
    df,
    PATH,
    folder="train_images",
    label_delim=" ",
    y_block=MultiCategoryBlock,
    item_tfms=Resize(224),
    batch_tfms=norm_tfms,
    bs=64,
    num_workers=max_workers,
    pin_memory=True,
)



## === cell 3
learn = cnn_learner(
    dls, resnet34, metrics=accuracy_multi
).to_fp16()  # mixed‑precision training

learn.freeze()
learn.fit_one_cycle(1, lr_max=1e-3)



## === cell 4
test_files = get_image_files(PATH / "test_images")
test_dl = dls.test_dl(sorted(test_files))
preds, _ = learn.get_preds(dl=test_dl)  # (num_test, num_classes)

vocab = dls.vocab
threshold = 0.5
labels = []
for prob in preds:
    idxs = (prob > threshold).nonzero(as_tuple=False).flatten()
    if len(idxs) == 0:
        idxs = [prob.argmax().item()]  # fallback to most probable class
    label_str = " ".join([vocab[i] for i in idxs])
    labels.append(label_str)

submission = pd.DataFrame(
    {"image": [p.name for p in sorted(test_files)], "labels": labels}
)

sample_sub = pd.read_csv(PATH / "sample_submission.csv")
assert len(submission) == len(sample_sub), "Row count mismatch in submission"

submission_path = Path("submission.csv")
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path} with {len(submission)} rows.")
