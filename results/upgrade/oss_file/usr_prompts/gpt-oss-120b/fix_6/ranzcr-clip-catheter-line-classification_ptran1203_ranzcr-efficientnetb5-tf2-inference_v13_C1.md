- `ETT - Borderline` - endotracheal tube placement borderline abnormal
- `ETT - Normal` - endotracheal tube placement normal
- `NGT - Abnormal` - nasogastric tube placement abnormal
- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

import os
import pandas as pd
import numpy as np
import cv2
from concurrent.futures import ThreadPoolExecutor
from sklearn.ensemble import GradientBoostingClassifier

BASE_DIR = "/kaggle/input/ranzcr-clip-catheter-line-classification"
SAMPLE_SUBMIT_PATH = os.path.join(BASE_DIR, "sample_submission.csv")
TRAIN_IMAGES_DIR = os.path.join(BASE_DIR, "train")
TEST_IMAGES_DIR = os.path.join(BASE_DIR, "test")
OUTPUT_SUBMIT_PATH = "submission.csv"

submission_df = pd.read_csv(SAMPLE_SUBMIT_PATH)
target_columns = [col for col in submission_df.columns if col != "StudyInstanceUID"]

FEATURE_DIM = 16  # 7 basic stats + 8 histogram bins + edge density


def _extract_single(uid, img_dir):
    """Read image and compute a richer set of simple image statistics."""
    img_path = os.path.join(img_dir, f"{uid}.jpg")
    img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        return np.full(FEATURE_DIM, 0.5, dtype=np.float32)

    mean_int = img.mean() / 255.0
    std_int = img.std() / 255.0
    max_int = img.max() / 255.0
    min_int = img.min() / 255.0
    median_int = np.median(img) / 255.0
    p25 = np.percentile(img, 25) / 255.0
    p75 = np.percentile(img, 75) / 255.0

    edges = cv2.Canny(img, 100, 200)
    edge_density = edges.sum() / (edges.size * 255.0)  # normalised

    hist, _ = np.histogram(img, bins=8, range=(0, 255), density=True)
    hist = hist.astype(np.float32)  # already sums to 1

    feats = np.array(
        [
            mean_int,
            std_int,
            max_int,
            min_int,
            median_int,
            p25,
            p75,
            edge_density,
            *hist,
        ],
        dtype=np.float32,
    )

    return feats


def extract_features_parallel(uids, img_dir):
    """Thread‑pool feature extraction with pre‑allocated output array."""
    n = len(uids)
    features = np.empty((n, FEATURE_DIM), dtype=np.float32)

    def worker(idx_uid):
        idx, uid = idx_uid
        features[idx] = _extract_single(uid, img_dir)

    with ThreadPoolExecutor(max_workers=os.cpu_count()) as executor:
        list(executor.map(worker, enumerate(uids)))
    return features



train_csv_path = os.path.join(BASE_DIR, "train.csv")
train_df = pd.read_csv(train_csv_path)

train_uids = train_df["StudyInstanceUID"].values
train_features = extract_features_parallel(train_uids, TRAIN_IMAGES_DIR)

models = {}
for col in target_columns:
    y = train_df[col].values
    clf = GradientBoostingClassifier(
        n_estimators=200, learning_rate=0.1, max_depth=3, random_state=42
    )
    clf.fit(train_features, y)
    models[col] = clf

test_uids = submission_df["StudyInstanceUID"].values
test_features = extract_features_parallel(test_uids, TEST_IMAGES_DIR)

for col in target_columns:
    prob = models[col].predict_proba(test_features)[:, 1]  # probability of class 1
    submission_df[col] = prob.astype(np.float32)



## === cell 1
submission_df.to_csv(OUTPUT_SUBMIT_PATH, index=False)
print(f"Submission file written to {OUTPUT_SUBMIT_PATH}")
