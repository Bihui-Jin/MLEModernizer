- `ETT - Normal` - endotracheal tube placement normal

- `NGT - Borderline` - nasogastric tube placement borderline abnormal
- `NGT - Incompletely Imaged` - nasogastric tube placement inconclusive due to imaging
- `NGT - Normal` - nasogastric tube placement borderline normal
- `CVC - Abnormal` - central venous catheter placement abnormal
- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

def get_model(
    base_cls,
    baseline_weight=None,
    init_weight=None,
    lr=0.001,
    optimizer=tf.optimizers.Adam,
):
    """Build EfficientNet backbone + dropout + sigmoid head."""
    base = base_cls(
        include_top=False,
        input_shape=(H, W, 3),
        pooling="avg",
        weights=baseline_weight,
    )  # e.g., "imagenet" or None
    x = tf.keras.layers.Dropout(0.3)(base.output)
    out = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base.input, outputs=out)

    model.compile(
        optimizer=optimizer(learning_rate=lr),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )

    if init_weight and os.path.exists(init_weight):
        try:
            model.load_weights(init_weight)
            print(f"Loaded fine‑tuned weights from {init_weight}")
        except Exception as e:
            print(f"Failed to load weights from {init_weight}: {e}")
    else:
        print("No fine‑tuned weight file found – using ImageNet weights.")
    return model




## === cell 2
def preprocess_image(image):
    """Resize, scale to [0,1] and normalize with ImageNet mean/std."""
    image = tf.image.resize(image, (H, W))
    image = tf.cast(image, tf.float32) / 255.0
    image = (image - mean) / std
    return image


def make_dataset_from_files(file_paths, labels=None, batch_size=16, shuffle=False):
    """
    Build a tf.data pipeline from JPG file paths.
    If `labels` is provided (as a NumPy array), it is paired with the images.
    """
    ds = tf.data.Dataset.from_tensor_slices(file_paths)
    ds = ds.map(lambda fp: tf.io.read_file(fp), num_parallel_calls=AUTOTUNE)
    ds = ds.map(
        lambda raw: tf.image.decode_jpeg(raw, channels=3), num_parallel_calls=AUTOTUNE
    )
    ds = ds.map(preprocess_image, num_parallel_calls=AUTOTUNE)

    if labels is not None:
        label_ds = tf.data.Dataset.from_tensor_slices(labels.astype(np.float32))
        ds = tf.data.Dataset.zip((ds, label_ds))

    if shuffle:
        ds = ds.shuffle(buffer_size=1024, reshuffle_each_iteration=True)

    ds = ds.batch(batch_size).prefetch(AUTOTUNE)
    return ds
