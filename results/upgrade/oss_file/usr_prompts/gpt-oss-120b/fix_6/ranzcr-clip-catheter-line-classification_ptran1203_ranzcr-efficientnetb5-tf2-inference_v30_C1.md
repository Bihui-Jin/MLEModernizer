- `CVC - Borderline` - central venous catheter placement borderline abnormal
- `CVC - Normal` - central venous catheter placement normal
- `Swan Ganz Catheter Present`
- `PatientID` - unique ID for each patient in the dataset

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import tensorflow as tf
import pandas as pd
import numpy as np
import glob

from tensorflow.keras import mixed_precision

mixed_precision.set_global_policy("mixed_float16")

W = H = 456
N_CLASSES = 11
AUTOTUNE = tf.data.experimental.AUTOTUNE

TARGET_COLS = [
    "CVC - Abnormal",
    "CVC - Borderline",
    "CVC - Normal",
    "ETT - Abnormal",
    "ETT - Borderline",
    "ETT - Normal",
    "NGT - Abnormal",
    "NGT - Borderline",
    "NGT - Incompletely Imaged",
    "NGT - Normal",
    "Swan Ganz Catheter Present",
]

MEAN = [0.485, 0.456, 0.406]
STD = [0.229, 0.224, 0.225]

BASE_INPUT = "/kaggle/input/ranzcr-clip-catheter-line-classification"
TRAIN_IMG_DIR = os.path.join(BASE_INPUT, "train")
TEST_IMG_DIR = os.path.join(BASE_INPUT, "test")
SAMPLE_SUBMIT_PATH = os.path.join(BASE_INPUT, "sample_submission.csv")




## === cell 1
def load_and_preprocess(path):
    """Read an image file, decode, resize, cast and normalise."""
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, (H, W))
    img = tf.cast(img, tf.float32) / 255.0
    img = (img - MEAN) / STD
    return img


def path_to_id(path):
    """Extract StudyInstanceUID from filename."""
    filename = tf.strings.split(path, os.sep)[-1]
    uid = tf.strings.regex_replace(filename, ".jpg$", "")
    return uid




## === cell 2
def build_model():
    """EfficientNetB5 backbone + dropout + sigmoid dense head."""
    base = tf.keras.applications.EfficientNetB5(
        include_top=False, input_shape=(H, W, 3), pooling="avg", weights="imagenet"
    )
    x = tf.keras.layers.Dropout(0.3)(base.output)
    out = tf.keras.layers.Dense(N_CLASSES, activation="sigmoid")(x)
    model = tf.keras.Model(inputs=base.input, outputs=out)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss="binary_crossentropy",
        metrics=[tf.keras.metrics.AUC(name="auc")],
    )
    return model




## === cell 3
train_df = pd.read_csv(os.path.join(BASE_INPUT, "train.csv"))
train_paths = [
    os.path.join(TRAIN_IMG_DIR, f"{uid}.jpg") for uid in train_df["StudyInstanceUID"]
]
train_labels = train_df[TARGET_COLS].values.astype(np.float32)

train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))


def _parse_train(path, label):
    img = load_and_preprocess(path)
    return img, label


train_ds = (
    train_ds.map(_parse_train, num_parallel_calls=AUTOTUNE)
    .shuffle(1024)
    .batch(64, drop_remainder=True)  # larger batch reduces number of steps
    .prefetch(AUTOTUNE)
)

model = build_model()
model.fit(train_ds, epochs=2, verbose=2)




## === cell 4
test_files = sorted(glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg")))
test_ds = tf.data.Dataset.from_tensor_slices(test_files)
test_ds = test_ds.map(
    lambda x: (load_and_preprocess(x), path_to_id(x)), num_parallel_calls=AUTOTUNE
)
test_ds = test_ds.cache().prefetch(AUTOTUNE)




## === cell 5
images_ds = test_ds.map(lambda img, uid: img).batch(128).prefetch(AUTOTUNE)
orig_preds = model.predict(images_ds, verbose=0)

flipped_images_ds = (
    test_ds.map(lambda img, uid: tf.image.flip_left_right(img))
    .batch(128)
    .prefetch(AUTOTUNE)
)
flip_preds = model.predict(flipped_images_ds, verbose=0)

preds_array = np.maximum(orig_preds, flip_preds)

id_list = [uid.numpy().decode("utf-8") for uid in test_ds.map(lambda img, uid: uid)]

submission = pd.DataFrame(preds_array, columns=TARGET_COLS)
submission.insert(0, "StudyInstanceUID", id_list)

submission.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv with shape:", submission.shape)
