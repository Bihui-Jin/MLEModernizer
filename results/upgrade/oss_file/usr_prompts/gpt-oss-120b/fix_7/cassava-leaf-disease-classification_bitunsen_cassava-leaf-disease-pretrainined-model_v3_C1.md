# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
os.environ["PROTOCOL_BUFFERS_PYTHON_FORCE_C_EXTENSION"] = "0"

import json
import numpy as np
import pandas as pd
from pathlib import Path
from PIL import Image  # only the needed class
import tensorflow as tf

tf.config.optimizer.set_jit(True)

gpus = tf.config.list_physical_devices("GPU")
if gpus:
    try:
        tf.config.experimental.set_memory_growth(gpus[0], True)
    except Exception:
        pass  # memory growth already configured or not supported

from tensorflow.keras import mixed_precision

mixed_precision.set_global_policy("mixed_float16")




## === cell 1
BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")




## === cell 2
with open(os.path.join(BASE_DIR, "label_num_to_disease_map.json")) as f:
    label_map = json.load(f)
print(json.dumps(label_map, indent=2))




## === cell 3
train_df = pd.read_csv(os.path.join(BASE_DIR, "train.csv"))
print(f"Training samples: {len(train_df)}")
label_list = sorted(train_df["label"].unique())
print("Labels:", label_list)




## === cell 4
IMG_SIZE = 224  # MobileNetV2 default input size
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE
EPOCHS = 6  # increased epochs for better learning
VALID_SPLIT = 0.2
SEED = 42




## === cell 5
def decode_image(filename, label):
    image_string = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(image_string, channels=3)
    image = tf.image.resize(image, [IMG_SIZE, IMG_SIZE])
    image = tf.image.convert_image_dtype(image, tf.float32)
    return image, label


train_filenames = [os.path.join(TRAIN_DIR, img_id) for img_id in train_df["image_id"]]
train_labels = train_df["label"].values

full_dataset = tf.data.Dataset.from_tensor_slices((train_filenames, train_labels))
full_dataset = full_dataset.map(decode_image, num_parallel_calls=AUTOTUNE)
full_dataset = full_dataset.shuffle(
    buffer_size=len(train_filenames), seed=SEED, reshuffle_each_iteration=False
)
cache_path = "/tmp/train_cache"
full_dataset = full_dataset.cache(cache_path)

val_size = int(VALID_SPLIT * len(train_filenames))
val_dataset = full_dataset.take(val_size)
train_dataset = full_dataset.skip(val_size)

train_dataset = train_dataset.batch(BATCH_SIZE).prefetch(AUTOTUNE)
val_dataset = val_dataset.batch(BATCH_SIZE).prefetch(AUTOTUNE)




## === cell 6
base_model = tf.keras.applications.MobileNetV2(
    input_shape=(IMG_SIZE, IMG_SIZE, 3), include_top=False, weights="imagenet"
)
base_model.trainable = True

global_average_layer = tf.keras.layers.GlobalAveragePooling2D()
prediction_layer = tf.keras.layers.Dense(5, activation="softmax")  # 5 classes

model = tf.keras.Sequential(
    [base_model, global_average_layer, tf.keras.layers.Dropout(0.2), prediction_layer]
)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()




## === cell 7
history = model.fit(train_dataset, epochs=EPOCHS, validation_data=val_dataset)




## === cell 8
test_filenames = [
    os.path.join(TEST_DIR, fname)
    for fname in os.listdir(TEST_DIR)
    if fname.lower().endswith(".jpg")
]
test_ids = [os.path.basename(p) for p in test_filenames]

test_ds = tf.data.Dataset.from_tensor_slices(test_filenames)


def load_test_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    img = tf.image.convert_image_dtype(img, tf.float32)  # fast scaling to [0,1]
    return img


test_ds = test_ds.map(load_test_image, num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE)

pred_probs = model.predict(test_ds)
pred_labels = np.argmax(pred_probs, axis=1)

submission_df = pd.DataFrame({"image_id": test_ids, "label": pred_labels})
submission_path = "submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}, shape: {submission_df.shape}")




## === cell 9
print(submission_df.head())
