# 5. Code solution

## === cell 0
import os
import glob
import pandas as pd
import numpy as np

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
import tensorflow as tf
import tensorflow.keras.layers as tfl
from tensorflow.keras.models import Sequential, Model
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from sklearn.model_selection import train_test_split

SEED = 42
tf.random.set_seed(SEED)
np.random.seed(SEED)




## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["label"] = df_train["label"].astype(str)
df_train["path"] = df_train["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))
df_train = df_train[df_train["path"].apply(os.path.isfile)].reset_index(drop=True)




## === cell 2
train_df, val_df = train_test_split(
    df_train,
    test_size=0.1,
    stratify=df_train["label"],
    random_state=SEED,
)

preprocess = tf.keras.applications.resnet50.preprocess_input
IMG_SIZE = (256, 256)
BATCH_SIZE = 64  # larger batch reduces per‑epoch overhead

train_idg = ImageDataGenerator(
    preprocessing_function=preprocess, horizontal_flip=True, rotation_range=15
)
val_idg = ImageDataGenerator(preprocessing_function=preprocess)

NUM_WORKERS = min(4, os.cpu_count() or 1)

train_gen = train_idg.flow_from_dataframe(
    dataframe=train_df,
    x_col="path",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="categorical",
    seed=SEED,
    shuffle=True,
    workers=NUM_WORKERS,
    use_multiprocessing=False,  # use threads instead of processes
)

val_gen = val_idg.flow_from_dataframe(
    dataframe=val_df,
    x_col="path",
    y_col="label",
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode="categorical",
    seed=SEED,
    shuffle=False,
    workers=NUM_WORKERS,
    use_multiprocessing=False,  # use threads instead of processes
)




## === cell 3
try:
    weight_path = "/kaggle/input/resnet50-trasnfer-learning-on-tpu/cassava_base.h5"
    my_model = tf.keras.models.load_model(weight_path)
except Exception:
    base = tf.keras.applications.ResNet50(
        include_top=False,
        weights="imagenet",
        input_shape=IMG_SIZE + (3,),
        pooling="avg",
    )
    base.trainable = False

    x = base.output
    x = tfl.Dense(256, activation="relu")(x)
    outputs = tfl.Dense(5, activation="softmax")(x)
    my_model = Model(inputs=base.input, outputs=outputs)
    my_model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
    )

my_model.fit(train_gen, epochs=5, validation_data=val_gen, verbose=1)




## === cell 4
test_images = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
df_test = pd.DataFrame(test_images, columns=["path"])


def make_test_gen(batch_size=64):
    test_idg = ImageDataGenerator(preprocessing_function=preprocess)
    return test_idg.flow_from_dataframe(
        dataframe=df_test,
        x_col="path",
        y_col=None,
        batch_size=batch_size,
        seed=SEED,
        shuffle=False,
        class_mode=None,
        target_size=IMG_SIZE,
        workers=NUM_WORKERS,
        use_multiprocessing=False,  # thread‑based loading
    )


test_gen = make_test_gen(batch_size=64)
pred_test = my_model.predict(test_gen, verbose=1)
pred_test_labels = np.argmax(pred_test, axis=-1)

final_submission = pd.DataFrame(
    {
        "image_id": df_test["path"].apply(lambda x: os.path.basename(x)),
        "label": pred_test_labels,
    }
)

final_csv_path = "submission.csv"
final_submission.to_csv(final_csv_path, index=False)
print(f"Submission written to {final_csv_path}")
