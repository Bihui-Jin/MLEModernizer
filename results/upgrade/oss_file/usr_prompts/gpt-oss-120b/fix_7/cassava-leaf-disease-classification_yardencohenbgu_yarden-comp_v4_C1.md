# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, models, optimizers
from tensorflow.keras.preprocessing.image import ImageDataGenerator

base_dir = "/kaggle/input/cassava-leaf-disease-classification"
if not os.path.isdir(base_dir):
    base_dir = "/kaggle/working"

train_dir = os.path.join(base_dir, "train_images")
test_dir = os.path.join(base_dir, "test_images")
train_csv_path = os.path.join(base_dir, "train.csv")
sample_submission_path = os.path.join(base_dir, "sample_submission.csv")

labels = pd.read_csv(train_csv_path)
sample_sub = pd.read_csv(sample_submission_path)

labels["label"] = labels["label"].astype(str)

batch_size = 32
img_height = 256
img_width = 256
num_classes = 5



## === cell 1
train_datagen = ImageDataGenerator(
    rescale=1.0 / 255,
    rotation_range=40,
    width_shift_range=0.2,
    height_shift_range=0.2,
    shear_range=0.2,
    zoom_range=0.2,
    horizontal_flip=True,
    validation_split=0.2,
)

val_datagen = ImageDataGenerator(rescale=1.0 / 255, validation_split=0.2)

train_ds = train_datagen.flow_from_dataframe(
    dataframe=labels,
    directory=train_dir,
    x_col="image_id",
    y_col="label",
    target_size=(img_height, img_width),
    batch_size=batch_size,
    class_mode="categorical",
    subset="training",
    shuffle=True,
    seed=42,
)

val_ds = val_datagen.flow_from_dataframe(
    dataframe=labels,
    directory=train_dir,
    x_col="image_id",
    y_col="label",
    target_size=(img_height, img_width),
    batch_size=batch_size,
    class_mode="categorical",
    subset="validation",
    shuffle=False,
    seed=42,
)



## === cell 2
model = models.Sequential(
    [
        layers.Conv2D(
            16, (3, 3), activation="relu", input_shape=(img_height, img_width, 3)
        ),
        layers.MaxPooling2D(2, 2),
        layers.Conv2D(32, (3, 3), activation="relu"),
        layers.MaxPooling2D(2, 2),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D(2, 2),
        layers.Dropout(0.2),
        layers.Flatten(),
        layers.Dense(128, activation="relu"),
        layers.Dense(num_classes, activation="softmax"),
    ]
)

model.compile(
    optimizer=optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()



## === cell 3
gpus = tf.config.list_physical_devices("GPU")
if gpus:
    try:
        for gpu in gpus:
            tf.config.experimental.set_memory_growth(gpu, True)
    except Exception:
        pass
print("GPUs available:", len(gpus))



## === cell 4
epochs = 10  # a modest increase to improve accuracy without changing core logic
model.fit(train_ds, validation_data=val_ds, epochs=epochs)



## === cell 5
test_datagen = ImageDataGenerator(rescale=1.0 / 255)

test_ds = test_datagen.flow_from_dataframe(
    dataframe=sample_sub,
    directory=test_dir,
    x_col="image_id",
    y_col=None,
    target_size=(img_height, img_width),
    batch_size=batch_size,
    class_mode=None,
    shuffle=False,
)

pred_probs = model.predict(test_ds, verbose=1)
pred_labels = pred_probs.argmax(axis=1)

submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": pred_labels})
submission_path = os.path.join(base_dir, "submission.csv")
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
