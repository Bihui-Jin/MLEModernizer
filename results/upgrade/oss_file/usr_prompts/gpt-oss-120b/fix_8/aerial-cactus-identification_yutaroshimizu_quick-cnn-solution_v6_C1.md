import os


def find_deepest_image_dir(root):
    """
    Recursively search for the deepest sub‑directory that contains image files.
    Returns the deepest directory with at least one .jpg file; if none found,
    returns the original root.
    """
    deepest_path = root
    max_depth = -1
    for cur_root, dirs, files in os.walk(root):
        if any(f.lower().endswith(".jpg") for f in files):
            depth = cur_root.count(os.sep)
            if depth > max_depth:
                deepest_path = cur_root
                max_depth = depth
    return deepest_path

from PIL import Image
import os
from torch.utils.data import Dataset
from torchvision import transforms
from torch.utils.data import DataLoader


class CustomDataset(Dataset):
    def __init__(self, path, df, transform=None):
        self.path = path  # directory containing the images
        self.df = df
        self.transform = transform

    def __len__(self):
        return len(self.df)

    def _find_image_path(self, img_id):
        img_path = os.path.join(self.path, img_id)
        if os.path.isfile(img_path):
            return img_path
        for root, _, files in os.walk(self.path):
            if img_id in files:
                return os.path.join(root, img_id)
        raise FileNotFoundError(f"Image file {img_id} not found in {self.path}")

    def __getitem__(self, i):
        img_id = self.df.iloc[i, 0]
        img_path = self._find_image_path(img_id)
        img = Image.open(img_path).convert("RGB")
        if self.df.shape[1] > 1:
            label = self.df.iloc[i, 1]
        else:
            label = -1  # dummy label for test set
        if self.transform:
            img = self.transform(img)
        return img, label
