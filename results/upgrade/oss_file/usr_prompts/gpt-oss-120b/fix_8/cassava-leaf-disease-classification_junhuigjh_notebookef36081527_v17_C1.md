train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(str)

label_to_index = {str(i): i for i in range(5)}
train_df["label_idx"] = train_df["label"].map(label_to_index)

from sklearn.model_selection import train_test_split

train_split, val_split = train_test_split(
    train_df,
    test_size=0.1,
    random_state=SEED,
    stratify=train_df["label_idx"],
)

base_resnet = ResNet50(weights="imagenet", include_top=False, input_shape=(224, 224, 3))
base_resnet.trainable = False
gap = GlobalAveragePooling2D()(base_resnet.output)
feature_extractor = Model(
    inputs=base_resnet.input, outputs=gap, name="feature_extractor"
)


def build_feature_dataset(df, augment=False, cache_path=None):
    file_paths = (
        df["image_id"].apply(lambda fname: os.path.join(TRAIN_IMG_DIR, fname)).values
    )
    labels = df["label_idx"].values.astype(np.int32)

    ds = tf.data.Dataset.from_tensor_slices((file_paths, labels))

    def _load_and_preprocess(path, label):
        img = tf.io.read_file(path)
        img = tf.image.decode_jpeg(img, channels=3)
        img = tf.image.resize(img, [224, 224])
        img = preprocess_input(img)
        return img, label

    ds = ds.map(_load_and_preprocess, num_parallel_calls=tf.data.AUTOTUNE)

    if augment:
        try:
            import tensorflow_addons as tfa

            def _augment(img, label):
                img = tf.image.random_flip_left_right(img, seed=SEED)
                angle_rad = tf.random.uniform([], -20, 20, seed=SEED) * (
                    3.14159265 / 180.0
                )
                img = tfa.image.rotate(img, angle_rad, fill_mode="reflect")
                scales = tf.random.uniform([], 0.8, 1.2, seed=SEED)
                new_size = tf.cast(scales * 224, tf.int32)
                img = tf.image.resize(img, [new_size, new_size])
                img = tf.image.resize_with_crop_or_pad(img, 224, 224)
                return img, label

        except ImportError:

            def _augment(img, label):
                img = tf.image.random_flip_left_right(img, seed=SEED)
                return img, label

        ds = ds.map(_augment, num_parallel_calls=tf.data.AUTOTUNE)

    def _extract_features(img, label):
        feats = feature_extractor(img, training=False)  # shape (2048,)
        label_onehot = tf.one_hot(label, depth=5)
        return feats, label_onehot

    ds = ds.map(_extract_features, num_parallel_calls=tf.data.AUTOTUNE)

    ds = ds.cache(cache_path) if cache_path else ds.cache()
    ds = ds.shuffle(buffer_size=len(df), seed=SEED, reshuffle_each_iteration=False)
    ds = ds.batch(32).prefetch(tf.data.AUTOTUNE)
    return ds


train_feature_dataset = build_feature_dataset(
    train_split, augment=True, cache_path="/tmp/train_feat_cache"
)
val_feature_dataset = build_feature_dataset(
    val_split, augment=False, cache_path="/tmp/val_feat_cache"
)

classifier = Sequential(
    [
        Input(shape=feature_extractor.output_shape[1:]),  # (2048,)
        Dense(5, activation="softmax"),
    ],
    name="classifier",
)

classifier.compile(
    optimizer=Adam(learning_rate=1e-4),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

classifier.fit(
    train_feature_dataset,
    epochs=5,
    validation_data=val_feature_dataset,
    verbose=1,
)
