base_path = "/kaggle/input/cassava-leaf-disease-classification"
train_csv_path = os.path.join(base_path, "train.csv")
train_img_dir = os.path.join(base_path, "train_images")
test_img_dir = os.path.join(base_path, "test_images")

train_df = pd.read_csv(train_csv_path)
train_df["label"] = train_df["label"].astype(str)  # keep original string type

train_paths = (
    train_df["image_id"].apply(lambda x: os.path.join(train_img_dir, x)).values
)
train_labels_int = train_df["label"].astype(int).values  # 0‑4

IMG_SIZE = (299, 299)
BATCH_SIZE = 64  # larger batch → fewer steps per epoch
NUM_CLASSES = 5


def _decode_and_preprocess(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = tf.cast(img, tf.float32) / 255.0
    return img


base_model = tf.keras.applications.Xception(
    weights="imagenet",
    include_top=False,
    input_shape=(299, 299, 3),
    pooling="avg",
)
base_model.trainable = False
FEATURE_DIM = base_model.output_shape[-1]  # 2048


def _load_train_feature(path, label):
    img = _decode_and_preprocess(path)
    img = tf.keras.applications.xception.preprocess_input(img)
    feat = base_model(img, training=False)  # (FEATURE_DIM,)
    label_onehot = tf.one_hot(label, NUM_CLASSES)
    return feat, label_onehot


full_dataset = tf.data.Dataset.from_tensor_slices((train_paths, train_labels_int))
full_dataset = full_dataset.map(
    _load_train_feature, num_parallel_calls=tf.data.AUTOTUNE
).cache()  # cache the computed features in memory

N = len(train_paths)
val_size = int(0.1 * N)

shuffled = full_dataset.shuffle(
    buffer_size=1024, seed=42, reshuffle_each_iteration=False
)

val_dataset = shuffled.take(val_size).batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
train_dataset = shuffled.skip(val_size).batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
