# 5. Code solution

## === cell 0
import os
import numpy as np
import pandas as pd
from pathlib import Path
import tensorflow as tf
from tensorflow.keras.applications import EfficientNetB4
from tensorflow.keras.layers import GlobalAveragePooling2D, Dense, Input, Rescaling
from tensorflow.keras.models import Model
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau
from tensorflow.keras.utils import to_categorical

np.random.seed(42)
tf.random.set_seed(42)

tf.config.optimizer.set_jit(True)

if tf.config.list_physical_devices("GPU"):
    from tensorflow.keras.mixed_precision import Policy, set_global_policy

    set_global_policy(Policy("mixed_float16"))



## === cell 1
DATA_ROOT = Path("../input/cassava-leaf-disease-classification")
TRAIN_IMG_DIR = DATA_ROOT / "train_images"
TEST_IMG_DIR = DATA_ROOT / "test_images"
TRAIN_CSV = DATA_ROOT / "train.csv"

df = pd.read_csv(TRAIN_CSV)
df = df.sample(frac=1, random_state=42).reset_index(drop=True)
df["label"] = df["label"].astype(str)

val_frac = 0.2
val_split = int(len(df) * (1 - val_frac))
train_df = df.iloc[:val_split]
val_df = df.iloc[val_split:]

IMG_SIZE = 512
BATCH_SIZE = 32
AUTOTUNE = tf.data.AUTOTUNE
NUM_CLASSES = 5


def _load_image(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_SIZE, IMG_SIZE])
    return img


def _process_train(path, label):
    img = _load_image(path)
    img = tf.cast(img, tf.float32) / 255.0
    return img, label


def _process_val(path, label):
    img = _load_image(path)
    img = tf.cast(img, tf.float32) / 255.0
    return img, label


def _process_test(path):
    img = _load_image(path)
    img = tf.cast(img, tf.float32) / 255.0
    return img


train_paths = [str(TRAIN_IMG_DIR / fname) for fname in train_df["image_id"]]
train_labels = to_categorical(train_df["label"].astype(int).values, NUM_CLASSES)

val_paths = [str(TRAIN_IMG_DIR / fname) for fname in val_df["image_id"]]
val_labels = to_categorical(val_df["label"].astype(int).values, NUM_CLASSES)

train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(_process_train, num_parallel_calls=AUTOTUNE)
train_ds = (
    train_ds.shuffle(buffer_size=1024, seed=42).batch(BATCH_SIZE).prefetch(AUTOTUNE)
)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
val_ds = val_ds.map(_process_val, num_parallel_calls=AUTOTUNE)
val_ds = val_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)

data_augmentation = tf.keras.Sequential(
    [
        tf.keras.layers.RandomFlip("horizontal"),
        tf.keras.layers.RandomRotation(0.1),
        tf.keras.layers.RandomZoom(0.1),
        tf.keras.layers.RandomTranslation(0.1, 0.1),
    ]
)

inputs = Input(shape=(IMG_SIZE, IMG_SIZE, 3))
x = data_augmentation(inputs)  # same augmentations as before
x = Rescaling(1.0)(x)  # images already scaled, keep layer for graph clarity
base_model = EfficientNetB4(weights="imagenet", include_top=False, input_tensor=x)
x = GlobalAveragePooling2D()(base_model.output)
outputs = Dense(NUM_CLASSES, activation="softmax")(x)
model = Model(inputs=inputs, outputs=outputs)

for layer in base_model.layers:
    layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

early_stop = EarlyStopping(
    monitor="val_accuracy", patience=3, restore_best_weights=True, verbose=1
)
reduce_lr = ReduceLROnPlateau(monitor="val_accuracy", factor=0.5, patience=2, verbose=1)

model.fit(
    train_ds,
    epochs=10,
    validation_data=val_ds,
    callbacks=[early_stop, reduce_lr],
    verbose=2,
)

for layer in base_model.layers[-30:]:
    layer.trainable = True

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(
    train_ds,
    epochs=5,
    validation_data=val_ds,
    callbacks=[early_stop, reduce_lr],
    verbose=2,
)



## === cell 2
test_images = sorted(
    [
        f
        for f in os.listdir(TEST_IMG_DIR)
        if f.lower().endswith((".png", ".jpg", ".jpeg"))
    ]
)

test_paths = [str(TEST_IMG_DIR / fname) for fname in test_images]
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(_process_test, num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=1)

submission = pd.DataFrame({"image_id": test_images, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)

print(f"Submission file written to {submission_path}.")
