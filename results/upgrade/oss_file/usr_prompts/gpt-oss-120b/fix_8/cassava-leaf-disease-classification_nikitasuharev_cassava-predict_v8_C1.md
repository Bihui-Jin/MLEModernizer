## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import numpy as np
import pandas as pd
import tensorflow as tf
from sklearn.model_selection import train_test_split

tf.random.set_seed(42)
np.random.seed(42)

print("Tensorflow version " + tf.__version__)



## === cell 1
from tensorflow.keras import backend as K


class FixedDropout(tf.keras.layers.Dropout):
    def _get_noise_shape(self, inputs):
        if self.noise_shape is None:
            return self.noise_shape
        symbolic_shape = K.shape(inputs)
        noise_shape = [
            symbolic_shape[axis] if shape is None else shape
            for axis, shape in enumerate(self.noise_shape)
        ]
        return tuple(noise_shape)




## === cell 2
def build_model():
    base = tf.keras.applications.MobileNetV2(
        input_shape=(224, 224, 3), include_top=False, weights="imagenet"
    )
    base.trainable = False

    inputs = tf.keras.Input(shape=(224, 224, 3))
    x = tf.keras.applications.mobilenet_v2.preprocess_input(inputs)
    x = base(x, training=False)
    x = tf.keras.layers.GlobalAveragePooling2D()(x)
    x = tf.keras.layers.Dropout(0.2)(x)
    outputs = tf.keras.layers.Dense(5, activation="softmax")(x)

    model = tf.keras.Model(inputs, outputs)
    model.compile(
        optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
    )
    return model




## === cell 3
import tempfile

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUBMIT = os.path.join(BASE_PATH, "sample_submission.csv")

train_df = pd.read_csv(TRAIN_CSV)
train_df["full_path"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)

train_meta, val_meta = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=42
)


def _load_image(path, label):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [224, 224])
    img = tf.cast(img, tf.float32)  # keep original 0‑255 range for preprocess_input
    return img, label


def _load_test_image(image_id):
    path = tf.strings.join([TEST_IMG_DIR, image_id], separator="/")
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [224, 224])
    img = tf.cast(img, tf.float32)  # keep original range
    return img


BATCH_SIZE = 32

train_cache_path = os.path.join(tempfile.gettempdir(), "train_cache.tfdata")
val_cache_path = os.path.join(tempfile.gettempdir(), "val_cache.tfdata")

train_ds = tf.data.Dataset.from_tensor_slices(
    (train_meta["full_path"].values, train_meta["label"].values)
)
train_ds = train_ds.map(
    _load_image, num_parallel_calls=tf.data.AUTOTUNE, deterministic=False
)
train_ds = train_ds.shuffle(1024)
train_ds = train_ds.batch(BATCH_SIZE)
train_ds = train_ds.cache(train_cache_path)  # cache batched tensors to disk
train_ds = train_ds.prefetch(tf.data.AUTOTUNE)

val_ds = tf.data.Dataset.from_tensor_slices(
    (val_meta["full_path"].values, val_meta["label"].values)
)
val_ds = val_ds.map(_load_image, num_parallel_calls=tf.data.AUTOTUNE)
val_ds = val_ds.batch(BATCH_SIZE)
val_ds = val_ds.cache(val_cache_path)  # cache batched tensors to disk
val_ds = val_ds.prefetch(tf.data.AUTOTUNE)



## === cell 4
tf.keras.backend.clear_session()  # clean any previous graph state
model = build_model()
model.fit(train_ds, validation_data=val_ds, epochs=6, verbose=1)



## === cell 5
submission = pd.read_csv(SAMPLE_SUBMIT)

test_ds = tf.data.Dataset.from_tensor_slices(submission["image_id"].values)
test_ds = test_ds.map(_load_test_image, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE)

pred_probs = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(pred_probs, axis=-1)

submission["label"] = pred_labels
submission.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv")
