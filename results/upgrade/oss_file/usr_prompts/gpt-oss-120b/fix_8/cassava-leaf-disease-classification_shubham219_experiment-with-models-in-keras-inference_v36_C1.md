# 5. Code solution

## === cell 0
import os
import glob
import pandas as pd
import numpy as np
from PIL import Image
from collections import Counter

SEED = 42
np.random.seed(SEED)




## === cell 1
import concurrent.futures

train_csv_path = "../input/cassava-leaf-disease-classification/train.csv"
train_images_dir = "../input/cassava-leaf-disease-classification/train_images/"
test_images_dir = "../input/cassava-leaf-disease-classification/test_images/"

train_df = pd.read_csv(train_csv_path)


def _extract_feat(img_path):
    """Return 6‑dim colour feature (mean+std) or None on error."""
    try:
        img = Image.open(img_path).convert("RGB")
        arr = np.asarray(img, dtype=np.float32)
        mean = arr.mean(axis=(0, 1))
        std = arr.std(axis=(0, 1))
        return np.concatenate([mean, std])
    except Exception:
        return None


train_paths = [
    os.path.join(train_images_dir, img_id) for img_id in train_df["image_id"]
]
train_labels_all = train_df["label"].astype(int).values

with concurrent.futures.ProcessPoolExecutor() as exe:
    train_feats_raw = list(exe.map(_extract_feat, train_paths, chunksize=32))

train_feats = []
train_labels = []
for ft, lb in zip(train_feats_raw, train_labels_all):
    if ft is not None:
        train_feats.append(ft)
        train_labels.append(lb)

train_feats = np.stack(train_feats)  # (N_train, 6)
train_labels = np.array(train_labels)  # (N_train,)

train_norms = np.einsum("ij,ij->i", train_feats, train_feats)  # shape (N_train,)

test_image_paths = sorted(glob.glob(os.path.join(test_images_dir, "*.jpg")))

with concurrent.futures.ProcessPoolExecutor() as exe:
    test_feats_raw = list(exe.map(_extract_feat, test_image_paths, chunksize=32))

fallback_label = int(train_df["label"].mode()[0])
k = 5  # number of neighbours for majority vote

pred_labels = []

for img_path, test_feat in zip(test_image_paths, test_feats_raw):
    if test_feat is None:
        pred_labels.append(fallback_label)
        continue

    test_norm = np.dot(test_feat, test_feat)
    dists = np.sqrt(
        np.maximum(train_norms - 2 * train_feats.dot(test_feat) + test_norm, 0.0)
    )

    if k < len(dists):
        knn_idx = np.argpartition(dists, k)[:k]
    else:
        knn_idx = np.arange(len(dists))

    knn_labels = train_labels[knn_idx]
    vote = Counter(knn_labels).most_common(1)[0][0]
    pred_labels.append(int(vote))

submission = pd.DataFrame(
    {"image_id": [os.path.basename(p) for p in test_image_paths], "label": pred_labels}
)




## === cell 2
output_path = "submission.csv"
submission.to_csv(output_path, index=False)
print(f"Submission file written to {output_path} with {len(submission)} rows.")
