**train_images** - The training set images.

## === cell 0
import os
import numpy as np
import pandas as pd



## === cell 1
train_path_candidates = [
    "../input/plant-pathology-2021-fgvc8/train.csv",
    "./input/plant-pathology-2021-fgvc8/train.csv",
    "./train.csv",
]

train_path = None
for p in train_path_candidates:
    if os.path.isfile(p):
        train_path = p
        break

if train_path is None:
    raise FileNotFoundError("train.csv not found in any expected location.")

train = pd.read_csv(train_path)
print("Loaded train rows:", len(train))
print(train.labels.value_counts().head())



## === cell 2
TEST_FOLDER = "../input/plant-pathology-2021-fgvc8/test_images/"
if not os.path.isdir(TEST_FOLDER):
    TEST_FOLDER = "./input/plant-pathology-2021-fgvc8/test_images/"

test_images = sorted(os.listdir(TEST_FOLDER))
sub = pd.DataFrame(test_images, columns=["image"])



## === cell 3
label_counts = {}
for lbls in train["labels"]:
    for lbl in lbls.split():
        label_counts[lbl] = label_counts.get(lbl, 0) + 1

most_common_label = max(label_counts, key=label_counts.get)
print(
    "Most common label:", most_common_label, "count:", label_counts[most_common_label]
)

sub["labels"] = most_common_label

submission_path = "submission.csv"
sub.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}, shape: {sub.shape}")
