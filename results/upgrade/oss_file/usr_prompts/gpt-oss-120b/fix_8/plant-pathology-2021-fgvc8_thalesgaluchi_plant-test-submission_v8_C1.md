**sample_submission.csv** - A sample submission file in the correct format.

- `image`

**train_images** - The training set images.

**test_images** - The test set images. This competition has a hidden test set: only three images are provided here as samples while the remaining 5,000 images will be available to your notebook once it is submitted.

## === cell 0
import os
import pandas as pd
import numpy as np
from collections import Counter




## === cell 1
train_path = "/kaggle/input/plant-pathology-2021-fgvc8/train.csv"
train_df = pd.read_csv(train_path)

label_counter = Counter()
for lbls in train_df["labels"]:
    for lbl in lbls.split():
        label_counter[lbl] += 1

labels_list = list(label_counter.keys())
freqs = np.array([label_counter[lbl] for lbl in labels_list], dtype=float)
probs = freqs / freqs.sum()

temperature = 5.0
adjusted_probs = probs ** (1.0 / temperature)
adjusted_probs = adjusted_probs / adjusted_probs.sum()

np.random.seed(42)




## === cell 2
sample_submission_path = (
    "/kaggle/input/plant-pathology-2021-fgvc8/sample_submission.csv"
)
sample_sub = pd.read_csv(sample_submission_path)
test_filenames = sample_sub["image"].tolist()
print(f"Number of test images according to sample submission: {len(test_filenames)}")




## === cell 3
submission_rows = [
    (img_name, np.random.choice(labels_list, p=adjusted_probs))
    for img_name in test_filenames
]




## === cell 4
submission_df = pd.DataFrame(submission_rows, columns=["image", "labels"])
submission_path = "./submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission file written to {submission_path}")
