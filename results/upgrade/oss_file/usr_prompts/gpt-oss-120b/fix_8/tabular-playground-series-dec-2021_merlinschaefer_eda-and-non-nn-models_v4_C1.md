test_path = "/kaggle/input/tabular-playground-series-dec-2021/test.csv"

df_train = pd.read_csv(train_path, low_memory=False)
df_test = pd.read_csv(test_path, low_memory=False)




## === cell 3
def reduce_mem_usage(df, verbose=True):
    """
    Fast identity version: skips downcasting to avoid expensive per‑column scans.
    The data values remain unchanged, preserving model behavior.
    """
    return df




## === cell 4
df_train = reduce_mem_usage(df_train)
df_test = reduce_mem_usage(df_test)




## === cell 5
def prep_test_data(df):
    """Prepare test data by dropping unused columns."""
    return df.drop(["Id", "Soil_Type7", "Soil_Type15"], axis=1)




## === cell 6
df_train = df_train.drop(["Id", "Soil_Type7", "Soil_Type15"], axis=1)



## === cell 7
features_train = df_train.drop("Cover_Type", axis=1).values  # (n_samples, n_features)
target_train = df_train["Cover_Type"].values  # (n_samples,)



## === cell 8
del df_train
gc.collect()



## === cell 9
rf_baseline = RandomForestClassifier(
    n_estimators=200, random_state=13, n_jobs=-1, max_features="auto"
)
