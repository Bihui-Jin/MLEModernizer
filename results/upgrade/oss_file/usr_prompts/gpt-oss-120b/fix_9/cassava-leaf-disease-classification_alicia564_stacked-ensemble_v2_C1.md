import os

tf.config.threading.set_intra_op_parallelism_threads(4)

gpus = tf.config.list_physical_devices("GPU")
if gpus:
    try:
        for gpu in gpus:
            tf.config.experimental.set_memory_growth(gpu, True)
    except Exception:
        pass
tf.keras.mixed_precision.set_global_policy("mixed_float16")

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_IMG_PATH = os.path.join(BASE_PATH, "train_images")
TEST_IMG_PATH = os.path.join(BASE_PATH, "test_images")
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")

train_df = pd.read_csv(TRAIN_CSV)
le = LabelEncoder()
train_df["label_enc"] = le.fit_transform(train_df["label"])

train_split, valid_split = train_test_split(
    train_df, test_size=0.2, stratify=train_df["label_enc"], random_state=42
)

BATCH_SIZE = 256


def make_dataset(df, shuffle=False, batch_size=BATCH_SIZE, cache_file=None):
    """Create an efficient tf.data pipeline.
    - Uses a file‑based cache when `cache_file` is provided to avoid RAM overflow.
    - Keeps all preprocessing identical to the original logic.
    """
    file_paths = df["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_PATH, x)).values
    labels = df["label_enc"].values
    ds = tf.data.Dataset.from_tensor_slices((file_paths, labels))

    def _process(path, label):
        img = tf.io.read_file(path)
        img = tf.image.decode_jpeg(img, channels=3)
        img = tf.image.resize(img, [224, 224])
        img = preprocess_input(img)
        label_onehot = tf.one_hot(label, depth=5)
        return img, label_onehot

    ds = ds.map(_process, num_parallel_calls=tf.data.AUTOTUNE)
    ds = ds.cache(cache_file)  # file‑based cache if path given
    if shuffle:
        ds = ds.shuffle(buffer_size=5000, seed=42)
    ds = ds.batch(batch_size, drop_remainder=False)
    ds = ds.prefetch(tf.data.AUTOTUNE)
    return ds


train_cache_path = "/tmp/train_cache.tfdata"
train_dataset = make_dataset(train_split, shuffle=True, cache_file=train_cache_path)
valid_dataset = make_dataset(valid_split, shuffle=False)  # in‑memory cache is fine

def make_test_dataset(file_list, batch_size=BATCH_SIZE):
    """Build a batched test dataset (no labels) with the same optimizations as training."""
    ds = tf.data.Dataset.from_tensor_slices(file_list)

    def _process(path):
        img = tf.io.read_file(path)
        img = tf.image.decode_jpeg(img, channels=3)
        img = tf.image.resize(img, [224, 224])
        img = preprocess_input(img)
        return img

    ds = ds.map(_process, num_parallel_calls=tf.data.AUTOTUNE)
    ds = ds.batch(batch_size, drop_remainder=False)
    ds = ds.prefetch(tf.data.AUTOTUNE)
    return ds
