# 5. Code solution

## === cell 0
import os
import json
import numpy as np
import pandas as pd
from PIL import Image
import tensorflow as tf
from tensorflow.keras import layers, models, applications, optimizers
from sklearn.model_selection import train_test_split

tf.config.optimizer.set_jit(True)

gpus = tf.config.list_physical_devices("GPU")
if gpus:
    try:
        for gpu in gpus:
            tf.config.experimental.set_memory_growth(gpu, True)
    except Exception:
        pass
cpu_count = os.cpu_count() or 4
tf.config.threading.set_inter_op_parallelism_threads(cpu_count)
tf.config.threading.set_intra_op_parallelism_threads(cpu_count)

tf.random.set_seed(42)
np.random.seed(42)




## === cell 1
BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")
IMG_HEIGHT = 224
IMG_WIDTH = 224
BATCH_SIZE = 256  # larger batch reduces steps per epoch
EPOCHS = 5
NUM_CLASSES = 5




## === cell 2
with open(os.path.join(BASE_DIR, "label_num_to_disease_map.json")) as f:
    label_map = json.load(f)
print(json.dumps(label_map, indent=2))




## === cell 3
train_df = pd.read_csv(os.path.join(BASE_DIR, "train.csv"))
train_df["label"] = train_df["label"].astype(int)




## === cell 4
train_paths, val_paths, train_labels, val_labels = train_test_split(
    train_df["image_id"].values,
    train_df["label"].values,
    test_size=0.2,
    stratify=train_df["label"],
    random_state=42,
)


def _process_path(file_path, label):
    """Read image, decode, resize and normalize."""
    full_path = tf.strings.join([TRAIN_DIR, "/", file_path])
    img = tf.io.read_file(full_path)
    img = tf.image.decode_jpeg(img, channels=3, dct_method="INTEGER_FAST")
    img = tf.image.resize(img, [IMG_HEIGHT, IMG_WIDTH])
    img = img / 255.0
    return img, label


def _process_path_test(file_path):
    full_path = tf.strings.join([TEST_DIR, "/", file_path])
    img = tf.io.read_file(full_path)
    img = tf.image.decode_jpeg(img, channels=3, dct_method="INTEGER_FAST")
    img = tf.image.resize(img, [IMG_HEIGHT, IMG_WIDTH])
    img = img / 255.0
    return img


def augment(image, label):
    image = tf.image.random_flip_left_right(image)
    image = tf.image.random_flip_up_down(image)
    image = tf.image.random_brightness(image, max_delta=0.2)
    image = tf.image.random_contrast(image, lower=0.8, upper=1.2)
    return image, label


TRAIN_CACHE_PATH = "/tmp/train_cache"
VAL_CACHE_PATH = "/tmp/val_cache"

train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = (
    train_ds.map(_process_path, num_parallel_calls=tf.data.AUTOTUNE)
    .cache(TRAIN_CACHE_PATH)  # <-- cache after decode/resize
    .shuffle(1024)
    .map(augment, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

val_ds = tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
val_ds = (
    val_ds.map(_process_path, num_parallel_calls=tf.data.AUTOTUNE)
    .cache(VAL_CACHE_PATH)  # <-- cache validation set
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)




## === cell 5
from tensorflow.keras import mixed_precision

if tf.config.list_physical_devices("GPU"):
    mixed_precision.set_global_policy("mixed_float16")

base_model = applications.EfficientNetB0(
    input_shape=(IMG_HEIGHT, IMG_WIDTH, 3), include_top=False, weights="imagenet"
)
base_model.trainable = False  # freeze backbone

model = models.Sequential(
    [
        base_model,
        layers.GlobalAveragePooling2D(),
        layers.Dropout(0.2),
        layers.Dense(NUM_CLASSES, activation="softmax"),
    ]
)

model.compile(
    optimizer=optimizers.Adam(),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()




## === cell 6
history = model.fit(train_ds, validation_data=val_ds, epochs=EPOCHS, verbose=2)




## === cell 7
test_df = pd.read_csv(os.path.join(BASE_DIR, "sample_submission.csv"))[["image_id"]]
test_paths = test_df["image_id"].values
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)

TEST_CACHE_PATH = "/tmp/test_cache"
test_ds = (
    test_ds.map(lambda x: _process_path_test(x), num_parallel_calls=tf.data.AUTOTUNE)
    .cache(TEST_CACHE_PATH)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)




## === cell 8
preds = model.predict(test_ds, verbose=0)
pred_labels = np.argmax(preds, axis=1)

submission = pd.DataFrame({"image_id": test_paths, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")




## === cell 9
print(submission.head())
