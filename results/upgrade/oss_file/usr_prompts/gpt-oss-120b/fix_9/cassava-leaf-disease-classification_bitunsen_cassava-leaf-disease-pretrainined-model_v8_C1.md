# 5. Code solution

## === cell 0
import os
import json
import numpy as np
import pandas as pd
import tensorflow as tf

tf.config.optimizer.set_jit(True)

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"
tf.random.set_seed(42)




## === cell 1
BASE_DIR = "/kaggle/input/cassava-leaf-disease-classification/"
TRAIN_DIR = os.path.join(BASE_DIR, "train_images")
TEST_DIR = os.path.join(BASE_DIR, "test_images")




## === cell 2
with open(os.path.join(BASE_DIR, "label_num_to_disease_map.json")) as f:
    label_map = json.load(f)
print("Label map:", label_map)




## === cell 3
train_df = pd.read_csv(os.path.join(BASE_DIR, "train.csv"))
print("Training samples:", train_df.shape[0])
train_df["label"] = train_df["label"].astype(str)




## === cell 4
IMG_HEIGHT = 224
IMG_WIDTH = 224
BATCH_SIZE = 256  # increased batch size to reduce iteration overhead
NUM_CLASSES = 5
EPOCHS = 5
SEED = 42

train_df["filepath"] = train_df["image_id"]

rotation_layer = tf.keras.layers.RandomRotation(0.04, seed=SEED)

file_paths = tf.strings.join([TRAIN_DIR, "/", train_df["image_id"].values])
labels_int = train_df["label"].astype(int).values

full_ds = tf.data.Dataset.from_tensor_slices((file_paths, labels_int))
full_ds = full_ds.shuffle(
    buffer_size=len(file_paths), seed=SEED, reshuffle_each_iteration=False
)

val_size = int(0.1 * len(file_paths))
val_ds = full_ds.take(val_size)
train_ds = full_ds.skip(val_size)


def _parse_image_no_aug(path, label):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_HEIGHT, IMG_WIDTH])
    img = tf.keras.applications.mobilenet_v2.preprocess_input(img)
    return img, label


def _augment(img, label):
    img = tf.image.random_flip_left_right(img, seed=SEED)
    img = rotation_layer(img)
    label_one_hot = tf.one_hot(label, NUM_CLASSES)
    return img, label_one_hot


train_ds = (
    train_ds.map(_parse_image_no_aug, num_parallel_calls=tf.data.AUTOTUNE)
    .cache()
    .shuffle(buffer_size=1000, seed=SEED)  # reshuffle each epoch
    .map(_augment, num_parallel_calls=tf.data.AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)

val_ds = (
    val_ds.map(_parse_image_no_aug, num_parallel_calls=tf.data.AUTOTUNE)
    .cache()
    .map(
        lambda img, label: (img, tf.one_hot(label, NUM_CLASSES)),
        num_parallel_calls=tf.data.AUTOTUNE,
    )
    .batch(BATCH_SIZE)
    .prefetch(tf.data.AUTOTUNE)
)




## === cell 5
base_model = tf.keras.applications.MobileNetV2(
    input_shape=(IMG_HEIGHT, IMG_WIDTH, 3),
    include_top=False,
    weights="imagenet",
    pooling="avg",
)

base_model.trainable = False

outputs = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(base_model.output)
model = tf.keras.Model(inputs=base_model.input, outputs=outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.summary()




## === cell 6
model.fit(train_ds, epochs=EPOCHS, validation_data=val_ds, verbose=2)

base_model.trainable = True
for layer in base_model.layers[:-20]:
    layer.trainable = False

model.compile(
    optimizer=tf.keras.optimizers.Adam(1e-5),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(train_ds, epochs=2, validation_data=val_ds, verbose=2)




## === cell 7
test_filenames = [f for f in os.listdir(TEST_DIR) if f.lower().endswith(".jpg")]
test_paths = tf.strings.join([TEST_DIR, "/", test_filenames])

test_ds = tf.data.Dataset.from_tensor_slices(test_paths)


def _load_test(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, [IMG_HEIGHT, IMG_WIDTH])
    img = tf.keras.applications.mobilenet_v2.preprocess_input(img)
    return img


test_ds = test_ds.map(_load_test, num_parallel_calls=tf.data.AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

preds = model.predict(test_ds, verbose=1)
pred_labels = np.argmax(preds, axis=1)

submission = pd.DataFrame({"image_id": test_filenames, "label": pred_labels})
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}, shape: {submission.shape}")
