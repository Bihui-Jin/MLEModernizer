## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

from tensorflow.keras import mixed_precision

mixed_precision.set_global_policy("mixed_float16")

import re
import numpy as np
import tensorflow as tf
from functools import partial

IMAGE_SIZE = (224, 224)  # Height, Width
BATCH_SIZE = 32
NUM_CLASSES = 5
AUTOTUNE = tf.data.AUTOTUNE

BASE_INPUT = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_TFREC_PATH = os.path.join(BASE_INPUT, "train_tfrecords")
TEST_TFREC_PATH = os.path.join(BASE_INPUT, "test_tfrecords")




## === cell 1
def dataset_sizes(filenames):
    """Return total number of samples encoded in TFRecord filenames."""
    n = [
        int(re.compile(r"-([0-9]*)\.").search(filename).group(1))
        for filename in filenames
    ]
    return np.sum(n)


def decode_img(img_bytes):
    """Decode JPEG bytes, resize, and normalize."""
    img = tf.io.decode_jpeg(img_bytes, channels=3)
    img = tf.image.resize(img, IMAGE_SIZE)
    img = tf.cast(img, tf.float32) / 255.0
    return img


def read_tfrecord(example, labeled):
    """Parse a single TFRecord example."""
    if labeled:
        tfrec_format = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "target": tf.io.FixedLenFeature([], tf.int64),
        }
    else:
        tfrec_format = {
            "image": tf.io.FixedLenFeature([], tf.string),
            "image_name": tf.io.FixedLenFeature([], tf.string),
        }
    example = tf.io.parse_single_example(example, tfrec_format)
    img = decode_img(example["image"])
    if labeled:
        label = tf.cast(example["target"], tf.int32)
        return img, label
    else:
        return img, example["image_name"]


def load_dataset(filenames, labeled=True, ordered=False):
    """Create a tf.data.Dataset from TFRecord files."""
    options = tf.data.Options()
    if not ordered:
        options.experimental_deterministic = False
    ds = tf.data.TFRecordDataset(filenames, num_parallel_reads=AUTOTUNE)
    ds = ds.with_options(options)
    ds = ds.map(partial(read_tfrecord, labeled=labeled), num_parallel_calls=AUTOTUNE)
    return ds


TEST_FILENAMES = tf.io.gfile.glob(os.path.join(TEST_TFREC_PATH, "ld_test*.tfrec"))
NUM_TEST_IMAGES = dataset_sizes(TEST_FILENAMES)


def get_test_data(ordered=False):
    """Return a batched test dataset yielding (image, image_id)."""
    ds = load_dataset(TEST_FILENAMES, labeled=False, ordered=ordered)
    ds = ds.cache()
    ds = ds.batch(BATCH_SIZE)
    ds = ds.prefetch(AUTOTUNE)
    return ds


def get_train_data(ordered=False):
    """Return a shuffled, batched training dataset."""
    train_filenames = tf.io.gfile.glob(
        os.path.join(TRAIN_TFREC_PATH, "ld_train*.tfrec")
    )
    ds = load_dataset(train_filenames, labeled=True, ordered=ordered)
    ds = ds.cache(filename="/tmp/train_cache")
    ds = ds.shuffle(10000)
    ds = ds.batch(BATCH_SIZE)
    ds = ds.prefetch(AUTOTUNE)
    return ds




## === cell 2
pretrained_path = "/kaggle/input/cassava-trained/EfficientNetB4_based(24).h5"

if tf.io.gfile.exists(pretrained_path):
    print("Loading pretrained model from:", pretrained_path)
    model = tf.keras.models.load_model(pretrained_path, compile=False)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )
else:
    print("Pretrained model not found – building a fallback EfficientNetB4 model.")
    base = tf.keras.applications.EfficientNetB4(
        include_top=False,
        weights="imagenet",
        input_shape=(*IMAGE_SIZE, 3),
    )
    x = tf.keras.layers.GlobalAveragePooling2D()(base.output)
    output = tf.keras.layers.Dense(NUM_CLASSES, activation="softmax")(x)
    model = tf.keras.Model(inputs=base.input, outputs=output)
    model.compile(
        optimizer=tf.keras.optimizers.Adam(),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"],
    )
    print(
        "Training the fallback model for a few epochs (this may take a few minutes)..."
    )
    train_ds = get_train_data()
    model.fit(train_ds, epochs=10, verbose=2)




## === cell 3
test_ds = get_test_data()

test_images_ds = test_ds.map(lambda img, img_id: img)

print("Running predictions on test set...")
probabilities = model.predict(test_images_ds, verbose=1)
predictions = np.argmax(probabilities, axis=-1).astype(int)
print("Predictions shape:", predictions.shape)




## === cell 4
print("Generating submission.csv file...")

test_ids_ds = test_ds.map(lambda img, img_id: img_id).unbatch()
test_ids = np.array([fid.decode("utf-8") for fid in test_ids_ds.as_numpy_iterator()])

submission_path = "submission.csv"
np.savetxt(
    submission_path,
    np.rec.fromarrays([test_ids, predictions]),
    fmt=["%s", "%d"],
    delimiter=",",
    header="image_id,label",
    comments="",
)

print(f"Submission file saved to {submission_path}")
