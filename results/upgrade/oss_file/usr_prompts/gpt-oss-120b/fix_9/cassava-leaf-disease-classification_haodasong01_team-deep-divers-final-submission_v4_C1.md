import os, json, random, time

## === cell 1
train_path = "/kaggle/input/cassava-leaf-disease-classification/train.csv"
test_path = "/kaggle/input/cassava-leaf-disease-classification/sample_submission.csv"
train_dir = "/kaggle/input/cassava-leaf-disease-classification/train_images/"
test_dir = "/kaggle/input/cassava-leaf-disease-classification/test_images/"

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

numeric_mode_label = int(train_df["label"].mode()[0])

train_df["label"] = train_df["label"].astype(str)
train_df["filepath"] = train_dir + train_df["image_id"].astype(str)
test_df["filepath"] = test_dir + test_df["image_id"].astype(str)

train_df, val_df = train_test_split(
    train_df, test_size=0.1, stratify=train_df["label"], random_state=42
)

from sklearn.ensemble import RandomForestClassifier
from PIL import Image


def extract_features(img_path, size=(128, 128)):
    """Resize image, flatten RGB pixels, and append channel mean/std."""
    try:
        img = Image.open(img_path).convert("RGB")
        img = img.resize(size)
        arr = np.asarray(img, dtype=np.float32) / 255.0  # (H, W, 3)
        flat = arr.flatten()
        channel_means = arr.mean(axis=(0, 1))
        channel_stds = arr.std(axis=(0, 1))
        stats = np.concatenate([channel_means, channel_stds])
        return np.concatenate([flat, stats])
    except Exception:
        dim = size[0] * size[1] * 3 + 6  # 6 stats = 3 means + 3 stds
        return np.zeros(dim, dtype=np.float32)


train_features = []
train_labels = []

for _, row in train_df.iterrows():
    feats = extract_features(row["filepath"])
    train_features.append(feats)
    train_labels.append(int(row["label"]))

X_train = np.stack(train_features)
y_train = np.array(train_labels)

rf = RandomForestClassifier(
    n_estimators=400,  # slightly larger forest for better capacity
    max_depth=None,
    random_state=42,
    n_jobs=-1,
    min_samples_leaf=1,
)
rf.fit(X_train, y_train)

test_features = [extract_features(p) for p in test_df["filepath"]]
X_test = np.stack(test_features)
pred_labels = rf.predict(X_test).astype(int)




## === cell 2
submission = pd.DataFrame({"image_id": test_df["image_id"], "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission written to {submission_path}")
