# 5. Code solution

## === cell 0
import os

os.environ["PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION"] = "python"

import random
import numpy as np
import pandas as pd

import tensorflow as tf
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.layers import (
    GlobalAveragePooling2D,
    Dense,
    RandomFlip,
    RandomRotation,
    RandomTranslation,
)
from tensorflow.keras.models import Model
from tensorflow.keras.callbacks import EarlyStopping
from sklearn.model_selection import train_test_split

from tensorflow.keras import mixed_precision

mixed_precision.set_global_policy("mixed_float16")

random.seed(42)
np.random.seed(42)
tf.random.set_seed(42)

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")



## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
train_df["filepath"] = train_df["image_id"].apply(
    lambda x: os.path.join(TRAIN_IMG_DIR, x)
)
train_df["label"] = train_df["label"].astype(str)

df_train, df_val = train_test_split(
    train_df,
    test_size=0.1,
    stratify=train_df["label"],
    random_state=42,
)

label_list = sorted(train_df["label"].unique())
label_to_idx = {lbl: i for i, lbl in enumerate(label_list)}
num_classes = len(label_list)


def df_to_arrays(df):
    paths = df["filepath"].values
    labels = df["label"].map(label_to_idx).values
    return paths, labels


train_paths, train_labels = df_to_arrays(df_train)
val_paths, val_labels = df_to_arrays(df_val)



## === cell 2
IMG_SIZE = (224, 224)
BATCH_SIZE = 64
AUTOTUNE = tf.data.AUTOTUNE
SHUFFLE_BUFFER = 2000  # moderate buffer to limit memory usage


def decode_and_preprocess(path, label):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    label = tf.one_hot(label, depth=num_classes)
    return img, label


augmentation = tf.keras.Sequential(
    [
        RandomFlip("horizontal"),
        RandomRotation(0.1111111),  # ~20 degrees
        RandomTranslation(0.1, 0.1),
    ]
)


def augment(image, label):
    image = augmentation(image, training=True)
    return image, label


train_ds = (
    tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
    .shuffle(SHUFFLE_BUFFER, seed=42, reshuffle_each_iteration=True)
    .map(decode_and_preprocess, num_parallel_calls=AUTOTUNE)
    .map(augment, num_parallel_calls=AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

val_ds = (
    tf.data.Dataset.from_tensor_slices((val_paths, val_labels))
    .map(decode_and_preprocess, num_parallel_calls=AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)



## === cell 3
base = ResNet50(weights="imagenet", include_top=False, input_shape=(224, 224, 3))
base.trainable = False  # keep backbone frozen for the first phase

x = GlobalAveragePooling2D()(base.output)
output = Dense(num_classes, activation="softmax", dtype="float32")(
    x
)  # force logits to float32
model = Model(inputs=base.input, outputs=output)



## === cell 4
model.compile(
    optimizer="adam",
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

early_stop = EarlyStopping(
    patience=8,  # longer patience for the extended epoch count
    restore_best_weights=True,
    verbose=1,
)

model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=30,  # more epochs to let the head learn thoroughly
    callbacks=[early_stop],
    verbose=2,
)



## === cell 5
for layer in base.layers[-10:]:  # unfreeze last ~10 layers
    layer.trainable = True

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-5),
    loss="categorical_crossentropy",
    metrics=["accuracy"],
)

model.fit(
    train_ds,
    validation_data=val_ds,
    epochs=5,
    callbacks=[early_stop],
    verbose=2,
)



## === cell 6
sample_sub = pd.read_csv(SAMPLE_SUB)  # keeps required order
sample_sub["filepath"] = sample_sub["image_id"].apply(
    lambda x: os.path.join(TEST_IMG_DIR, x)
)
test_paths = sample_sub["filepath"].values


def decode_only(path):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    return img


test_ds = (
    tf.data.Dataset.from_tensor_slices(test_paths)
    .map(lambda p: decode_only(p), num_parallel_calls=AUTOTUNE)
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

test_probs = model.predict(test_ds, verbose=0)
test_pred = np.argmax(test_probs, axis=1)

submission = pd.DataFrame({"image_id": sample_sub["image_id"], "label": test_pred})
submission.to_csv("submission.csv", index=False)
print("Submission saved to submission.csv")
