## === cell 0
import os, json, numpy as np, pandas as pd, tensorflow as tf
from tensorflow.keras import layers, models, mixed_precision

tf.random.set_seed(42)
np.random.seed(42)

ROOT_DIR = "/kaggle/input/cassava-leaf-disease-classification"
if not os.path.isdir(ROOT_DIR):
    ROOT_DIR = os.path.abspath("../input/cassava-leaf-disease-classification")
TRAIN_IMG_DIR = os.path.join(ROOT_DIR, "train_images")
TEST_IMG_DIR = os.path.join(ROOT_DIR, "test_images")
TRAIN_CSV = os.path.join(ROOT_DIR, "train.csv")
SAMPLE_SUB = os.path.join(ROOT_DIR, "sample_submission.csv")

mixed_precision.set_global_policy("mixed_float16")

tf.config.threading.set_intra_op_parallelism_threads(4)
tf.config.threading.set_inter_op_parallelism_threads(4)




## === cell 1
train_df = pd.read_csv(TRAIN_CSV)
train_df["filepath"] = TRAIN_IMG_DIR + "/" + train_df["image_id"]
train_paths = train_df["filepath"].values
train_labels = train_df["label"].values.astype(np.int32)




## === cell 2
IMG_SIZE = 224
BATCH_SIZE = 64  # unchanged batch size
AUTOTUNE = tf.data.AUTOTUNE
CACHE_PATH = "/tmp/train_cache"  # disk‑based cache to avoid memory blow‑up


def _decode_and_resize(filename):
    """Read, decode and resize an image without augmentation."""
    image = tf.io.read_file(filename)
    image = tf.image.decode_jpeg(image, channels=3)
    image = tf.image.resize(image, [IMG_SIZE, IMG_SIZE])
    image = image / 255.0
    return image


def _augment(image, label):
    """Apply the same light augmentation used originally."""
    image = tf.image.random_flip_left_right(image)
    return image, label


train_ds = tf.data.Dataset.from_tensor_slices((train_paths, train_labels))
train_ds = train_ds.map(
    lambda x, y: (_decode_and_resize(x), y), num_parallel_calls=AUTOTUNE
)
train_ds = train_ds.cache(CACHE_PATH)  # cache decoded images on disk
train_ds = train_ds.map(_augment, num_parallel_calls=AUTOTUNE)
train_ds = (
    train_ds.shuffle(buffer_size=10000, seed=42)
    .batch(BATCH_SIZE, drop_remainder=True)
    .prefetch(AUTOTUNE)
)




## === cell 3
model = models.Sequential(
    [
        layers.Conv2D(
            32, (3, 3), activation="relu", input_shape=(IMG_SIZE, IMG_SIZE, 3)
        ),
        layers.MaxPooling2D(),
        layers.Conv2D(64, (3, 3), activation="relu"),
        layers.MaxPooling2D(),
        layers.Conv2D(128, (3, 3), activation="relu"),
        layers.MaxPooling2D(),
        layers.Flatten(),
        layers.Dense(128, activation="relu"),
        layers.Dropout(0.3),
        layers.Dense(5, activation="softmax"),
    ]
)
model.compile(
    optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"]
)




## === cell 4
EPOCHS = 10  # keep original epoch count
model.fit(train_ds, epochs=EPOCHS, verbose=1)




## === cell 5
test_files = sorted([f for f in os.listdir(TEST_IMG_DIR) if f.lower().endswith(".jpg")])
test_paths = [os.path.join(TEST_IMG_DIR, f) for f in test_files]
test_ds = tf.data.Dataset.from_tensor_slices(test_paths)
test_ds = test_ds.map(lambda x: _decode_and_resize(x), num_parallel_calls=AUTOTUNE)
test_ds = test_ds.batch(BATCH_SIZE).prefetch(AUTOTUNE)




## === cell 6
pred_probs = model.predict(test_ds, verbose=0)
preds = np.argmax(pred_probs, axis=1)




## === cell 7
assert len(test_files) == len(
    preds
), "Length mismatch between test files and predictions"
submission = pd.DataFrame({"image_id": test_files, "label": preds})
submission_path = os.path.join("/kaggle/working", "submission.csv")
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path} (rows: {len(submission)})")
