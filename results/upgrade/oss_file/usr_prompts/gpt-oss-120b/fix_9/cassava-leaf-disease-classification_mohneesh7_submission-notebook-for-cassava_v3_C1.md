## === cell 0
import os
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score

SEED = 42
np.random.seed(SEED)

TRAIN_DIR = "/kaggle/input/cassava-leaf-disease-classification/train_images/"
TEST_DIR = "/kaggle/input/cassava-leaf-disease-classification/test_images/"
TRAIN_CSV = "/kaggle/input/cassava-leaf-disease-classification/train.csv"

train_df = pd.read_csv(TRAIN_CSV)
train_df["label"] = train_df["label"].astype(int)

IMG_SIZE = (64, 64)  # small size for fast processing


def load_and_preprocess(image_id, base_dir):
    path = os.path.join(base_dir, image_id)
    img = Image.open(path).convert("RGB")
    img = img.resize(IMG_SIZE)
    return np.asarray(img).astype(np.uint8)


train_images = []
train_labels = []
for idx, row in train_df.iterrows():
    img_array = load_and_preprocess(row["image_id"], TRAIN_DIR)
    train_images.append(img_array.flat)  # flatten
    train_labels.append(row["label"])
train_X = np.stack(train_images)
train_y = np.array(train_labels)

X_tr, X_val, y_tr, y_val = train_test_split(
    train_X, train_y, test_size=0.1, random_state=SEED, stratify=train_y
)

clf = LogisticRegression(
    multi_class="multinomial",
    solver="lbfgs",
    max_iter=200,
    n_jobs=4,
    random_state=SEED,
)
clf.fit(X_tr, y_tr)

val_pred = clf.predict(X_val)
print("Validation accuracy:", accuracy_score(y_val, val_pred))
