# 5. Code solution

## === cell 0
import os
import glob
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score

SEED = 42
np.random.seed(SEED)

NUM_WORKERS = max(1, os.cpu_count() - 1)



## === cell 1
BASE_PATH = "../input/cassava-leaf-disease-classification"
TRAIN_CSV = os.path.join(BASE_PATH, "train.csv")
TRAIN_IMG_DIR = os.path.join(BASE_PATH, "train_images")
TEST_IMG_DIR = os.path.join(BASE_PATH, "test_images")
SAMPLE_SUB = os.path.join(BASE_PATH, "sample_submission.csv")

df_train = pd.read_csv(TRAIN_CSV)
df_train["label"] = df_train["label"].astype(str)
df_train["path"] = df_train["image_id"].apply(lambda x: os.path.join(TRAIN_IMG_DIR, x))

train_df, val_df = train_test_split(
    df_train,
    test_size=0.1,
    random_state=SEED,
    stratify=df_train["label"],
)



## === cell 2
IMG_SIZE = (64, 64)  # smaller size for faster processing


def load_images(df):
    """Load and resize images from a dataframe containing a 'path' column."""
    images = []
    for p in df["path"]:
        img = Image.open(p).convert("RGB").resize(IMG_SIZE, Image.BILINEAR)
        arr = np.asarray(img, dtype=np.float32) / 255.0  # normalize
        images.append(arr.flatten())
    return np.stack(images)




## === cell 3
label_encoder = LabelEncoder()
label_encoder.fit(df_train["label"])

y_train = label_encoder.transform(train_df["label"])
y_val = label_encoder.transform(val_df["label"])

X_train = load_images(train_df)
X_val = load_images(val_df)



## === cell 4
model = LogisticRegression(
    multi_class="multinomial",
    solver="saga",
    max_iter=200,
    n_jobs=NUM_WORKERS,
    random_state=SEED,
)

model.fit(X_train, y_train)

val_pred = model.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy: {val_acc:.4f}")



## === cell 5
test_images = glob.glob(os.path.join(TEST_IMG_DIR, "*.jpg"))
df_test = pd.DataFrame({"path": test_images})

X_test = load_images(df_test)

test_pred_int = model.predict(X_test)
test_pred_labels = label_encoder.inverse_transform(test_pred_int)

submission = pd.DataFrame(
    {
        "image_id": df_test["path"].apply(lambda p: os.path.basename(p)),
        "label": test_pred_labels,
    }
)

sample_sub = pd.read_csv(SAMPLE_SUB)
submission = submission[sample_sub.columns]

submission.to_csv("submission.csv", index=False)
print("Submission file written to submission.csv")
