# 5. Code solution

## === cell 0
import os, json, numpy as np, pandas as pd, cv2
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score
from concurrent.futures import ThreadPoolExecutor  # lighter threads for I/O bound work

np.random.seed(42)

BASE_PATH = "/kaggle/input/cassava-leaf-disease-classification"
train_dir = os.path.join(BASE_PATH, "train_images")
train_csv = os.path.join(BASE_PATH, "train.csv")
label_json = os.path.join(BASE_PATH, "label_num_to_disease_map.json")
test_dir = os.path.join(BASE_PATH, "test_images")

with open(label_json) as f:
    label_name = json.load(f)

data_label = pd.read_csv(train_csv)




## === cell 1
def load_image(path):
    """Read, resize, normalize and flatten an image."""
    img = cv2.imread(path, cv2.IMREAD_COLOR)
    if img is None:
        return None
    img = cv2.resize(img, (100, 100))
    img = img.astype(np.float32) / 255.0
    return img.flatten()  # 100*100*3 = 30000 features


train_paths = [os.path.join(train_dir, img_id) for img_id in data_label["image_id"]]
train_labels = data_label["label"].astype(int).tolist()

max_workers = min(8, os.cpu_count() or 1)

num_samples = len(train_paths)
feature_dim = 100 * 100 * 3
X_full = np.empty((num_samples, feature_dim), dtype=np.float32)
valid_mask = np.zeros(num_samples, dtype=bool)

with ThreadPoolExecutor(max_workers=max_workers) as executor:
    for idx, vec in enumerate(executor.map(load_image, train_paths)):
        if vec is not None:
            X_full[idx] = vec
            valid_mask[idx] = True

X = X_full[valid_mask]
y = np.array([train_labels[i] for i, ok in enumerate(valid_mask) if ok], dtype=np.int64)



## === cell 2
X_train, X_val, y_train, y_val = train_test_split(
    X, y, test_size=0.1, random_state=42, stratify=y
)

clf = LogisticRegression(
    multi_class="multinomial",
    solver="saga",
    max_iter=100,
    n_jobs=-1,
    verbose=0,
    penalty="l2",
    C=1.0,
    random_state=42,
)
clf.fit(X_train, y_train)

val_pred = clf.predict(X_val)
val_acc = accuracy_score(y_val, val_pred)
print(f"Validation accuracy: {val_acc:.5f}")



## === cell 3
test_images = os.listdir(test_dir)
test_paths = [os.path.join(test_dir, img_name) for img_name in test_images]

num_test = len(test_paths)
test_full = np.empty((num_test, feature_dim), dtype=np.float32)
test_valid_mask = np.zeros(num_test, dtype=bool)

with ThreadPoolExecutor(max_workers=max_workers) as executor:
    for idx, vec in enumerate(executor.map(load_image, test_paths)):
        if vec is not None:
            test_full[idx] = vec
            test_valid_mask[idx] = True

if test_valid_mask.any():
    test_matrix = test_full[test_valid_mask]
    test_names = np.array(test_images)[test_valid_mask]
    pred_labels = clf.predict(test_matrix).astype(int).tolist()
    test_names = test_names.tolist()
else:
    test_names, pred_labels = [], []

submission_df = pd.DataFrame({"image_id": test_names, "label": pred_labels})
submission_path = "/kaggle/working/submission.csv"
submission_df.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
