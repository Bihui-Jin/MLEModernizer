DISEASES = [
    "scab",
    "frog_eye_leaf_spot",
    "complex",
    "rust",
    "powdery_mildew",
    "healthy",
]
num_classes = len(DISEASES)

DISEASE_TO_IDX = {d: i for i, d in enumerate(DISEASES)}


def multi_hot(label_str):
    arr = np.zeros(num_classes, dtype=np.float32)
    for token in label_str.split():
        idx = DISEASE_TO_IDX.get(token)
        if idx is not None:
            arr[idx] = 1.0
    return arr


train_labels = np.stack(train_df["labels"].apply(multi_hot).values)




## === cell 3
IMG_SIZE = (224, 224)
AUTOTUNE = tf.data.experimental.AUTOTUNE
BATCH_SIZE = 256  # increased to reduce iteration overhead


def decode_image(path, label=None):
    img = tf.io.read_file(path)
    img = tf.image.decode_jpeg(img, channels=3)
    img = tf.image.resize(img, IMG_SIZE)
    img = img / 255.0
    if label is None:
        return img
    else:
        return img, label


base_model = applications.ResNet50(
    weights="imagenet", include_top=False, input_shape=IMG_SIZE + (3,)
)
base_model.trainable = False  # freeze ImageNet weights

feature_extractor = models.Sequential(
    [base_model, layers.GlobalAveragePooling2D()], name="feature_extractor"
)

raw_train_ds = (
    tf.data.Dataset.from_tensor_slices(train_paths)
    .map(lambda p: decode_image(p), num_parallel_calls=AUTOTUNE)
    .cache()  # cache decoded images in memory
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

raw_test_ds = (
    tf.data.Dataset.from_tensor_slices(test_paths_ordered)
    .map(lambda p: decode_image(p), num_parallel_calls=AUTOTUNE)
    .cache()
    .batch(BATCH_SIZE)
    .prefetch(AUTOTUNE)
)

train_features = feature_extractor.predict(raw_train_ds, verbose=0)
test_features = feature_extractor.predict(raw_test_ds, verbose=0)

classifier = models.Sequential(
    [
        layers.Input(shape=train_features.shape[1:]),
        layers.Dense(num_classes, activation="sigmoid"),
    ],
    name="classifier",
)

classifier.compile(
    optimizer=optimizers.Adam(),
    loss="binary_crossentropy",
    metrics=[tf.keras.metrics.AUC(name="auc")],
)
