## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

## Submission Format
For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value
00001,0.5
00013,0.5
00015,0.5
etc.
```

- **sample_submission.csv** - a sample submission file in the correct format

## === cell 0
import os
import pandas as pd
import numpy as np



## === cell 1
_possible_paths = [
    os.path.join(
        os.getcwd(), "input", "rsna-miccai-brain-tumor-radiogenomic-classification"
    ),
    "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification",
]
BASE_INPUT = None
for p in _possible_paths:
    if os.path.isdir(p):
        BASE_INPUT = p
        break
if BASE_INPUT is None:
    raise FileNotFoundError("Could not locate the dataset input directory.")

TRAIN_LABELS_PATH = os.path.join(BASE_INPUT, "train_labels.csv")
TRAIN_DIR = os.path.join(BASE_INPUT, "train")
TEST_DIR = os.path.join(BASE_INPUT, "test")

train_df = pd.read_csv(TRAIN_LABELS_PATH)
global_mean = train_df["MGMT_value"].mean()
print(f"Global mean MGMT_value from training data: {global_mean:.5f}")




## === cell 2
def total_folder_size(path):
    """Return total size in bytes of all files under the given directory."""
    total = 0
    for root, _, files in os.walk(path):
        for f in files:
            fp = os.path.join(root, f)
            try:
                total += os.path.getsize(fp)
            except OSError:
                pass
    return total


train_cases = train_df["BraTS21ID"].astype(str).tolist()
train_sizes = []
train_labels = []
for case_id in train_cases:
    case_path = os.path.join(TRAIN_DIR, case_id)
    if os.path.isdir(case_path):
        size = total_folder_size(case_path)
        train_sizes.append(size)
        label = train_df.loc[
            train_df["BraTS21ID"] == int(case_id), "MGMT_value"
        ].values[0]
        train_labels.append(label)
    else:
        print(f"Warning: training folder for case {case_id} not found, skipping.")
train_sizes = np.array(train_sizes, dtype=float)
train_labels = np.array(train_labels, dtype=float)

if len(train_sizes) >= 2:
    slope, intercept = np.polyfit(train_sizes, train_labels, 1)
    print(f"Fitted linear model: pred = {slope:.3e} * size + {intercept:.5f}")
else:
    slope, intercept = 0.0, global_mean
    print("Not enough data for regression; falling back to global mean.")



## === cell 3
test_cases = sorted(
    [
        name
        for name in os.listdir(TEST_DIR)
        if os.path.isdir(os.path.join(TEST_DIR, name))
    ]
)
print(f"Found {len(test_cases)} test cases.")

test_sizes = np.array(
    [total_folder_size(os.path.join(TEST_DIR, cid)) for cid in test_cases], dtype=float
)

if len(test_sizes) > 0 and (slope != 0.0 or intercept != global_mean):
    raw_preds = slope * test_sizes + intercept
    preds = np.clip(raw_preds, 0.0, 1.0)
else:
    preds = np.full(len(test_cases), global_mean, dtype=float)

preds = np.where(np.isnan(preds), global_mean, preds)

submission = pd.DataFrame({"BraTS21ID": test_cases, "MGMT_value": preds})
print("Sample of submission DataFrame:")
print(submission.head())



## === cell 4
submission_path = "submission.csv"
submission.to_csv(submission_path, index=False)
print(f"Submission saved to {submission_path}")
