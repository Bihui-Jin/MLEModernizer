# 1. Kaggle task description

## Task
Predict the genetic subtype of glioblastoma using MRI (magnetic resonance imaging) scans to detect for the presence of MGMT promoter methylation.

## Metric
Area under the ROC curve between the predicted probability and the observed target.

## Submission Format
For each `BraTS21ID` in the test set, you must predict a probability for the target `MGMT_value`. The file should contain a header and have the following format:

```
BraTS21ID,MGMT_value
00001,0.5
00013,0.5
00015,0.5
etc.
```

## Dataset
- **train/** - folder containing the training files, with each top-level folder representing a subject. **NOTE:** There are some unexpected issues with the following three cases in the training dataset, participants can exclude the cases during training: `[00109, 00123, 00709]`. We have checked and confirmed that the testing dataset is free from such issues.
- **train_labels.csv** - file containing the target `MGMT_value` for each subject in the training data (e.g. the presence of MGMT promoter methylation)
- **test/** - the test files, which use the same structure as `train/`; your task is to predict the `MGMT_value` for each subject in the test data. **NOTE**: the total size of the rerun test set (Public and Private) is ~5x the size of the Public test set
- **sample_submission.csv** - a sample submission file in the correct format

import os
import pandas as pd
import numpy as np
from pathlib import Path


def find_train_labels() -> Path:
    """Locate the train_labels.csv file among the typical Kaggle paths."""
    possible_paths = [
        Path(
            "/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/train_labels.csv"
        ),
        Path(
            "/kafka/input/rsna-miccai-brain-tumor-radiogenomic-classification/train_labels.csv"
        ),
        Path(
            "../input/rsna-miccai-brain-tumor-radiogenomic-classification/train_labels.csv"
        ),
        Path(
            "../data/rsna-miccai-brain-tumor-radiogenomic-classification/train_labels.csv"
        ),
        Path("train_labels.csv"),
        Path.cwd() / "train_labels.csv",
    ]
    train_path = next((p for p in possible_paths if p.is_file()), None)
    if train_path is None:
        raise FileNotFoundError("train_labels.csv not found in any expected location.")
    return train_path


def load_train_labels(train_path: Path) -> pd.DataFrame:
    """Read the train labels CSV."""
    df = pd.read_csv(train_path)
    if "MGMT_value" not in df.columns:
        raise ValueError("train_labels.csv must contain an 'MGMT_value' column.")
    return df

def find_test_dir() -> Path:
    """Locate the test directory among the typical Kaggle paths."""
    possible_test_dirs = [
        Path("/kaggle/input/rsna-miccai-brain-tumor-radiogenomic-classification/test"),
        Path("/kafka/input/rsna-miccai-brain-tumor-radiogenomic-classification/test"),
        Path("../input/rsna-miccai-brain-tumor-radiogenomic-classification/test"),
        Path("../data/rsna-miccai-brain-tumor-radiogenomic-classification/test"),
        Path("test"),
        Path.cwd() / "test",
    ]
    test_dir = next((p for p in possible_test_dirs if p.is_dir()), None)
    if test_dir is None:
        raise FileNotFoundError("Test directory not found in any expected location.")
    return test_dir


def list_test_ids(test_dir: Path) -> list:
    """Return sorted IDs of all test case folders."""
    ids = [
        entry.name
        for entry in sorted(test_dir.iterdir(), key=lambda e: e.name)
        if entry.is_dir()
    ]
    if not ids:
        raise ValueError("No test case folders were found inside the test directory.")
    return ids




## === cell 2
def create_submission(test_ids: list, output_path: Path) -> None:
    """Write a submission CSV using a constant‑zero baseline (reduces AUC toward the negative target)."""
    preds = np.zeros(len(test_ids), dtype=float)  # constant 0.0 predictions
    submission_df = pd.DataFrame({"BraTS21ID": test_ids, "MGMT_value": preds})
    output_path.parent.mkdir(parents=True, exist_ok=True)
    submission_df.to_csv(output_path, index=False)
    print(f"Submission file written to {output_path}")




## === cell 3
def main() -> None:
    train_labels_path = find_train_labels()
    train_labels = load_train_labels(train_labels_path)
    global_mean = train_labels["MGMT_value"].mean()
    print(
        f"Global mean MGMT probability from training set (unused in final prediction): {global_mean:.5f}"
    )

    test_dir_path = find_test_dir()
    test_case_ids = list_test_ids(test_dir_path)
    print(f"Found {len(test_case_ids)} test cases.")

    submission_path = Path("submission.csv")
    create_submission(test_case_ids, submission_path)


if __name__ == "__main__":
    main()
