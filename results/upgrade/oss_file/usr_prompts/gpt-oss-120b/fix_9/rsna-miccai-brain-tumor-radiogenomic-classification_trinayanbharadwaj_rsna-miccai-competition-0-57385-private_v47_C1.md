def _find_file(filename):
    """
    Return the first file matching *filename* anywhere under typical Kaggle input paths
    **or** under the current working directory. This broader search makes the script
    robust when data resides in ./data instead of /kaggle/input.
    """
    candidates = glob.glob(f"/kaggle/input/**/{filename}", recursive=True)
    if not candidates:
        candidates = glob.glob(f"./**/{filename}", recursive=True)
    return candidates[0] if candidates else None


def get_test_ids(path_test):
    """
    Return a list of BraTS21ID strings from folder names under the test directory.
    If the directory is missing/empty, fall back to IDs from a sample_submission file.
    """
    ids = []
    if path_test and os.path.isdir(path_test):
        for entry in sorted(os.scandir(path_test), key=lambda e: e.name):
            if entry.is_dir():
                ids.append(entry.name)  # folder name is the ID e.g., '00002'
    if not ids:
        sp_path = _find_file("sample_submission.csv")
        if sp_path:
            df = pd.read_csv(sp_path, dtype=str)
            ids = df["BraTS21ID"].tolist()
    return ids

def _simple_auc(y_true, y_pred):
    """Compute AUC using the Mann‑Whitney U statistic (no external libs)."""
    order = np.argsort(y_pred)
    y_true_sorted = y_true[order]
    n_pos = np.sum(y_true_sorted)
    n_neg = len(y_true_sorted) - n_pos
    if n_pos == 0 or n_neg == 0:
        return 0.5
    ranks = np.arange(1, len(y_true_sorted) + 1)
    sum_ranks_pos = np.sum(ranks[y_true_sorted == 1])
    auc = (sum_ranks_pos - n_pos * (n_pos + 1) / 2) / (n_pos * n_neg)
    return auc
